/*
 * ==========================================
 *  第09课：算术小天才——算术运算符
 *  知识点：加减乘除取余、整数除法、运算符优先级
 *  C++趣味学习系列
 * ==========================================
 */

#include <iostream>
using namespace std;

int main() {
    // =====================================
    // 示例1：五种算术运算初体验
    //   +  加法：把两个数加在一起
    //   -  减法：从一个数中减去另一个
    //   *  乘法：把两个数相乘
    //   /  除法：两个整数相除，结果只保留整数！
    //   %  取余：求除法剩下的余数
    // =====================================
    cout << "========== 示例1：五种算术运算 ==========" << endl;
    int a = 15, b = 4;
    cout << "a = " << a << ", b = " << b << endl;
    cout << endl;

    cout << "加法：a + b = " << a + b << endl;
    cout << "减法：a - b = " << a - b << endl;
    cout << "乘法：a * b = " << a * b << endl;

    // 注意：整数除法！15/4 在数学上是 3.75，但在 C++ 中整数除法只保留整数部分 3
    cout << "除法：a / b = " << a / b;
    cout << "  （15÷4=3.75，整数除法结果=3，小数被丢弃！）" << endl;

    // 取余：15÷4=3余3，所以余数是3
    cout << "取余：a % b = " << a % b;
    cout << "  （15÷4=3...余3）" << endl;
    cout << endl;

    // =====================================
    // 示例2：分糖果问题
    //   用「/」算每人分到几颗糖
    //   用「%」算分完之后还剩几颗糖
    // =====================================
    cout << "========== 示例2：分糖果问题 ==========" << endl;

    int candies, kids;
    cout << "请输入糖果总数：";
    cin >> candies;
    cout << "请输入小朋友人数：";
    cin >> kids;

    // 除法算每人分几颗
    int eachGets = candies / kids;
    // 取余算剩下几颗
    int leftOver = candies % kids;

    cout << endl;
    cout << "一共有 " << candies << " 颗糖果" << endl;
    cout << "分给 " << kids << " 个小朋友" << endl;
    cout << "每个小朋友分到：" << eachGets << " 颗" << endl;
    cout << "还剩下：" << leftOver << " 颗" << endl;

    // 判断是否正好分完
    if (leftOver == 0) {
        cout << "太棒了！正好分完，每人一样多！" << endl;
    } else {
        cout << "还剩下 " << leftOver << " 颗，老师帮你保管吧~" << endl;
    }
    cout << endl;

    // =====================================
    // 示例3：运算符优先级大挑战
    //   规则：括号最优先 → 乘除取余第二 → 加减最后
    //   同级运算符：从左到右依次计算
    // =====================================
    cout << "========== 示例3：优先级大挑战 ==========" << endl;

    // 对比：有没有括号，差别很大！
    cout << "3 + 5 * 2   = " << 3 + 5 * 2;
    cout << "  （先算5*2=10，再算3+10=13）" << endl;

    cout << "(3 + 5) * 2 = " << (3 + 5) * 2;
    cout << "  （先算括号3+5=8，再算8*2=16）" << endl;
    cout << endl;

    // 同级运算符，从左到右
    cout << "20 - 8 - 3 = " << 20 - 8 - 3;
    cout << "  （20-8=12，12-3=9）" << endl;

    cout << "24 / 3 * 2 = " << 24 / 3 * 2;
    cout << "  （24/3=8，8*2=16）" << endl;
    cout << endl;

    // 综合挑战：复杂表达式
    int result = (10 + 2) * 3 % 5 - 4 / 2;
    cout << "(10+2)*3%5 - 4/2 = " << result << endl;
    cout << "分步计算：" << endl;
    cout << "  ① (10+2) = 12" << endl;
    cout << "  ② 12 * 3 = 36" << endl;
    cout << "  ③ 36 % 5 = 1  （36÷5=7余1）" << endl;
    cout << "  ④ 4 / 2 = 2" << endl;
    cout << "  ⑤ 1 - 2 = -1" << endl;
    cout << endl;

    // =====================================
    // 练习1 参考答案：分糖果小能手
    //   17颗糖果分给5个小朋友
    // =====================================
    cout << "========== 练习1答案：分糖果 ==========" << endl;
    int candies1 = 17;
    int kids1 = 5;
    cout << candies1 << " 颗糖分给 " << kids1 << " 个小朋友：" << endl;
    cout << "每人分到：" << candies1 / kids1 << " 颗" << endl;
    cout << "还剩下：" << candies1 % kids1 << " 颗" << endl;
    cout << endl;

    // =====================================
    // 练习2 参考答案：优先级大考验
    //   计算 (3+5)*2 - 8/4
    // =====================================
    cout << "========== 练习2答案：优先级 ==========" << endl;
    cout << "(3+5)*2 - 8/4 = " << (3 + 5) * 2 - 8 / 4 << endl;
    cout << "分步计算：" << endl;
    cout << "  ① 3+5 = 8" << endl;
    cout << "  ② 8*2 = 16" << endl;
    cout << "  ③ 8/4 = 2" << endl;
    cout << "  ④ 16-2 = 14" << endl;
    cout << endl;

    // =====================================
    // 练习3 参考答案：三位数数位分离
    //   百位 = n / 100
    //   十位 = n / 10 % 10
    //   个位 = n % 10
    // =====================================
    cout << "========== 练习3答案：数位分离 ==========" << endl;
    int num = 368;

    int bai = num / 100;        // 368 / 100 = 3（百位）
    int shi = num / 10 % 10;    // 368/10=36, 36%10=6（十位）
    int ge = num % 10;          // 368 % 10 = 8（个位）

    cout << "数字 " << num << " 的数位分离结果：" << endl;
    cout << "  百位：" << bai << endl;
    cout << "  十位：" << shi << endl;
    cout << "  个位：" << ge << endl;

    // 验证：百位×100 + 十位×10 + 个位 应该等于原数字
    cout << "  验证：" << bai << "×100 + " << shi << "×10 + " << ge;
    cout << " = " << bai * 100 + shi * 10 + ge << endl;
    cout << endl;

    // =====================================
    // 挑战任务 参考答案：超级计算器
    //   输入两个整数，输出加减乘除（保留小数）和取余
    // =====================================
    cout << "========== 挑战任务：超级计算器 ==========" << endl;

    int num1, num2;
    cout << "请输入第一个整数：";
    cin >> num1;
    cout << "请输入第二个整数：";
    cin >> num2;

    cout << endl;
    cout << "===== 超级计算器运算结果 =====" << endl;

    // 加法
    cout << num1 << " + " << num2 << " = " << num1 + num2 << endl;

    // 减法
    cout << num1 << " - " << num2 << " = " << num1 - num2 << endl;

    // 乘法
    cout << num1 << " × " << num2 << " = " << num1 * num2 << endl;

    // 除法（用 double 保留小数，避免整数除法截断）
    cout << num1 << " ÷ " << num2 << " = " << double(num1) / num2;
    cout << "  （精确商，保留小数）" << endl;

    // 取余
    cout << num1 << " % " << num2 << " = " << num1 % num2;
    cout << "  （余数）" << endl;

    cout << endl;

    // =====================================
    // 程序结束
    // =====================================
    cout << "========== 程序结束 ==========" << endl;
    cout << "恭喜！你已经掌握了五种算术运算符！" << endl;
    cout << "记住口诀：括号优先，乘除取余其二，加减最后" << endl;

    return 0;  // 告诉操作系统：程序正常结束啦
}
