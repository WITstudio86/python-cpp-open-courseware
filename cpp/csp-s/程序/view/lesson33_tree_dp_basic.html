#include <bits/stdc++.h>
using namespace std;

// ============================================================
// 通用配置
// ============================================================
const int MAXN = 100005;               // 最大节点数

// ============================================================
// 题目1：没有上司的舞会 —— 最大权独立集
// ============================================================
// 洛谷 P1352 / AcWing 285
// 题目描述：
//   某公司有 n 个职员，编号 1~n，关系构成一棵以1为根的树。
//   父节点是子节点的直接上司。每个职员有快乐指数 H_i。
//   要求：如果一个职员的上司来了，这个职员就不来。
//   求最大的快乐指数总和。
//
// 状态设计：
//   dp[u][0]：不选节点 u 时，以 u 为根的子树的最大快乐指数
//   dp[u][1]：选节点 u 时，以 u 为根的子树的最大快乐指数
//
// 状态转移：
//   dp[u][0] = Σ max(dp[v][0], dp[v][1])    // u不来，儿子随意
//   dp[u][1] = H[u] + Σ dp[v][0]            // u来，儿子都不能来
//
// 时间复杂度：O(n)
// 空间复杂度：O(n)
// ============================================================

namespace Problem1_MaxIndependentSet {

int n;                                  // 节点数
int h[MAXN];                            // 快乐指数（节点权值）
vector<int> g[MAXN];                    // 邻接表
int dp[MAXN][2];                        // DP数组

void dfs(int u, int fa) {
    // ===== 初始化：只考虑节点自身的价值 =====
    dp[u][0] = 0;                       // 不选u：初始值为0
    dp[u][1] = h[u];                    // 选u：初始值为u的快乐指数

    // ===== 后序遍历：先处理所有儿子 =====
    for (int v : g[u]) {
        if (v == fa) continue;          // 避免回到父节点
        dfs(v, u);                      // 递归处理子树

        // 状态转移
        // u不来：儿子v可以来也可以不来，取最优
        dp[u][0] += max(dp[v][0], dp[v][1]);

        // u来：儿子v都不能来
        dp[u][1] += dp[v][0];
    }
}

void solve() {
    // ===== 输入 =====
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> h[i];
    }

    // 建树（无向边）
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    // ===== DFS计算DP值 =====
    dfs(1, 0);                          // 以1为根，父节点设为0

    // ===== 输出答案 =====
    // 根节点可以选或不选，取最大值
    cout << max(dp[1][0], dp[1][1]) << '\n';
}

}  // namespace Problem1_MaxIndependentSet


// ============================================================
// 题目2：战略游戏 —— 最小点覆盖
// ============================================================
// 洛谷 P2016 / AcWing 323
// 题目描述：
//   树上有 n 个节点，Bob要放置最少的士兵来观察所有的边。
//   每个士兵放在节点上可以观察到所有与该节点相连的边。
//   求最少需要多少士兵。
//
// 状态设计：
//   dp[u][0]：不选节点 u 时，覆盖以 u 为根的子树所有边的最少士兵数
//   dp[u][1]：选节点 u 时，覆盖以 u 为根的子树所有边的最少士兵数
//
// 状态转移：
//   dp[u][0] = Σ dp[v][1]                      // u不放，儿子必须全放
//   dp[u][1] = 1 + Σ min(dp[v][0], dp[v][1])   // u放，儿子随意
//
// 重要性质（树作为二分图）：
//   最大独立集 + 最小点覆盖 = n
//   最大匹配数 = 最小点覆盖数（Kőnig定理）
//
// 时间复杂度：O(n)
// 空间复杂度：O(n)
// ============================================================

namespace Problem2_MinVertexCover {

int n;                                  // 节点数
vector<int> g[MAXN];                    // 邻接表
int dp[MAXN][2];                        // DP数组

void dfs(int u, int fa) {
    // ===== 初始化 =====
    dp[u][0] = 0;                       // 不选u：初始需要0个士兵
    dp[u][1] = 1;                       // 选u：至少需要1个士兵（u自己）

    // ===== 后序遍历：先处理所有儿子 =====
    for (int v : g[u]) {
        if (v == fa) continue;
        dfs(v, u);

        // 状态转移
        // u不放士兵：儿子v必须全部放士兵，否则边(u,v)无人看守
        dp[u][0] += dp[v][1];

        // u放士兵：儿子v放不放都行，取较优值
        dp[u][1] += min(dp[v][0], dp[v][1]);
    }
}

void solve() {
    // ===== 输入 =====
    // 注意：n可以为0（空树），需要特判
    cin >> n;
    if (n == 0) {
        cout << 0 << '\n';
        return;
    }

    // 建树（无向边）
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        // 题干中节点编号可能从0开始，统一 +1 处理为 1-based
        u++; v++;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    // ===== DFS计算DP值 =====
    dfs(1, 0);

    // ===== 输出答案 =====
    cout << min(dp[1][0], dp[1][1]) << '\n';
}

/**
 * ============================================================
 * 测试用例
 * ============================================================
 *
 * 题目1 测试用例（没有上司的舞会）：
 *   输入：
 *     7
 *     1 1 1 1 1 1 1
 *     1 3
 *     2 3
 *     6 4
 *     7 4
 *     4 5
 *     3 5
 *   输出：5
 *   解释：选择节点 1,2,6,7,3（或等价方案），最大独立集大小为5
 *
 * 题目2 测试用例（战略游戏）：
 *   输入（简化格式——边列表）：
 *     4
 *     0 1
 *     1 2
 *     2 3
 *   输出：2
 *   解释：在节点1和3放置士兵，可以覆盖所有边
 *
 *   输入（洛谷原始格式）：
 *     4
 *     0:(1) 1
 *     1:(2) 2 3
 *     2:(0)
 *     3:(0)
 *   输出：1
 *   解释：在节点1放置一个士兵即可覆盖所有边
 *
 * ============================================================
 * 知识点总结
 * ============================================================
 *
 * 1. 树形DP的核心：
 *    - 利用树的后序遍历（DFS）确保子问题先求解
 *    - 状态通常分为"选/不选当前节点"
 *
 * 2. 最大独立集（没有上司的舞会）：
 *    - dp[u][0] = Σ max(dp[v][0], dp[v][1])
 *    - dp[u][1] = w[u] + Σ dp[v][0]
 *    - 应用：选点不相邻的最大价值
 *
 * 3. 最小点覆盖（战略游戏）：
 *    - dp[u][0] = Σ dp[v][1]
 *    - dp[u][1] = 1 + Σ min(dp[v][0], dp[v][1])
 *    - 性质：最大独立集 + 最小点覆盖 = n（在树上）
 *
 * 4. 易错提醒：
 *    - DFS必须传入父节点参数 fa，避免死循环
 *    - 先递归子节点，再更新父节点（后序遍历）
 *    - 注意权值可能为负数（dp[u][1] 初始化为 w[u] 而非 1）
 *    - 使用邻接表而非邻接矩阵
 *
 * ============================================================
 */
