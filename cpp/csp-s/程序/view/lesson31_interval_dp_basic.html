#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <climits>
using namespace std;

const int INF = 0x3f3f3f3f;

// ==================== 0. 辅助：打印二维DP表（调试用） ====================
void print_dp(const vector<vector<int>>& dp, int n, const string& title = "") {
    if (!title.empty()) cout << "=== " << title << " ===" << endl;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            if (dp[i][j] >= INF / 2) cout << "INF\t";
            else cout << dp[i][j] << "\t";
        }
        cout << endl;
    }
    cout << endl;
}

// ==================== 1. 石子合并（直线版）- 最小代价 ====================
// 时间复杂度 O(n³)，空间复杂度 O(n²)
// 核心：区间DP五步转移法
int merge_stones_min(const vector<int>& a) {
    int n = a.size();

    // 前缀和预处理
    vector<int> pre(n + 1, 0);
    for (int i = 1; i <= n; ++i) pre[i] = pre[i-1] + a[i-1];
    // 区间和 O(1) 查询
    auto sum = [&](int l, int r) { return pre[r+1] - pre[l]; };

    // dp[l][r] = 合并区间[l, r]的最小代价
    vector<vector<int>> dp(n, vector<int>(n, INF));
    for (int i = 0; i < n; ++i) dp[i][i] = 0;  // 单堆代价为0

    // 区间DP五步法
    for (int len = 2; len <= n; ++len) {           // ① 枚举区间长度（从小到大）
        for (int l = 0; l + len - 1 < n; ++l) {    // ② 枚举左端点
            int r = l + len - 1;                    // ③ 计算右端点
            for (int k = l; k < r; ++k) {           // ④ 枚举分割点 k ∈ [l, r-1]
                // ⑤ 状态转移：将区间[l,k]和[k+1,r]合并
                dp[l][r] = min(dp[l][r],
                    dp[l][k] + dp[k+1][r] + sum(l, r));
            }
        }
    }
    return dp[0][n-1];
}

// 石子合并 - 最大代价版本
int merge_stones_max(const vector<int>& a) {
    int n = a.size();
    vector<int> pre(n + 1, 0);
    for (int i = 1; i <= n; ++i) pre[i] = pre[i-1] + a[i-1];
    auto sum = [&](int l, int r) { return pre[r+1] - pre[l]; };

    vector<vector<int>> dp(n, vector<int>(n, 0));  // 最大值初始化为0
    for (int i = 0; i < n; ++i) dp[i][i] = 0;

    for (int len = 2; len <= n; ++len) {
        for (int l = 0; l + len - 1 < n; ++l) {
            int r = l + len - 1;
            for (int k = l; k < r; ++k) {
                dp[l][r] = max(dp[l][r],
                    dp[l][k] + dp[k+1][r] + sum(l, r));
            }
        }
    }
    return dp[0][n-1];
}

// ==================== 2. 最优三角剖分 ====================
// n边形顶点编号1..n，w[i][j]是弦(i,j)的权值
// dp[l][r]表示从顶点l到顶点r组成的多边形的剖分最小权值和
// 转移：dp[l][r] = min(dp[l][k] + dp[k][r] + w[l][k] + w[k][r])  for k ∈ (l+1, r-1)
int optimal_triangulation(const vector<vector<int>>& w, int n) {
    vector<vector<int>> dp(n + 1, vector<int>(n + 1, INF));

    // 初始化：单个顶点和相邻顶点的剖分代价为0
    for (int i = 1; i <= n; ++i) {
        dp[i][i] = 0;
        if (i < n) dp[i][i+1] = 0;
    }

    for (int len = 3; len <= n; ++len) {
        for (int l = 1; l + len - 1 <= n; ++l) {
            int r = l + len - 1;
            // 枚举中间顶点 k 作为三角形的顶点
            for (int k = l + 1; k < r; ++k) {
                dp[l][r] = min(dp[l][r],
                    dp[l][k] + dp[k][r] + w[l][k] + w[k][r]);
            }
        }
    }
    return dp[1][n];
}

// ==================== 3. 矩阵链乘法 (Matrix Chain Multiplication) ====================
// 给定n个矩阵的维度p[0..n]（第i个矩阵为p[i-1]×p[i]），求最少乘法次数
// dp[l][r] = min(dp[l][k] + dp[k+1][r] + p[l-1]*p[k]*p[r])
int matrix_chain_order(const vector<int>& p) {
    int n = p.size() - 1;  // n个矩阵
    vector<vector<int>> dp(n + 1, vector<int>(n + 1, INF));
    for (int i = 1; i <= n; ++i) dp[i][i] = 0;

    for (int len = 2; len <= n; ++len) {
        for (int l = 1; l + len - 1 <= n; ++l) {
            int r = l + len - 1;
            for (int k = l; k < r; ++k) {
                int cost = dp[l][k] + dp[k+1][r] + p[l-1] * p[k] * p[r];
                dp[l][r] = min(dp[l][r], cost);
            }
        }
    }
    return dp[1][n];
}

// ==================== 4. 回文串相关 - 最少插入次数使之成为回文 ====================
// dp[l][r] = 使子串s[l..r]成为回文串所需的最少插入次数
int min_insertions_to_palindrome(const string& s) {
    int n = s.size();
    vector<vector<int>> dp(n, vector<int>(n, 0));

    for (int len = 2; len <= n; ++len) {
        for (int l = 0; l + len - 1 < n; ++l) {
            int r = l + len - 1;
            if (s[l] == s[r]) {
                dp[l][r] = dp[l+1][r-1];  // 两端相同，不需要插入
            } else {
                dp[l][r] = min(dp[l+1][r], dp[l][r-1]) + 1;
            }
        }
    }
    return dp[0][n-1];
}

// 判断子串s[l..r]是否为回文（区间DP预处理）
vector<vector<bool>> is_palindrome(const string& s) {
    int n = s.size();
    vector<vector<bool>> dp(n, vector<bool>(n, false));

    // 长度为1的子串都是回文
    for (int i = 0; i < n; ++i) dp[i][i] = true;

    for (int len = 2; len <= n; ++len) {
        for (int l = 0; l + len - 1 < n; ++l) {
            int r = l + len - 1;
            if (s[l] == s[r] && (len == 2 || dp[l+1][r-1])) {
                dp[l][r] = true;
            }
        }
    }
    return dp;
}

// ==================== 5. 真题1：石子合并（直线版）【洛谷 P1775 风格】 ====================
void solve_merge_stones() {
    cout << "\n========== 真题1：石子合并（直线版） ==========" << endl;
    int n;
    cout << "请输入石子堆数 n: ";
    cin >> n;
    vector<int> a(n);
    cout << "请输入每堆石子的数量: ";
    for (int i = 0; i < n; ++i) cin >> a[i];

    cout << "最小合并代价: " << merge_stones_min(a) << endl;
}

// ==================== 6. 真题2：矩阵链乘法 ====================
void solve_matrix_chain() {
    cout << "\n========== 真题2：矩阵链乘法 ==========" << endl;
    int n;
    cout << "请输入矩阵数量 n: ";
    cin >> n;
    vector<int> p(n + 1);
    cout << "请输入 " << n + 1 << " 个维度值: ";
    for (int i = 0; i <= n; ++i) cin >> p[i];

    cout << "最少乘法次数: " << matrix_chain_order(p) << endl;
}

// ==================== 7. 测试函数 ====================
void run_all_tests() {
    cout << "==================== 第31课：区间DP基础 - 全部测试 ====================" << endl;

    // 测试1：石子合并 - 最小代价
    {
        vector<int> a = {4, 5, 9, 4};
        int ans = merge_stones_min(a);
        cout << "【测试1】石子合并(最小) - 输入: [4,5,9,4]" << endl;
        cout << "  期望输出: 44, 实际输出: " << ans << " -> "
             << (ans == 44 ? "✓ PASS" : "✗ FAIL") << endl;
    }

    // 测试2：石子合并 - 最大代价
    {
        vector<int> a = {4, 5, 9, 4};
        int ans = merge_stones_max(a);
        cout << "【测试2】石子合并(最大) - 输入: [4,5,9,4]" << endl;
        cout << "  期望输出: 54, 实际输出: " << ans << " -> "
             << (ans == 54 ? "✓ PASS" : "✗ FAIL") << endl;
    }

    // 测试3：石子合并 - 简单情况
    {
        vector<int> a = {1, 2, 3};
        int ans = merge_stones_min(a);
        cout << "【测试3】石子合并(最小) - 输入: [1,2,3]" << endl;
        // 方案1: (1,2)=3, 再(3,3)=6 -> 3+6=9
        // 方案2: (2,3)=5, 再(1,5)=6 -> 5+6=11
        // 最小9
        cout << "  期望输出: 9, 实际输出: " << ans << " -> "
             << (ans == 9 ? "✓ PASS" : "✗ FAIL") << endl;
    }

    // 测试4：矩阵链乘法
    {
        vector<int> p = {10, 100, 5, 50};  // 3个矩阵: 10x100, 100x5, 5x50
        int ans = matrix_chain_order(p);
        cout << "【测试4】矩阵链乘法 - 维度: [10,100,5,50]" << endl;
        // 方案A: (A1*A2)*A3 = 10*100*5 + 10*5*50 = 5000+2500 = 7500
        // 方案B: A1*(A2*A3) = 100*5*50 + 10*100*50 = 25000+50000 = 75000
        cout << "  期望输出: 7500, 实际输出: " << ans << " -> "
             << (ans == 7500 ? "✓ PASS" : "✗ FAIL") << endl;
    }

    // 测试5：矩阵链乘法 - 简单情况
    {
        vector<int> p = {1, 2, 3, 4};  // 3个矩阵: 1x2, 2x3, 3x4
        int ans = matrix_chain_order(p);
        cout << "【测试5】矩阵链乘法 - 维度: [1,2,3,4]" << endl;
        // (1,2,3)*4: (1*2*3)+1*3*4=6+12=18  OR 1*(2,3,4): 2*3*4+1*2*4=24+8=32
        cout << "  期望输出: 18, 实际输出: " << ans << " -> "
             << (ans == 18 ? "✓ PASS" : "✗ FAIL") << endl;
    }

    // 测试6：最少插入次数使之成为回文
    {
        string s = "abca";
        int ans = min_insertions_to_palindrome(s);
        cout << "【测试6】最少插入回文 - 输入: \"abca\"" << endl;
        // "abcba" 插入1个 -> 1
        cout << "  期望输出: 1, 实际输出: " << ans << " -> "
             << (ans == 1 ? "✓ PASS" : "✗ FAIL") << endl;
    }

    // 测试7：最少插入次数 - 已经回文
    {
        string s = "aba";
        int ans = min_insertions_to_palindrome(s);
        cout << "【测试7】最少插入回文 - 输入: \"aba\"" << endl;
        cout << "  期望输出: 0, 实际输出: " << ans << " -> "
             << (ans == 0 ? "✓ PASS" : "✗ FAIL") << endl;
    }

    // 测试8：最少插入次数 - 全不同
    {
        string s = "abcde";
        int ans = min_insertions_to_palindrome(s);
        cout << "【测试8】最少插入回文 - 输入: \"abcde\"" << endl;
        // "abcdedcba" 需要插入4个
        cout << "  期望输出: 4, 实际输出: " << ans << " -> "
             << (ans == 4 ? "✓ PASS" : "✗ FAIL") << endl;
    }

    // 测试9：最优三角剖分（简单示例）
    {
        int n = 4;  // 四边形
        vector<vector<int>> w(n + 1, vector<int>(n + 1, 1));  // 所有弦权值为1
        int ans = optimal_triangulation(w, n);
        cout << "【测试9】最优三角剖分 - 四边形, 所有弦权值=1" << endl;
        // dp[1][4] = min(dp[1][2]+dp[2][4]+w[1][2]+w[2][4], dp[1][3]+dp[3][4]+w[1][3]+w[3][4])
        //          = min(0+2+1+1, 2+0+1+1) = 4
        cout << "  期望输出: 4, 实际输出: " << ans << " -> "
             << (ans == 4 ? "✓ PASS" : "✗ FAIL") << endl;
    }

    // 测试10：回文判断预处理
    {
        string s = "ababa";
        auto pal = is_palindrome(s);
        cout << "【测试10】回文判断 - 输入: \"ababa\"" << endl;
        cout << "  s[0..4]=\"ababa\" 是回文: " << (pal[0][4] ? "✓ PASS" : "✗ FAIL") << endl;
        cout << "  s[0..2]=\"aba\" 是回文: " << (pal[0][2] ? "✓ PASS" : "✗ FAIL") << endl;
        cout << "  s[1..3]=\"bab\" 是回文: " << (pal[1][3] ? "✓ PASS" : "✗ FAIL") << endl;
    }

    cout << "\n==================== 全部测试完成 ====================" << endl;
}
