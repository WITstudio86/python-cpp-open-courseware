⚙️ next 数组的递推构建过程

递推算法（双指针 i 和 j）

vector<int> get_next(const string& P) { int m = P.size(); vector<int> next(m, 0); int j = 0; // j = 已匹配的前缀长度 for (int i = 1; i < m; i++) { while (j > 0 && P[i] != P[j]) j = next[j - 1]; // 失配 → 回退 if (P[i] == P[j]) j++; // 匹配 → 前进 next[i] = j; } return next; }

以 P = "ababac" 演示递推

i=1: P[1]='b', j=0 → P[1]!=P[0] → next[1]=0
i=2: P[2]='a', j=0 → P[2]==P[0] → j=1, next[2]=1
i=3: P[3]='b', j=1 → P[3]==P[1] → j=2, next[3]=2
i=4: P[4]='a', j=2 → P[4]==P[2] → j=3, next[4]=3
i=5: P[5]='c', j=3 → P[5]!=P[3] → j=next[2]=1
    P[5]!=P[1] → j=next[0]=0 → next[5]=0
结果: next = [0, 0, 1, 2, 3, 0]

复杂度分析：j 每次最多 +1，总增加 ≤ m 次，故回退总次数 ≤ m。构建 next 数组为 O(m)。

🔄 最小循环节定理

定理

对长度为 len 的字符串 s， 设 cycle = len - next[len-1] 若 len % cycle == 0，则： → s 由长度为 cycle 的循环节重复 k = len/cycle 次构成 → cycle 是 最小循环节 长度

直观理解

next[len-1] 是 最长公共前后缀 的长度，那么：

s = [ 公共前缀部分 ][ 独有后缀部分 ] ← next[len-1] →← cycle →

如果 len 能被 cycle 整除，说明「独有后缀」可以完整地反复拼接出整个字符串。

示例

字符串lennext[len-1]cycle可循环？说明
"abcabcabc"963✅ k=3循环节 "abc"
"ababab"642✅ k=3循环节 "ab"
"abcab"523❌5%3≠0，需补1个字符
"aaaa"431✅ k=4循环节 "a"

补充成循环串

当 len % cycle != 0 时，需要添加 cycle - len % cycle 个字符（从串首开始按 cycle 循环取）才能使其成为完整的循环串。

【讲解】🔄 最小循环节定理 结合讲义《六、最小循环节定理》强调： · 定理：对于长度为 len 的字符串 s，若 len % (len - next[len-1]) == 0，则 s 可以由长度为 len - next[len-1] 的循环节重复构成，且该长度为**最小循环节**。 · next[len-1] 是 s 的最长公共前后缀长度 · len - next[len-1] 是「不在公共前后缀中」的那段长度 · 如果 len 能被这个长度整除，说明整个字符串由这个循环节重复构成 · s = "abcabcabc"，len = 9，next[8] = 6 幻灯片上可见要点： · len % cycle == 0 · 最小循环节 · next[len-1] 是 最长公共前后缀 的长度，那么： · 如果 len 能被 cycle 整除，说明「独有后缀」可以完整地反复拼接出整个字符串。 讲解时放慢关键术语；用 1 个生活类比 + 1 个反例加深印象。

💻 代码模板

get_next + kmp_search

vector<int> get_next(const string& P) { int m = P.size(); vector<int> next(m, 0); int j = 0; for (int i = 1; i < m; i++) { while (j > 0 && P[i] != P[j]) j = next[j - 1]; if (P[i] == P[j]) j++; next[i] = j; } return next; } vector<int> kmp(const string& T, const string& P) { auto next = get_next(P); int j = 0; vector<int> res; for (int i = 0; i < T.size(); i++) { while (j > 0 && T[i] != P[j]) j = next[j - 1]; if (T[i] == P[j]) j++; if (j == P.size()) { res.push_back(i - j + 1); j = next[j - 1]; } } return res; }

最小循环节

int min_cycle_len(const string& s) { int n = s.size(); auto next = get_next(s); int cycle = n - next[n - 1]; if (n % cycle == 0) return cycle; return n; // 无完整循环节 } // 补充最少字符使其成为循环串 int chars_needed(const string& s) { int n = s.size(); auto next = get_next(s); int cycle = n - next[n - 1]; return (cycle - n % cycle) % cycle; }

扩展：前缀出现次数

vector<int> prefix_cnt(const string& s) { auto next = get_next(s); int n = s.size(); vector<int> cnt(n, 1); for (int i = n - 1; i >= 0; i--) if (next[i] > 0) cnt[next[i]-1] += cnt[i]; return cnt; }

📝 真题精练 1：KMP 匹配统计

题目：给定文本串 T 和模式串 P，统计 P 在 T 中出现的次数（允许重叠）。|T| ≤ 10⁶，|P| ≤ 10⁵。

AC 代码

#include <bits/stdc++.h> using namespace std; vector<int> get_next(const string& P) { int m = P.size(); vector<int> next(m, 0); int j = 0; for (int i = 1; i < m; i++) { while (j > 0 && P[i] != P[j]) j = next[j - 1]; if (P[i] == P[j]) j++; next[i] = j; } return next; } int kmp_count(const string& T, const string& P) { int n = T.size(), m = P.size(); if (m == 0) return 0; auto next = get_next(P); int j = 0, cnt = 0; for (int i = 0; i < n; i++) { while (j > 0 && T[i] != P[j]) j = next[j - 1]; if (T[i] == P[j]) j++; if (j == m) { cnt++; j = next[j - 1]; } } return cnt; } int main() { ios::sync_with_stdio(false); cin.tie(nullptr); string T, P; cin >> T >> P; cout << kmp_count(T, P) << '\n'; }

⏱ 时间复杂度：O(|T| + |P|)，空间 O(|P|)

📝 真题精练 2：最小循环节

题目：给定字符串 s，若 s 由某个子串重复构成，输出循环节及 k；否则输出最少需添加多少字符。|s| ≤ 10⁶。

AC 代码

#include <bits/stdc++.h> using namespace std; // get_next 同上（省略） int main() { ios::sync_with_stdio(false); cin.tie(nullptr); string s; cin >> s; int n = s.size(); auto next = get_next(s); int cycle = n - next[n - 1]; if (n % cycle == 0 && cycle < n) { cout << "循环节: " << s.substr(0, cycle) << "\n重复次数 k = " << n / cycle; } else { int need = cycle - n % cycle; cout << "需要添加 " << need << " 个字符"; } }

🎯 本课小结

核心知识点要点复杂度
暴力匹配问题 最坏 O(n×m)，i 和 j 双重回溯 O(n×m)
next 数组 next[i] = P[0..i] 的最长公共前后缀长度 O(m) 构建
KMP 匹配 i 不回退，失配时 j = next[j-1] O(n+m)
最小循环节 cycle = len - next[len-1]，需 len % cycle == 0 O(n)
前缀出现次数 从后向前累加 cnt[next[i]-1] += cnt[i] O(n)

⚠️ 易错点

  • 失配时是 j = next[j-1]，不是 next[j]
  • while 条件必须写 j > 0 防越界
  • 最小循环节判别需 len % cycle == 0
  • 完全匹配后 continue 还是 j = next[j-1] 取决于是否允许重叠

🏆 实战建议

  • next 数组从 0 开始定义最简洁
  • get_next 和 kmp_search 结构几乎相同，可一起记忆
  • 循环节问题是 CSP-S 高频考点
  • KMP 的扩展（AC 自动机）是省选难度内容
