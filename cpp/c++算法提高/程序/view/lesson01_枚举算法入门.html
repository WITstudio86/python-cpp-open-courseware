/*
 * ================================================================
 * C++算法提高 第01课：枚举算法入门——穷举的力量
 * ================================================================
 *
 * 算法知识点：
 *   - 枚举算法的核心思想：逐一尝试所有可能的答案
 *   - 单层循环枚举：遍历一个变量的所有可能取值
 *   - 双层循环嵌套枚举：遍历两个变量的所有组合
 *   - 枚举三要素：枚举对象、枚举范围、判断条件
 *
 * 经典例题：
 *   例题1 - 百钱买百鸡：公鸡5元，母鸡3元，小鸡1元3只，100元买100只鸡
 *   例题2 - 水仙花数：三位数，各位数字立方和等于该数本身
 */

#include <iostream>
#include <cmath>
#include <iomanip>
using namespace std;

// ================================================================
// 辅助函数：判断一个三位数是否为水仙花数
// 水仙花数定义：各位数字的立方和等于该数本身
// 例如：153 = 1^3 + 5^3 + 3^3 = 1 + 125 + 27 = 153
// ================================================================
bool isNarcissus(int n) {
    int a = n / 100;          // 提取百位数字（整数除法）
    int b = n / 10 % 10;      // 提取十位数字（先除10再去个位）
    int c = n % 10;           // 提取个位数字（模10运算）
    // 计算各位立方和并与原数比较
    return (a * a * a + b * b * b + c * c * c == n);
}

// ================================================================
// 例题1：百钱买百鸡
// 题目：公鸡一只5钱，母鸡一只3钱，小鸡三只1钱
//       用100钱恰好买100只鸡，求所有可能的买法
//
// 枚举思路：
//   - 枚举对象：公鸡数量 gong，母鸡数量 mu
//   - 枚举范围：gong ∈ [0, 20]（最多买20只公鸡，5×20=100）
//              mu ∈ [0, 33]（最多买33只母鸡，3×33=99）
//   - 小鸡数量由总数推导：xiao = 100 - gong - mu
//   - 判断条件：xiao >= 0 && xiao % 3 == 0 && 5*gong + 3*mu + xiao/3 == 100
// ================================================================
void solveBaiJi() {
    cout << "══════════════════════════════════════" << endl;
    cout << "  例题1：百钱买百鸡" << endl;
    cout << "══════════════════════════════════════" << endl;
    cout << "公鸡5元/只，母鸡3元/只，小鸡1元3只" << endl;
    cout << "100元买100只鸡，问各有几只？" << endl;
    cout << "--------------------------------------" << endl;

    int count = 0;  // 解的数量计数器

    // 外层循环：枚举公鸡数量
    for (int gong = 0; gong <= 20; gong++) {
        // 内层循环：枚举母鸡数量
        for (int mu = 0; mu <= 33; mu++) {
            // 小鸡数量由总数关系直接计算
            int xiao = 100 - gong - mu;

            // 判断是否满足条件：
            // 1. 小鸡数量不能为负
            // 2. 小鸡数量必须是3的倍数（因为1元3只，花的钱必须是整数元）
            // 3. 总价等于100元
            if (xiao >= 0 && xiao % 3 == 0) {
                if (5 * gong + 3 * mu + xiao / 3 == 100) {
                    count++;
                    cout << "解" << count << ": ";
                    cout << "公鸡 " << setw(2) << gong << " 只, ";
                    cout << "母鸡 " << setw(2) << mu << " 只, ";
                    cout << "小鸡 " << setw(2) << xiao << " 只";
                    // 验证总价
                    int totalPrice = 5 * gong + 3 * mu + xiao / 3;
                    cout << "  [总价: " << totalPrice << "元]";
                    cout << endl;
                }
            }
        }
    }

    cout << "--------------------------------------" << endl;
    cout << "共有 " << count << " 组解" << endl;
    cout << endl;
}

// ================================================================
// 例题2：水仙花数
// 题目：找出所有三位水仙花数（各位数字立方和等于该数本身）
//
// 枚举思路：
//   - 枚举对象：每个三位数 n
//   - 枚举范围：n ∈ [100, 999]
//   - 判断条件：百位³ + 十位³ + 个位³ == n
//
// 补充：还有四位数的"四叶玫瑰数"（各位四次方和），
//       五位数的"五角星数"等，道理相同。
// ================================================================
void solveNarcissus() {
    cout << "══════════════════════════════════════" << endl;
    cout << "  例题2：水仙花数" << endl;
    cout << "══════════════════════════════════════" << endl;
    cout << "三位数，各位数字立方和等于该数本身" << endl;
    cout << "--------------------------------------" << endl;

    int count = 0;

    // 枚举所有三位数：100 到 999
    for (int n = 100; n <= 999; n++) {
        // 分解出每一位数字
        int a = n / 100;          // 百位：如 153 / 100 = 1
        int b = n / 10 % 10;      // 十位：如 153 / 10 = 15, 15 % 10 = 5
        int c = n % 10;           // 个位：如 153 % 10 = 3

        // 计算各位立方和
        int sum = a * a * a + b * b * b + c * c * c;

        // 判断是否为水仙花数
        if (sum == n) {
            count++;
            cout << n << " = " << a << "³ + "
                 << b << "³ + " << c << "³";
            cout << " = " << a*a*a << " + "
                 << b*b*b << " + " << c*c*c;
            cout << endl;
        }
    }

    cout << "--------------------------------------" << endl;
    cout << "共有 " << count << " 个三位水仙花数" << endl;
    cout << endl;
}

// ================================================================
// 附加演示：单层枚举示例——找出1~100中能被7整除的数
// 帮助学生理解最基本的枚举结构
// ================================================================
void demoSimpleEnum() {
    cout << "══════════════════════════════════════" << endl;
    cout << "  演示：单层枚举——1~100中能被7整除的数" << endl;
    cout << "══════════════════════════════════════" << endl;

    int count = 0;
    for (int i = 1; i <= 100; i++) {
        if (i % 7 == 0) {
            cout << i << " ";
            count++;
            // 每行输出10个数
            if (count % 10 == 0) cout << endl;
        }
    }
    cout << endl;
    cout << "共 " << count << " 个" << endl;
    cout << endl;
}
