/*
 * ================================================================
 * C++算法提高 第13课：归并排序——合并有序序列
 * ================================================================
 *
 * 算法知识点：
 *   - 归并排序的分治策略：分（Divide）→ 治（Conquer）→ 合（Combine）
 *   - 合并两个有序数组（merge）的双指针法
 *   - 递归分解：不断二分直到子数组长度为1（天然有序）
 *   - 利用归并排序在合并过程中统计逆序对
 *
 * 经典例题：
 *   例题1 - 求逆序对：在归并过程中统计 i<j 且 a[i]>a[j] 的数量
 *   例题2 - 归并排序验证：使用归并排序对 n 个整数排序
 *
 * 复杂度：
 *   时间复杂度：O(n log n)，始终稳定
 *   空间复杂度：O(n)，需要额外的临时数组
 */

#include <iostream>
#include <vector>
#include <iomanip>
using namespace std;

typedef long long ll;  // 用于逆序对计数，结果可能超出 int 范围

// ================================================================
// 函数：合并两个有序子数组
// ================================================================
// 参数说明：
//   arr   - 原始数组（引用传递，结果写回 arr）
//   left  - 左子数组的起始下标
//   mid   - 左子数组的结束下标
//   right - 右子数组的结束下标
//
// 前置条件：arr 的 [left, mid] 和 [mid+1, right] 两个区间
//           已经分别是有序的（从小到大排列）
//
// 算法思路（双指针法）：
//   1. 创建临时数组 temp，大小为 (right - left + 1)
//   2. i 指向左子数组开头，j 指向右子数组开头，k 指向 temp 开头
//   3. 比较 arr[i] 和 arr[j]：
//      - 如果 arr[i] <= arr[j]，取 arr[i] 放入 temp，i++、k++
//      - 否则取 arr[j] 放入 temp，j++、k++
//   4. 重复步骤 3，直到某个子数组的元素全部取完
//   5. 把剩余元素全部复制到 temp 末尾
//   6. 将 temp 的内容拷贝回 arr 的 [left, right] 区间
// ================================================================
void merge(vector<int>& arr, int left, int mid, int right) {
    // 临时数组：存放合并后的有序结果
    vector<int> temp(right - left + 1);

    // i —— 遍历左子数组 [left, mid] 的指针
    // j —— 遍历右子数组 [mid+1, right] 的指针
    // k —— 填充临时数组 temp 的指针
    int i = left;
    int j = mid + 1;
    int k = 0;

    // ---- 阶段1：两个子数组都还有元素时，谁小取谁 ----
    while (i <= mid && j <= right) {
        if (arr[i] <= arr[j]) {
            // 左子数组的当前元素更小（或相等），取左边
            // 使用 <= 保证了归并排序的稳定性：
            // 相等时优先取左边的，保持原有相对顺序不变
            temp[k] = arr[i];
            i++;  // 左指针后移
        } else {
            // 右子数组的当前元素更小，取右边
            temp[k] = arr[j];
            j++;  // 右指针后移
        }
        k++;  // temp 指针始终后移
    }

    // ---- 阶段2：左子数组还有剩余元素，全部搬过去 ----
    // （此时右子数组已经空了，左子数组剩余的元素天然有序且都比已放入的大）
    while (i <= mid) {
        temp[k] = arr[i];
        i++;
        k++;
    }

    // ---- 阶段3：右子数组还有剩余元素，全部搬过去 ----
    // （和阶段2对称，这两个 while 循环只有一个会真正执行）
    while (j <= right) {
        temp[k] = arr[j];
        j++;
        k++;
    }

    // ---- 阶段4：将合并结果拷贝回原数组 ----
    // temp[0] 对应 arr[left]，temp[t] 对应 arr[left + t]
    for (int t = 0; t < (int)temp.size(); t++) {
        arr[left + t] = temp[t];
    }
}

// ================================================================
// 函数：归并排序（递归实现）
// ================================================================
// 参数说明：
//   arr   - 待排序的数组（引用传递，排序结果直接反映在原数组上）
//   left  - 当前需要排序的区间左端点
//   right - 当前需要排序的区间右端点
//
// 递归过程详解：
//   1. 终止条件：left >= right，即区间长度 ≤ 1
//      单个元素天然有序，不需要排序，直接返回
//   2. 分解：计算中间位置 mid，将区间分成两半
//   3. 递归排序左半：mergeSort(arr, left, mid)
//   4. 递归排序右半：mergeSort(arr, mid + 1, right)
//   5. 合并：merge(arr, left, mid, right)
//      此时左右两半都已有序，合并成一个有序的整体
//
// 递推过程可以想象成一棵树：
//   - 递归调用一路向下"分"，直到区间长度为1
//   - 然后从叶子节点开始向上"合"，逐层返回
//
// 例如排序 [8, 4, 5, 7, 1, 3, 6, 2]：
//   分：[8,4,5,7,1,3,6,2]
//        → [8,4,5,7] + [1,3,6,2]
//        → [8,4] + [5,7] + [1,3] + [6,2]
//        → [8] + [4] + [5] + [7] + [1] + [3] + [6] + [2]
//   合：[4,8] + [5,7] → [4,5,7,8]
//        [1,3] + [2,6] → [1,2,3,6]
//        [4,5,7,8] + [1,2,3,6] → [1,2,3,4,5,6,7,8]
// ================================================================
void mergeSort(vector<int>& arr, int left, int right) {
    // 递归的"终止条件"（也叫"基线条件"）
    // 当区间里只有 0 或 1 个元素时，已经是有序的，不需要再分
    if (left >= right) {
        return;
    }

    // 计算中间位置
    // 使用 left + (right - left) / 2 而不是 (left + right) / 2
    // 是为了防止 left + right 可能 int 溢出（当数组很大时）
    int mid = left + (right - left) / 2;

    // 【分治第一步：分】
    // 递归排序左半部分 [left, mid]
    // 这个调用会一直深入下去，直到 left >= mid
    mergeSort(arr, left, mid);

    // 递归排序右半部分 [mid+1, right]
    // 同样会一直深入，直到 mid+1 >= right
    mergeSort(arr, mid + 1, right);

    // 【分治第三步：合】
    // 当左右两半都已经排好序后，将它们合并
    // 注意：这里 mergeSort 采用的是"后序遍历"的顺序
    // 先递归处理左右子树，再处理当前层（合并）
    merge(arr, left, mid, right);
}

// ================================================================
// 例题1：求逆序对
// ================================================================
// 题目：对于一个序列 a，如果 i < j 且 a[i] > a[j]，则称
//       (a[i], a[j]) 为一个逆序对。给定序列，求逆序对总数。
//
// 解题核心思路：
//   在归并排序的"合并"过程中，当 arr[i] > arr[j] 时：
//   - 由于左子数组已经有序，arr[i] 后面的元素都比 arr[i] 更大
//   - 因此左子数组中从 i 到 mid 的所有元素，都和 arr[j] 构成逆序对
//   - 一次就统计出 (mid - i + 1) 个逆序对！
//
// 例如：左 [3,5,8]  右 [2,6,9]
//   比较 3 和 2：3 > 2 → 逆序对 +3  (3,2) (5,2) (8,2)
//   比较 3 和 6：3 < 6 → 正常合并
//   比较 5 和 6：5 < 6 → 正常合并
//   比较 8 和 6：8 > 6 → 逆序对 +1  (8,6)
//   比较 8 和 9：8 < 9 → 正常合并
//   最终逆序对总数 = 3 + 1 = 4
//
// 时间复杂度：O(n log n)，空间复杂度：O(n)
// 注意：逆序对数量最多可达 n(n-1)/2 ≈ 5×10^9（n=10^5时），
//       超出 int 范围（约 2.1×10^9），必须使用 long long
// ================================================================

// 合并函数，同时统计逆序对
ll mergeAndCount(vector<int>& arr, int left, int mid, int right) {
    vector<int> temp(right - left + 1);
    int i = left;
    int j = mid + 1;
    int k = 0;
    ll count = 0;  // 记录本次合并过程中发现的逆序对数量

    // 双指针合并，同时计数
    while (i <= mid && j <= right) {
        if (arr[i] <= arr[j]) {
            // 左边 <= 右边，不构成逆序对，正常合并
            temp[k] = arr[i];
            i++;
        } else {
            // 左边 > 右边，构成逆序对！
            // 关键：因为左子数组有序，所以 arr[i], arr[i+1], ..., arr[mid]
            //       全部大于 arr[j]，一共 (mid - i + 1) 个元素
            temp[k] = arr[j];
            // ★ 核心语句：累加跨区逆序对的数量 ★
            count += (ll)(mid - i + 1);
            j++;
        }
        k++;
    }

    // 剩余元素直接复制（此时不会产生新的逆序对）
    while (i <= mid) {
        temp[k] = arr[i];
        i++;
        k++;
    }
    while (j <= right) {
        temp[k] = arr[j];
        j++;
        k++;
    }

    // 拷贝回原数组
    for (int t = 0; t < (int)temp.size(); t++) {
        arr[left + t] = temp[t];
    }

    return count;
}

// 递归排序并统计逆序对
// 返回值：区间 [left, right] 内的逆序对总数
ll mergeSortAndCount(vector<int>& arr, int left, int right) {
    // 终止条件：区间只有 0 或 1 个元素，没有逆序对
    if (left >= right) {
        return 0;
    }

    int mid = left + (right - left) / 2;

    // 分治统计逆序对：
    //   1. 左半部分内部的逆序对
    //   2. 右半部分内部的逆序对
    //   3. 横跨左右两部分的逆序对（在合并阶段统计）

    ll leftCount = mergeSortAndCount(arr, left, mid);      // 左边内部的逆序对
    ll rightCount = mergeSortAndCount(arr, mid + 1, right); // 右边内部的逆序对
    ll crossCount = mergeAndCount(arr, left, mid, right);   // 跨左右的逆序对

    // 三者之和就是逆序对总数
    return leftCount + rightCount + crossCount;
}

// 例题1的包装函数
void solveInversePair() {
    cout << "══════════════════════════════════════" << endl;
    cout << "  例题1：求逆序对" << endl;
    cout << "══════════════════════════════════════" << endl;
    cout << "题目：对于一个序列 a，如果 i<j 且 a[i]>a[j]，" << endl;
    cout << "      则称 (a[i], a[j]) 为一个逆序对。" << endl;
    cout << "      给定序列，求其中逆序对的数量。" << endl;
    cout << "--------------------------------------" << endl;

    // 测试用例1：样例
    cout << "【测试1】序列: 3 1 4 2 5" << endl;
    vector<int> test1 = {3, 1, 4, 2, 5};
    ll ans1 = mergeSortAndCount(test1, 0, (int)test1.size() - 1);
    cout << "逆序对数量: " << ans1 << " (预期: 3)" << endl;
    cout << "逆序对详解: (3,1) (3,2) (4,2)" << endl;

    // 测试用例2：完全逆序（最大逆序对数量）
    cout << "\n【测试2】完全逆序序列: 5 4 3 2 1" << endl;
    vector<int> test2 = {5, 4, 3, 2, 1};
    ll ans2 = mergeSortAndCount(test2, 0, (int)test2.size() - 1);
    cout << "逆序对数量: " << ans2 << " (预期: 10, 即 C(5,2))" << endl;

    // 测试用例3：完全有序（没有逆序对）
    cout << "\n【测试3】完全有序序列: 1 2 3 4 5" << endl;
    vector<int> test3 = {1, 2, 3, 4, 5};
    ll ans3 = mergeSortAndCount(test3, 0, (int)test3.size() - 1);
    cout << "逆序对数量: " << ans3 << " (预期: 0)" << endl;

    // 测试用例4：含重复元素
    cout << "\n【测试4】含重复元素: 2 3 2 3 2" << endl;
    vector<int> test4 = {2, 3, 2, 3, 2};
    // 逆序对: (2[0],2[2]), (2[0],2[4]), (3[1],2[2]),
    //        (3[1],2[4]), (2[2],2[4]?不,相等不算), (3[3],2[4])
    // 实际：3的索引1 vs 2的索引2,4; 3的索引3 vs 2的索引4; 2的索引0 vs 无
    // 准确：位置1,2:3>2; 位置1,4:3>2; 位置3,4:3>2;
    //       位置0,2:2>=2不构成; 位置0,4:2>=2不构成; 位置2,4:2>=2不构成
    // 共3个逆序对
    ll ans4 = mergeSortAndCount(test4, 0, (int)test4.size() - 1);
    cout << "逆序对数量: " << ans4 << " (预期: 3)" << endl;
    cout << "逆序对: (3₁,2₂) (3₁,2₄) (3₃,2₄)" << endl;

    cout << "--------------------------------------" << endl;
    cout << "算法分析:" << endl;
    cout << "  核心思想: 在归并排序的合并过程中统计逆序对" << endl;
    cout << "  当 arr[i] > arr[j] 时，累加 (mid - i + 1)" << endl;
    cout << "  时间复杂度: O(n log n)，空间复杂度: O(n)" << endl;
    cout << "  注意: 答案可能超过 int 范围，必须使用 long long" << endl;
    cout << endl;
}

    cout << "      比较 43 vs 82 → 取43" << endl;

    cout << "      最终结果: [3, 9, 10, 27, 38, 43, 82]" << endl;

    cout << "--------------------------------------" << endl;
    cout << "总结：归并排序 = 递归分解（拆到单个元素）" << endl;
    cout << "                + 两两合并（保持有序）" << endl;
    cout << "      每次合并需要 O(n) 时间，共 log n 层" << endl;
    cout << "      总时间复杂度 = O(n log n)" << endl;
    cout << endl;
}
