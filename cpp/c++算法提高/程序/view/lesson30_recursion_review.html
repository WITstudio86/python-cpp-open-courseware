// ================================================================
// 例题1：幂次方表示
// ================================================================
// 问题描述：
//   任何一个正整数都可以表示为2的幂次方之和。
//   进一步要求：指数也要用同样的方式表示。
//   例如：137 = 2(2(2)+2+2(0)) + 2(2+2(0)) + 2(0)
//
// 递归分解思路：
//   1. 将 n 转换为二进制，找出所有为 1 的位
//   2. 对于每一位 i（i 是指数）：
//      - i == 0 → "2(0)"
//      - i == 1 → "2"（省略括号，约定简写）
//      - i >= 2 → "2(" + 递归分解(i) + ")"
//   3. 各项用 "+" 连接
//
// 递归终止条件：
//   - n == 0 → 返回 "0"
//   - n == 1 → 返回 ""（表示 2^1 只写 "2"）
//
// 关键技巧：
//   - 从高位向低位扫描二进制，保证输出顺序
//   - 使用 first 标志控制 "+" 的输出
// ================================================================

// 递归函数：返回数字 n（作为指数时）的幂次方表示字符串
// 注意：这个函数返回的是"指数部分"的表示
// 调用者需要自行添加 "2(" 和 ")" 或直接使用 "2"
string powerRepr(int n) {
    // 【终止条件1】指数为 0，返回 "0"
    if (n == 0) return "0";

    // 【终止条件2】指数为 1，返回空字符串
    // 因为约定 2^1 简写为 "2"，不需要写成 "2(1)"
    if (n == 1) return "";

    string result = "";           // 存储当前数字的幂次方表示
    bool first = true;            // 标记是否是第一项（控制 + 号）

    // 从高位向低位扫描二进制位
    // 20000 < 2^15 = 32768，扫描 15 到 0 位足够
    for (int i = 15; i >= 0; i--) {
        // 检查第 i 位是否为 1（即 2^i 是否在 n 的二进制表示中）
        if (n & (1 << i)) {
            // 如果不是第一项，先输出 "+" 连接符
            if (!first) result += "+";
            first = false;

            // 根据指数 i 的大小，分三种情况处理
            if (i == 0) {
                // 2^0：直接写 "2(0)"
                result += "2(0)";
            } else if (i == 1) {
                // 2^1：约定简写为 "2"
                result += "2";
            } else {
                // 2^i (i >= 2)：指数 i 需要递归分解
                // 格式：2( 递归分解(i) )
                result += "2(" + powerRepr(i) + ")";
            }
        }
    }

    return result;
}

// 幂次方表示的入口函数
void solvePowerRepresentation() {
    cout << "╔══════════════════════════════════════════════╗" << endl;
    cout << "║     例题1：幂次方表示                         ║" << endl;
    cout << "║     将正整数表示为2的幂次方之和               ║" << endl;
    cout << "╚══════════════════════════════════════════════╝" << endl;
    cout << endl;

    // 测试多个数字
    int testNumbers[] = {137, 7, 3, 1315};
    int testCount = sizeof(testNumbers) / sizeof(testNumbers[0]);

    for (int t = 0; t < testCount; t++) {
        int n = testNumbers[t];

        // 先验算：验证二进制分解
        cout << "  ───────────────────────────────────────" << endl;
        cout << "  数字 n = " << n << endl;
        cout << "  二进制: ";
        bool firstBit = true;
        for (int i = 15; i >= 0; i--) {
            if (n & (1 << i)) {
                if (!firstBit) cout << " + ";
                firstBit = false;
                cout << "2^" << i;
            }
        }
        cout << endl;

        // 输出幂次方表示
        string repr = powerRepr(n);
        cout << "  幂次方表示: " << repr << endl;
    }
    cout << "  ───────────────────────────────────────" << endl;
    cout << endl;
}


// ================================================================
// 例题2：FBI树
// ================================================================
// 问题描述：
//   给定一个长度为 2^N 的 "01" 串，递归构造 FBI 树：
//   1. 根节点对应整个字符串，根据字符串内容判断类型：
//      - 全 '0' → B 节点（Bomb）
//      - 全 '1' → I 节点（Intelligence）
//      - 混合   → F 节点（FBI）
//   2. 如果字符串长度 > 1，将其对半分为左右子串，
//      分别构造左子树和右子树
//   3. 构造完成后，输出后序遍历序列（左→右→根）
//
// 递归构建过程（以 "10001011" 为例）：
//
//                      F("10001011")
//                    /               \
//            F("1000")              F("1011")
//           /        \            /        \
//      F("10")     B("00")    F("10")    I("11")
//      /    \      /    \     /    \     /    \
//   I("1") B("0") B("0") B("0") I("1") B("0") I("1") I("1")
//
// 后序遍历输出：I B F B B F I B I F I I F F
//
// 递归终止条件：
//   当区间长度为 1（l == r）时，该位置直接是叶子节点，
//   类型为 B（字符为 '0'）或 I（字符为 '1'）
// ================================================================

string fbiString;  // 存储输入的 01 串

// 递归构建 FBI 树并输出后序遍历
// 参数 l, r：当前处理的字符串区间 [l, r]（闭区间）
void buildFBI(int l, int r) {
    // 【递归步骤】如果区间长度大于 1，先递归处理左右子树
    // 后序遍历要求：先左子树，再右子树，最后根节点
    if (l < r) {
        int mid = (l + r) / 2;           // 对半分割
        buildFBI(l, mid);                 // 递归构建左子树
        buildFBI(mid + 1, r);             // 递归构建右子树
    }

    // 【判断节点类型】扫描当前区间 [l, r] 内的所有字符
    bool hasZero = false;   // 区间内是否包含字符 '0'
    bool hasOne  = false;   // 区间内是否包含字符 '1'

    for (int i = l; i <= r; i++) {
        if (fbiString[i] == '0') hasZero = true;
        if (fbiString[i] == '1') hasOne  = true;
    }

    // 【后序遍历】输出当前节点的类型（B / I / F）
    if (hasZero && hasOne) {
        cout << 'F';          // 既有 0 又有 1 → F 节点
    } else if (hasZero) {
        cout << 'B';          // 只有 0 → B 节点
    } else {
        cout << 'I';          // 只有 1 → I 节点
    }
}
