 *
 * 算法知识点：

 *   - 基准选择策略：随机选择，避免最坏情况
 *   - 快速选择（quickSelect）：利用分区思想在 O(n) 平均时间
 *     内找到数组中第 k 大/小的元素
 *
 * 经典例题：

#include <iostream>
#include <algorithm>
#include <cstdlib>
#include <ctime>
#include <iomanip>
using namespace std;

const int MAXN = 100005;  // 最大数据规模

// ================================================================
// partition 函数：分区操作（快速排序的核心）
// ================================================================
// 功能描述：
//   选取一个基准元素 pivot，将数组 arr[left...right] 重新排列，
//   使得所有小于 pivot 的元素都在 pivot 左边，
//   所有大于等于 pivot 的元素都在 pivot 右边。
//
// 实现方法：双指针法
//   - 指针 i：指向"小于基准区域"的最后一个元素位置
//   - 指针 j：遍历数组，寻找比基准小的元素
//   - 当 j 找到一个比基准小的元素时，i 右移一位，交换 arr[i] 和 arr[j]
//   - 遍历结束后，将基准放到 i+1 的位置（即两个区域的交界处）
//
// 参数：
//   arr   - 待分区的数组
//   left  - 当前处理区间的左边界（包含）
//   right - 当前处理区间的右边界（包含）
//
// 返回值：
//   pivot 元素最终所在的下标位置
//
// 时间复杂度：O(n)，其中 n = right - left + 1
// 空间复杂度：O(1)，只使用了常数个额外变量
//
// 注意事项（重要！）：
//   1. 使用随机选择基准来避免最坏情况（已经有序的数组）
//   2. 先将随机选中的基准交换到最右边，简化后续处理
//   3. swap 是 C++ 标准库函数，在 <algorithm> 头文件中
// ================================================================
int partition(int arr[], int left, int right) {
    // ----- 步骤1：随机选取基准元素 -----
    // 为什么要随机？避免在数组已经有序时退化为 O(n²)
    // rand() % (right - left + 1) 生成 [0, range-1] 的随机数
    // left + ... 将其转换为 [left, right] 范围内的实际下标
    int randomIndex = left + rand() % (right - left + 1);

    // 将随机选中的基准元素交换到区间的最右边
    // 这样做的好处：pivot 始终在 arr[right]，后续代码可以统一处理
    swap(arr[randomIndex], arr[right]);

    // ----- 步骤2：获取基准值 -----
    int pivot = arr[right];  // 基准元素的值

    // ----- 步骤3：双指针分区 -----
    // i 的含义：所有 <= i 的位置，其元素值都小于 pivot
    // 初始时，没有元素被确认小于 pivot，所以 i = left - 1
    int i = left - 1;

    // j 从 left 开始遍历到 right-1（不包括基准元素本身）
    // 每找到一个小于 pivot 的元素，就把它交换到"小于区域"
    for (int j = left; j < right; j++) {
        // 如果当前元素 arr[j] 小于基准值 pivot
        if (arr[j] < pivot) {
            i++;  // 扩大"小于区域"的边界

            // 将 arr[j] 交换到"小于区域"的新位置
            // 注意：当 i == j 时，交换没有实际效果，但保持代码简洁
            swap(arr[i], arr[j]);

            /*
             * 分区过程的可视化说明：
             *
             * [小于pivot的区域] [大于等于pivot的区域] [未处理的区域] [pivot]
             *  ↑                ↑                    ↑             ↑
             * left ... i       i+1 ... j-1          j ... right-1 right
             *
             * 当 arr[j] < pivot 时：
             *   i 右移一位 → 将 arr[j] 交换到 i 的位置
             *   → 小于区域扩大
             *
             * 当 arr[j] >= pivot 时：
             *   什么都不做，j 继续右移
             *   → 大于等于区域自然扩大
             */
        }
    }

    // ----- 步骤4：将基准放到正确位置 -----
    // 遍历结束后，i+1 的位置就是 pivot 应该在的位置
    // 此时 arr[i+1] 是大于等于区域中的第一个元素
    // 将 pivot（在 arr[right]）与 arr[i+1] 交换
    swap(arr[i + 1], arr[right]);

    // 返回基准元素的最终位置
    // 此时：左半部分 [left, i] 全部 < pivot
    //      右半部分 [i+2, right] 全部 >= pivot
    //      中间 arr[i+1] = pivot
    return i + 1;
}

// ================================================================
// quickSort 函数：递归快速排序
// ================================================================
// 功能描述：
//   对数组 arr[left...right] 进行从小到大排序。
//   采用分治策略：
//     1. 分区：用 partition 将数组分为左右两部分
//     2. 递归：对左半部分和右半部分分别调用 quickSort
//
// 参数：
//   arr   - 待排序数组
//   left  - 当前排序区间的左边界（包含）
//   right - 当前排序区间的右边界（包含）
//
// 递归终止条件：
//   left >= right 时，区间为空或只有一个元素，无需再排
//
// 时间复杂度：平均 O(n log n)，最坏 O(n²)（但随机基准避免了最坏情况）
// 空间复杂度：O(log n)（递归调用栈深度）
//
// 稳定性说明：
//   快速排序是不稳定的排序算法。
//   原因：在分区过程中，相同值的元素的相对位置可能被 swap 打乱。
//   例如：[2a, 2b, 1] 分区后可能变为 [1, 2b, 2a]。
// ================================================================
void quickSort(int arr[], int left, int right) {
    // 递归终止条件：当前区间为空或只有一个元素
    // left == right：只有一个元素，自然有序
    // left > right：空区间（可能发生在 pivot 在边界时）
    if (left >= right) return;

    // ----- 分区操作：获取基准元素的最终位置 -----
    // partition 会重新排列 arr[left...right]
    // 并返回 pivot 最终所在的下标
    int pivotIndex = partition(arr, left, right);

    // ----- 递归：分别排序左右两部分 -----
    // 左半部分：从 left 到 pivotIndex - 1
    //   （这些元素都小于 pivot，但内部可能无序）
    quickSort(arr, left, pivotIndex - 1);

    // 右半部分：从 pivotIndex + 1 到 right
    //   （这些元素都大于等于 pivot，但内部可能无序）
    quickSort(arr, pivotIndex + 1, right);

    // 注意：pivot 本身不需要再参与递归！
    // 因为它已经位于最终排好序时的正确位置。
}

// ================================================================
// quickSelect 函数：快速选择算法
// ================================================================
// 功能描述：
//   在数组 arr[left...right] 中找到排名第 k 小的元素。
//   k 从 1 开始计数（k=1 表示最小的元素，k=n 表示最大的元素）。
//
// 算法原理：
//   利用快速排序的 partition 思想，但每次只需要递归一边。
//   分区后，pivot 的排名（rank）可以确定：
//     - 如果 rank == k → pivot 就是答案
//     - 如果 rank > k  → 答案在左半部分，递归左边
//     - 如果 rank < k  → 答案在右半部分，递归右边（注意更新 k 值）
//
// 示例说明：
//   数组 [3, 1, 4, 1, 5, 9]，找第 2 小（k=2）
//   分区后：[1, 1, 3, 4, 5, 9], pivot=3 在位置 2, rank=3
//   rank(3) > k(2) → 答案在左半部分 [1, 1]
//   在 [1, 1] 中找第 2 小 → 答案是 1
//
// 参数：
//   arr   - 待查找的数组（注意：函数会修改数组内容）
//   left  - 当前查找区间的左边界（包含）
//   right - 当前查找区间的右边界（包含）
//   k     - 要查找的排名（从1开始，表示第k小的元素）
//
// 返回值：
//   数组中第 k 小的元素的值
//
// 时间复杂度：平均 O(n)，最坏 O(n²)（通过随机基准优化）
// 空间复杂度：O(log n)（递归调用栈）
//
// 对比说明：
//   - 先排序再取第k个：O(n log n)，总需要完全排序
//   - 快速选择：平均 O(n)，不需要完全排序
//   - 当只需要几个特定排名的元素时，快速选择远优于全排序
// ================================================================
int quickSelect(int arr[], int left, int right, int k) {
    // 递归终止条件：区间只有一个元素
    if (left == right) {
        return arr[left];  // 唯一的元素就是答案
    }

    // ----- 分区操作 -----
    // 使用与快速排序相同的 partition 函数
    int pivotIndex = partition(arr, left, right);

    // ----- 计算 pivot 在当前区间中的排名 -----
    // pivot 在当前区间 [left, right] 中排第几（从1开始）
    // 例如：left=0, pivotIndex=2 → rank = 2-0+1 = 3（排第3）
    int rank = pivotIndex - left + 1;

    // ----- 根据排名决定下一步 -----
    if (rank == k) {
        // 情况1：pivot 的排名恰好等于 k
        // 说明 pivot 就是我们要找的第 k 小的元素
        return arr[pivotIndex];
    } else if (rank > k) {
        // 情况2：pivot 的排名比 k 大
        // 说明第 k 小的元素在 pivot 的左边
        // 递归查找左半部分，k 值不变
        return quickSelect(arr, left, pivotIndex - 1, k);
    } else {
        // 情况3：pivot 的排名比 k 小
        // 说明第 k 小的元素在 pivot 的右边
        // 注意：在右半部分中，排名需要减去 (rank) 个元素
        //       因为左半部分和 pivot 已经包含了 rank 个更小的元素
        return quickSelect(arr, pivotIndex + 1, right, k - rank);
    }
}

// ================================================================
// 辅助函数：打印数组
// ================================================================
// 功能：打印数组 arr[0...n-1] 的所有元素，用空格分隔
void printArray(int arr[], int n) {
    for (int i = 0; i < n; i++) {
        cout << arr[i];
        if (i < n - 1) cout << " ";
    }
    cout << endl;
}

// ================================================================
// 辅助函数：将数组内容复制到另一个数组
// ================================================================
void copyArray(int dest[], int src[], int n) {
    for (int i = 0; i < n; i++) {
        dest[i] = src[i];
    }
}

// ================================================================
// 例题1：第k大的数（使用快速选择算法）
// ================================================================
// 题目描述：
//   给定一个长度为 n 的整数序列，请找出其中第 k 大的数。
//   k 从 1 开始计数。例如：
//     k=1 表示找最大的数
//     k=2 表示找第二大的数
//     ...
//     k=n 表示找最小的数
//
// 算法思路：
//   "第k大" 等于 "第 (n - k + 1) 小"。
//   例如：数组 [3, 1, 4, 1, 5, 9], n=6
//     第1大 (k=1) = 第6小 (6-1+1=6) = 9
//     第2大 (k=2) = 第5小 (6-2+1=5) = 5
//   然后用 quickSelect 找第 (n - k + 1) 小的元素即可。
//
// 输入格式：
//   第一行：两个整数 n 和 k (1 ≤ k ≤ n ≤ 100000)
//   第二行：n 个整数，每个在 int 范围内
//
// 输出格式：
//   一个整数，表示第 k 大的数
//
// 样例输入：
//   6 2
//   3 1 4 1 5 9
//
// 样例输出：
//   5
//
// 样例解释：
//   数组排序（从小到大）：[1, 1, 3, 4, 5, 9]
//   第2大的数是 5
// ================================================================
void solveProblem1_KthLargest() {
    cout << "╔══════════════════════════════════════════════╗" << endl;
    cout << "║  例题1：第k大的数（快速选择算法）             ║" << endl;
    cout << "╚══════════════════════════════════════════════╝" << endl;
    cout << endl;

    cout << "数组: [3, 1, 4, 1, 5, 9]  (n=6)" << endl;

    int arr1[] = {3, 1, 4, 1, 5, 9};
    int n1 = 6;

    // 测试 k=1..6 的所有情况
    for (int k = 1; k <= n1; k++) {
        // 复制数组（因为 quickSelect 会修改原数组）
        int temp[MAXN];
        copyArray(temp, arr1, n1);

        // 第k大 = 第 (n - k + 1) 小
        int ans = quickSelect(temp, 0, n1 - 1, n1 - k + 1);
        cout << "  第 " << k << " 大的数: " << ans << endl;
    }
    cout << endl;

    // 演示完整的查找过程
    cout << "--- 演示：找第2大的数（即第5小的数） ---" << endl;
    int demo[MAXN];
    copyArray(demo, arr1, n1);
    cout << "  原数组: ";
    printArray(demo, n1);

    // 第一轮分区
    cout << "  第1轮分区..." << endl;
    int result = quickSelect(demo, 0, n1 - 1, n1 - 2 + 1);
    cout << "  第2大的数 = " << result << endl;
    cout << "  （快速选择只做了部分排序，比全排序更高效！）" << endl;
    cout << endl;
}

// ================================================================
// 例题2：快速排序实现
// ================================================================
// 题目描述：
//   使用快速排序算法对 n 个整数从小到大排序。
//
// 输入格式：
//   第一行：一个整数 n (1 ≤ n ≤ 100000)
//   第二行：n 个整数，以空格分隔
//
// 输出格式：
//   一行 n 个整数，为从小到大排序后的结果，用空格分隔
//
// 样例输入：
//   5
//   38 27 43 3 9
//
// 样例输出：
//   3 9 27 38 43
// ================================================================
void solveProblem2_QuickSort() {
    cout << "╔══════════════════════════════════════════════╗" << endl;
    cout << "║  例题2：快速排序实现                          ║" << endl;
    cout << "╚══════════════════════════════════════════════╝" << endl;
    cout << endl;

    int arr2[] = {38, 27, 43, 3, 9, 82, 55, 10};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);

    cout << "排序前: ";
    printArray(arr2, n2);

    // 调用快速排序
    quickSort(arr2, 0, n2 - 1);

    cout << "排序后: ";
    printArray(arr2, n2);
    cout << endl;

    // ================================================================
    // 扩展演示：不同数据规模下的快速排序性能对比
    // ================================================================
    cout << "--- 快速排序性能演示 ---" << endl;
    cout << "（简单计时，仅为演示性质）" << endl;
    cout << endl;

    // 测试1：随机数据
    int testArr[MAXN];
    int testSizes[] = {1000, 10000, 50000};

    for (int t = 0; t < 3; t++) {
        int size = testSizes[t];

        // 生成随机数据
        for (int i = 0; i < size; i++) {
            testArr[i] = rand() % 1000000;
        }

        // 计时开始
        clock_t start = clock();
        quickSort(testArr, 0, size - 1);
        clock_t end = clock();

        // 验证排序是否正确
        bool sorted = true;
        for (int i = 1; i < size; i++) {
            if (testArr[i] < testArr[i - 1]) {
                sorted = false;
                break;
            }
        }

        double elapsed = double(end - start) / CLOCKS_PER_SEC;
        cout << "  n=" << setw(6) << size
             << "  耗时: " << fixed << setprecision(6) << elapsed << " 秒"
             << "  排序" << (sorted ? "正确 ✓" : "错误 ✗") << endl;
    }
    cout << endl;

    // ================================================================
    // 扩展演示：对比随机基准 vs 固定基准（选最后一个）
    // ================================================================
    cout << "--- 基准选择策略对比 ---" << endl;
    cout << "（演示为什么随机化很重要）" << endl;
    cout << endl;

    int sizeDemo = 10000;
    int arrRandom[MAXN], arrFixed[MAXN];

    // 情况1：已排序数组（最坏情况 for 固定基准）
    for (int i = 0; i < sizeDemo; i++) {
        arrRandom[i] = i + 1;
        arrFixed[i] = i + 1;
    }

    cout << "  已排序数组 (n=" << sizeDemo << ")：" << endl;

    // 固定基准（选最后一个元素，会退化）
    // 这里用 partition_fixed 模拟（实际上 partition 已经用了随机基准，所以只需展示随机化效果）
    cout << "    随机基准快速排序 → 性能稳定 O(n log n)" << endl;
    cout << "    固定基准（选最后一个）→ 可能退化为 O(n²)" << endl;
    cout << "    结论：随机化基准可以有效避免最坏情况！" << endl;
    cout << endl;
}

// ================================================================
// 附加演示：分区操作的可视化过程
// ================================================================
// 功能：单步演示 partition 的详细过程，帮助学生理解双指针法
void demoPartition() {
    cout << "╔══════════════════════════════════════════════╗" << endl;
    cout << "║  演示：分区（partition）操作详解              ║" << endl;
    cout << "╚══════════════════════════════════════════════╝" << endl;
    cout << endl;

    // 使用演示用的固定数据（便于跟踪变化过程）
    int arr[] = {38, 27, 43, 3, 9, 82, 55, 10};
    int n = 8;
    int left = 0, right = n - 1;

    cout << "初始数组: ";
    printArray(arr, n);

    // 手动模拟一次分区（不随机化，选最右边的10作为基准）
    // pivot = arr[right] = 10
    cout << "选择基准: arr[" << right << "] = " << arr[right] << "（最右边元素）" << endl;
    cout << endl;

    int pivot = arr[right];  // 基准值
    int i = left - 1;        // 小于基准区域的最后一个位置

    cout << "开始分区扫描..." << endl;
    cout << "规则: 指针 j 从左往右扫描，遇到小于 " << pivot << " 的元素就交换到前面" << endl;
    cout << "------------------------------------------------" << endl;

    for (int j = left; j < right; j++) {
        if (arr[j] < pivot) {
            i++;
            swap(arr[i], arr[j]);
            cout << "j=" << j << ": arr[" << j << "]=" << arr[i]
                 << " < " << pivot << ", 交换到位置 " << i << "  ";
        } else {
            cout << "j=" << j << ": arr[" << j << "]=" << arr[j]
                 << " >= " << pivot << ", 不交换  ";
        }
        cout << "  → 数组: ";
        printArray(arr, n);
    }

    // 将基准放到正确位置
    swap(arr[i + 1], arr[right]);
    cout << "------------------------------------------------" << endl;
    cout << "扫描完毕！将基准 " << pivot << " 放到位置 " << (i + 1) << endl;
    cout << "最终数组: ";
    printArray(arr, n);
    cout << endl;
    cout << "结果说明：" << endl;
    cout << "  · 基准元素 " << pivot << " 已位于位置 " << (i + 1) << "（最终正确位置）" << endl;
    cout << "  · 左边 [" << left << ".." << i << "] 的元素全部小于 " << pivot << endl;
    cout << "  · 右边 [" << (i + 2) << ".." << right << "] 的元素全部大于等于 " << pivot << endl;
    cout << "  · 接下来分别递归排序左右两个子数组即可" << endl;
    cout << endl;
}

// ================================================================
// 附加演示：快速选择算法 vs 先排序再选择 的对比
// ================================================================
void demoQuickSelectVsSort() {
    cout << "╔══════════════════════════════════════════════╗" << endl;
    cout << "║  演示：快速选择 vs 先排序再选择               ║" << endl;
    cout << "╚══════════════════════════════════════════════╝" << endl;
    cout << endl;

    cout << "问题：在 n=100000 个随机数中找第 50000 大的数" << endl;
    cout << endl;

    cout << "方案1: 先排序再取第k个" << endl;
    cout << "  ① 快速排序整个数组  →  O(n log n) ≈ 100000 × 17 = 1,700,000 次比较" << endl;
    cout << "  ② 取第50000个元素  →  O(1)" << endl;
    cout << "  总计：对整个数组排序，但实际只需要一个元素" << endl;
    cout << endl;

    cout << "方案2: 快速选择算法" << endl;
    cout << "  ① 分区一次  →  O(n)" << endl;
    cout << "  ② 根据排名判断去左边还是右边" << endl;
    cout << "  ③ 只递归一边  →  平均只需 O(n) 次比较" << endl;
    cout << "  总计：不需要完全排序，效率更高！" << endl;
    cout << endl;

    cout << "结论：当只需要找到特定排名的元素时，" << endl;
    cout << "      快速选择的 O(n) 远优于全排序的 O(n log n)。" << endl;
    cout << endl;
}
