/*
 * ================================================================
 * C++算法提高 第22课：背包贪心——部分背包
 * ================================================================
 *
 * 算法知识点：
 *   - 部分背包（分数背包）：物品可以分割，贪心策略有效
 *   - 性价比排序：按 value/weight 从高到低排序，优先选单位价值最高的
 *   - 最优装载：按重量从小到大排序，优先装轻的，最大化数量
 *   - 贪心核心：确定正确的贪心标准（评价函数）
 *
 * 经典例题：
 *   例题1 - 金银岛：N种金属，每种有重量和总价值，背包容量C，求最大价值
 *   例题2 - 最优装载：N个集装箱，轮船载重C，最大化装载数量
 *
 * 对比 0/1 背包：
 *   0/1背包：物品不可分割 → 需要DP（动态规划），O(N*C)
 *   部分背包：物品可以分割 → 贪心即可，O(N log N)
 *   关键区别在于"是否可分割"改变了问题的性质
 */

#include <iostream>
#include <algorithm>
#include <cstdio>
#include <iomanip>
using namespace std;

// ================================================================
// 金属结构体：用于"金银岛"问题（部分背包）
// 包含重量、总价值和性价比（单位重量的价值）
// ================================================================
struct Metal {
    double weight;  // 金属的重量（千克）
    double value;   // 金属的总价值（金币）
    double ratio;   // 性价比 = value / weight（单位重量的价值）

    // 默认构造函数（sort 需要）
    Metal() : weight(0), value(0), ratio(0) {}

    // 带参数的构造函数
    Metal(double w, double v) : weight(w), value(v) {
        ratio = v / w;  // 自动计算性价比
    }

    /*
     * 重载小于号，用于 sort 排序
     * 排序规则：性价比高的排在前面（降序排列）
     * 注意：返回 ratio > other.ratio，所以 sort 会得到降序
     *
     * 为什么要按性价比排序？
     * 因为物品可以分割，每个单位重量应尽可能带来更多价值。
     * 优先选择单位价值最高的物品，是部分背包的最优策略。
     */
    bool operator<(const Metal& other) const {
        return ratio > other.ratio;  // > 号实现降序排列
    }
};

// ================================================================
// 例题1：金银岛（部分背包/分数背包）
// 题目描述：
//   金银岛上有 N 种金属，第 i 种金属的总重量为 w_i 千克，
//   总价值为 v_i 金币。背包容量为 C 千克，金属可以切割。
//   求最多能带走多少金币？
//
// 输入格式：
//   第一行：N C（金属种类数 背包容量）
//   接下来 N 行：w_i v_i（重量 价值）
//
// 输出格式：
//   一个实数，保留两位小数，表示最大金币数
//
// 贪心策略：按性价比（价值/重量）从高到低排序，
//           依次装入，装不下的取部分即可
//
// 时间复杂度：O(N log N) — 排序是瓶颈
// 空间复杂度：O(N) — 存储金属数组
// ================================================================
void solveJinYinDao() {
    cout << "══════════════════════════════════════" << endl;
    cout << "  例题1：金银岛（部分背包）" << endl;
    cout << "══════════════════════════════════════" << endl;
    cout << "金属可以切割，按性价比贪心选取" << endl;
    cout << "--------------------------------------" << endl;

    // ---- 测试数据 ----
    int N = 3;          // 3种金属
    double C = 50.0;    // 背包容量50千克

    // 输入数据：[重量, 价值]
    // 金属1：10kg 60金币 → 性价比 6.0
    // 金属2：20kg 100金币 → 性价比 5.0
    // 金属3：30kg 120金币 → 性价比 4.0
    Metal metals[] = {
        Metal(10, 60),
        Metal(20, 100),
        Metal(30, 120)
    };

    // 打印输入数据
    cout << "输入数据：" << endl;
    cout << "N = " << N << ", 背包容量 C = " << C << " 千克" << endl;
    for (int i = 0; i < N; i++) {
        printf("  金属%d: 重量 %.0f kg, 总价值 %.0f 金币, 性价比 %.2f\n",
               i + 1, metals[i].weight, metals[i].value, metals[i].ratio);
    }
    cout << endl;

    // ---- 第一步：按性价比从高到低排序 ----
    // sort 使用 Metal::operator<，即按 ratio 降序
    sort(metals, metals + N);

    cout << "按性价比排序后：" << endl;
    for (int i = 0; i < N; i++) {
        printf("  金属%d: 重量 %.0f kg, 价值 %.0f 金币, 性价比 %.2f\n",
               i + 1, metals[i].weight, metals[i].value, metals[i].ratio);
    }
    cout << endl;

    // ---- 第二步：贪心装入背包 ----
    double totalValue = 0.0;  // 总价值
    double remain = C;        // 背包剩余容量

    cout << "贪心装入过程：" << endl;
    for (int i = 0; i < N; i++) {
        if (remain <= 0) break;  // 背包已满，提前结束

        if (remain >= metals[i].weight) {
            // 【情况1】剩余容量足够装下整个金属 → 全部装入
            totalValue += metals[i].value;
            cout << "  全部装入金属" << (i + 1) << "：";
            printf("重量 %.0f kg, 获得 %.0f 金币, ",
                   metals[i].weight, metals[i].value);
            remain -= metals[i].weight;
            printf("剩余容量 %.0f kg\n", remain);
        } else {
            // 【情况2】剩余容量不够装整个 → 装入部分
            // 部分价值 = 性价比 × 装入的重量
            double partValue = metals[i].ratio * remain;
            totalValue += partValue;
            cout << "  部分装入金属" << (i + 1) << "：";
            printf("取 %.0f kg (共 %.0f kg), 获得 %.2f 金币, ",
                   remain, metals[i].weight, partValue);
            remain = 0;
            cout << "背包已满！" << endl;
        }
    }

    cout << "--------------------------------------" << endl;
    printf("最大总价值：%.2f 金币\n", totalValue);
    cout << endl;
}

// ================================================================
// 例题2：最优装载
// 题目描述：
//   货轮载重量为 C，码头有 N 个集装箱，第 i 个重 w_i 吨。
//   每个集装箱要么全装要么不装（不能切割）。
//   目标是最大化装载的集装箱数量（不关心价值）。
//   求最多能装多少个集装箱？
//
// 输入格式：
//   第一行：N C（集装箱数量 载重量）
//   第二行：N个整数 w_1 ... w_N（每个集装箱的重量）
//
// 输出格式：
//   一个整数，表示最多装载的集装箱数量
//
// 贪心策略：按重量从小到大排序，先装轻的，能装多少装多少
//   因为每个集装箱对"数量"的贡献都是1，但消耗的载重不同，
//   所以应该优先选择消耗载重最少的（最轻的）集装箱。
//
// 时间复杂度：O(N log N) — 排序
// 空间复杂度：O(N)
// ================================================================
void solveYouZuiZhuangZai() {
    cout << "══════════════════════════════════════" << endl;
    cout << "  例题2：最优装载" << endl;
    cout << "══════════════════════════════════════" << endl;
    cout << "按重量从小到大排序，优先装轻的" << endl;
    cout << "--------------------------------------" << endl;

    // ---- 测试数据 ----
    int N = 6;      // 6个集装箱
    int C = 50;     // 载重量50吨

    // 集装箱重量：30, 10, 15, 20, 25, 5
    int weights[] = {30, 10, 15, 20, 25, 5};

    // 打印输入数据
    cout << "输入数据：" << endl;
    cout << "N = " << N << ", 载重量 C = " << C << " 吨" << endl;
    cout << "集装箱重量：[ ";
    for (int i = 0; i < N; i++) {
        cout << weights[i] << " ";
    }
    cout << "]" << endl;
    cout << endl;

    // ---- 第一步：按重量从小到大排序 ----
    sort(weights, weights + N);

    cout << "按重量排序后：[ ";
    for (int i = 0; i < N; i++) {
        cout << weights[i] << " ";
    }
    cout << "]" << endl;
    cout << endl;

    // ---- 第二步：贪心装载 ----
    int count = 0;      // 已装载数量
    int totalW = 0;     // 已装载总重量

    cout << "贪心装载过程：" << endl;
    for (int i = 0; i < N; i++) {
        if (totalW + weights[i] <= C) {
            // 还能装下当前的集装箱
            totalW += weights[i];
            count++;
            cout << "  装入集装箱" << (i + 1) << "：重量 " << weights[i]
                 << " 吨, 累计重量 " << totalW
                 << " 吨, 累计数量 " << count << " 个" << endl;
        } else {
            // 装不下了，更重的更装不下，直接退出
            cout << "  集装箱" << (i + 1) << "（重量 " << weights[i]
                 << " 吨）装不下，停止装载。" << endl;
            break;
        }
    }

    cout << "--------------------------------------" << endl;
    cout << "最多装载数量：" << count << " 个" << endl;
    cout << "总重量：" << totalW << " 吨（载重量 " << C << " 吨）" << endl;
    cout << endl;
}

// ================================================================
// 附加演示：0/1背包 vs 部分背包反例
// 用一个具体例子说明为什么0/1背包不能用贪心（性价比排序）
// 帮助学生理解两种背包问题的本质区别
// ================================================================
void demoZeroOneVsFractional() {
    cout << "══════════════════════════════════════" << endl;
    cout << "  演示：为什么0/1背包不能用贪心？" << endl;
    cout << "══════════════════════════════════════" << endl;

    // 假设有3件物品，背包容量 50
    // 物品A: 重量10, 价值60, 性价比=6
    // 物品B: 重量20, 价值100, 性价比=5
    // 物品C: 重量30, 价值120, 性价比=4
    //
    // 若按贪心（性价比排序）：
    //   装A(10,60) → 装B(20,100) → 剩余20装不下C(30)
    //   总价值 = 60 + 100 = 160
    //
    // 但如果我们放弃B，选择C：
    //   装A(10,60) → 装C(30,120) → 总价值 = 180
    //   180 > 160！说明贪心不是最优！
    //
    // 为什么部分背包可以用贪心？
    //   因为可以切割，A+B+部分C = 60+100+80 = 240，这是最优的

    cout << "假设背包容量 50，三件物品：" << endl;
    cout << "  物品A: 重量 10, 价值 60, 性价比 = 6.0" << endl;
    cout << "  物品B: 重量 20, 价值 100, 性价比 = 5.0" << endl;
    cout << "  物品C: 重量 30, 价值 120, 性价比 = 4.0" << endl;
    cout << endl;

    cout << "【0/1背包（不可分割）】贪心策略：" << endl;
    cout << "  选A(10,60) + B(20,100) = 总价值 160" << endl;
    cout << "  但如果选A(10,60) + C(30,120) = 总价值 180" << endl;
    cout << "  → 贪心结果 160 < 最优结果 180，贪心失效！" << endl;
    cout << endl;

    cout << "【部分背包（可以分割）】贪心策略：" << endl;
    cout << "  选A(10,60) + B(20,100) + C的部分(20, 80) = 240" << endl;
    cout << "  → 这就是最优解，贪心有效！" << endl;
    cout << endl;

    cout << "结论：物品能否分割，决定了贪心策略是否有效。" << endl;
    cout << "      不可分割时，需要考虑[替换]效应，需要DP求解。" << endl;
    cout << "      可分割时，不存在替换问题，贪心就是最优。" << endl;
    cout << endl;
}
