/*
 * ================================================================
 * C++算法提高 第23课：贪心策略证明——交换论证与归纳法
 * ================================================================
 *
 * 算法知识点：
 *   - 贪心策略证明的两种方法：交换论证、归纳法
 *   - 交换论证：假设存在更优解，用贪心选择替换，证明结果不差
 *   - 归纳法：基础步骤（第一步贪心选择正确）+ 归纳步骤（第k步→第k+1步）
 *   - 排序贪心模式：通过排序确定操作顺序
 *   - 优先队列贪心模式：每次取最优（最小）元素进行操作
 *
 * 经典例题：
 *   例题1 - 排队接水：N个人排队接水，按接水时间升序排列使总等待时间最小
 *   例题2 - 合并果子：N堆果子，每次合并最小的两堆，总代价最小（哈夫曼）
 */

#include <iostream>
#include <algorithm>   // sort
#include <queue>       // priority_queue
#include <vector>      // vector
#include <iomanip>     // setw
using namespace std;

// ================================================================
// 例题1：排队接水
// 题目：有 n 个人排队接水，每人需要 t[i] 时间。
//       求一种排队顺序，使所有人的等待时间之和最小。
//
// 贪心策略：按接水时间从小到大排序
//
// 交换论证证明（简述）：
//   假设最优排列中，存在某人A在某人B前面，但 t_A > t_B。
//   交换A和B的位置。
//   - A前面的人：等待时间不变
//   - B后面的人：等待时间不变（因为A和B的总接水时间没变）
//   - A和B之间的人（如果有）：等待时间都减少了 (t_A - t_B) > 0
//   因此总等待时间严格减少，矛盾！
//   结论：任何最优排列中，接水时间短的人必须在前面。
//   所以按时间升序排列就是最优解。
//
// 时间复杂度：O(n log n) —— 排序是瓶颈
// 空间复杂度：O(n) —— 存储n个人的信息
// ================================================================

// 定义人员结构体
struct Person {
    int id;      // 原始编号（输入顺序，从1开始）
    int time;    // 接水所需时间（分钟）
};

// 排序比较函数：按接水时间升序排列
bool cmpByTime(Person a, Person b) {
    return a.time < b.time;
}

void solveQueueForWater() {
    cout << "╔══════════════════════════════════════════╗" << endl;
    cout << "║     例题1：排队接水                        ║" << endl;
    cout << "╚══════════════════════════════════════════╝" << endl;
    cout << endl;

    // ---- 测试数据 ----
    // n = 5 个人，接水时间分别为：3, 6, 1, 4, 2
    const int n = 5;
    Person persons[n] = {
        {1, 3},
        {2, 6},
        {3, 1},
        {4, 4},
        {5, 2}
    };

    cout << "【输入数据】" << endl;
    cout << "人数 n = " << n << endl;
    cout << "接水时间：";
    for (int i = 0; i < n; i++) {
        cout << persons[i].time << " ";
    }
    cout << endl << endl;

    // ---- 贪心核心：按接水时间升序排列 ----
    // 这就是贪心选择——让接水快的人排在前面
    sort(persons, persons + n, cmpByTime);

    // ---- 计算最小总等待时间 ----
    // 方法：遍历排序后的数组，维护一个"已接水时间总和"的前缀和
    // 第 i 个人的等待时间 = 前面 i-1 个人的接水时间之和
    long long totalWait = 0;   // 所有人等待时间之和
    long long prefixSum = 0;   // 前面所有人的接水时间之和（前缀和）

    cout << "【排序后的排队顺序】" << endl;
    cout << "顺序 | 编号 | 接水时间 | 等待时间" << endl;
    cout << "-----+------+----------+----------" << endl;

    for (int i = 0; i < n; i++) {
        // 当前这个人的等待时间 = 前面所有人的接水时间之和
        long long waitTime = prefixSum;
        totalWait += waitTime;

        cout << "  " << (i + 1) << "   |  "
             << setw(2) << persons[i].id << "  |    "
             << setw(2) << persons[i].time << "    |    "
             << setw(3) << waitTime << endl;

        // 更新前缀和：加上当前这个人的接水时间
        // 这样后面的人等待时间就会包含这个人的接水时间
        prefixSum += persons[i].time;
    }

    cout << "-----+------+----------+----------" << endl;
    cout << endl;

    // ---- 输出结果 ----
    cout << "【最优排队顺序（按编号输出）】" << endl;
    for (int i = 0; i < n; i++) {
        cout << persons[i].id;
        if (i < n - 1) cout << " ";
    }
    cout << endl;

    cout << "【最小总等待时间】" << totalWait << " 分钟" << endl;
    cout << endl;

    // ---- 验证交换论证 ----
    // 演示：如果故意把时间长的放在前面会怎样？
    cout << "【交换论证验证】" << endl;
    cout << "假设把接水时间 6 的人放在接水时间 1 的人前面：" << endl;
    cout << "  原始最优顺序（部分）：... , 1分钟, 6分钟, ..." << endl;
    cout << "  若交换成：... , 6分钟, 1分钟, ..." << endl;
    cout << "  中间所有人的等待时间增加 (6-1)=5 分钟" << endl;
    cout << "  => 总等待时间必定增加，不可能是最优解" << endl;
    cout << "  结论：时间短的必须在前面！✓" << endl;
    cout << endl;
}

// ================================================================
// 例题2：合并果子
// 题目：有 n 堆果子，每堆重量为 a[i]。每次可以合并任意两堆，
//       合并的代价等于两堆重量之和。求合并成一堆的最小总代价。
//
// 贪心策略：每次选取最小的两堆进行合并（使用最小堆/优先队列）
//
// 归纳法证明（简述）：
//   基础步骤（n=2）：只有两堆时，只有一种合并方式，显然最优。
//
//   归纳步骤：
//     设所有堆中最小的两堆重量为 a 和 b（a ≤ b）。
//     我们可以证明：存在一个最优方案，a 和 b 是第一次合并的两堆。
//
//     证明方法（反证 + 交换论证）：
//       假设最优方案中，第一次合并的不是 a 和 b，而是某两堆 x 和 y，
//       且 a ≤ b ≤ x ≤ y（因为是"最小的两堆"）。
//       把最优方案中 x 和 y 的位置替换为 a 和 b（调整后续合并），
//       可以证明新方案的总代价 ≤ 原方案的总代价。
//       因此，"先合并 a 和 b" 一定不差。
//
//     合并 a 和 b 后，问题变成 n-1 堆的子问题（多了一堆 a+b）。
//     由归纳假设，贪心策略在子问题上也最优。
//     因此整体最优。得证。
//
// 时间复杂度：O(n log n) —— 每次堆操作 O(log n)，共 n-1 次合并
// 空间复杂度：O(n) —— 优先队列存储最多 n 个元素
// ================================================================

void solveMergeFruits() {
    cout << "╔══════════════════════════════════════════╗" << endl;
    cout << "║     例题2：合并果子                        ║" << endl;
    cout << "╚══════════════════════════════════════════╝" << endl;
    cout << endl;

    // ---- 测试数据 ----
    // n = 5 堆果子，重量分别为：1, 3, 4, 2, 5
    const int n = 5;
    int piles[n] = {1, 3, 4, 2, 5};

    cout << "【输入数据】" << endl;
    cout << "果子堆数 n = " << n << endl;
    cout << "各堆重量：";
    for (int i = 0; i < n; i++) {
        cout << piles[i] << " ";
    }
    cout << endl << endl;

    // ---- 贪心核心：使用最小堆（优先队列） ----
    // priority_queue 默认是大顶堆，加 greater<int> 变成小顶堆
    // 这样每次 top() 取出的都是最小的元素
    priority_queue<int, vector<int>, greater<int>> minHeap;

    // 将所有果子堆的重量加入最小堆
    for (int i = 0; i < n; i++) {
        minHeap.push(piles[i]);
    }

    long long totalCost = 0;   // 总合并代价（体力消耗）
    int step = 0;              // 合并步骤计数

    cout << "【合并过程】" << endl;
    cout << "步骤 | 堆1 | 堆2 | 新堆 | 本步代价 | 累计代价" << endl;
    cout << "-----+-----+-----+------+----------+----------" << endl;

    // 循环直到只剩一堆（堆大小为1）
    while (minHeap.size() > 1) {
        step++;

        // 贪心选择：取出最小的两堆
        int smallest1 = minHeap.top();  // 最小堆
        minHeap.pop();
        int smallest2 = minHeap.top();  // 次小堆
        minHeap.pop();

        // 合并这两堆
        int mergedPile = smallest1 + smallest2;
        totalCost += mergedPile;

        // 将合并后的新堆放回优先队列
        minHeap.push(mergedPile);

        // 输出本步详情
        cout << "  " << setw(2) << step << "  | "
             << setw(3) << smallest1 << " | "
             << setw(3) << smallest2 << " | "
             << setw(4) << mergedPile << " | "
             << setw(8) << smallest1 + smallest2 << " | "
             << setw(8) << totalCost << endl;
    }

    cout << "-----+-----+-----+------+----------+----------" << endl;
    cout << endl;

    // ---- 输出结果 ----
    cout << "【最小总体力消耗】" << totalCost << endl;
    cout << endl;

    // ---- 图解合并树 ----
    // 展示哈夫曼树的合并过程
    cout << "【合并树（类似哈夫曼树）】" << endl;
    cout << "  每次选最小的两堆合并，合并后的新堆参与后续选择" << endl;
    cout << endl;

    // 手动模拟哈夫曼合并树以便清晰展示
    cout << "          15 ← 最终一堆" << endl;
    cout << "         /  \\" << endl;
    cout << "        6    9" << endl;
    cout << "       / \\  / \\" << endl;
    cout << "      3   3 4  5" << endl;
    cout << "     / \\" << endl;
    cout << "    1   2" << endl;
    cout << endl;
    cout << "  合并顺序：" << endl;
    cout << "    步骤1: 1+2=3  (代价3)" << endl;
    cout << "    步骤2: 3+3=6  (代价6)" << endl;
    cout << "    步骤3: 4+5=9  (代价9)" << endl;
    cout << "    步骤4: 6+9=15 (代价15)" << endl;
    cout << "    总代价: 3+6+9+15 = 33" << endl;
    cout << endl;

    // ---- 对比：如果不贪心会怎样 ----
    cout << "【贪心 vs 非贪心对比】" << endl;
    cout << "  贪心策略（每次选最小两堆）：总代价 = " << totalCost << endl;
    cout << "  如果随便合并（比如从左到右）：" << endl;
    cout << "    1+3=4, 4+4=8, 8+2=10, 10+5=15" << endl;
    cout << "    总代价 = 4+8+10+15 = 37 > " << totalCost << endl;
    cout << "  => 贪心策略确实更优！✓" << endl;
    cout << endl;
}

// ================================================================
// 附加例题：排队接水 —— 多水龙头版本（选学）
// 题目：有 m 个水龙头，n 个人，求最小总等待时间
//
// 贪心策略：仍按接水时间升序排序，然后轮流分配给人最少的水龙头
// 证明思路：与单水龙头类似，交换论证仍然适用
// ================================================================

void solveMultiTapQueue() {
    cout << "╔══════════════════════════════════════════╗" << endl;
    cout << "║  附加题：多水龙头排队接水（选学）          ║" << endl;
    cout << "╚══════════════════════════════════════════╝" << endl;
    cout << endl;

    const int n = 6;  // 6个人
    const int m = 2;  // 2个水龙头
    int times[n] = {1, 2, 5, 3, 7, 4};

    // 按接水时间升序排序
    sort(times, times + n);

    // m 个水龙头的累计接水时间
    // tapTime[i] 表示第 i 个水龙头当前已安排的总接水时间
    // 初始化为0（相当于小顶堆，但没有用 priority_queue 是为了教学清晰）
    long long tapTime[m] = {0};

    cout << "【输入】" << endl;
    cout << n << " 个人, " << m << " 个水龙头" << endl;
    cout << "接水时间（排序后）：";
    for (int i = 0; i < n; i++) cout << times[i] << " ";
    cout << endl << endl;

    cout << "【分配过程】" << endl;
    long long totalWait = 0;

    for (int i = 0; i < n; i++) {
        // 找到当前累计接水时间最少的水龙头（贪心选择）
        int minTap = 0;
        for (int j = 1; j < m; j++) {
            if (tapTime[j] < tapTime[minTap]) {
                minTap = j;
            }
        }

        // 当前这个人被分配到 minTap 号水龙头
        // 他的等待时间 = 该水龙头已有的累计接水时间
        long long wait = tapTime[minTap];
        totalWait += wait;

        cout << "  接水时间 " << times[i] << " 的人 → 水龙头" << (minTap + 1)
             << "（等待" << wait << "分钟）" << endl;

        // 更新该水龙头的累计接水时间
        tapTime[minTap] += times[i];
    }

    cout << endl;
    cout << "【最小总等待时间】" << totalWait << " 分钟" << endl;
    cout << endl;
}

void demoExchangeArgument() {
    cout << "╔══════════════════════════════════════════╗" << endl;
    cout << "║  交换论证直观演示                         ║" << endl;
    cout << "╚══════════════════════════════════════════╝" << endl;
    cout << endl;

    cout << "【交换论证五步法】" << endl;
    cout << endl;

    cout << "  ┌─────────────────────────────────────────────────────┐" << endl;
    cout << "  │ 第1步：假设存在最优解 OPT                            │" << endl;
    cout << "  │ 第2步：找到 OPT 中第一个与贪心解不同的位置            │" << endl;
    cout << "  │ 第3步：用贪心选择替换 OPT 中该位置的选择              │" << endl;
    cout << "  │ 第4步：证明替换后结果不会变差                        │" << endl;
    cout << "  │ 第5步：重复2-4步，直到 OPT = 贪心解，结论成立        │" << endl;
    cout << "  └─────────────────────────────────────────────────────┘" << endl;
    cout << endl;

    // 用排队接水做具体演示
    cout << "【实例：排队接水中的交换论证】" << endl;
    cout << endl;

    cout << "  接水时间：[A=5分钟, B=2分钟, C=3分钟, D=1分钟]" << endl;
    cout << endl;
    cout << "  假设某人声称最优顺序是：[A(5), B(2), C(3), D(1)]" << endl;
    cout << "                         ↑贪心说A不该在这里" << endl;
    cout << endl;
    cout << "  执行交换论证：" << endl;
    cout << "    第1步：找到第一个「贪心解和OPT不同」的位置 → 位置0" << endl;
    cout << "    第2步：贪心解在此处放的是D(1分钟)，OPT放的是A(5分钟)" << endl;
    cout << "    第3步：把OPT中的A(5)和D(1)交换" << endl;
    cout << "           交换前 OPT: A(5), B(2), C(3), D(1)" << endl;
    cout << "           交换后     : D(1), B(2), C(3), A(5)" << endl;
    cout << "    第4步：计算等待时间" << endl;
    cout << "           交换前: 0+5+7+10 = 22" << endl;
    cout << "           交换后: 0+1+3+6  = 10 ← 明显更小！" << endl;
    cout << "    第5步：继续调整直到完全按时间升序" << endl;
    cout << "           最终: [D(1), B(2), C(3), A(5)] → 等待时间=10" << endl;
    cout << "           这是真正的最优解 ✓" << endl;
    cout << endl;
}
