// 一行代码判断闰年 —— 先判400，再判4且不判100
bool isLeap(int y) {
    return (y % 400 == 0) ||
           (y % 4 == 0 && y % 100 != 0);
}
// 注意：除号 / 取余 %，不要混用！
⚡ 关键点：判断顺序有讲究！先判断 400（因为能被400整除的一定能被4和100整除），再判断4且不判100。

totalDays 方法 —— 累计总天数

// 数组查表：monthDays[1~12]，2月初始值 = 28
int monthDays[13] = {0, 31, 28, 31, 30, 31, 30,
                        31, 31, 30, 31, 30, 31};

// 获取指定年月的天数
int getMonthDays(int y, int m) {
    if (m == 2 && isLeap(y)) return 29;  // 2月特判
    return monthDays[m];                  // 其余直接查表
}

// 计算从公元1年1月1日到 y-m-d 的总天数
long long totalDays(int y, int m, int d) {
    long long days = 0;
    for (int i = 1; i < y; i++)          // 累加整年
        days += isLeap(i) ? 366 : 365;
    for (int i = 1; i < m; i++)          // 累加整月
        days += getMonthDays(y, i);
    return days + d;                      // 加上当月天数
}

星期推算：锚点 + 取模

锚点: 1900年1月1日 = 星期一 对于任意日期 (y, m, d): ┌──────────────────────────────────────────────┐ │ diff = totalDays(y, m, d) - totalDays(1900, 1, 1) │ │ weekday = diff % 7 │ │ │ │ 余数: 0=星期一 1=星期二 2=星期三 3=星期四 │ │ 4=星期五 5=星期六 6=星期日 │ └──────────────────────────────────────────────┘
🎯 核心技巧：两个日期的天数差 = totalDays(日期A) - totalDays(日期B)。只要写好 totalDays，日期差、星期几问题迎刃而解！

例题一：黑色星期五

统计某年范围内，13号恰好是星期五的日期数量

题目描述

给定正整数 N（N ≥ 1900），求 1900年1月1日 ~ N年12月31日 之间，有多少个日期满足：日期 = 13号 且 星期 = 星期五。

解题步骤

已知: 1900-01-01 = 星期一 (weekday = 0) Step 1: 写好 isLeap()、getMonthDays()、totalDays() 三个基础函数 Step 2: 遍历所有年份 (1900 ≤ y ≤ N) 遍历所有月份 (1 ≤ m ≤ 12) 计算 (y, m, 13) 的星期几 diff = totalDays(y, m, 13) - totalDays(1900, 1, 1) 若 diff % 7 == 4 → count++ Step 3: 输出 count 
// 核心统计逻辑
int countBlackFriday(int N) {
    int count = 0;
    long long base = totalDays(1900, 1, 1);
    for (int y = 1900; y <= N; y++) {
        for (int m = 1; m <= 12; m++) {
            long long diff = totalDays(y, m, 13) - base;
            if (diff % 7 == 4) count++;  // 4 = 星期五
        }
    }
    return count;
}
// 复杂度: O(N) — 需要遍历每个月

例题二：日期差问题

计算两个日期之间相隔的天数

题目描述

输入两个日期 (y1, m1, d1) 和 (y2, m2, d2)，保证日期1 早于或等于 日期2。求两个日期之间相隔多少天（不含两端或含两端，按题目要求）。

解法一：totalDays 相减法（推荐）

// 最优雅的写法：直接相减
long long dateDiff(int y1, int m1, int d1,
                    int y2, int m2, int d2) {
    long long t1 = totalDays(y1, m1, d1);
    long long t2 = totalDays(y2, m2, d2);
    return t2 - t1;  // 不含起始日，含结束日
}
// 复杂度: O(y2 - y1)  →  年份跨度越大越慢
// 但 y2-y1 通常 ≤ 200，完全可以接受

解法二：逐日模拟法（直观但慢）

d1=1, m1=1, y1=2023 → d2=31, m2=12, y2=2023 用 while 逐日推进，适合日期跨度小的场景： 天数=0; while(日期1 != 日期2) { 日期1++; 天数++; } ──────────────────────────────────────────── 优点: 逻辑简单，不容易出错 缺点: 跨年大跨度时极慢（可能循环数十万次）

两种方法对比

方法时间复杂度适用场景推荐度
totalDays 相减O(年数)所有场景★★★★★
逐日模拟O(天数)跨度 <1000天★★☆☆☆
数学公式法O(1)竞速优化★★★☆☆
💡 实战建议：考试中直接用 totalDays 相减法，代码短、不易错、性能足够。只有日期跨度极大（如几万年）时才需要考虑优化。

⚡ 常见错误警示

  1. 忘了 1900年是平年，不是闰年！
  2. 数组索引：monthDays[0] 不用，月份从 下标1 开始
  3. 星期编号：确认 0 是周一还是周日，不同题目可能不同
  4. 计算星期时别忘了减去 锚点日期的 totalDays
  5. totalDays 的返回值要用 long long，int 会溢出
  6. 判断闰年用 取余 % 而不是除法 /
