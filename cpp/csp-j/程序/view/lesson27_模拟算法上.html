#include <iostream>
#include <string>
#include <algorithm>   // reverse
using namespace std;

// ============================================================
// 演示 1：闰年判断
//
// 闰年规则（格里高利历）：
//   1. 能被 4 整除但不能被 100 整除 → 闰年
//   2. 能被 400 整除 → 闰年
//   3. 其他 → 平年
//
// 示例：
//   2000 是闰年（能被400整除）
//   1900 不是闰年（能被100整除但不能被400整除）
//   2024 是闰年（能被4整除且不能被100整除）
//   2023 不是闰年
// ============================================================

/**
 * @brief 判断给定年份是否为闰年
 *
 * 闰年规则：
 *   (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)
 *
 * @param year  年份（正整数）
 * @return      true 如果是闰年，否则 false
 */
bool isLeap(int year) {
    // 能被400整除的一定是闰年（如2000年）
    if (year % 400 == 0) return true;
    // 能被100整除但不能被400整除的不是闰年（如1900年）
    if (year % 100 == 0) return false;
    // 能被4整除的是闰年
    if (year % 4 == 0) return true;
    // 其他情况为平年
    return false;
}

/**
 * @brief 演示闰年判断
 */
void demoIsLeap() {
    cout << "【演示1】闰年判断：" << endl;
    cout << "  规则：能被4整除但不能被100整除，或能被400整除" << endl;
    cout << endl;

    // 选取几个有代表性的年份进行测试
    int testYears[] = {2000, 1900, 2024, 2023, 2100, 2400};
    int numTests = sizeof(testYears) / sizeof(testYears[0]);

    for (int i = 0; i < numTests; i++) {
        int y = testYears[i];
        cout << "  " << y << " 年 → ";
        if (isLeap(y)) {
            cout << "闰年（366天）" << endl;
        } else {
            cout << "平年（365天）" << endl;
        }
    }
    cout << endl;
}

// ============================================================
// 演示 2：日期计算（两个日期之间的天数差）
//
// 计算方法：将每个日期换算为"从公元1年1月1日起的天数"，
// 然后两个天数相减即可得到日期差。
// （这里简化为从 year 年 1 月 1 日起的天数 + 跨年的天数）
//
// 这个方法的优点是不需要特殊处理跨月、跨年的边界情况，
// 只需要一个"日期 → 天序号"的转换函数。
// ============================================================

/**
 * @brief 获取某年某月的天数
 *
 * 需要根据闰年/平年来确定 2 月的天数：
 *   闰年 2 月 29 天，平年 2 月 28 天。
 *   其他月份天数固定：1,3,5,7,8,10,12 月 31 天，
 *                      4,6,9,11 月 30 天。
 *
 * @param year   年份
 * @param month  月份（1~12）
 * @return       该月天数
 */
int daysInMonth(int year, int month) {
    // 各月天数（默认平年，2月先填28）
    int monthDays[] = {0, 31, 28, 31, 30, 31, 30,
                          31, 31, 30, 31, 30, 31};
    // 闰年2月特殊处理
    if (month == 2 && isLeap(year)) {
        return 29;
    }
    return monthDays[month];
}

/**
 * @brief 计算从公元1年1月1日到给定日期的总天数
 *
 * 算法：
 *   1. 累加从 1 年到 year-1 年的全年天数（365 或 366）。
 *   2. 累加 year 年从 1 月到 month-1 月的天数。
 *   3. 加上 day。
 *
 * @param year   年份
 * @param month  月份（1~12）
 * @param day    日（1~31，假设输入合法）
 * @return       从公元1年1月1日起的总天数
 */
int dateToDays(int year, int month, int day) {
    int total = 0;

    // 第一步：累加从 1 年到 year-1 年的所有天数
    for (int y = 1; y < year; y++) {
        total += isLeap(y) ? 366 : 365;
    }

    // 第二步：累加 year 年 1 月到 month-1 月的天数
    for (int m = 1; m < month; m++) {
        total += daysInMonth(year, m);
    }

    // 第三步：加上本月的天数
    total += day;

    return total;
}

/**
 * @brief 计算两个日期之间的天数差（绝对值）
 *
 * 将两个日期分别转换为天序号，然后相减取绝对值。
 *
 * @param y1, m1, d1  第一个日期
 * @param y2, m2, d2  第二个日期
 * @return            两个日期之间的天数差（非负）
 */
int daysBetween(int y1, int m1, int d1, int y2, int m2, int d2) {
    int days1 = dateToDays(y1, m1, d1);
    int days2 = dateToDays(y2, m2, d2);
    return abs(days1 - days2);  // 返回绝对差值
}

/**
 * @brief 演示日期计算
 */
void demoDateCalculation() {
    cout << "【演示2】日期计算（两个日期之间的天数差）：" << endl;
    cout << "  方法：将日期转换为从公元1年1月1日起的天数，再相减" << endl;
    cout << endl;

    // 示例 1：同一年内
    int y1 = 2024, m1 = 1, d1 = 1;
    int y2 = 2024, m2 = 12, d2 = 31;
    int diff = daysBetween(y1, m1, d1, y2, m2, d2);
    cout << "  示例1: " << y1 << "/" << m1 << "/" << d1
         << " 到 " << y2 << "/" << m2 << "/" << d2
         << " 相差 " << diff << " 天" << endl;
    // 2024年是闰年，共366天，1月1日到12月31日相差365天
    cout << "    （2024是闰年，全年366天，首尾相差365天）" << endl;

    // 示例 2：跨年
    y1 = 2023; m1 = 12; d1 = 31;
    y2 = 2024; m2 = 1;  d2 = 1;
    diff = daysBetween(y1, m1, d1, y2, m2, d2);
    cout << "  示例2: " << y1 << "/" << m1 << "/" << d1
         << " 到 " << y2 << "/" << m2 << "/" << d2
         << " 相差 " << diff << " 天" << endl;

    // 示例 3：出生天数
    y1 = 2010; m1 = 1; d1 = 1;
    y2 = 2024; m2 = 6; d2 = 1;
    diff = daysBetween(y1, m1, d1, y2, m2, d2);
    cout << "  示例3: " << y1 << "/" << m1 << "/" << d1
         << " 到 " << y2 << "/" << m2 << "/" << d2
         << " 相差 " << diff << " 天" << endl;

    cout << endl;
}

// ============================================================
// 演示 3：高精度加法（字符串模拟竖式加法）
//
// 问题背景：C++ 中 unsigned long long 最大也只有约 1.8×10^19，
// 当需要计算上百位的整数加法时，内置类型无法满足。
// 高精度计算（又称大数运算）用字符串存储每一位数字，
// 模仿手工竖式计算的过程来逐位运算。
//
// 竖式加法过程（以 123 + 4567 为例）：
//      1 2 3
//   +  4 5 6 7
//   -----------
//   =  4 6 9 0
//
// 从最低位（个位）开始逐位相加，进位 carry 保存到高位。
//
// 关键步骤：
//   1. 将两个字符串反转（方便从低位开始处理）。
//   2. 逐位相加，记录进位。
//   3. 处理最高位的进位。
//   4. 将结果反转回来。
// ============================================================

/**
 * @brief 高精度加法：计算两个非负大整数的和
 *
 * 算法步骤（模拟竖式加法）：
 *   1. 将两个数字字符串 s1, s2 反转，使低位在前。
 *   2. 设 carry = 0（进位），从 i=0 开始：
 *      - digit = (s1[i]的数字) + (s2[i]的数字) + carry
 *      - 当前位结果 = digit % 10（取个位）
 *      - 新进位 carry = digit / 10（取十位及以上）
 *   3. 循环直到两个字符串都处理完且 carry == 0。
 *   4. 将结果字符串反转回来。
 *
 * 时间复杂度：O(max(len1, len2))
 *
 * @param a  第一个加数（十进制数字字符串，如 "123456789"）
 * @param b  第二个加数（十进制数字字符串）
 * @return   两数之和的字符串表示
 */
string highPrecisionAdd(string a, string b) {
    // 第一步：反转字符串，让低位（个位）在索引 0 处
    // 这样我们就可以用一个简单的 for 循环从 0 开始逐位处理。
    reverse(a.begin(), a.end());
    reverse(b.begin(), b.end());

    string result = "";           // 存储计算结果（也是低位在前）
    int carry = 0;                // 进位：初始为 0
    int lenA = a.length();        // 第一个数的长度
    int lenB = b.length();        // 第二个数的长度
    int maxLen = max(lenA, lenB); // 最长的长度

    // 第二步：逐位相加
    for (int i = 0; i < maxLen; i++) {
        // 取当前位的数字，如果已超出长度则视为 0
        int digitA = (i < lenA) ? (a[i] - '0') : 0;  // 字符'0'~'9'减去'0'得到数值0~9
        int digitB = (i < lenB) ? (b[i] - '0') : 0;

        int sum = digitA + digitB + carry;  // 当前位之和 + 低位的进位
        carry = sum / 10;                    // 新的进位（向高位进多少）
        int currentDigit = sum % 10;         // 当前位结果（0~9）

        // 将当前位转为字符追加到结果中
        result += (char)(currentDigit + '0');
    }

    // 第三步：处理最高位可能的进位
    // 例如 999 + 1 = 1000，最高位会多出一位 1
    if (carry > 0) {
        result += (char)(carry + '0');
    }

    // 第四步：反转回来，得到正常的高位在前的数字字符串
    reverse(result.begin(), result.end());

    return result;
}

/**
 * @brief 演示高精度加法
 */
void demoHighPrecisionAdd() {
    cout << "【演示3】高精度加法（字符串模拟竖式加法）：" << endl;
    cout << "  方法：将两个数字以字符串存储，逐位模拟手工竖式加法" << endl;
    cout << endl;

    // 示例 1：普通加法
    string a1 = "123", b1 = "4567";
    string sum1 = highPrecisionAdd(a1, b1);
    cout << "  示例1: " << a1 << " + " << b1 << " = " << sum1 << endl;
    cout << "    竖式验证:   123" << endl;
    cout << "             + 4567" << endl;
    cout << "             ------" << endl;
    cout << "             = 4690 ✓" << endl;

    // 示例 2：有连续进位的加法
    string a2 = "999", b2 = "1";
    string sum2 = highPrecisionAdd(a2, b2);
    cout << "  示例2: " << a2 << " + " << b2 << " = " << sum2 << endl;
    cout << "    （连续进位示例：999 + 1 = 1000）" << endl;

    // 示例 3：超大整数加法（50 位）
    string a3 = "123456789012345678901234567890";
    string b3 = "987654321098765432109876543210";
    string sum3 = highPrecisionAdd(a3, b3);
    cout << "  示例3: 50位大整数加法：" << endl;
    cout << "    加数1: " << a3 << endl;
    cout << "    加数2: " << b3 << endl;
    cout << "    结果:  " << sum3 << endl;

    // 示例 4：不等长大整数加法
    string a4 = "10000000000000000000000000000000000000000000000000";  // 10^49
    string b4 = "1";
    string sum4 = highPrecisionAdd(a4, b4);
    cout << "  示例4: 10^49 + 1 = " << sum4 << endl;

    cout << endl;
}
