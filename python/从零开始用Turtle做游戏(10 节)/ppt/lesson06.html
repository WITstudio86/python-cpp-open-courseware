  3. listen 后绑键，ontimer(tick, 20)

第 1 / 5 步 · 打

左右躲开

player 放底部。press_left 改 vx。tick 末尾要 ontimer。

import turtle
import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.title("第 6 课 · 躲避陨石")
screen.bgcolor("black")
screen.tracer(0)
vx = 0
player = turtle.Turtle()
player.shape("turtle")
player.color("deepskyblue")
player.shapesize(1.4, 1.4)
player.penup()
player.goto(0, -230)
player.setheading(90)

def press_left():
    global vx
    vx = -8
def press_right():
    global vx
    vx = 8
def stop():
    global vx
    vx = 0
def tick():
    player.setx(player.xcor() + vx)
    if player.xcor() > 370:
        player.setx(370)
    if player.xcor() < -370:
        player.setx(-370)
    screen.update()
    screen.ontimer(tick, 20)

screen.listen()
screen.onkeypress(press_left, "Left")
screen.onkeypress(press_right, "Right")
screen.onkeyrelease(stop, "Left")
screen.onkeyrelease(stop, "Right")
screen.ontimer(tick, 20)
turtle.done()
运行后你该看到： 按住左右，蓝色海龟在底部滑动。天上还没有陨石。
坑： 忘了 global vx，海龟原地不动。

第 2 / 5 步 · 讲

6 颗陨石下雨

6 个橙圆从天上往下掉，掉出底部就送回天上。

为什么： 不要写 m1 m2 m3。创建、append、for 更新，是从这课开始的标准套路。
  1. player 下面 meteors = []，for 循环 6 次 append
  2. 加 reset_meteor
  3. tick 里 for m in meteors：y 减 4，掉出就 reset
运行后你该看到： 6 颗橙圆往下砸，掉出窗口后又从天上出现。这时撞上还没事。

第 2 / 5 步 · 打

6 颗陨石下雨

用列表装 6 颗橙圆。掉出 -300 就 reset_meteor。每只都要 penup。

import turtle
import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.title("第 6 课 · 躲避陨石")
screen.bgcolor("black")
screen.tracer(0)
vx = 0
player = turtle.Turtle()
player.shape("turtle")
player.color("deepskyblue")
player.shapesize(1.4, 1.4)
player.penup()
player.goto(0, -230)
player.setheading(90)
meteors = []
for _ in range(6):
    m = turtle.Turtle()
    m.shape("circle")
    m.color("orange")
    m.penup()
    m.goto(random.randint(-360, 360), random.randint(80, 360))
    meteors.append(m)

def reset_meteor(m):
    m.goto(random.randint(-360, 360), random.randint(280, 420))
def press_left():
    global vx
    vx = -8
def press_right():
    global vx
    vx = 8
def stop():
    global vx
    vx = 0
def tick():
    player.setx(player.xcor() + vx)
    if player.xcor() > 370:
        player.setx(370)
    if player.xcor() < -370:
        player.setx(-370)
    for m in meteors:
        m.sety(m.ycor() - 4)
        if m.ycor() < -300:
            reset_meteor(m)
    screen.update()
    screen.ontimer(tick, 20)

screen.listen()
screen.onkeypress(press_left, "Left")
screen.onkeypress(press_right, "Right")
screen.onkeyrelease(stop, "Left")
screen.onkeyrelease(stop, "Right")
screen.ontimer(tick, 20)
turtle.done()
运行后你该看到： 6 颗橙圆往下砸，掉出窗口后又从天上出现。这时撞上还没事。
坑： 陨石忘了 penup，满屏橙线。

第 3 / 5 步 · 讲

撞到就结束

碰到任意一颗陨石，写出「被砸中了」，游戏停。

为什么： 和第 5 课石头一样：distance < 24。alive 关掉后 tick 开头 return。
  1. 加 alive = True 和 hud
  2. tick 开头 if not alive
  3. for 循环里再判断 distance < 24
运行后你该看到： 撞上橙圆立刻出现「被砸中了」，海龟不再动。

第 3 / 5 步 · 打

撞到就结束

player.distance(m) < 24 就结束。tick 开头判断 alive。

import turtle
import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.title("第 6 课 · 躲避陨石")
screen.bgcolor("black")
screen.tracer(0)
alive = True
vx = 0
player = turtle.Turtle()
player.shape("turtle")
player.color("deepskyblue")
player.shapesize(1.4, 1.4)
player.penup()
player.goto(0, -230)
player.setheading(90)
meteors = []
for _ in range(6):
    m = turtle.Turtle()
    m.shape("circle")
    m.color("orange")
    m.penup()
    m.goto(random.randint(-360, 360), random.randint(80, 360))
    meteors.append(m)
hud = turtle.Turtle()
hud.hideturtle()
hud.penup()
hud.color("white")

def reset_meteor(m):
    m.goto(random.randint(-360, 360), random.randint(280, 420))
def press_left():
    global vx
    vx = -8
def press_right():
    global vx
    vx = 8
def stop():
    global vx
    vx = 0
def tick():
    global alive
    if not alive:
        screen.update()
        return
    player.setx(player.xcor() + vx)
    if player.xcor() > 370:
        player.setx(370)
    if player.xcor() < -370:
        player.setx(-370)
    for m in meteors:
        m.sety(m.ycor() - 4)
        if m.ycor() < -300:
            reset_meteor(m)
        if player.distance(m) < 24:
            alive = False
            hud.goto(0, 20)
            hud.write("被砸中了！", align="center", font=("Arial", 26, "bold"))
            screen.update()
            return
    screen.update()
    screen.ontimer(tick, 20)

screen.listen()
screen.onkeypress(press_left, "Left")
screen.onkeypress(press_right, "Right")
screen.onkeyrelease(stop, "Left")
screen.onkeyrelease(stop, "Right")
screen.ontimer(tick, 20)
turtle.done()
运行后你该看到： 撞上橙圆立刻出现「被砸中了」，海龟不再动。
坑： 阈值写成 80，站着也被判撞。
第 6 课 · 越来越难
结束时 return，不要再 ontimer。

第 4 / 5 步 · 讲

活得越久分越高

头顶出现存活分；分数越高陨石掉得越快。

为什么： 每 5 帧 +1 分。speed = 4 + score//15。一上来就很快没法玩。
  1. 加 score 和 frames
  2. tick 里 frames+1，每 5 帧 score+1
  3. 下落速度改成 4 + score // 15；陨石 left(4) 旋转
运行后你该看到： 头顶分数往上加。撑得越久，陨石越快。
第 6 课 · 越来越难
整数除法写在黑板上。大约 8 分钟。

第 4 / 5 步 · 打

活得越久分越高

每 5 帧加 1 分。速度 4 + score // 15。

import turtle
import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.title("第 6 课 · 躲避陨石")
screen.bgcolor("black")
screen.tracer(0)
score = 0
alive = True
vx = 0
frames = 0
player = turtle.Turtle()
player.shape("turtle")
player.color("deepskyblue")
player.shapesize(1.4, 1.4)
player.penup()
player.goto(0, -230)
player.setheading(90)
meteors = []
for _ in range(6):
    m = turtle.Turtle()
    m.shape("circle")
    m.color("orange")
    m.penup()
    m.goto(random.randint(-360, 360), random.randint(80, 360))
    meteors.append(m)
hud = turtle.Turtle()
hud.hideturtle()
hud.penup()
hud.color("white")

def reset_meteor(m):
    m.goto(random.randint(-360, 360), random.randint(280, 420))
def press_left():
    global vx
    vx = -8
def press_right():
    global vx
    vx = 8
def stop():
    global vx
    vx = 0
def draw_hud():
    hud.clear()
    hud.goto(0, 250)
    hud.write(f"存活分 {score}", align="center", font=("Arial", 18, "bold"))
def tick():
    global alive, score, frames
    if not alive:
        screen.update()
        return
    frames = frames + 1
    if frames % 5 == 0:
        score = score + 1
    player.setx(player.xcor() + vx)
    if player.xcor() > 370:
        player.setx(370)
    if player.xcor() < -370:
        player.setx(-370)
    speed = 4 + score // 15
    for m in meteors:
        m.sety(m.ycor() - speed)
        m.left(4)
        if m.ycor() < -300:
            reset_meteor(m)
        if player.distance(m) < 24:
            alive = False
            draw_hud()
            hud.goto(0, 20)
            hud.write(f"被砸中了！最终分 {score}", align="center", font=("Arial", 26, "bold"))
            screen.update()
            return
    draw_hud()
    screen.update()
    screen.ontimer(tick, 20)

screen.listen()
screen.onkeypress(press_left, "Left")
screen.onkeypress(press_right, "Right")
screen.onkeyrelease(stop, "Left")
screen.onkeyrelease(stop, "Right")
screen.ontimer(tick, 20)
turtle.done()
运行后你该看到： 头顶分数往上加。撑得越久，陨石越快。
坑： speed 写成 4 + score，几秒就没法躲。要用 // 15。
第 6 课 · 越来越难
hud 每帧重画。撞到时把最终分写出来。

第 5 / 5 步 · 讲

按 R 再来一局

结束后按 R：分数清零、陨石回天上、游戏继续。

为什么： 重开要复位所有状态：分数、帧、alive、速度、位置。漏一个就会残局。
  1. 加 restart，里面 alive=True，再调用 tick()
  2. 绑小写 r
运行后你该看到： 撞死后按 R，又能躲。头顶提示「R 重开」。
第 6 课 · 越来越难
这是过关线。让全班死一次再重开。大约 8 分钟。
