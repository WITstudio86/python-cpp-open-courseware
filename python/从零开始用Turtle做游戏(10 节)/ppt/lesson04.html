第 1 / 5 步 · 打

篮子和金币

篮子在 (0, -240)，金币在 (0, 260)。tracer(0) 后面要有 update。

import turtle
import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.title("第 4 课 · 接住金币")
screen.bgcolor("navy")
screen.tracer(0)
basket = turtle.Turtle()
basket.shape("square")
basket.color("gold")
basket.shapesize(1, 5)
basket.penup()
basket.goto(0, -240)

coin.shape("circle")
coin.color("yellow")
coin.penup()
coin.goto(0, 260)
screen.update()
turtle.done()

第 2 / 5 步 · 讲

按住篮子会走

第 2 / 5 步 · 打

按住篮子会走

press_left 把 vx 改成 -8。tick 末尾要 ontimer。松开用 onkeyrelease。

import turtle
import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.title("第 4 课 · 接住金币")
screen.bgcolor("navy")
screen.tracer(0)
vx = 0
basket = turtle.Turtle()
basket.shape("square")
basket.color("gold")
basket.shapesize(1, 5)
basket.penup()
basket.goto(0, -240)
coin = turtle.Turtle()
coin.shape("circle")
coin.color("yellow")
coin.penup()
coin.goto(0, 260)

def press_left():
    global vx
    vx = -8
def press_right():
    global vx
    vx = 8
def stop():
    global vx
    vx = 0
def tick():
    basket.setx(basket.xcor() + vx)
    if basket.xcor() > 350:
        basket.setx(350)
    if basket.xcor() < -350:
        basket.setx(-350)
    screen.update()
    screen.ontimer(tick, 20)

screen.listen()
screen.onkeypress(press_left, "Left")
screen.onkeypress(press_right, "Right")
screen.onkeyrelease(stop, "Left")
screen.onkeyrelease(stop, "Right")
screen.ontimer(tick, 20)
turtle.done()
运行后你该看到： 按住方向键篮子连续走。松开立刻停。金币还停在天上。
坑： 函数里改 vx 忘了 global vx，篮子原地不动。
第 4 课 · 让画面自己动
tick 末尾一定再约自己，否则只动一帧。

金币往下掉

金币每帧往下掉；掉出底部就回到天上随机位置。

为什么： 下落就是 sety 减小。掉出屏幕不要结束，先送回天上。

  2. tick 里 coin.sety(... - 5)
  3. y < -280 就 reset_coin()
运行后你该看到： 黄圆不断下落，掉出窗口后又从天上随机位置出现。

import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.title("第 4 课 · 接住金币")

def reset_coin():
    coin.goto(random.randint(-340, 340), 280)
def press_left():

    vx = -8
def press_right():

    vx = 8
def stop():

    vx = 0
def tick():

    if basket.xcor() > 350:

    if basket.xcor() < -350:
        basket.setx(-350)
    coin.sety(coin.ycor() - 5)
    if coin.ycor() < -280:
        reset_coin()
    screen.update()
    screen.ontimer(tick, 20)

screen.listen()
screen.onkeypress(press_left, "Left")
screen.onkeypress(press_right, "Right")
screen.onkeyrelease(stop, "Left")
screen.onkeyrelease(stop, "Right")
screen.ontimer(tick, 20)

第 4 / 5 步 · 讲

接到和漏掉

靠近篮子：分数 +1；掉出底部：漏掉 +1。头顶显示两个数字。

为什么： 接到用 distance。先判断掉出，再判断接到，顺序不要反。tick 开头 global score, missed。
  1. vx 上面加 score 和 missed，再加 hud
  2. tick 开头 global；掉出 missed+1；靠近则 score+1
运行后你该看到： 接到黄圆分数 +1 并换一枚。漏掉头顶漏掉数 +1。

第 4 / 5 步 · 打

接到和漏掉

tick 开头 global score, missed。靠近篮子 distance < 70 就加分。

import turtle
import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.title("第 4 课 · 接住金币")
screen.bgcolor("navy")
screen.tracer(0)
score = 0
missed = 0
vx = 0
basket = turtle.Turtle()
basket.shape("square")
basket.color("gold")
basket.shapesize(1, 5)
basket.penup()
basket.goto(0, -240)
coin = turtle.Turtle()
coin.shape("circle")
coin.color("yellow")
coin.penup()
coin.goto(0, 260)
hud = turtle.Turtle()
hud.hideturtle()
hud.penup()
hud.color("white")

def reset_coin():
    coin.goto(random.randint(-340, 340), 280)
def press_left():
    global vx
    vx = -8
def press_right():
    global vx
    vx = 8
def stop():
    global vx
    vx = 0
def draw_hud():
    hud.clear()
    hud.goto(0, 250)
    hud.write(f"接到 {score}    漏掉 {missed}", align="center", font=("Arial", 20, "bold"))
def tick():
    global score, missed
    basket.setx(basket.xcor() + vx)
    if basket.xcor() > 350:
        basket.setx(350)
    if basket.xcor() < -350:
        basket.setx(-350)
    coin.sety(coin.ycor() - 5)
    if coin.ycor() < -280:
        missed = missed + 1
        reset_coin()
    if coin.ycor() < -210 and basket.distance(coin) < 70:
        score = score + 1
        reset_coin()
    draw_hud()
    screen.update()
    screen.ontimer(tick, 20)

screen.listen()
screen.onkeypress(press_left, "Left")
screen.onkeypress(press_right, "Right")
screen.onkeyrelease(stop, "Left")
screen.onkeyrelease(stop, "Right")
screen.ontimer(tick, 20)
turtle.done()
运行后你该看到： 接到黄圆分数 +1 并换一枚。漏掉头顶漏掉数 +1。
坑： tick 里不加 global score，一接到就报错。
第 4 课 · 让画面自己动
distance < 70 和 y < -210 一起判断，避免天上就算接到。

第 5 / 5 步 · 讲

漏 8 个就结束

漏满 8 个写出结束语；接到越多掉得越快。

为什么： alive 关掉后 tick 开头就 return。速度用 5 + score//3，整数除法。
  1. 加 alive 和 game_over
  2. tick 开头 if not alive: return
  3. missed >= 8 就结束；下落速度改成 5 + score//3
运行后你该看到： 漏满 8 个出现「金币漏光了」。接到越多，后面掉得越快。

第 5 / 5 步 · 打

漏 8 个就结束

漏满 8 个 game_over 并 return。下落改成 5 + score // 3。

import turtle
import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.title("第 4 课 · 接住金币")
screen.bgcolor("navy")
screen.tracer(0)
score = 0
missed = 0
alive = True
vx = 0
basket = turtle.Turtle()
basket.shape("square")
basket.color("gold")
basket.shapesize(1, 5)
basket.penup()
basket.goto(0, -240)
coin = turtle.Turtle()
coin.shape("circle")
coin.color("yellow")
coin.penup()
coin.goto(0, 260)
hud = turtle.Turtle()
hud.hideturtle()
hud.penup()
hud.color("white")

def reset_coin():
    coin.goto(random.randint(-340, 340), 280)
def press_left():
    global vx
    vx = -8
def press_right():
    global vx
    vx = 8
def stop():
    global vx
    vx = 0
def draw_hud():
    hud.clear()
    hud.goto(0, 250)
    hud.write(f"接到 {score}    漏掉 {missed}/8", align="center", font=("Arial", 20, "bold"))
def game_over(msg):
    global alive
    alive = False
    hud.goto(0, 20)
    hud.write(msg, align="center", font=("Arial", 28, "bold"))
def tick():
    global score, missed
    if not alive:
        screen.update()
        return
    basket.setx(basket.xcor() + vx)
    if basket.xcor() > 350:
        basket.setx(350)
    if basket.xcor() < -350:
        basket.setx(-350)
    coin.sety(coin.ycor() - (5 + score // 3))
    if coin.ycor() < -280:
        missed = missed + 1
        reset_coin()
        if missed >= 8:
            draw_hud()
            game_over("金币漏光了……再开一局试试")
            screen.update()
            return
    if coin.ycor() < -210 and basket.distance(coin) < 70:
        score = score + 1
        reset_coin()
    draw_hud()
    screen.update()
    screen.ontimer(tick, 20)

screen.listen()
screen.onkeypress(press_left, "Left")
screen.onkeypress(press_right, "Right")
screen.onkeyrelease(stop, "Left")
screen.onkeyrelease(stop, "Right")
screen.ontimer(tick, 20)
turtle.done()
运行后你该看到： 漏满 8 个出现「金币漏光了」。接到越多，后面掉得越快。
坑： 结束之后还 ontimer，字会被下一帧清掉。game_over 后要 return。

hud 改成 漏掉 n/8。game_over 后一定 return。

🐞 改错时刻

这三处，课上必踩

❌ tick 里忘了 ontimer

✅ 末尾再约一次
否则只动一帧

❌ 忘了 global vx

✅ 在函数里写 global vx
否则改的是局部变量

❌ tracer(0) 不 update

✅ 每帧 update()
否则全黑
