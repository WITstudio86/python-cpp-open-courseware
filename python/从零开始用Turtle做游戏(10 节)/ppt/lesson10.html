第 1 / 5 步 · 讲

菜单和空格开始

先看到菜单。按空格，大字消失，进入游戏状态。

为什么： state 是字符串。tick 开头如果不是 play 也要继续 ontimer，否则循环停住，再按空格没人接。
  1. state = "menu"，写 show_menu / start_game / fire_or_start
  2. tick 里 if state != "play" 仍要 ontimer
  3. 空格绑 fire_or_start
运行后你该看到： 打开是「太空大战 / 按空格开始」。按空格字消失，底部出现三角。
第 10 课 · 完整作品
这就是热身。让每人按空格进出一次。大约 8 分钟。

第 1 / 5 步 · 打

菜单和空格开始

state 先是 menu。空格调用 start_game。tick 不是 play 也要再约自己。

import turtle
import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.title("第 10 课 · 太空大战")
screen.bgcolor("black")
screen.tracer(0)
state = "menu"
player = turtle.Turtle()
player.shape("triangle")
player.color("deepskyblue")
player.penup()
player.goto(0, -240)
player.setheading(90)
hud = turtle.Turtle()
hud.hideturtle()
hud.penup()
hud.color("white")

def show_menu():
    global state
    state = "menu"
    hud.clear()
    hud.goto(0, 80)
    hud.write("太空大战", align="center", font=("Arial", 42, "bold"))
    hud.goto(0, -20)
    hud.write("按空格开始", align="center", font=("Arial", 22, "bold"))
    screen.update()
def start_game():
    global state
    state = "play"
    hud.clear()
    player.goto(0, -240)
    player.showturtle()
def fire_or_start():
    if state == "menu":
        start_game()
def tick():
    if state != "play":
        screen.update()
        screen.ontimer(tick, 20)
        return
    screen.update()
    screen.ontimer(tick, 20)

show_menu()
screen.listen()
screen.onkeypress(fire_or_start, "space")
screen.ontimer(tick, 20)
turtle.done()
运行后你该看到： 打开是「太空大战 / 按空格开始」。按空格字消失，底部出现三角。
坑： 菜单里 tick 不 ontimer，按空格开始后画面不再刷新。
第 10 课 · 完整作品
菜单时也要 ontimer。空格先只负责开始。

第 2 / 5 步 · 打

移动和射击

空格：菜单开始，游戏射击。子弹 for shot in bullets[:] 再删除。

import turtle
import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.title("第 10 课 · 太空大战")
screen.bgcolor("black")
screen.tracer(0)
state = "menu"
vx = 0
cool = 0
player = turtle.Turtle()
player.shape("triangle")
player.color("deepskyblue")
player.penup()
player.goto(0, -240)
player.setheading(90)
bullets = []
hud = turtle.Turtle()
hud.hideturtle()
hud.penup()
hud.color("white")

def show_menu():
    global state
    state = "menu"
    hud.clear()
    hud.goto(0, 80)
    hud.write("太空大战", align="center", font=("Arial", 42, "bold"))
    hud.goto(0, -20)
    hud.write("按空格开始", align="center", font=("Arial", 22, "bold"))
    screen.update()
def start_game():
    global state
    state = "play"
    hud.clear()
    player.goto(0, -240)
    player.showturtle()
def make_bullet():
    shot = turtle.Turtle()
    shot.shape("square")
    shot.shapesize(0.3, 0.7)
    shot.color("yellow")
    shot.penup()
    shot.goto(player.xcor(), player.ycor() + 18)
    shot.setheading(90)
    bullets.append(shot)
def press_left():
    global vx
    if state == "play":
        vx = -8
def press_right():
    global vx
    if state == "play":
        vx = 8
def stop():
    global vx
    vx = 0
def fire_or_start():
    global cool
    if state == "menu":
        start_game()
        return
    if state == "play" and cool <= 0 and len(bullets) < 5:
        make_bullet()
        cool = 8
def tick():
    global cool
    if state != "play":
        screen.update()
        screen.ontimer(tick, 20)
        return
    if cool > 0:
        cool = cool - 1
    player.setx(player.xcor() + vx)
    if player.xcor() > 370:
        player.setx(370)
    if player.xcor() < -370:
        player.setx(-370)
    for shot in bullets[:]:
        shot.forward(14)
        if shot.ycor() > 310:
            shot.hideturtle()
            bullets.remove(shot)
    screen.update()
    screen.ontimer(tick, 20)

show_menu()
screen.listen()
screen.onkeypress(press_left, "Left")
screen.onkeypress(press_right, "Right")
screen.onkeyrelease(stop, "Left")
screen.onkeyrelease(stop, "Right")
screen.onkeypress(fire_or_start, "space")
screen.ontimer(tick, 20)
turtle.done()
运行后你该看到： 开始后能左右走、空格射击。天上还没有敌人。
坑： 空格不判断 state，菜单里也会射出子弹。

press_left 里 if state == "play"。子弹一定 penup。

import random

screen = turtle.Screen()
screen.setup(800, 600)
screen.title("第 10 课 · 太空大战")

screen.tracer(0)
state = "menu"

enemies = []
hud = turtle.Turtle()
hud.hideturtle()
hud.penup()
hud.color("white")

def make_enemy():
    e = turtle.Turtle()
    e.shape("turtle")
    e.color("red")
    e.penup()
    e.goto(random.randint(-340, 340), random.randint(140, 280))
    e.setheading(270)
    enemies.append(e)
def make_bullet():
    shot = turtle.Turtle()
    shot.shape("square")
    shot.shapesize(0.3, 0.7)
    shot.color("yellow")
    shot.penup()
    shot.goto(player.xcor(), player.ycor() + 18)
    shot.setheading(90)
    bullets.append(shot)
def hide_all():
    for shot in bullets:
        shot.hideturtle()
    bullets.clear()
    for e in enemies:
        e.hideturtle()
    enemies.clear()
def show_menu():
    global state
    state = "menu"
    hide_all()
    hud.clear()
    hud.goto(0, 80)
    hud.write("太空大战", align="center", font=("Arial", 42, "bold"))

    screen.update()
def start_game():
    global state, score
    state = "play"
    score = 0
    hide_all()
    player.goto(0, -240)
    for _ in range(4):
        make_enemy()
    hud.clear()
def press_left():
    global vx
    if state == "play":
        vx = -8
def press_right():
    global vx
    if state == "play":
        vx = 8
def stop():
    global vx
    vx = 0
def fire_or_start():
    global cool

        return

def draw_hud():

    hud.goto(0, 250)
    hud.write(f"得分 {score}", align="center", font=("Arial", 18, "bold"))

    if state != "play":
        screen.update()
        screen.ontimer(tick, 20)

        e.sety(e.ycor() - 2)
        e.left(3)
        for shot in bullets[:]:

                score = score + 10
                shot.hideturtle()
                bullets.remove(shot)
                e.hideturtle()
                enemies.remove(e)
