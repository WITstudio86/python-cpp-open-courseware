"""
第3节：键盘是我的遥控器
========================
学习目标：使用键盘方向键控制屏幕上的矩形移动
核心知识点：
  1. pygame.key.get_pressed() — 获取键盘按键状态，实现平滑连续移动
  2. 边界检测 — 确保矩形不会移出窗口
  3. 游戏主循环结构
"""

import pygame
import sys

# ==================== 初始化 Pygame ====================
pygame.init()

# ==================== 窗口设置 ====================
WINDOW_WIDTH = 800   # 窗口宽度
WINDOW_HEIGHT = 600  # 窗口高度
WINDOW_TITLE = "第3节：键盘是我的遥控器"

screen = pygame.display.set_mode((WINDOW_WIDTH, WINDOW_HEIGHT))
pygame.display.set_caption(WINDOW_TITLE)

# ==================== 颜色定义（RGB） ====================
# 注意：在计算机中，颜色由红(R)、绿(G)、蓝(B)三个分量组成，每个分量取值范围 0~255
WHITE = (255, 255, 255)      # 白色 - 背景色
BLACK = (0, 0, 0)            # 黑色
RED = (255, 85, 85)          # 柔和红色
GREEN = (85, 200, 85)        # 柔和绿色
BLUE = (85, 130, 255)        # 柔和蓝色
ORANGE = (255, 165, 65)      # 橙色
PURPLE = (160, 80, 220)      # 紫色
TEAL = (65, 195, 195)        # 青色

BACKGROUND_COLOR = (30, 30, 50)  # 深蓝灰色背景，让画面更有质感

# ==================== 矩形（玩家角色）设置 ====================
RECT_WIDTH = 80    # 矩形宽度
RECT_HEIGHT = 60   # 矩形高度

# 矩形的初始位置：放在窗口正中央
rect_x = (WINDOW_WIDTH - RECT_WIDTH) // 2
rect_y = (WINDOW_HEIGHT - RECT_HEIGHT) // 2

# 矩形移动速度（像素/帧）
# 数值越大移动越快；5 是一个适中的速度，方便精准控制
RECT_SPEED = 5

# 矩形颜色：使用紫色，鲜艳醒目
RECT_COLOR = PURPLE


def check_boundary(x, y, width, height):
    """
    边界检测函数：确保矩形不会移出窗口

    参数：
        x: 矩形当前的 x 坐标（左上角）
        y: 矩形当前的 y 坐标（左上角）
        width: 矩形的宽度
        height: 矩形的高度

    返回：
        (new_x, new_y): 修正后的坐标，保证矩形完全在窗口内

    原理说明：
        矩形的左上角坐标不能为负数（不能跑出左/上边界），
        矩形的右下角坐标（x+宽, y+高）不能超出窗口的宽和高（不能跑出右/下边界）。
    """
    # 左边界：x 不能小于 0
    if x < 0:
        x = 0
    # 右边界：(x + 宽度) 不能超过窗口宽度
    if x + width > WINDOW_WIDTH:
        x = WINDOW_WIDTH - width
    # 上边界：y 不能小于 0
    if y < 0:
        y = 0
    # 下边界：(y + 高度) 不能超过窗口高度
    if y + height > WINDOW_HEIGHT:
        y = WINDOW_HEIGHT - height

    return x, y


def draw_scene(screen, rect_x, rect_y):
    """
    绘制整个游戏画面

    参数：
        screen: Pygame 窗口表面
        rect_x: 矩形的 x 坐标
        rect_y: 矩形的 y 坐标
    """
    # 1. 填充背景色（这一步会覆盖上一帧的所有内容）
    screen.fill(BACKGROUND_COLOR)

    # 2. 绘制装饰性网格线（让画面更有层次感，纯视觉效果，不影响游戏逻辑）
    grid_color = (50, 50, 70)
    for x in range(0, WINDOW_WIDTH, 50):
        pygame.draw.line(screen, grid_color, (x, 0), (x, WINDOW_HEIGHT), 1)
    for y in range(0, WINDOW_HEIGHT, 50):
        pygame.draw.line(screen, grid_color, (0, y), (WINDOW_WIDTH, y), 1)

    # 3. 绘制玩家控制的矩形
    #    先画一个稍大的阴影矩形，再画主矩形，形成立体效果
    shadow_offset = 4
    shadow_rect = pygame.Rect(rect_x + shadow_offset, rect_y + shadow_offset, RECT_WIDTH, RECT_HEIGHT)
    pygame.draw.rect(screen, (10, 10, 30), shadow_rect)  # 阴影

    main_rect = pygame.Rect(rect_x, rect_y, RECT_WIDTH, RECT_HEIGHT)
    pygame.draw.rect(screen, RECT_COLOR, main_rect)       # 主体

    # 4. 绘制矩形的高光边框（让矩形更好看）
    border_color = (210, 140, 255)  # 浅紫色边框
    pygame.draw.rect(screen, border_color, main_rect, 3)  # 线宽为 3 像素

    # 5. 绘制窗口四角的边界指示标记（提示玩家边界位置）
    corner_size = 30
    corner_color = (100, 100, 150)
    # 左上角 L 形标记
    pygame.draw.line(screen, corner_color, (0, 0), (corner_size, 0), 2)
    pygame.draw.line(screen, corner_color, (0, 0), (0, corner_size), 2)
    # 右上角 L 形标记
    pygame.draw.line(screen, corner_color, (WINDOW_WIDTH - corner_size, 0), (WINDOW_WIDTH, 0), 2)
    pygame.draw.line(screen, corner_color, (WINDOW_WIDTH, 0), (WINDOW_WIDTH, corner_size), 2)
    # 左下角 L 形标记
    pygame.draw.line(screen, corner_color, (0, WINDOW_HEIGHT - corner_size), (0, WINDOW_HEIGHT), 2)
    pygame.draw.line(screen, corner_color, (0, WINDOW_HEIGHT), (corner_size, WINDOW_HEIGHT), 2)
    # 右下角 L 形标记
    pygame.draw.line(screen, corner_color, (WINDOW_WIDTH - corner_size, WINDOW_HEIGHT), (WINDOW_WIDTH, WINDOW_HEIGHT), 2)
    pygame.draw.line(screen, corner_color, (WINDOW_WIDTH, WINDOW_HEIGHT - corner_size), (WINDOW_WIDTH, WINDOW_HEIGHT), 2)

    # 6. 显示提示文字（按 ESC 退出）
    font = pygame.font.SysFont("SimHei", 18)
    hint_text = font.render("方向键移动 | ESC 退出", True, (180, 180, 200))
    screen.blit(hint_text, (15, 15))


def main():
    """
    游戏主循环：程序的核心，每一帧都在这里处理输入、更新状态、绘制画面
    """
    # 将矩形的坐标声明为全局变量，方便在函数内修改
    global rect_x, rect_y

    # 创建时钟对象，用于控制帧率
    clock = pygame.time.Clock()

    # 游戏运行标志：当它为 False 时，主循环结束，程序退出
    running = True

    # ==================== 主循环 ====================
    while running:

        # ---------- 第 1 步：处理事件 ----------
        # 遍历 Pygame 事件队列中的所有事件
        for event in pygame.event.get():
            # 如果用户点击了窗口的关闭按钮
            if event.type == pygame.QUIT:
                running = False
            # 如果用户按下了 ESC 键，也退出游戏
            if event.type == pygame.KEYDOWN:
                if event.key == pygame.K_ESCAPE:
                    running = False

        # ---------- 第 2 步：获取键盘输入并更新位置 ----------
        # pygame.key.get_pressed() 返回一个列表，记录当前所有按键的按下状态
        # 与 KEYDOWN 事件不同，get_pressed() 是"实时"的——按住不放就会持续返回 True
        # 这就是实现"平滑连续移动"的关键！
        keys = pygame.key.get_pressed()

        # 方向键 ← 左移：减小 x 坐标
        if keys[pygame.K_LEFT]:
            rect_x -= RECT_SPEED

        # 方向键 → 右移：增大 x 坐标
        if keys[pygame.K_RIGHT]:
            rect_x += RECT_SPEED

        # 方向键 ↑ 上移：减小 y 坐标（注意：屏幕坐标系中 y 轴向下！）
        if keys[pygame.K_UP]:
            rect_y -= RECT_SPEED

        # 方向键 ↓ 下移：增大 y 坐标
        if keys[pygame.K_DOWN]:
            rect_y += RECT_SPEED

        # ---------- 第 3 步：边界检测 ----------
        # 移动之后，检查矩形是否超出了窗口边界
        # 如果超出，自动修正到边界位置，让矩形始终可见
        rect_x, rect_y = check_boundary(rect_x, rect_y, RECT_WIDTH, RECT_HEIGHT)

        # ---------- 第 4 步：绘制画面 ----------
        draw_scene(screen, rect_x, rect_y)

        # ---------- 第 5 步：刷新屏幕 ----------
        # pygame.display.flip() 将后台绘制的内容一次性显示到屏幕上
        pygame.display.flip()

        # ---------- 第 6 步：控制帧率 ----------
        # clock.tick(60) 确保每秒最多运行 60 帧
        # 这样在不同性能的电脑上，游戏速度都是一致的
        clock.tick(60)

    # ==================== 退出程序 ====================
    pygame.quit()
    sys.exit()


# ==================== 程序入口 ====================
# 当直接运行本文件时（非 import），执行 main() 函数启动游戏
if __name__ == "__main__":
    main()
