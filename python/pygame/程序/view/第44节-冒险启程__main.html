# ==================== 经验曲线与升级 ====================
def exp_to_next_level(level):
    """计算升至下一级所需经验值（线性曲线）"""
    return level * 50

def level_up():
    """升级：属性全面增长，满血满蓝"""
    global level_up_anim_timer
    player["level"] += 1
    player["max_hp"] += 20
    player["max_mp"] += 10
    player["atk"] += 5
    player["def"] += 3
    # 升级福利：满血满蓝！
    player["hp"] = player["max_hp"]
    player["mp"] = player["max_mp"]
    level_up_anim_timer = 120  # 升级动画持续120帧（2秒）
    add_message(f"升级！达到 {player['level']} 级！属性全面增长！", C["gold"])

def gain_exp(amount):
    """获得经验值，自动检测升级（while循环处理连续升级）"""
    player["exp"] += amount
    needed = exp_to_next_level(player["level"])
    while player["exp"] >= needed:
        player["exp"] -= needed
        level_up()
        needed = exp_to_next_level(player["level"])

def add_message(text, color=None):
    """添加一条消息到屏幕显示"""
    if color is None:
        color = C["white"]
    message_lines.append({"text": text, "color": color, "timer": 180})
    # 最多保留5条消息
    if len(message_lines) > 5:
        message_lines.pop(0)

# ==================== 物品使用 ====================
def use_item(index, in_battle=False):
    """使用背包中的物品
    Args:
        index: 物品在inventory中的索引
        in_battle: 是否在战斗中使用
    Returns:
        使用是否成功
    """
    global atk_boost
    if index < 0 or index >= len(inventory):
        add_message("背包格子是空的！", C["gray"])
        return False

    item_name = inventory[index]
    item_info = ITEMS[item_name]

    if item_info["effect"] == "hp":
        old_hp = player["hp"]
        player["hp"] = min(player["hp"] + item_info["value"], player["max_hp"])
        healed = player["hp"] - old_hp
        add_message(f"使用 {item_name}，恢复了 {healed} 点生命！", C["green"])

    elif item_info["effect"] == "mp":
        old_mp = player["mp"]
        player["mp"] = min(player["mp"] + item_info["value"], player["max_mp"])
        restored = player["mp"] - old_mp
        add_message(f"使用 {item_name}，恢复了 {restored} 点魔法！", C["blue_light"])

    elif item_info["effect"] == "atk":
        if in_battle:
            atk_boost += item_info["value"]
            add_message(f"使用 {item_name}，本次战斗攻击力+{item_info['value']}！", C["orange"])
        else:
            add_message(f"{item_name} 只能在战斗中使用！", C["gray"])
            return False

    inventory.pop(index)  # 使用后从背包移除
    return True

# ==================== 战斗系统 ====================
def init_battle():
    """初始化战斗：选择当前要挑战的怪物"""
    global current_monster, battle_log, atk_boost, is_defending, game_state
    if defeated_count >= len(MONSTERS):
        game_state = "victory"
        add_message("你已经击败了所有怪物！和平降临大地！", C["gold"])
        return

    # 复制怪物数据（避免修改原模板）
    m = MONSTERS[defeated_count]
    current_monster = {
        "name": m["name"], "hp": m["hp"], "max_hp": m["max_hp"],
        "atk": m["atk"], "def": m["def"], "exp": m["exp"],
        "gold": m["gold"], "color": m["color"], "type": m["type"]
    }
    battle_log = []
    atk_boost = 0
    is_defending = False
    game_state = "battle"
    add_message(f"遭遇了 {current_monster['name']}！准备战斗！", C["red"])

def do_player_attack():
    """玩家攻击"""
    global is_defending
    is_defending = False
    damage = max(1, (player["atk"] + atk_boost) - current_monster["def"])
    # 10% 暴击率
    is_crit = random.random() < 0.1
    if is_crit:
        damage = int(damage * 1.8)
    current_monster["hp"] -= damage
    crit_text = "暴击！" if is_crit else ""
    add_message(f"你攻击了 {current_monster['name']}，造成 {damage} 点伤害！{crit_text}", C["blue_light"])

def do_monster_attack():
    """怪物攻击"""
    if is_defending:
        # 防御状态：防御力翻倍
        damage = max(1, current_monster["atk"] - player["def"] * 2)
    else:
        damage = max(1, current_monster["atk"] - player["def"])
    player["hp"] -= damage
    def_text = "（你正在防御！）" if is_defending else ""
    add_message(f"{current_monster['name']} 攻击了你，造成 {damage} 点伤害！{def_text}", C["red"])

def check_battle_end():
    """检查战斗是否结束"""
    global game_state, defeated_count
    if current_monster["hp"] <= 0:
        # 战斗胜利
        add_message(f"击败了 {current_monster['name']}！", C["green"])
        exp_gain = current_monster["exp"]
        gold_gain = current_monster["gold"]
        gain_exp(exp_gain)
        player["gold"] += gold_gain
        add_message(f"获得经验 {exp_gain}，金币 {gold_gain}！", C["gold"])
        defeated_count += 1

        # 50%概率掉落物品
        if random.random() < 0.5:
            item = random.choice(["生命药水", "魔法药水", "攻击卷轴"])
            inventory.append(item)
            add_message(f"获得了 [{item}]！", C["purple"])

        game_state = "battle_win"
        return True

    if player["hp"] <= 0:
        player["hp"] = 0
        add_message("你被击败了...", C["red_dark"])
        game_state = "game_over"
        return True

    return False

def player_flee():
    """尝试逃跑"""
    global game_state
    if random.random() < 0.5:
        add_message("逃跑成功！", C["green"])
        game_state = "menu"
    else:
        add_message("逃跑失败！", C["red"])
        do_monster_attack()
        check_battle_end()

# ==================== 存档系统 ====================
SAVE_FILE = "rpg_save.json"

def save_game():
    """将游戏数据保存为JSON文件"""
    data = {
        "player": player,
        "inventory": inventory,
        "defeated_count": defeated_count,
    }
    try:
        with open(SAVE_FILE, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        add_message("存档成功！游戏进度已保存。", C["green"])
    except Exception as e:
        add_message(f"存档失败：{e}", C["red"])

def load_game():
    """从JSON文件读取游戏数据"""
    global player, inventory, defeated_count
    if not os.path.exists(SAVE_FILE):
        add_message("未找到存档文件！", C["gray"])
        return False
    try:
        with open(SAVE_FILE, "r", encoding="utf-8") as f:
            data = json.load(f)
        player = data["player"]
        inventory = data["inventory"]
        defeated_count = data.get("defeated_count", 0)
        add_message("读档成功！欢迎回来！", C["green"])
        return True
    except Exception as e:
        add_message(f"读档失败：{e}", C["red"])
        return False


# ==================== 绘图函数 ====================

def draw_text(surf, text, x, y, font, color, center=False, shadow=False):
    """绘制文字，支持居中和阴影"""
    if shadow:
        shadow_surf = font.render(text, True, (0, 0, 0))
        sr = shadow_surf.get_rect()
        if center:
            sr.center = (x + 2, y + 2)
        else:
            sr.topleft = (x + 2, y + 2)
        surf.blit(shadow_surf, sr)
    text_surf = font.render(text, True, color)
    tr = text_surf.get_rect()
    if center:
        tr.center = (x, y)
    else:
        tr.topleft = (x, y)
    surf.blit(text_surf, tr)

def draw_bar(surf, x, y, w, h, current, max_val, fg_color, bg_color=None):
    """绘制状态条（血条/蓝条/经验条）
    Args:
        surf: 目标Surface
        x, y: 左上角坐标
        w, h: 宽高
        current: 当前值
        max_val: 最大值
        fg_color: 前景色
        bg_color: 背景色（可选）
    """
    if bg_color is None:
        bg_color = C["gray_dark"]
    # 背景
    pygame.draw.rect(surf, bg_color, (x, y, w, h), border_radius=4)
    # 填充
    if max_val > 0:
        ratio = max(0, min(1, current / max_val))
        fill_w = int(w * ratio)
        if fill_w > 0:
            pygame.draw.rect(surf, fg_color, (x, y, fill_w, h), border_radius=4)
    # 边框
    pygame.draw.rect(surf, C["gold_dim"], (x, y, w, h), 2, border_radius=4)

def draw_button(surf, rect, text, font, text_color, bg_color, hover=False, border_color=None):
    """绘制按钮
    Args:
        rect: pygame.Rect
        hover: 是否鼠标悬停（悬停时颜色更亮）
    """
    if border_color is None:
        border_color = C["gold"] if hover else C["gold_dim"]
    if hover:
        bg_color = tuple(min(255, c + 30) for c in bg_color)
    pygame.draw.rect(surf, bg_color, rect, border_radius=10)
    pygame.draw.rect(surf, border_color, rect, 3, border_radius=10)
    draw_text(surf, text, rect.centerx, rect.centery, font, text_color, center=True)

def draw_panel(surf, x, y, w, h, alpha=200):
    """绘制半透明面板"""
    panel = pygame.Surface((w, h), pygame.SRCALPHA)
    panel.fill((*C["bg_panel"], alpha))
    surf.blit(panel, (x, y))
    pygame.draw.rect(surf, C["gold_dim"], (x, y, w, h), 2, border_radius=8)

def draw_monster_sprite(surf, x, y, monster):
    """根据怪物类型绘制不同的怪物图形"""
    mtype = monster.get("type", "slime")
    color = monster.get("color", (100, 200, 100))
    cx, cy = x, y

    if mtype == "slime":
        # 史莱姆 — 半圆形绿色软泥怪
        body_rect = pygame.Rect(cx - 40, cy - 20, 80, 55)
        pygame.draw.ellipse(surf, color, body_rect)
        pygame.draw.ellipse(surf, tuple(max(0, c - 40) for c in color), body_rect, 3)
        # 阴影/高光
        pygame.draw.ellipse(surf, tuple(min(255, c + 50) for c in color), (cx - 25, cy - 30, 30, 20))
        # 眼睛
        pygame.draw.circle(surf, C["white"], (cx - 14, cy - 8), 8)
        pygame.draw.circle(surf, C["white"], (cx + 14, cy - 8), 8)
        pygame.draw.circle(surf, (20, 20, 20), (cx - 12, cy - 8), 4)
        pygame.draw.circle(surf, (20, 20, 20), (cx + 12, cy - 8), 4)
        # 微笑
        pygame.draw.arc(surf, (20, 20, 20), (cx - 10, cy - 5, 20, 10), 0, math.pi, 2)

    elif mtype == "skeleton":
        # 骷髅兵 — 白色骨架
        # 头颅
        pygame.draw.circle(surf, color, (cx, cy - 20), 20)
        pygame.draw.circle(surf, (140, 130, 110), (cx, cy - 20), 20, 2)
        # 眼睛洞
        pygame.draw.ellipse(surf, (30, 25, 20), (cx - 12, cy - 28, 8, 10))
        pygame.draw.ellipse(surf, (30, 25, 20), (cx + 4, cy - 28, 8, 10))
        # 鼻子洞
        pygame.draw.polygon(surf, (30, 25, 20), [(cx - 2, cy - 18), (cx + 2, cy - 18), (cx, cy - 14)])
        # 牙齿
        for i in range(4):
            pygame.draw.line(surf, (30, 25, 20), (cx - 8 + i * 5, cy - 10), (cx - 8 + i * 5, cy - 4), 2)
        # 脊柱
        pygame.draw.line(surf, color, (cx, cy), (cx, cy + 40), 5)
        # 肋骨
        for i in range(3):
            pygame.draw.line(surf, color, (cx, cy + 8 + i * 12), (cx - 18, cy + 14 + i * 12), 3)
            pygame.draw.line(surf, color, (cx, cy + 8 + i * 12), (cx + 18, cy + 14 + i * 12), 3)
        # 手臂（骨头）
        pygame.draw.line(surf, color, (cx - 18, cy + 10), (cx - 35, cy + 30), 4)
        pygame.draw.line(surf, color, (cx + 18, cy + 10), (cx + 35, cy + 30), 4)
        # 武器（斧头）
        pygame.draw.rect(surf, C["gray"], (cx + 30, cy + 20, 4, 25), border_radius=1)
        pygame.draw.polygon(surf, C["gray_dark"], [(cx + 18, cy + 24), (cx + 48, cy + 20), (cx + 44, cy + 35)])
        # 腿骨
        pygame.draw.line(surf, color, (cx - 8, cy + 40), (cx - 14, cy + 70), 5)
        pygame.draw.line(surf, color, (cx + 8, cy + 40), (cx + 14, cy + 70), 5)

    elif mtype == "dragon":
        # 恶龙 — 红色巨龙
        # 身体
        body_rect = pygame.Rect(cx - 45, cy - 15, 90, 55)
        pygame.draw.ellipse(surf, color, body_rect)
        pygame.draw.ellipse(surf, tuple(max(0, c - 40) for c in color), body_rect, 3)
        # 肚皮
        belly_color = tuple(min(255, c + 40) for c in color)
        pygame.draw.ellipse(surf, belly_color, (cx - 25, cy + 5, 50, 25))

        # 头部
        pygame.draw.circle(surf, color, (cx + 40, cy - 30), 22)

# ==================== 主循环 ====================
def main():
    global game_state, level_up_anim_timer, message_lines, message_timer, atk_boost, is_defending

    running = True

    while running:
        dt = clock.tick(60)  # 60 FPS

        # 更新消息计时器
        for msg in message_lines[:]:
            msg["timer"] -= 1
            if msg["timer"] <= 0:
                message_lines.remove(msg)

        # 更新升级动画
        if level_up_anim_timer > 0:
            level_up_anim_timer -= 1

        # ============ 事件处理 ============
        mouse_pos = pygame.mouse.get_pos()
        mouse_clicked = False

        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False
                break

            if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
                mouse_clicked = True

            if event.type == pygame.KEYDOWN:
                # 全局快捷键
                if event.key == pygame.K_s:
                    save_game()
                elif event.key == pygame.K_l:
                    load_game()

                # 状态相关快捷键
                if game_state == "menu":
                    if event.key == pygame.K_e or event.key == pygame.K_SPACE:
                        init_battle()
                    elif event.key == pygame.K_i:
                        game_state = "inventory"

                elif game_state == "battle":
                    if event.key == pygame.K_1 or event.key == pygame.K_a:
                        # 攻击
                        do_player_attack()
                        if not check_battle_end() and game_state == "battle":
                            do_monster_attack()
                            check_battle_end()
                    elif event.key == pygame.K_2 or event.key == pygame.K_d:
                        # 防御
                        is_defending = True
                        add_message("你摆出了防御姿态！（本回合防御力翻倍）", C["blue"])
                        do_monster_attack()
                        check_battle_end()
                    elif event.key == pygame.K_3 or event.key == pygame.K_i:
                        # 打开背包使用物品
                        game_state = "inventory"
                    elif event.key == pygame.K_4 or event.key == pygame.K_f:
                        player_flee()

                elif game_state == "battle_win":
                    if event.key == pygame.K_SPACE or event.key == pygame.K_RETURN:
                        game_state = "menu"

                elif game_state == "inventory":
                    if event.key == pygame.K_ESCAPE:
                        # 如果在战斗中打开的背包，返回战斗；否则返回菜单
                        game_state = "battle" if current_monster and current_monster["hp"] > 0 else "menu"

                elif game_state == "game_over":
                    if event.key == pygame.K_r:
                        # 重新开始
                        reset_game()

                elif game_state == "victory":
                    if event.key == pygame.K_ESCAPE:
                        running = False

        if not running:
            break

        # ============ 鼠标点击处理 ============
        if mouse_clicked:
            if game_state == "menu":
                # 探索按钮
                explore_rect = pygame.Rect(310, 350, 180, 55)
                if explore_rect.collidepoint(mouse_pos):
                    init_battle()
                # 背包按钮
                inv_rect = pygame.Rect(310, 420, 180, 55)
                if inv_rect.collidepoint(mouse_pos):
                    game_state = "inventory"
                # 存档按钮
                save_rect = pygame.Rect(310, 490, 180, 55)
                if save_rect.collidepoint(mouse_pos):
                    save_game()
                # 读档按钮
                load_rect = pygame.Rect(510, 350, 100, 55)
                if load_rect.collidepoint(mouse_pos):
                    load_game()

            elif game_state == "battle":
                btn_w, btn_h = 120, 40
                btn_y = 540
                # 攻击
                if pygame.Rect(40, btn_y, btn_w, btn_h).collidepoint(mouse_pos):
                    do_player_attack()
                    if not check_battle_end() and game_state == "battle":
                        do_monster_attack()
                        check_battle_end()
                # 防御
                elif pygame.Rect(170, btn_y, btn_w, btn_h).collidepoint(mouse_pos):
                    is_defending = True
                    add_message("你摆出了防御姿态！", C["blue"])
                    do_monster_attack()
                    check_battle_end()
                # 道具
                elif pygame.Rect(300, btn_y, btn_w, btn_h).collidepoint(mouse_pos):
                    game_state = "inventory"
                # 逃跑
                elif pygame.Rect(430, btn_y, btn_w, btn_h).collidepoint(mouse_pos):
                    player_flee()

            elif game_state == "battle_win":
                cont_rect = pygame.Rect(300, 360, 200, 50)
                if cont_rect.collidepoint(mouse_pos):
                    game_state = "menu"

            elif game_state == "inventory":
                # 物品使用
                cols = 3
                cell_w, cell_h = 180, 120
                gap_x, gap_y = 20, 20
                start_x = (W - (cols * cell_w + (cols - 1) * gap_x)) // 2
                start_y = 110
                for i, item_name in enumerate(inventory):
                    col = i % cols
                    row = i // cols
                    cx = start_x + col * (cell_w + gap_x)
                    cy = start_y + row * (cell_h + gap_y)
                    use_btn_x = cx + cell_w // 2 - 35
                    use_btn_y = cy + cell_h - 36
                    use_rect = pygame.Rect(use_btn_x, use_btn_y, 70, 28)
                    if use_rect.collidepoint(mouse_pos):
                        # 判断是否在战斗中使用
                        in_battle = (game_state == "battle" or
                                    (current_monster and current_monster["hp"] > 0))
                        use_item(i, in_battle=in_battle)
                        break
                # 返回按钮
                back_rect = pygame.Rect(350, 530, 100, 40)
                if back_rect.collidepoint(mouse_pos):
                    if current_monster and current_monster["hp"] > 0 and player["hp"] > 0:
                        game_state = "battle"
                    else:
                        game_state = "menu"

            elif game_state == "game_over":
                restart_rect = pygame.Rect(300, 320, 200, 50)
                if restart_rect.collidepoint(mouse_pos):
                    reset_game()

        # ============ 渲染 ============
        if game_state == "menu":
            draw_menu_screen(screen)

        elif game_state == "battle":
            draw_battle_screen(screen)

        elif game_state == "battle_win":
            draw_battle_win_screen(screen)

        elif game_state == "inventory":
            draw_inventory_screen(screen)

        elif game_state == "game_over":
            draw_game_over_screen(screen)

        elif game_state == "victory":
            draw_victory_screen(screen)

        # 升级动画叠加
        if level_up_anim_timer > 0:
            alpha = int(level_up_anim_timer / 120 * 150)
            draw_level_up_overlay(screen, alpha)

        # 消息显示
        draw_messages(screen)

        pygame.display.flip()

    pygame.quit()


def reset_game():
    """重置游戏到初始状态"""
    global player, inventory, defeated_count, game_state, battle_log
    global atk_boost, is_defending, level_up_anim_timer, message_lines
    player = {
        "name": "勇者", "hp": 100, "max_hp": 100,
        "mp": 50, "max_mp": 50, "atk": 15, "def": 5,
        "exp": 0, "level": 1, "gold": 0,
    }
    inventory = []
    defeated_count = 0
    game_state = "menu"
    battle_log = []
    atk_boost = 0
    is_defending = False
    level_up_anim_timer = 0
    message_lines = []
    add_message("新的冒险开始了！", C["gold"])


# ==================== 程序入口 ====================
if __name__ == "__main__":
    main()
