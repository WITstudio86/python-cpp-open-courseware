"""
第13节：翻开看看 —— 4x4卡牌网格
功能：创建4x4卡牌网格（16张牌，8对），点击翻开卡牌，不同颜色表示不同图案
知识点：二维列表、卡牌状态管理、pygame.draw.rect绘制、坐标换算
"""

import pygame
import random
import sys

# ═══════════════════════════════════════
# 初始化Pygame
# ═══════════════════════════════════════
pygame.init()

# 窗口设置
SCREEN_W = 800
SCREEN_H = 600
screen = pygame.display.set_mode((SCREEN_W, SCREEN_H))
pygame.display.set_caption("第13节：翻开看看 —— 4x4卡牌网格")

# 时钟（控制帧率）
clock = pygame.time.Clock()

# ═══════════════════════════════════════
# 卡牌尺寸和网格计算
# ═══════════════════════════════════════
ROWS = 4        # 4行
COLS = 4        # 4列
CARD_W = 120    # 卡牌宽度
CARD_H = 120    # 卡牌高度
GAP = 20        # 卡牌之间的间隙

# 计算整个网格的宽度和高度
GRID_W = COLS * CARD_W + (COLS - 1) * GAP
GRID_H = ROWS * CARD_H + (ROWS - 1) * GAP

# 让网格在窗口中居中
OFFSET_X = (SCREEN_W - GRID_W) // 2
OFFSET_Y = (SCREEN_H - GRID_H) // 2

# ═══════════════════════════════════════
# 8种卡牌颜色（8对配对）
# ═══════════════════════════════════════
COLORS = [
    (255, 50, 50),     # 红色
    (50, 130, 255),    # 蓝色
    (50, 200, 50),     # 绿色
    (255, 200, 0),     # 黄色
    (200, 50, 255),    # 紫色
    (255, 130, 0),     # 橙色
    (0, 200, 200),     # 青色
    (255, 100, 180)    # 粉色
]

# 每个颜色对应的形状名称（用于在牌上画小图标）
SHAPE_NAMES = ["◆", "●", "▲", "★", "■", "⬟", "✦", "♥"]


def create_board():
    """
    创建4x4棋盘，随机放置8对颜色。
    每张牌是一个字典：{"color": (R,G,B), "state": "hidden"/"revealed"/"matched"}
    """
    # 生成8对颜色编号：[0,0, 1,1, 2,2, ..., 7,7]
    pairs = []
    for i in range(8):
        pairs.append(i)  # 第一张
        pairs.append(i)  # 第二张（配对）

    # 洗牌！打乱顺序，让每次游戏都不一样
    random.shuffle(pairs)

    # 填入4x4的二维列表
    board = []
    for row in range(ROWS):
        current_row = []
        for col in range(COLS):
            # 从洗好的列表中按顺序取出颜色编号
            index = row * COLS + col
            color_index = pairs[index]
            card = {
                "color": COLORS[color_index],
                "shape_index": color_index,       # 形状编号（与颜色对应）
                "state": "hidden"                 # 初始状态：隐藏
            }
            current_row.append(card)
        board.append(current_row)

    return board


def get_card_at_pos(mouse_x, mouse_y):
    """
    根据鼠标坐标，返回点击的是哪张牌的行列位置。
    点击空白处或间隙则返回 None。

    核心公式：col = (mouse_x - OFFSET_X) // (CARD_W + GAP)
    """
    # 先快速判断是否在网格大致范围内
    if mouse_x < OFFSET_X or mouse_y < OFFSET_Y:
        return None

    # 整数除法换算成行列
    col = (mouse_x - OFFSET_X) // (CARD_W + GAP)
    row = (mouse_y - OFFSET_Y) // (CARD_H + GAP)

    # 行列必须在有效范围（0~3）
    if not (0 <= row < ROWS and 0 <= col < COLS):
        return None

    # 精确判断：点的是牌面本身，而不是间隙
    card_x = OFFSET_X + col * (CARD_W + GAP)
    card_y = OFFSET_Y + row * (CARD_H + GAP)
    if card_x <= mouse_x <= card_x + CARD_W and card_y <= mouse_y <= card_y + CARD_H:
        return (row, col)

    return None


def draw_card_shape(screen, x, y, w, h, shape_index, color):
    """在卡牌中央绘制一个小形状图标，让不同颜色更容易区分"""
    cx = x + w // 2   # 中心X
    cy = y + h // 2   # 中心Y
    r = 25            # 形状半径/大小

    if shape_index == 0:   # 红色 —— 菱形
        points = [(cx, cy - r), (cx + r, cy), (cx, cy + r), (cx - r, cy)]
        pygame.draw.polygon(screen, (255, 255, 255), points)

    elif shape_index == 1: # 蓝色 —— 圆形
        pygame.draw.circle(screen, (255, 255, 255), (cx, cy), r)

    elif shape_index == 2: # 绿色 —— 三角形
        points = [(cx, cy - r), (cx + r, cy + r), (cx - r, cy + r)]
        pygame.draw.polygon(screen, (255, 255, 255), points)

    elif shape_index == 3: # 黄色 —— 五角星
        import math
        points = []
        for i in range(10):
            angle = math.pi / 2 + i * math.pi / 5
            radius = r if i % 2 == 0 else r * 0.4
            px = cx + radius * math.cos(angle)
            py = cy - radius * math.sin(angle)
            points.append((px, py))
        pygame.draw.polygon(screen, (255, 255, 255), points)

    elif shape_index == 4: # 紫色 —— 正方形
        pygame.draw.rect(screen, (255, 255, 255), (cx - r, cy - r, 2 * r, 2 * r))

    elif shape_index == 5: # 橙色 —— 六边形
        import math
        points = []
        for i in range(6):
            angle = math.pi / 2 + i * math.pi / 3
            px = cx + r * math.cos(angle)
            py = cy - r * math.sin(angle)
            points.append((px, py))
        pygame.draw.polygon(screen, (255, 255, 255), points)

    elif shape_index == 6: # 青色 —— 四角星
        points = []
        for i in range(8):
            angle = math.pi / 2 + i * math.pi / 4
            radius = r if i % 2 == 0 else r * 0.3
            px = cx + radius * math.cos(angle)
            py = cy - radius * math.sin(angle)
            points.append((px, py))
        pygame.draw.polygon(screen, (255, 255, 255), points)

    elif shape_index == 7: # 粉色 —— 爱心
        # 简化的爱心形状（两个圆 + 三角形）
        pygame.draw.circle(screen, (255, 255, 255), (cx - r // 2, cy - r // 3), r // 2)
        pygame.draw.circle(screen, (255, 255, 255), (cx + r // 2, cy - r // 3), r // 2)
        points = [(cx - r, cy - r // 3), (cx + r, cy - r // 3), (cx, cy + r)]
        pygame.draw.polygon(screen, (255, 255, 255), points)


def draw_board(screen, board):
    """绘制整个棋盘（16张卡牌）"""
    # 清空背景
    screen.fill((25, 25, 45))

    # 绘制标题文字
    font = pygame.font.Font(None, 36)
    title = font.render("第13节：翻开看看 —— 点击卡牌翻开！", True, (200, 200, 220))
    screen.blit(title, (SCREEN_W // 2 - title.get_width() // 2, 15))

    # 计算已翻开的数量
    revealed_count = 0
    for row in range(ROWS):
        for col in range(COLS):
            if board[row][col]["state"] == "revealed":
                revealed_count += 1

    # 显示翻开计数
    count_text = font.render(f"已翻开: {revealed_count} / 16", True, (255, 255, 255))
    screen.blit(count_text, (20, SCREEN_H - 45))

    # 逐行逐列绘制卡牌
    for row in range(ROWS):
        for col in range(COLS):
            # 计算这张牌在屏幕上的坐标
            x = OFFSET_X + col * (CARD_W + GAP)
            y = OFFSET_Y + row * (CARD_H + GAP) + 15  # 标题占一点空间

            card = board[row][col]

            if card["state"] == "hidden":
                # ── 隐藏状态：灰色牌背，中间显示"?" ──
                # 牌面
                pygame.draw.rect(screen, (60, 60, 100), (x, y, CARD_W, CARD_H), border_radius=10)
                # 边框（亮一点，有立体感）
                pygame.draw.rect(screen, (130, 130, 200), (x, y, CARD_W, CARD_H), 3, border_radius=10)
                # 中间画"?"
                q_font = pygame.font.Font(None, 60)
                q_text = q_font.render("?", True, (180, 180, 220))
                q_rect = q_text.get_rect(center=(x + CARD_W // 2, y + CARD_H // 2))
                screen.blit(q_text, q_rect)

            elif card["state"] == "revealed":
                # ── 翻开状态：显示卡牌的真实颜色 ──
                # 彩色牌面
                pygame.draw.rect(screen, card["color"], (x, y, CARD_W, CARD_H), border_radius=10)
                # 白色边框
                pygame.draw.rect(screen, (255, 255, 255), (x, y, CARD_W, CARD_H), 3, border_radius=10)
                # 画形状图标
                draw_card_shape(screen, x, y, CARD_W, CARD_H, card["shape_index"], card["color"])

            elif card["state"] == "matched":
                # ── 已匹配状态：颜色变暗（备用，下一节课用到） ──
                gray_color = tuple(c // 3 for c in card["color"])
                pygame.draw.rect(screen, gray_color, (x, y, CARD_W, CARD_H), border_radius=10)
                pygame.draw.rect(screen, (80, 80, 80), (x, y, CARD_W, CARD_H), 3, border_radius=10)


# ═══════════════════════════════════════
# 主程序入口
# ═══════════════════════════════════════
def main():
    # 创建棋盘
    board = create_board()
    running = True

    print("=" * 50)
    print("第13节：翻开看看 —— 4x4卡牌网格")
    print("点击卡牌翻开查看颜色！")
    print("=" * 50)

    while running:
        # ── 事件处理 ──
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False

            elif event.type == pygame.MOUSEBUTTONDOWN:
                # 获取鼠标点击位置
                mouse_x, mouse_y = event.pos

                # 换算出点击的是哪张牌
                result = get_card_at_pos(mouse_x, mouse_y)

                if result is not None:
                    row, col = result
                    card = board[row][col]

                    # 只有隐藏状态的牌才能翻开
                    if card["state"] == "hidden":
                        card["state"] = "revealed"
                        print(f"翻开了第 {row} 行第 {col} 列的牌！颜色编号：{card['shape_index']}")

            elif event.type == pygame.KEYDOWN:
                if event.key == pygame.K_r:
                    # 按R键重新开始
                    board = create_board()
                    print("游戏重新开始！")

        # ── 绘制画面 ──
        draw_board(screen, board)
        pygame.display.flip()
        clock.tick(60)  # 限制帧率60FPS

    pygame.quit()
    sys.exit()


# 运行主程序
if __name__ == "__main__":
    main()
