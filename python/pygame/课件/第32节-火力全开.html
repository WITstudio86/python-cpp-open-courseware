🎁 道具掉落系统 — 工作原理

敌人被消灭 │ ▼ roll = random.random() ← 生成 0~1 随机数 │ ├── roll < 0.15（小兵15%） → 掉落道具！ │ ├── roll < 1.00（Boss 100%） → 必定掉落！ │ ▼ 随机选择道具类型： 🔫 武器升级（50%） | 💚 生命恢复（30%） | 🛡️ 护盾（20%） │ ▼ 道具从敌人位置下落 → 玩家碰到 → 拾取生效！

核心思想：概率控制 + 随机类型

🎁 道具掉落 — 代码实现

# 道具精灵类 class PowerUp(pygame.sprite.Sprite): def __init__(self, x, y, power_type): super().__init__() # 根据类型绘制不同颜色 if power_type == "weapon": color = (255, 200, 0) # 金色 elif power_type == "heal": color = (100, 255, 100) # 绿色 else: color = (100, 180, 255) # 蓝色 # ... 绘制圆形道具 ... self.speed_y = 2 # 下落速度 def try_drop_item(x, y, is_boss=False): if is_boss or random.random() < 0.15: roll = random.random() if roll < 0.5: return PowerUp(x, y, "weapon") elif roll < 0.8: return PowerUp(x, y, "heal") else: return PowerUp(x, y, "shield") return None

👹 Boss 状态机 — 代码实现

def update_phase(self): # 根据血量百分比判断当前阶段 percent = self.hp / self.max_hp if percent > 0.5: self.phase = 1 # 阶段1：轻松模式 elif percent > 0.2: self.phase = 2 # 阶段2：加强模式 else: self.phase = 3 # 暴怒阶段：终极模式 def shoot(self): if self.phase == 1: # 扇形弹幕 3 发 elif self.phase == 2: # 扇形 5 发 + 圆形弹交替 elif self.phase == 3: # 密集扇形 + 追踪弹 # 追踪弹：计算朝向玩家的方向向量 dx = player.rect.centerx - self.rect.centerx dy = player.rect.centery - self.rect.centery dist = math.sqrt(dx**2 + dy**2) vx = 3 * dx / dist vy = 3 * dy / dist

💥 爆炸特效 ① — 粒子飞溅

粒子三要素：
▸ 位置 (x, y)
▸ 速度 (vx, vy) — 随机方向
▸ 生命 (life) — 逐帧递减

颜色渐变：
白 → 黄 → 橙 → 红
中心最热、边缘冷却

class Particle(Sprite): def update(self): self.rect.x += self.vx self.rect.y += self.vy self.life -= 1 # 渐隐：透明度降低 alpha = 255 * (self.life / self.max_life) self.image.set_alpha(alpha) if self.life <= 0: self.kill()

关键：set_alpha() 控制透明度 → 粒子慢慢消失

# Boss 死亡大爆炸：4 层圆圈从中心向外扩散 rings = [ {"color": (255, 255, 255), "max_r": 200}, # 白 ← 最外层 {"color": (255, 255, 0), "max_r": 160}, # 黄 {"color": (255, 165, 0), "max_r": 120}, # 橙 {"color": (255, 0, 0), "max_r": 80}, # 红 ← 最内层 ] for ring in rings: progress = 1 - (life / max_life) # 扩散进度 0→1 radius = ring["max_r"] * progress # 用 SRCALPHA 创建透明 surface temp = pygame.Surface((w, h), pygame.SRCALPHA) pygame.draw.circle(temp, color_with_alpha, center, radius, 4)

4 层 × 扩散 + 渐隐 = 逼真爆炸效果！
