def _draw_full_bunker(self):
    # 深绿色底
    self.image.fill((0, 130, 0))

    # 画砖块线 —— 交错排列
    for row in range(5):
        offset = (row % 2) * 12
        for col in range(5):
            bx = col * 22 + offset
            by = row * 12 + 2
            pygame.draw.rect(self.image,
                (0,100,0), (bx, by, 18, 9), 1)

💻 代码实现：掩体局部破坏
Python — Bunker.damage() 方法
def damage(self, hit_x, hit_y, blast_radius=10):
    # 第 1 步：计算破坏区域（确保不越界）
    left = max(0, hit_x - blast_radius)
    top = max(0, hit_y - blast_radius)
    width = min(blast_radius * 2, self.width - left)
    height = min(blast_radius * 2, self.height - top)
    damage_rect = pygame.Rect(left, top, width, height)

    # 第 2 步：🔑 subsurface 切出子区域
    # damaged 和 self.image 共享同一块内存！
    damaged = self.image.subsurface(damage_rect)

    # 第 3 步：在子区域上画弹坑
    pygame.draw.circle(damaged, (0,0,0),
        (blast_radius, blast_radius), blast_radius)

    # 减少耐久度
    self.health -= 12
    if self.health <= 0:
        self.kill()

progress = elapsed / max_elapsed  # 0.0 → 1.0
