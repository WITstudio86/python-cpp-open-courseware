"我们先来画一个塔防游戏的地图。塔防游戏最关键的就是一条敌人走的路线。怎么表示这条路线呢？——用一串坐标点！就像导航软件里的路线，由一个个关键位置连起来。"

```python
WIDTH = 800

DARK_BROWN = (100, 60, 25)    # 路线边缘

# 路径是一串坐标，敌人会依次走过这些点
# 这里设计一条 S 形路线：先右 → 向下 → 再右 → 向下 → 再右
path = [
    (0, 100),      # 起点：屏幕左侧
    (200, 100),    # 第一个拐弯：向右走到这里
    (200, 300),    # 第二个拐弯：向下走
    (500, 300),    # 第三个拐弯：向右走
```

**讲解要点：**
- `path` 列表中的每个元素是一个 `(x, y)` 元组，表示一个"定位点"

- 中间的点就是"拐弯"的地方 —— 就像路口

```
敌人就像跟着导航走：先走到第一个路口，再走到第二个路口……
```

```python
def draw_map():
    """绘制游戏地图：绿色草地 + 棕色路线"""
    # 1. 画草地背景（铺满整个窗口）
    screen.fill(GREEN)

    # 2. 画网格线（让草地看起来更丰富）
    for x in range(0, WIDTH, 40):
        pygame.draw.line(screen, DARK_GREEN, (x, 0), (x, HEIGHT), 1)
    for y in range(0, HEIGHT, 40):
        pygame.draw.line(screen, DARK_GREEN, (0, y), (WIDTH, y), 1)

    # 3. 画路线：用粗线条把路径点依次连起来
    if len(path) >= 2:
        # 先画路线的深色边缘（宽一点，作为边框）
        pygame.draw.lines(screen, DARK_BROWN, False, path, PATH_WIDTH + 6)
        # 再画路线的浅色主体
        pygame.draw.lines(screen, BROWN, False, path, PATH_WIDTH)

    # 4. 标记路径点（画小圆，帮助学生理解路径点的位置）
    for point in path:
        pygame.draw.circle(screen, YELLOW, point, 6)
        pygame.draw.circle(screen, BLACK, point, 6, 1)

    # 5. 画起点和终点的特殊标记
    # 起点：绿色旗帜（三角形）
    start = path[0]
    pygame.draw.polygon(screen, (0, 200, 0),
        [(start[0], start[1] - 12), (start[0] + 10, start[1] - 4),
         (start[0], start[1] + 4)])
    # 终点：红色标记
    end = path[-1]
    pygame.draw.circle(screen, (200, 0, 0), end, 10)
    pygame.draw.line(screen, WHITE, (end[0] - 5, end[1] - 5),
                     (end[0] + 5, end[1] + 5), 2)
    pygame.draw.line(screen, WHITE, (end[0] + 5, end[1] - 5),
                     (end[0] - 5, end[1] + 5), 2)
```

**讲解要点：**
- `pygame.draw.lines()` 可以把一串点用线段连起来，非常适合画路线
- `False` 参数表示不闭合（路线是开放的，不是封闭图形）
- 先画深色边框再画浅色主体，能让路线看起来有立体感

#### 步骤3：敌人沿路径移动（8分钟）【本节课核心难点】

**教师话故：**

"路线画好了，接下来要让敌人沿着路线走。敌人怎么知道该往哪走呢？——它要一直朝向当前的目标点走，走到了就换下一个目标点。"

```python
class Enemy(pygame.sprite.Sprite):
    """敌人类：沿路径点依次移动"""
    def __init__(self):
        super().__init__()
        # 创建敌人图像：红色圆形
        self.image = pygame.Surface((20, 20), pygame.SRCALPHA)
        pygame.draw.circle(self.image, RED, (10, 10), 10)
        pygame.draw.circle(self.image, (180, 30, 30), (10, 10), 10, 2)
        self.rect = self.image.get_rect()

        # 速度：每帧移动多少个像素
        self.speed = 2

        # 当前所在的路径点索引（从0开始，即起点）
        self.target_index = 1  # 第一个目标是 path[1]

        # 设置初始位置：路径起点
        self.rect.center = path[0]

    def update(self):
        """每帧调用：向当前目标点移动"""
        # 1. 获取当前目标点
        target = path[self.target_index]

        # 2. 计算敌人当前位置到目标点的方向
        dx = target[0] - self.rect.centerx
        dy = target[1] - self.rect.centery

        # 3. 计算距离
        distance = math.sqrt(dx * dx + dy * dy)

        # 4. 如果距离很近，说明到达了目标点 → 切换到下一个目标
        if distance < self.speed:
            self.target_index += 1
            # 如果已经走完所有点，敌人到达终点 → 移除
            if self.target_index >= len(path):
                self.kill()  # 从精灵组中移除
                return

            # 更新目标，重新计算方向
            target = path[self.target_index]
            dx = target[0] - self.rect.centerx
            dy = target[1] - self.rect.centery
            distance = math.sqrt(dx * dx + dy * dy)

        # 5. 沿方向移动一步（单位向量 × 速度）
        if distance > 0:
            self.rect.x += (dx / distance) * self.speed
            self.rect.y += (dy / distance) * self.speed
```

**核心算法讲解（可视化演示）：**

```
假设敌人从 path[0]=(0, 100) 出发，target_index=1，目标是 path[1]=(200, 100)

第1帧：
  敌人位置: (0, 100)
  目标: (200, 100)
  dx = 200 - 0 = 200, dy = 100 - 100 = 0
  distance = 200
  移动方向: (200/200, 0/200) = (1, 0) → 水平向右
  新位置: x + 1*2 = 2, y + 0*2 = 100

第50帧：
  敌人位置: (100, 100)
  目标: (200, 100)
  dx = 100, dy = 0, distance = 100
  继续向右移动...

第100帧：
  敌人位置: (198, 100)  （几乎到达目标）
  目标: (200, 100)
  distance = 2 < self.speed(2)  ✓ 判定为"到达"
  target_index 变成 2
  新目标: path[2] = (200, 300)
  dx = 200-198=2, dy = 300-100=200 → 开始向下走！
```

**讲解关键点：**
- `target_index` 就像一个"导航指针"，指向当前要去第几个路径点
- 每一帧都朝着当前目标走一小步
- 当距离小于速度时，说明"到了"，切换目标
- 到达最后一个路径点后，敌人消失（走出屏幕了）

**踩坑提示：**
- 速度不能太大，否则可能"跳过"路径点（直接穿过拐弯处）
- `dx/distance` 和 `dy/distance` 是"单位方向向量"，确保斜着走的速度也是恒定的
- 到达目标点后要重新计算 `dx` 和 `dy`，否则用的是旧方向

```python
class Tower(pygame.sprite.Sprite):
    """炮塔类：放置在路边，用于防御"""
    def __init__(self, x, y):
        super().__init__()
        # 创建炮塔图像：蓝色方形，中间有炮管
        self.image = pygame.Surface((36, 36), pygame.SRCALPHA)
        # 炮塔底座（方形）
        pygame.draw.rect(self.image, BLUE, (3, 3, 30, 30))
        pygame.draw.rect(self.image, (40, 80, 200), (3, 3, 30, 30), 2)
        # 炮管（小圆形）
        pygame.draw.circle(self.image, (100, 180, 255), (18, 18), 8)
        pygame.draw.circle(self.image, BLUE, (18, 18), 4)
        self.rect = self.image.get_rect()
        self.rect.center = (x, y)

# 金币系统
gold = 200           # 初始金币
TOWER_COST = 50      # 放置一个炮塔需要 50 金币

def can_place_tower(x, y):
    """检查某个位置是否可以放置炮塔"""
    # 1. 检查是否在路线上（不能放在路上）
    for i in range(len(path) - 1):
        p1 = path[i]
        p2 = path[i + 1]
        # 计算点到线段的距离
        # 简单判断：用矩形区域近似（方便学生理解）
        min_x = min(p1[0], p2[0]) - PATH_WIDTH // 2
        max_x = max(p1[0], p2[0]) + PATH_WIDTH // 2
        min_y = min(p1[1], p2[1]) - PATH_WIDTH // 2
        max_y = max(p1[1], p2[1]) + PATH_WIDTH // 2
        if min_x <= x <= max_x and min_y <= y <= max_y:
            return False  # 在路线上，不能放
    return True           # 可以放置

# 在事件循环中处理点击
for event in pygame.event.get():
    # ... 其他事件
    if event.type == pygame.MOUSEBUTTONDOWN:
        mouse_x, mouse_y = event.pos
        # 检查金币是否够、位置是否合法
        if gold >= TOWER_COST and can_place_tower(mouse_x, mouse_y):
            tower = Tower(mouse_x, mouse_y)
            tower_group.add(tower)
            gold -= TOWER_COST  # 扣除金币
```

**讲解要点：**
- `MOUSEBUTTONDOWN` 是鼠标点击事件，`event.pos` 返回点击位置的 `(x, y)`
- `can_place_tower()` 遍历每段路径，检查点击位置是否和路线重叠
- 只有金币够 **且** 位置不在路线上，才能放置炮塔
- 判断点是否在路线上的简化方法：构建每段路线的矩形范围，检查点是否在矩形内

```python
# 挑战3提示：点到线段的最短距离
def point_to_line_distance(px, py, x1, y1, x2, y2):
    """计算点(px,py)到线段(x1,y1)-(x2,y2)的最短距离"""
    # 线段向量
    dx = x2 - x1
    dy = y2 - y1
    # 线段长度的平方
    line_len_sq = dx * dx + dy * dy
    if line_len_sq == 0:
        return math.sqrt((px - x1) ** 2 + (py - y1) ** 2)
    # 计算投影参数 t
    t = max(0, min(1, ((px - x1) * dx + (py - y1) * dy) / line_len_sq))
    # 最近点坐标
    near_x = x1 + t * dx
    near_y = y1 + t * dy
    return math.sqrt((px - near_x) ** 2 + (py - near_y) ** 2)
```

## 💡 常见问题与解决

- **问题1：敌人走到拐弯处就停住了**
  → 检查 `target_index` 是否正确递增；检查到达条件 `distance < self.speed` 是否太严格。可以将条件改为 `distance < self.speed + 1` 给一点容差。

- **问题2：敌人移动方向不对，斜着走的时候速度变快了**
  → 没有做"归一化"处理。必须用 `dx/distance` 和 `dy/distance` 来保证在任何方向的速度都是恒定的 `speed`。如果直接用 `dx` 和 `dy` 加到坐标上，斜着走时实际速度是 `sqrt(dx^2 + dy^2)`，比 `speed` 大。

- **问题3：炮塔可以放在路线上**
  → 检查 `can_place_tower()` 函数中的矩形范围判断是否正确。注意 `min_x`、`max_x` 等变量的计算是否考虑了路径宽度。

- **问题4：点击放置炮塔没反应**
  → 检查事件类型是否是 `pygame.MOUSEBUTTONDOWN`（注意大小写）；检查金币是否足够；检查 `can_place_tower()` 是否返回 `True`。

- **问题5：敌人走出屏幕后程序报错**
  → `target_index >= len(path)` 时访问 `path[target_index]` 会报索引越界。必须在 `target_index += 1` 后先检查是否越界，如果是则 `kill()` 并 `return`。

- **问题6：多个敌人同时出现时，只有一个在动**
  → 检查是否把所有敌人放入了 `pygame.sprite.Group`，并用 `group.update()` 来更新所有敌人。如果逐个手动调用 `update()`，容易遗漏。
