### 二、核心讲解（25分钟）

#### 步骤1：敌机组列表管理（7分钟）

**核心原理**：和子弹列表一样，用列表 `enemies` 管理所有敌机。每隔一定帧数（如 60 帧 = 1 秒），在屏幕顶部随机位置生成一架新敌机。

```python
import pygame
import sys
import random

pygame.init()
WIDTH, HEIGHT = 800, 600
screen = pygame.display.set_mode((WIDTH, HEIGHT))
pygame.display.set_caption("敌人来袭")
clock = pygame.time.Clock()

# --- 颜色常量 ---
BLACK = (10, 10, 30)
WHITE = (255, 255, 255)
RED = (255, 50, 50)
GREEN = (50, 255, 50)
YELLOW = (255, 255, 0)
BLUE = (50, 150, 255)
DARK_RED = (150, 0, 0)

# --- 玩家飞机 ---
player_width, player_height = 50, 60
player_x = WIDTH // 2 - player_width // 2
player_y = HEIGHT - player_height - 30
player_speed = 5

# --- 子弹系统 ---
bullets = []
bullet_width, bullet_height = 6, 15
bullet_speed = 8
bullet_cooldown = 0
COOLDOWN_MAX = 15

# --- 敌机系统 ---
enemies = []               # 敌机列表
enemy_width, enemy_height = 40, 40
enemy_speed = 3            # 敌机向下移动速度
enemy_timer = 0            # 敌机生成计时器
ENEMY_SPAWN_INTERVAL = 60  # 每60帧（约1秒）生成一架敌机

# --- 得分 ---
score = 0

# --- 生命值 ---
max_hp = 100               # 最大生命值
hp = max_hp                 # 当前生命值

# --- 游戏状态 ---
game_over = False

running = True
while running:
    # --- 事件处理 ---
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        if event.type == pygame.KEYDOWN:
            # Game Over 状态下按 R 重新开始
            if game_over and event.key == pygame.K_r:
                # 重置所有变量
                bullets.clear()
                enemies.clear()
                player_x = WIDTH // 2 - player_width // 2
                player_y = HEIGHT - player_height - 30
                score = 0
                hp = max_hp
                enemy_timer = 0
                game_over = False

            # 空格发射子弹（仅在游戏未结束时）
            if not game_over and event.key == pygame.K_SPACE and bullet_cooldown == 0:
                new_bullet = {
                    "x": player_x + player_width // 2 - bullet_width // 2,
                    "y": player_y
                }
                bullets.append(new_bullet)
                bullet_cooldown = COOLDOWN_MAX

    # --- 游戏逻辑更新（仅在未 Game Over 时） ---
    if not game_over:
        # 冷却计时
        if bullet_cooldown > 0:
            bullet_cooldown -= 1

        # 玩家移动
        keys = pygame.key.get_pressed()
        if keys[pygame.K_LEFT] and player_x > 0:
            player_x -= player_speed
        if keys[pygame.K_RIGHT] and player_x < WIDTH - player_width:
            player_x += player_speed
        if keys[pygame.K_UP] and player_y > 0:
            player_y -= player_speed
        if keys[pygame.K_DOWN] and player_y < HEIGHT - player_height:
            player_y += player_speed

        # --- 敌机生成 ---
        enemy_timer += 1
        if enemy_timer >= ENEMY_SPAWN_INTERVAL:
            enemy_timer = 0  # 重置计时器
            # 在屏幕顶部随机位置生成敌机
            new_enemy = {
                "x": random.randint(0, WIDTH - enemy_width),
                "y": -enemy_height  # 从屏幕上方外面出现，效果更自然
            }
            enemies.append(new_enemy)

        # --- 更新子弹位置 + 移除出界子弹 ---
        for bullet in bullets:
            bullet["y"] -= bullet_speed
        # 移除飞出屏幕的子弹（列表推导式写法）
        bullets = [b for b in bullets if b["y"] + bullet_height > 0]

        # --- 更新敌机位置 ---
        for enemy in enemies:
            enemy["y"] += enemy_speed

        # --- 碰撞检测1：子弹 vs 敌机 ---
        player_rect = pygame.Rect(player_x, player_y, player_width, player_height)
        bullets_to_remove = []
        enemies_to_remove = []

        for bullet in bullets:
            bullet_rect = pygame.Rect(bullet["x"], bullet["y"], bullet_width, bullet_height)
            for enemy in enemies:
                enemy_rect = pygame.Rect(enemy["x"], enemy["y"], enemy_width, enemy_height)
                if bullet_rect.colliderect(enemy_rect):
                    # 子弹和敌机碰撞
                    bullets_to_remove.append(bullet)
                    enemies_to_remove.append(enemy)
                    score += 10  # 击毁敌机 +10分
                    break  # 一颗子弹只消灭一架敌机

        # 清除碰撞的子弹和敌机
        for b in bullets_to_remove:
            if b in bullets:
                bullets.remove(b)
        for e in enemies_to_remove:
            if e in enemies:
                enemies.remove(e)

        # --- 碰撞检测2：敌机 vs 玩家 ---
        player_rect = pygame.Rect(player_x, player_y, player_width, player_height)
        enemies_hit_player = []
        for enemy in enemies:
            enemy_rect = pygame.Rect(enemy["x"], enemy["y"], enemy_width, enemy_height)
            if player_rect.colliderect(enemy_rect):
                enemies_hit_player.append(enemy)
                hp -= 20  # 被敌机撞到扣20点血

        # 清除撞到玩家的敌机
        for e in enemies_hit_player:
            enemies.remove(e)

        # --- 移除飞出屏幕底部的敌机（及时清理，避免列表膨胀） ---
        enemies = [e for e in enemies if e["y"] < HEIGHT + 50]

        # --- Game Over 判定 ---
        if hp <= 0:
            hp = 0
            game_over = True

    # ========== 绘制部分 ==========
    screen.fill(BLACK)

    # 绘制玩家飞机
    player_rect = pygame.Rect(player_x, player_y, player_width, player_height)
    pygame.draw.rect(screen, BLUE, player_rect)
    pygame.draw.polygon(screen, (80, 180, 255), [
        (player_x - 15, player_y + player_height // 2),
        (player_x + player_width + 15, player_y + player_height // 2),
        (player_x + player_width // 2, player_y)
    ])

    # 绘制子弹
    for bullet in bullets:
        pygame.draw.rect(screen, YELLOW,
                         (bullet["x"], bullet["y"], bullet_width, bullet_height))

    # 绘制敌机（红色倒三角 + 矩形）
    for enemy in enemies:
        ex, ey = enemy["x"], enemy["y"]
        # 敌机机身
        pygame.draw.rect(screen, RED, (ex, ey, enemy_width, enemy_height))
        # 敌机标志（小三角形）
        pygame.draw.polygon(screen, (255, 100, 100), [
            (ex - 10, ey + enemy_height // 2),
            (ex + enemy_width + 10, ey + enemy_height // 2),
            (ex + enemy_width // 2, ey + enemy_height + 10)
        ])

    # --- 绘制血条 ---
    bar_width = 200
    bar_height = 20
    bar_x = 10
    bar_y = 10
    # 红色底框（血量背景）
    pygame.draw.rect(screen, DARK_RED, (bar_x, bar_y, bar_width, bar_height))
    # 绿色血条（当前血量，宽度按比例缩放）
    current_bar_width = int(bar_width * (hp / max_hp))
    if current_bar_width > 0:
        pygame.draw.rect(screen, GREEN, (bar_x, bar_y, current_bar_width, bar_height))
    # 血条边框
    pygame.draw.rect(screen, WHITE, (bar_x, bar_y, bar_width, bar_height), 2)

    # 显示血量数字
    font = pygame.font.SysFont("simhei", 22)
    hp_text = font.render(f"HP: {hp}/{max_hp}", True, WHITE)
    screen.blit(hp_text, (bar_x + bar_width + 15, bar_y - 2))

    # 显示得分
    score_text = font.render(f"Score: {score}", True, WHITE)
    screen.blit(score_text, (WIDTH - 150, bar_y - 2))

    # --- Game Over 画面 ---
    if game_over:
        # 半透明遮罩
        overlay = pygame.Surface((WIDTH, HEIGHT))
        overlay.set_alpha(180)
        overlay.fill((0, 0, 0))
        screen.blit(overlay, (0, 0))

        # Game Over 文字
        go_font = pygame.font.SysFont("simhei", 72)
        go_text = go_font.render("GAME OVER", True, RED)
        go_rect = go_text.get_rect(center=(WIDTH // 2, HEIGHT // 2 - 40))
        screen.blit(go_text, go_rect)

        # 最终得分
        final_score_text = font.render(f"最终得分: {score}", True, WHITE)
        fs_rect = final_score_text.get_rect(center=(WIDTH // 2, HEIGHT // 2 + 30))
        screen.blit(final_score_text, fs_rect)

        # 重新开始提示
        restart_text = font.render("按 R 键重新开始", True, (200, 200, 200))
        rt_rect = restart_text.get_rect(center=(WIDTH // 2, HEIGHT // 2 + 70))
        screen.blit(restart_text, rt_rect)

    pygame.display.flip()
    clock.tick(60)

pygame.quit()
sys.exit()
```

**⭐ 重点讲解：**
- **敌机生成定时器**：`enemy_timer` 每帧 +1，达到 `ENEMY_SPAWN_INTERVAL` 时重置并生成新敌机。
- **敌机从屏幕外出现**：`"y": -enemy_height` 让敌机从上方逐渐飞入，效果更自然。
- 敌机飞出底部也要移除：`enemies = [e for e in enemies if e["y"] < HEIGHT + 50]`

#### 步骤2：多组碰撞检测（6分钟）

**核心原理**：敌机 vs 子弹、敌机 vs 玩家，两组碰撞检测互相独立。

```python
# --- 碰撞检测1：子弹 vs 敌机 ---
for bullet in bullets:
    bullet_rect = pygame.Rect(bullet["x"], bullet["y"], bullet_width, bullet_height)
    for enemy in enemies:
        enemy_rect = pygame.Rect(enemy["x"], enemy["y"], enemy_width, enemy_height)
        if bullet_rect.colliderect(enemy_rect):
            bullets_to_remove.append(bullet)
            enemies_to_remove.append(enemy)
            score += 10
            break  # 一颗子弹只能击中一架敌机

# --- 碰撞检测2：敌机 vs 玩家 ---
player_rect = pygame.Rect(player_x, player_y, player_width, player_height)
for enemy in enemies:
    enemy_rect = pygame.Rect(enemy["x"], enemy["y"], enemy_width, enemy_height)
    if player_rect.colliderect(enemy_rect):
        enemies_hit_player.append(enemy)
        hp -= 20  # 扣血
```

**⭐ 重点讲解：**
- **两层循环**：外层遍历子弹，内层遍历敌机，每对组合都检测碰撞。
- **`break` 的用途**：一颗子弹击中一架敌机后跳出内层循环，避免一颗子弹同时消灭多架敌机。
- **碰撞顺序**：先检测子弹 vs 敌机，再检测敌机 vs 玩家，逻辑清晰。

**对比图示：**
```
子弹列表: [B1, B2, B3, ...]
敌机列表: [E1, E2, E3, ...]

检测 B1 vs E1, E2, E3  ← 外层循环 B1
检测 B2 vs E1, E2, E3  ← 外层循环 B2
检测 B3 vs E1, E2, E3  ← 外层循环 B3
```

#### 步骤3：生命值与血条系统（6分钟）

**核心原理**：用变量 `hp` 存储血量，被敌机撞到时扣血。血条是用两个矩形绘制而成：红色底框 + 绿色填充（宽度按比例缩放）。

```python
max_hp = 100     # 最大血量
hp = max_hp      # 当前血量

# 被敌机撞到：
hp -= 20         # 扣20点

# --- 绘制血条 ---
bar_width = 200
bar_height = 20
bar_x = 10
bar_y = 10

# 1. 红色底框（满血时的背景）
pygame.draw.rect(screen, (150, 0, 0), (bar_x, bar_y, bar_width, bar_height))

# 2. 绿色当前血量（宽度 = 总宽度 × 血量百分比）
current_width = int(bar_width * (hp / max_hp))
pygame.draw.rect(screen, (50, 255, 50), (bar_x, bar_y, current_width, bar_height))

# 3. 白色边框
pygame.draw.rect(screen, (255, 255, 255), (bar_x, bar_y, bar_width, bar_height), 2)

# 4. 文字显示
hp_text = font.render(f"HP: {hp}/{max_hp}", True, (255, 255, 255))
screen.blit(hp_text, (bar_x + bar_width + 15, bar_y - 2))
```

**⭐ 重点讲解：**
- **比例计算**：`hp / max_hp` 得出血量百分比（0 到 1），乘以总宽度得到绿色条的宽度。
- **`int()` 转换**：宽度必须是整数，`int()` 确保像素值是整数。
- **绘图顺序**：先画底框 → 再画填充 → 最后画边框，这样边框在最上层。

#### 步骤4：Game Over 判定与重新开始（6分钟）

**核心原理**：当血量归零时，设置 `game_over = True`。在游戏结束状态下，绘制遮罩 + 结束文字，并可按键重置。

```python
# Game Over 判定
if hp <= 0:
    hp = 0         # 防止负数
    game_over = True

# 在事件处理中：
if game_over and event.key == pygame.K_r:
    # 重置所有游戏变量
    bullets.clear()
    enemies.clear()
    player_x = WIDTH // 2 - player_width // 2
    player_y = HEIGHT - player_height - 30
    score = 0
    hp = max_hp
    enemy_timer = 0
    game_over = False

# 绘制部分用 if not game_over: 包裹游戏逻辑
if not game_over:
    # ... 移动、碰撞检测等 ...

# Game Over 遮罩
if game_over:
    overlay = pygame.Surface((WIDTH, HEIGHT))
    overlay.set_alpha(180)  # 半透明
    overlay.fill((0, 0, 0))
    screen.blit(overlay, (0, 0))
    # 绘制 Game Over 文字和重新开始提示...
```

**⭐ 重点讲解：**
- **重置所有变量**：重新开始时，必须把子弹、敌机、分数、血量、飞机位置全部恢复。
- **`bullets.clear()`**：清空列表，比重新赋值 `bullets = []` 更安全（因为其他地方可能引用了原列表）。
- **半透明遮罩**：`set_alpha()` 设置透明度，180 表示约 70% 不透明，效果很好。

## 💡 常见问题与解决

| 问题 | 原因 | 解决 |
|------|------|------|
| 敌机生成太快/太慢 | `ENEMY_SPAWN_INTERVAL` 设置不当 | 调整间隔值，60=1秒，30=0.5秒 |
| 一颗子弹消灭多架敌机 | 内层循环没加 `break` | 命中一架后 `break` 跳出内层循环 |
| 血条不按比例缩放 | `hp/max_hp` 是整数除法 | Python 3 中用 `/` 保证浮点除法 |
| Game Over 后飞机还能移动 | 没加 `if not game_over` 判断 | 用 `if not game_over:` 包裹所有游戏逻辑 |
| 重新开始后子弹还在 | 没清空子弹和敌机列表 | 按 R 时调用 `bullets.clear()` 和 `enemies.clear()` |
| 碰撞检测"穿过"了 | 物体移动速度太快，一帧跨过了碰撞区域 | 降低速度或增大碰撞体积 |
