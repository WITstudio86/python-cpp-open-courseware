LEVEL 1 — MATH CHALLENGE

🧮 数学试炼场 —— 完整代码

def level_1_math_challenge():
    print("\n🧮 ═══ 第1关：数学试炼场 ═══ 🧮")
    print("  规则：5道随机算术题，答对3题即可过关！")
    print("  每题1分，加油！\n")

    score = 0                         # 初始化得分
    total_questions = 5                # 总题数

    for i in range(1, total_questions + 1):
        # 随机选择运算类型：1=加法 2=减法 3=乘法
        op = random.randint(1, 3)

        if op == 1:      # 加法
            a, b = random.randint(10, 99), random.randint(10, 99)
            answer = a + b
            symbol = "+"
        elif op == 2:    # 减法（确保不出现负数）
            a = random.randint(20, 99)
            b = random.randint(1, a)         # b <= a
            answer = a - b
            symbol = "-"
        else:             # 乘法（数字小一点）
            a, b = random.randint(2, 12), random.randint(2, 9)
            answer = a * b
            symbol = "×"

        # 提问 + 判断
        user_ans = input(f"第{i}题：{a} {symbol} {b} = ? ")

        if user_ans.isdigit() or (user_ans.startswith("-") and user_ans[1:].isdigit()):
            user_ans = int(user_ans)
            if user_ans == answer:
                print("   ✅ 正确！+1分\n")
                score += 1
            else:
                print(f"   ❌ 可惜，答案是 {answer}\n")
        else:
            print(f"   ❌ 请输入数字哦（答案：{answer}）\n")

    # ── 判定过关 ──
    print(f"═══ 得分：{score}/{total_questions} ═══")
    if score >= 3:
        print("🎉 恭喜过关！获得 ★智慧之星★ 勋章！🎉\n")
    else:
        print("💪 还差一点，再来一次你一定行！\n")
设计要点：① 减法用 randint(1, a) 确保 minuend >= subtrahend，不出现负数；② 乘法数字控制在 2-12 范围，避免心算太难；③ isdigit() 处理非数字输入，提升程序的"抗打击"能力。

LEVEL 2 — CIPHER BREAK

🔐 密码破译局 —— 完整代码

凯撒密码规则

每个字母在字母表中后移 3 位：A → D, B → E, C → F ... X → A, Y → B, Z → C（循环）
例："HELLO WORLD" 加密后 → "KHOOR ZRUOG"
解密则相反：前移 3 位：D → A, E → B, C → Z（循环） 

def level_2_cipher_break():
    print("\n🔐 ═══ 第2关：密码破译局 ═══ 🔐")
    print("  规则：下面的密文用凯撒密码加密（字母后移3位）")
    print("  你的任务：破译出原文！\n")

    # 题库：密文 → 原文的映射（支持多次挑战）
    puzzles = [
        ("KHOOR ZRUOG", "HELLO WORLD"),
        ("SBWKRQ", "PYTHON"),
        ("JUHDW MRE", "GREAT JOB"),
        ("BRX FDQ GR LW", "YOU CAN DO IT"),
    ]

    # 随机选一题
    cipher, plain = random.choice(puzzles)
    print(f"📜 密文：{cipher}")
    print("💡 提示：凯撒密码，每个字母后移3位（循环）\n")

    user_guess = input("🔍 请输入原文（大写）：").strip().upper()

    if user_guess == plain:
        print("\n🎉 破译成功！获得 ★破译勋章★ 🎉")
        print("   你已掌握凯撒密码的破译方法！\n")
    else:
        print(f"\n❌ 方向不对哦。密文\"{cipher}\"的原文是\"{plain}\"")
        print("   💡 提示：试试在纸上写下字母表，每个字母往回数3个。\n")

Python 自动解密算法（拓展）

def caesar_decrypt(cipher_text, shift=3):
    """凯撒密码解密：每个字母前移 shift 位"""
    result = ""
    for char in cipher_text:
        if char.isalpha():   # 只处理字母
            # 确定基准 ord('A')=65 还是 ord('a')=97
            base = ord('A') if char.isupper() else ord('a')
            # (当前编号 - 偏移量 + 26) % 26：向前移并循环
            pos = (ord(char) - base - shift + 26) % 26
            result += chr(pos + base)
        else:
            result += char      # 空格、标点保持原样
    return result

# 测试
print(caesar_decrypt("KHOOR ZRUOG"))  # → HELLO WORLD

解密核心原理：ord(char) - base 将字母转为 0-25 编号，(pos - shift + 26) % 26 前移 shift 位并循环回绕，chr(pos + base) 转回字符。+26 再 % 26 确保负数也能正确循环——这是处理循环问题的经典技巧！

# 修改 shift 参数即可切换加密规则
def caesar_encrypt(plain_text, shift=5):
    result = ""
    for char in plain_text:
        if char.isalpha():
            base = ord('A') if char.isupper() else ord('a')
            pos = (ord(char) - base + shift) % 26
            result += chr(pos + base)
        else:
            result += char
    return result

# 示例：将你的悄悄话加密后发给同桌
secret = caesar_encrypt("PYTHON IS FUN")  # → "UDYMTS NX KZS"
print(f"加密后：{secret}")
