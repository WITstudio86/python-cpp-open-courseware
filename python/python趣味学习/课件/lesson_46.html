核心概念

KEY CONCEPTS

Turtle 烟花动画的五个核心概念

动画原理

像翻页动画书：每一帧"画 → 清屏 → 移动 → 再画"，快速切换帧产生"动"的错觉。Turtle 中用 clear() 清除上一帧，用 update() 一次性显示。

turtle.clear()

擦除当前海龟画过的所有痕迹（但不改变海龟的位置和方向）。在每帧开始前调用，清除上一帧画面，为画新一帧做准备。

tracer(0) + update()

turtle.tracer(0, 0) 关闭自动刷新（第一个0=延迟0ms，第二个0=完全关闭自动更新）。每帧画完后手动调用 turtle.update()，所有绘制同时出现，避免闪烁。

粒子系统

每个烟花拆成 30~50 个"小光点"（粒子），每个粒子有独立的：角度 angle、速度 speed、生命 life。在循环中分别更新每个粒子的位置，形成扩散效果。

随机颜色

每朵烟花的颜色通过 random.choice(["red","gold","cyan",...]) 随机选取，或用 random.random() 生成随机 RGB 值，让每朵烟花都独一无二。

物理要点：粒子运动公式 —— dx = math.cos(angle) * speed，dy = math.sin(angle) * speed。speed 逐帧衰减（空气阻力）speed *= 0.96，dy 要减去重力 dy -= 0.1 * step，size 随 step 减小实现淡出。

粒子系统 + 动画循环

# ═══════════════════════════════════════════
# 粒子系统和动画循环的核心逻辑
# ═══════════════════════════════════════════
import turtle, math, random, time

def boom(x, y):
    # 选择随机颜色
    colors = ["red", "gold", "cyan", "magenta",
              "orange", "lime", "hotpink", "yellow"]
    color = random.choice(colors)

    # 生成粒子列表：每个粒子 = [x, y, angle, speed, life, color]
    particles = []
    for i in range(40):  # 40 个粒子
        angle = random.uniform(0, 2 * math.pi)  # 随机角度 0~360°
        speed = random.uniform(2, 8)             # 随机初速度
        particles.append([x, y, angle, speed, 30, color])
        #                    0  1   2      3     4    5

    # 动画循环：逐帧更新每个粒子的位置
    for step in range(30):  # 30 帧扩散动画
        t.clear()            # 清空画布，准备画新一帧

        for p in particles:
            # 计算移动量（三角函数分解速度）
            dx = math.cos(p[2]) * p[3]   # x 方向移动量
            dy = math.sin(p[2]) * p[3]   # y 方向移动量
            dy -= 0.1 * step             # 重力影响

            # 更新粒子位置
            p[0] += dx                   # 新 x 坐标
            p[1] += dy                   # 新 y 坐标
            p[3] *= 0.96                 # 速度衰减（空气阻力）

            # 绘制这个粒子
            size = max(1, int(p[4] - step * 0.8))  # 越来越小
            t.penup()
            t.goto(p[0], p[1])
            t.pendown()
            t.dot(size, p[5])            # 画圆点

        turtle.update()                  # 一次性显示整帧
        time.sleep(0.03)                 # 控制帧率
# ═══════════════════════════════════════════
# 主程序 — 设置画面 + 循环燃放烟花
# ═══════════════════════════════════════════
turtle.bgcolor("midnightblue")   # 深蓝色夜空背景
turtle.title("🎆 烟花动画 — Turtle Fireworks")
turtle.tracer(0, 0)               # 关闭自动刷新，手动控制帧
turtle.hideturtle()               # 隐藏海龟箭头
turtle.colormode(255)
t = turtle.Turtle()
t.hideturtle()
t.speed(0)                        # 最快速度

# 循环燃放 10 朵烟花（随机位置）
for i in range(10):
    x = random.randint(-350, 350)  # 随机 x 位置
    y = random.randint(-200, 200)  # 随机 y 位置
    boom(x, y)
    time.sleep(0.3)                # 烟花间隔

# 最后保持画面
t.clear()
t.penup()
t.goto(0, -60)
t.pencolor("white")
t.write("🎆 烟花动画完成！", align="center",
        font=("Arial", 36, "bold"))
turtle.update()
turtle.done()

#!/usr/bin/env python3
"""Turtle 烟花动画 — Fireworks Animation
完整可运行版本。复制此文件直接运行即可！
"""
import turtle
import math
import random
import time

# ── 设置画面 ──
turtle.bgcolor("midnightblue")
turtle.title("🎆 烟花动画 — Turtle Fireworks")
turtle.tracer(0, 0)          # 关闭自动刷新，手动控制
turtle.hideturtle()
turtle.colormode(255)

t = turtle.Turtle()
t.hideturtle()
t.speed(0)
t.penup()

# 可用的烟花颜色列表
FIREWORK_COLORS = [
    "red", "gold", "cyan", "magenta",
    "orange", "lime", "hotpink", "yellow",
    "violet", "aqua", "salmon", "lawngreen"
]


def rise_then_burst(x, y):
    """上升光点 → 到达目标位置 → 爆炸"""
    # 从底部发射一个光点上升到爆炸位置
    start_y = -300
    for h in range(20):
        t.clear()
        t.goto(x, start_y + (y - start_y) * h / 19)
        t.dot(6, "white")
        turtle.update()
        time.sleep(0.02)
    # 到达后爆炸
    boom(x, y)


def boom(x, y):
    """在 (x, y) 位置燃放一朵烟花"""
    color = random.choice(FIREWORK_COLORS)
    num_particles = random.randint(35, 55)   # 粒子数量 35~55
    max_life = random.randint(25, 35)        # 扩散帧数 25~35

    # 生成粒子
    particles = []
    for i in range(num_particles):
        angle = random.uniform(0, 2 * math.pi)
        speed = random.uniform(2.5, 9)       # 初始速度
        particles.append({
            "x": x, "y": y,
            "angle": angle,
            "speed": speed,
            "life": max_life,
            "color": color
        })

    # 逐帧动画
    for step in range(max_life):
        t.clear()

        for p in particles:
            # 物理计算
            dx = math.cos(p["angle"]) * p["speed"]
            dy = math.sin(p["angle"]) * p["speed"]
            dy -= 0.12 * step                # 重力
            p["x"] += dx
            p["y"] += dy
            p["speed"] *= 0.96             # 空气阻力

            # 粒子逐渐变小淡出
            size = max(1, int((p["life"] - step) * 0.25))
            t.goto(p["x"], p["y"])
            t.dot(size, p["color"])

        turtle.update()
        time.sleep(0.03)


def rainbow_boom(x, y):
    """彩色渐变烟花：外层红色，内层黄色，多层颜色"""
    num_particles = 40
    max_life = 30
    layers = [
        ("red", "orange", 0.6, 1.0),          # 外层 红色
        ("orange", "yellow", 0.3, 0.6),      # 中层 橙色
        ("yellow", "gold", 0.0, 0.3),        # 内层 黄色
    ]

    particles = []
    for i in range(num_particles):
        angle = random.uniform(0, 2 * math.pi)
        speed = random.uniform(2, 8)
        ratio = random.random()  # 0~1
        for c1, c2, lo, hi in layers:
            if lo <= ratio < hi:
                color = c1 if random.random() > 0.5 else c2
                break
        particles.append([x, y, angle, speed, max_life, color])

    for step in range(max_life):
        t.clear()
        for p in particles:
            dx = math.cos(p[2]) * p[3]
            dy = math.sin(p[2]) * p[3]
            dy -= 0.12 * step
            p[0] += dx
            p[1] += dy
            p[3] *= 0.96
            size = max(1, int((p[4] - step) * 0.25))
            t.goto(p[0], p[1])
            t.dot(size, p[5])
        turtle.update()
        time.sleep(0.03)


# ── 主程序 ──
if __name__ == "__main__":
    print("🎆 Turtle 烟花动画开始！")
    print("   请切换到 Turtle 窗口观看动画")
    time.sleep(0.5)

    # 循环燃放 10 朵烟花
    for i in range(10):
        x = random.randint(-350, 350)
        y = random.randint(-150, 250)

        # 随机选择烟花类型
        fireworks_type = random.choice(["normal", "rising", "rainbow"])
        if fireworks_type == "normal":
            boom(x, y)
        elif fireworks_type == "rising":
            rise_then_burst(x, y)
        else:
            rainbow_boom(x, y)

        time.sleep(0.2)

    # 结束画面
    t.clear()
    t.goto(0, -80)
    t.pencolor("white")
    t.write("🎆 烟花动画完成！🎆",
            align="center",
            font=("Arial", 40, "bold"))
    t.goto(0, -140)
    t.write("你已掌握粒子系统 + 动画帧控制！",
            align="center",
            font=("Arial", 18, "normal"))
    turtle.update()
    print("✅ 动画播放完毕！")
    turtle.done()
