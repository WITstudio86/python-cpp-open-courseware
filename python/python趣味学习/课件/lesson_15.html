重点概念

KEY CONCEPTS

新知识点

random.randint(a, b)

生成一个 a 到 b 之间（包含两端）的随机整数。需要先 import random。

and（与）

两个条件都成立才为 True。如 score > 0 and score < 100。

or（或）

两个条件只要有一个成立就为 True。如 answer == "y" or answer == "yes"。

not（非）

取反。True 变 False，False 变 True。如 not is_pass。

import random

# 生成 1 到 100 之间的随机整数
secret = random.randint(1, 100)
print(secret)  # 每次运行结果可能不同
and 用法示例：age > 6 and age < 18 —— 年龄在 7-17 之间
or 用法示例：day == "周六" or day == "周日" —— 是周末
not 用法示例：not is_raining —— 没下雨

猜数字游戏 —— 完整版

import random

secret = random.randint(1, 100)
count = 0  # 记录猜的次数

print("=" * 40)
print("🎮 猜数字游戏")
print("电脑想好了一个 1-100 的数字，来猜吧！")
print("=" * 40)

while True:
    guess = int(input("请输入你猜的数字："))
    count += 1

    # 判断输入是否合法
    if guess < 1 or guess > 100:
        print("请猜 1 到 100 之间的数字哦~")
        continue

    # 判断大小
    if guess > secret:
        print("📈 猜大了！再试试~")
    elif guess < secret:
        print("📉 猜小了！再试试~")
    else:
        print(f"🎉 猜对了！就是 {secret}！")
        print(f"你一共猜了 {count} 次！")
        if count <= 3:
            print("天才！这么快就猜中了！")
        elif count <= 7:
            print("不错哦，表现很好！")
        else:
            print("终于猜中了，下次加油！")
        break
