# 一、我们已经用过的内置函数
print("=== 一、Python 内置函数（老朋友！）===")

# print() —— 显示信息
print("你好，这是我用 print() 显示的")

# input() —— 获取用户输入
# name = input("你叫什么名字？")

# len() —— 计算长度
fruits = ["苹果", "香蕉", "橘子"]
print(f"列表有 {len(fruits)} 个元素")

# max()、min()、sum() —— 数值计算
scores = [85, 92, 78, 95]
print(f"最高分: {max(scores)}")
print(f"最低分: {min(scores)}")
print(f"总分: {sum(scores)}")

# type() —— 查看数据类型
print(f"fruits 的类型: {type(fruits)}")

# 三、有函数：优雅！
print("\n=== 三、有函数：简洁清晰 ===")

def print_line():
    """打印一条分隔线"""
    print("=" * 30)

# 看！只需要调用函数就行了
print_line()
print("语文成绩单")
print_line()
print("小明: 95 分")
print("小红: 88 分")
print_line()

print_line()
print("数学成绩单")
print_line()
print("小明: 92 分")
print("小红: 90 分")
print_line()

print("\n上面的代码只写了一次分隔线代码！")
print("如果想把 = 改成 -，只需要改一个地方！")

# 重构后
print("\n【重构后】使用函数，简洁多了：")

def print_separator():
    """打印分隔线"""
    print("=" * 30)

def print_chapter(title):
    """打印章节标题"""
    print_separator()
    print(title)
    print_separator()

print_chapter("第1章")
print("内容...")
print_separator()
print()
print_chapter("第2章  ")
print("内容...")
print_separator()

print("\n对比一下：重构前 10 行，重构后同样效果但更清晰！")
print("而且如果要改分隔线样式，重构后只改一个地方就行！")
