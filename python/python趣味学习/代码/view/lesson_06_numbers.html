# ===== 1. 认识 int 和 float =====
print("===== 1. int 和 float =====")

# int（整数）—— 没有小数点
my_age = 11
students_count = 45
temperature = -5
print("age:", my_age, "→ 类型:", type(my_age))

# float（小数）—— 有小数点
my_height = 1.45
pi = 3.14159
price = 9.90
print("height:", my_height, "→ 类型:", type(my_height))

# 重要发现：除法 / 的结果永远是 float
print("\n10 / 5 =", 10 / 5, "→ 类型:", type(10 / 5))
# 虽然 10/5=2，但结果是 2.0（float）！

# ===== 2. 四则运算 =====
print("\n===== 2. 四则运算 =====")

a = 17
b = 5

print(f"{a} + {b} = {a + b}")    # 加法
print(f"{a} - {b} = {a - b}")    # 减法
print(f"{a} × {b} = {a * b}")    # 乘法（用 * 而不是 ×）
print(f"{a} ÷ {b} = {a / b}")    # 除法
print(f"{a} 的 {b} 次方 = {a ** b}")  # 幂运算

# ===== 3. 整除（//）和取余（%）=====
print("\n===== 3. 整除和取余 =====")

candies = 17
kids = 5

# 整除 // ：每人分几颗（扔掉剩下不够分的）
each_get = candies // kids
print(f"{candies} 颗糖，{kids} 个人，每人 {each_get} 颗")

# 取余 % ：分完后还剩几颗
remainder = candies % kids
print(f"还剩 {remainder} 颗")

# 验证：每人分到的 × 人数 + 剩下的 = 总数
print(f"验证：{each_get} × {kids} + {remainder} = {each_get * kids + remainder}")

# 更多例子
print("\n--- 更多整除和取余例子 ---")
print("10 // 3 =", 10 // 3, "（商）  10 % 3 =", 10 % 3, "（余数）")
print("20 // 7 =", 20 // 7, "（商）  20 % 7 =", 20 % 7, "（余数）")
print("100 // 9 =", 100 // 9, "（商） 100 % 9 =", 100 % 9, "（余数）")

# ===== 4. 幂运算（**）=====
print("\n===== 4. 幂运算 =====")

print("2 的 3 次方 =", 2 ** 3)          # 8
print("3 的 2 次方 =", 3 ** 2)          # 9
print("10 的 3 次方 =", 10 ** 3)        # 1000
print("5 的 5 次方 =", 5 ** 5)          # 3125
print("2 的 100 次方 =", 2 ** 100)      # 超大数字！

# ===== 5. 运算优先级 =====
print("\n===== 5. 运算优先级 =====")

# 和数学课一样：先乘除，后加减，括号最优先
print("2 + 3 * 4 =", 2 + 3 * 4)         # 3*4=12, 2+12=14
print("(2 + 3) * 4 =", (2 + 3) * 4)     # 2+3=5, 5*4=20
print("10 - 8 / 2 =", 10 - 8 / 2)       # 8/2=4.0, 10-4.0=6.0
print("(10 - 8) / 2 =", (10 - 8) / 2)   # 10-8=2, 2/2=1.0

# ===== 6. 生活中的数学 =====
print("\n===== 6. 生活中的数学 =====")

# 场景1：班级平均分
print("--- 班级平均分 ---")
chinese = 92
math = 85
english = 96
total = chinese + math + english
average = total / 3
print(f"语文:{chinese}  数学:{math}  英语:{english}")
print(f"总分:{total}  平均分:{average:.1f}")

# 场景2：购物找零
print("\n--- 购物找零 ---")
my_money = 100
book = 25.5
pen = 8.5
snack = 12
total_cost = book + pen + snack
change = my_money - total_cost
print(f"带了:{my_money}元")
print(f"书:{book} + 笔:{pen} + 零食:{snack} = {total_cost}元")
print(f"找零:{change}元")

# 场景3：分座位
print("\n--- 分座位 ---")
total_students = 42
seats_per_row = 6
full_rows = total_students // seats_per_row
left_students = total_students % seats_per_row
print(f"{total_students}人, 每排{seats_per_row}人")
print(f"坐满 {full_rows} 排, 还剩 {left_students} 人")

# 场景4：面积和周长
print("\n--- 教室面积 ---")
length = 8
width = 6
area = length * width
perimeter = (length + width) * 2
print(f"长{length}m, 宽{width}m")
print(f"面积:{area}平方米, 周长:{perimeter}米")

# ===== 7. 奇偶数判断 =====
print("\n===== 7. 奇偶数判断 =====")

for num in [7, 8, 15, 24, 100]:
    if num % 2 == 0:
        print(f"{num} 是偶数（除以2余0）")
    else:
        print(f"{num} 是奇数（除以2余1）")
