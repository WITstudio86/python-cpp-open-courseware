# ============================================================
# 第29节：def —— 定义自己的函数
# 适合：10-12 岁 Python 初学者
# ============================================================

# 一、第一个函数：say_hello()
print("=== 一、我的第一个函数 ===")

def say_hello():
    """一个简单的打招呼函数（这是文档字符串，解释函数用途）"""
    print("=" * 20)
    print("你好呀！")
    print("欢迎来到 Python 世界！")
    print("=" * 20)

# 调用函数
say_hello()
print()  # 空一行
say_hello()  # 可以多次调用！
print()
say_hello()  # 调用多少次都行！

# 二、def 语法的四个要素
print("\n=== 二、def 语法要素 ===")

print("定义一个函数有四个部分：")
print("  1. def     —— Python 关键字，表示'我要定义函数'")
print("  2. 函数名  —— 自己取名字，规则跟变量一样")
print("  3. ():     —— 小括号和冒号，必须写！")
print("  4. 函数体  —— 缩进的代码，函数要执行的命令")
print()
print("格式：")
print("  def 函数名():")
print("      函数体（要有缩进！）")

# 三、函数命名规则
print("\n=== 三、函数命名规则 ===")

print("✅ 好的函数名：")
print("  say_hello()  —— 一看就是打招呼")
print("  print_line() —— 一看就是打印分隔线")
print("  calculate_total() —— 一看就是计算总数")
print("  get_average() —— 一看就是获取平均值")
print()
print("❌ 坏的函数名：")
print("  a()    —— 不知道干什么")
print("  f1()   —— 太模糊")
print("  xyz()  —— 莫名其妙")
print()
print("命名规则：")
print("  - 只能字母、数字、下划线")
print("  - 不能数字开头")
print("  - 要有意义！")

# 四、定义和调用顺序
print("\n=== 四、定义和调用顺序 ===")

print("重要！必须先定义函数，再调用它！")
print()

# 定义函数
def greet():
    print("你好！")

# 调用函数（在定义之后）
greet()
print("✅ 正确：先定义，后调用")
print()

print("如果颠倒顺序会怎样？")
print("  greet()  ← 先调用")
print("  def greet(): ← 后定义")
print("  ❌ 报错！NameError: name 'greet' is not defined")

# 五、缩进规则
print("\n=== 五、缩进规则 ===")

print("函数体内的代码必须缩进（通常 4 个空格）：")
print()

def demonstrate_indent():
    print("  这一行缩进了 4 个空格 —— 属于函数")
    print("  这一行也缩进了 4 个空格 —— 属于函数")
    print("  注意：必须统一缩进量！")

print("这一行没有缩进 —— 不属于函数")
demonstrate_indent()

# 六、多个函数配合
print("\n=== 六、多个函数配合 ===")

def print_star_line():
    """打印星星分隔线"""
    print("⭐" * 15)

def print_dash_line():
    """打印横线分隔线"""
    print("-" * 30)

def welcome():
    """欢迎界面"""
    print_star_line()
    print("    欢迎来到 Python 乐园！")
    print_star_line()

def goodbye():
    """告别界面"""
    print_dash_line()
    print("    感谢使用，再见！")
    print_dash_line()

# 使用函数
welcome()
print()
print("这里是程序的主要内容...")
print()
goodbye()
