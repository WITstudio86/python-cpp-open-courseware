# ============================================================
# 课程 37：循环 + Turtle = 美丽花纹
# 学习目标：
#   1. 理解"循环 + 旋转 = 花纹"的魔法公式
#   2. 学会每次循环中改变颜色和角度
#   3. 能画出万花筒花纹（36 个圆，每次转 10°）
#   4. 能画出旋转正方形花纹
#   5. 探索颜色渐变的效果
# ============================================================

import turtle

screen = turtle.Screen()
screen.title("第37课：循环 + Turtle = 美丽花纹")
screen.bgcolor("black")  # 深色背景，让花纹更醒目！

t = turtle.Turtle()
t.shape("turtle")
t.speed(0)  # 最快速度，花纹复杂时不用等待
t.pensize(2)

# =====================================================
# 第一部分：魔法公式 —— 循环 + 旋转
# =====================================================

# 核心思想：
#   1. 在 for 循环中画一个图形（比如圆、正方形）
#   2. 每次循环结束后，海龟旋转一点角度
#   3. 重复很多次 → 美丽的对称花纹就出现了！


# =====================================================
# 第二部分：万花筒花纹 —— 36 个圆
# =====================================================

# 万花筒花纹：画 36 个圆，每次转 10°
# 360 ÷ 36 = 10，所以转满一圈刚好 36 个圆

t.penup()
t.goto(0, 0)  # 从画面中心开始
t.pendown()

# 定义颜色列表（彩虹色）
colors = ["red", "orange", "yellow", "green", "cyan", "blue", "purple", "magenta"]

for i in range(36):
    # 每次循环换一种颜色
    # i % 8 的结果在 0-7 之间循环，对应 8 种颜色
    t.color(colors[i % 8])

    # 画一个圆
    t.circle(100)

    # 旋转 10 度，准备画下一个圆
    t.right(10)

# 看！36 个不同颜色的圆，每次转 10°，形成了一个美丽的万花筒！

# 暂停一下，让我们看看效果
t.hideturtle()


# =====================================================
# 第三部分：旋转正方形花纹
# =====================================================

# 清除画布，画一个新的花纹
t.clear()
t.penup()
t.goto(0, 0)
t.pendown()
t.showturtle()

for i in range(36):
    t.color(colors[i % 8])
    t.pensize(2)

    # 画一个小正方形
    for j in range(4):
        t.forward(80)
        t.right(90)

    # 旋转 10 度
    t.right(10)

# 36 个正方形叠在一起旋转，形成了漂亮的花纹！


# =====================================================
# 第四部分：旋转三角形花纹
# =====================================================

t.clear()
t.penup()
t.goto(0, 0)
t.pendown()

for i in range(36):
    t.color(colors[i % 8])

    # 画一个三角形
    for j in range(3):
        t.forward(80)
        t.right(120)

    t.right(10)

# 三角形也一样可以形成美丽花纹！


# =====================================================
# 第五部分：改变大小 —— 逐渐变大的花纹
# =====================================================

t.clear()
t.penup()
t.goto(0, 0)
t.pendown()

# 这次我们让圆越来越大！
for i in range(20):
    t.color(colors[i % 8])

    # 半径从 20 慢慢增大到 200
    radius = 20 + i * 9  # i=0 时半径 20，i=19 时半径 191

    t.circle(radius)
    t.right(18)  # 360 ÷ 20 = 18


# =====================================================
# 第六部分：星星花纹（画星星 + 旋转）
# =====================================================

t.clear()
t.penup()
t.goto(0, 0)
t.pendown()

for i in range(18):  # 18 颗星星
    t.color(colors[i % 8])

    # 画一颗小五角星
    t.begin_fill()
    for j in range(5):
        t.forward(50)
        t.right(144)
    t.end_fill()

    # 旋转 20 度
    t.right(20)


# =====================================================
# 第七部分：万花筒进阶 —— 两个图形交替
# =====================================================

t.clear()
t.penup()
t.goto(0, 0)
t.pendown()

for i in range(36):
    t.color(colors[i % 8])

    if i % 2 == 0:
        # 偶数次：画圆
        t.circle(100)
    else:
        # 奇数次：画正方形
        for j in range(4):
            t.forward(80)
            t.right(90)

    t.right(10)

# 圆和正方形交替出现，花纹更加丰富！

t.hideturtle()
turtle.done()
