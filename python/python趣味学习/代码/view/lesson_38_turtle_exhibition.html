# ============================================================
# 课程 38：综合项目 —— Turtle 作品展
# 学习目标：
#   1. 综合运用前面 5 课所学的所有 Turtle 知识
#   2. 学会把一个复杂的画面拆分成多个简单的部分
#   3. 培养"规划 -> 实现 -> 完善"的编程思维
#   4. 完成一幅完整的 Turtle 画作

import turtle

# ---------- 准备工作 ----------

screen = turtle.Screen()
screen.title("第38课：Turtle 作品展 —— 太阳花与几何抽象画")
screen.bgcolor("#F0F8FF")  # 淡蓝色背景（爱丽丝蓝）

t = turtle.Turtle()
t.shape("turtle")
t.speed(0)  # 最快速度
t.pensize(2)

# =====================================================
# 作品一：太阳花图案
# =====================================================

# 太阳花的结构：
#   1. 花心（中间一个圆）
#   2. 花瓣（多个椭圆/弧线围绕花心）
#   3. 花茎和叶子

# --- 1.1 画花心 ---
t.penup()
t.goto(-150, 50)  # 把花画在画面左侧
t.pendown()

t.color("orange", "yellow")  # 橙色边框 + 黄色填充
t.begin_fill()
t.circle(30)  # 花心半径 30
t.end_fill()

# --- 1.2 画花瓣 ---
# 用 12 个弧线围绕花心，模拟花瓣
t.penup()
t.goto(-150, 80)  # 从花心上方开始
t.pendown()

petal_colors = ["#FF6347", "#FF4500", "#FF8C00", "#FFA500",
                "#FFD700", "#FFEA00"]

for i in range(12):
    # 每个花瓣用不同深浅的橙/黄色
    t.color(petal_colors[i % 6])

    # 画一个弧线当花瓣（半圆向外弯）
    t.begin_fill()
    t.circle(50, 60)   # 画 60 度的弧线
    t.left(120)        # 转回来
    t.circle(50, 60)   # 再画 60 度弧线，形成花瓣形状
    t.end_fill()

    t.right(150)  # 调整角度准备画下一个花瓣

# --- 1.3 画花茎 ---
t.penup()
t.goto(-150, 20)  # 花心下方
t.pendown()
t.color("green")
t.pensize(6)
t.setheading(270)  # 朝下
t.forward(180)     # 花茎长度

# --- 1.4 画叶子 ---
# 左叶子
t.penup()
t.goto(-150, -80)
t.pendown()
t.setheading(315)  # 朝右下
t.color("green", "lightgreen")
t.begin_fill()
t.circle(40, 90)   # 叶子的一半
t.left(90)
t.circle(40, 90)   # 叶子的另一半
t.end_fill()

# 右叶子
t.penup()
t.goto(-150, -100)
t.pendown()
t.setheading(225)  # 朝左下
t.color("green", "lightgreen")
t.begin_fill()
t.circle(40, 90)
t.left(90)
t.circle(40, 90)
t.end_fill()


# =====================================================
# 作品二：几何抽象画（画面右侧）
# =====================================================

# 几何抽象画由多个彩色几何图形组成：
#   - 彩色正方形（不同大小、不同颜色）
#   - 彩色圆
#   - 彩色三角形
#   - 彩色五角星

# --- 2.1 画一个大彩色正方形 ---
t.penup()
t.goto(120, 180)
t.pendown()
t.setheading(0)   # 恢复朝向
t.color("#4169E1", "#87CEEB")  # 皇家蓝边框 + 天蓝填充
t.pensize(3)

t.begin_fill()
for i in range(4):
    t.forward(100)
    t.right(90)
t.end_fill()


# --- 2.2 在正方形上叠一个旋转的小正方形 ---
t.penup()
t.goto(170, 130)
t.pendown()
t.color("#FF6347", "#FFA07A")  # 番茄红边框 + 浅鲑鱼色填充

t.right(45)  # 旋转 45 度，让正方形"歪着"
t.begin_fill()
for i in range(4):
    t.forward(70)
    t.right(90)
t.end_fill()

t.setheading(0)  # 恢复朝向


# --- 2.3 画一个彩色三角形 ---
t.penup()
t.goto(140, 20)
t.pendown()
t.color("#32CD32", "#98FB98")  # 酸橙绿边框 + 淡绿填充

t.begin_fill()
for i in range(3):
    t.forward(80)
    t.right(120)
t.end_fill()


# --- 2.4 画一个彩色圆 ---
t.penup()
t.goto(180, -80)
t.pendown()
t.color("#FF69B4", "#FFB6C1")  # 热粉红边框 + 浅粉填充

t.begin_fill()
t.circle(45)
t.end_fill()


# --- 2.5 画一颗彩色小五角星 ---
t.penup()
t.goto(250, 150)
t.pendown()
t.color("#FFD700", "#FFFACD")  # 金色边框 + 柠檬绸填充
t.pensize(2)

t.begin_fill()
for i in range(5):
    t.forward(40)
    t.right(144)
t.end_fill()


# =====================================================
# 作品三：装饰边框和标题
# =====================================================

# --- 3.1 画装饰性边框 ---
t.penup()
t.goto(-350, 260)
t.pendown()
t.color("#6A5ACD")  # 石板蓝
t.pensize(4)

# 画一个圆角效果的大边框（用弧线模拟）
for i in range(2):
    t.forward(700)
    t.circle(10, 90)
    t.forward(500)
    t.circle(10, 90)


# --- 3.2 写标题 ---
t.penup()
t.goto(0, 280)
t.pendown()
t.color("#2F4F4F")  # 暗石板灰
t.write("★ Turtle 作品展 ★", align="center", font=("Arial", 20, "bold"))


# --- 3.3 画装饰性小圆点（画面底部） ---
t.penup()
t.goto(-250, -220)
t.pendown()

dot_colors = ["red", "orange", "yellow", "green", "blue", "purple", "pink"]

for i, c in enumerate(dot_colors):
    t.color(c, c)
    t.begin_fill()
    t.circle(8)
    t.end_fill()
    t.penup()
    t.forward(70)
    t.pendown()


# =====================================================
# 作品四：万花筒小花（画面底部中间）
# =====================================================

t.penup()
t.goto(250, -120)
t.pendown()
t.pensize(2)

flower_colors = ["#FF69B4", "#FF1493", "#DB7093", "#C71585"]

for i in range(8):
    t.color(flower_colors[i % 4])
    t.begin_fill()
    t.circle(20, 90)
    t.left(90)
    t.circle(20, 90)
    t.end_fill()
    t.right(135)


t.hideturtle()
turtle.done()
