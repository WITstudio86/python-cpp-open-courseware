# 一、认识 return
print("=== 一、认识 return ===")

def add(a, b):
    """两个数相加，返回结果"""
    result = a + b
    return result  # 把结果交回去！

# 用变量接收返回值
answer = add(3, 5)
print(f"3 + 5 = {answer}")

# 直接使用返回值
print(f"10 + 20 = {add(10, 20)}")

# 返回值可以参与运算
total = add(1, 2) + add(3, 4)
print(f"(1+2) + (3+4) = {total}")

# 二、print 和 return 的区别（非常重要！）
print("\n=== 二、print vs return（非常重要的区别！）===")

# print 版本：只显示，不返回值
def add_print(a, b):
    """用 print 显示结果 —— 只给人看"""
    print(f"{a} + {b} = {a + b}")

# return 版本：返回值 —— 给程序继续用
def add_return(a, b):
    """用 return 返回结果 —— 给程序用"""
    return a + b

print("--- add_print  ---")
add_print(3, 5)  # 屏幕显示：3 + 5 = 8
r1 = add_print(3, 5)
print(f"add_print 的返回值：{r1}")  # None！没有返回值！

print("\n--- add_return ---")
r2 = add_return(3, 5)
print(f"add_return 的返回值：{r2}")  # 8！有返回值！
print(f"可以拿去算：{r2 * 2}")       # 16！可以继续用！

print("\n记忆口诀：")
print("  print 是'显示出来'（给人看的）")
print("  return 是'交回来'（给程序用的）")

# 三、return 会终止函数
print("\n=== 三、return 终止函数 ===")

def grade(score):
    """根据分数返回评级 —— 用 return 提前结束"""
    if score < 0 or score > 100:
        return "分数错误！"  # 函数直接结束
    if score >= 90:
        return "优秀"         # 符合条件就返回，不再往后走
    if score >= 80:
        return "良好"
    if score >= 60:
        return "及格"
    return "不及格"           # 前面都不满足才到这里

# 测试
for s in [95, 85, 72, 55, 120]:
    print(f"  {s} 分 → {grade(s)}")

# 演示 return 后面的代码不执行
def demo():
    print("这行会执行")
    return "我回来了"
    print("这行永远不会执行！")  # 不会执行

print("\nreturn 后面的代码不会执行：")
result = demo()
print(f"返回值：{result}")

# 四、返回多个值
print("\n=== 四、返回多个值 ===")

def get_stats(numbers):
    """返回列表的统计信息（多个值用逗号分隔）"""
    total = sum(numbers)
    average = total / len(numbers)
    maximum = max(numbers)
    minimum = min(numbers)
    return total, average, maximum, minimum

scores = [88, 92, 76, 95, 83]
t, avg, hi, lo = get_stats(scores)
print(f"总分：{t}")
print(f"平均：{avg:.1f}")
print(f"最高：{hi}")
print(f"最低：{lo}")

# 五、有返回值 vs 无返回值
print("\n=== 五、有返回值 vs 无返回值 ===")

# 无返回值（或返回 None）
def do_something():
    """做一些事，但不返回结果"""
    print("我做完了！")

# 有返回值
def calculate_something(x):
    """计算并返回结果"""
    return x * x + 2 * x + 1  # (x+1)²

do_something()  # 做完了，但没东西给你

result = calculate_something(3)
print(f"计算结果：{result}")     # 16，可以继续用
print(f"结果再乘 2：{result * 2}")  # 32

def subtract(a, b):
    """返回 a - b 的结果"""
    return a - b

def multiply(a, b):
    """返回 a × b 的结果"""
    return a * b

r = subtract(10, 4)

r = multiply(6, 8)
print(f"6 × 8 = {r}")

# 练习2：判断函数
print("\n【练习2】判断函数：")

def is_positive(n):
    """如果 n > 0 返回 True，否则返回 False"""
    return n > 0

def is_even(n):
    """判断偶数"""
    return n % 2 == 0

for n in [-3, 0, 5, 7, 8]:
    print(f"  {n} 是正数吗？{is_positive(n)}")
    if n > 0:
        print(f"  {n} 是偶数吗？{is_even(n)}")

# 练习3：print vs return
print("\n【练习3】print vs return 对比：")

# 用 print（没法继续用结果）
def grade_print(score):
    """用 print 显示评级"""
    if score >= 90:
        print("优秀")
    elif score >= 60:
        print("及格")
    else:
        print("不及格")

# 用 return（结果可以继续用）
def grade_return(score):
    """用 return 返回评级"""
    if score >= 90:
        return "优秀"
    elif score >= 60:
        return "及格"
    else:
        return "不及格"

print("grade_print 结果：", end="")
grade_print(95)

r = grade_return(95)
print(f"grade_return 结果：{r}")
print(f"可以拼接：最终评级是{r}！")

def analyze_scores(scores):
    """分析成绩列表，返回一份完整的报告字典"""
    if len(scores) == 0:
        return None

    total = sum(scores)
    avg = total / len(scores)
    highest = max(scores)
    lowest = min(scores)

    pass_count = 0
    excellent_count = 0
    for s in scores:
        if s >= 90:
            excellent_count += 1
        if s >= 60:
            pass_count += 1

    return {
        "总分": total,
        "平均分": round(avg, 1),
        "最高分": highest,
        "最低分": lowest,
        "及格人数": pass_count,
        "优秀人数": excellent_count,
        "总人数": len(scores),
        "及格率": round(pass_count / len(scores) * 100, 1)
    }

def get_grade(score):
    """返回评级"""
    if score >= 90:
        return "优秀"
    elif score >= 80:
        return "良好"
    elif score >= 60:
        return "及格"
    else:
        return "不及格"

# 测试成绩
math_scores = [95, 88, 92, 76, 83, 90, 45, 100, 67, 82]
report = analyze_scores(math_scores)

print("📊 数学成绩分析报告")
print("=" * 25)
if report:
    for key, value in report.items():
        print(f"  {key}：{value}")
    print("=" * 25)

    # 显示每个人评级
    print("\n个人评级：")
    for i, s in enumerate(math_scores, 1):
        print(f"  {i}. {s} 分 → {get_grade(s)}")

# 让用户输入自己的成绩
print("\n--- 分析你的成绩 ---")
print("请输入成绩，用空格分隔（如：90 85 78 92）：")
user_input = input("> ").strip()
if user_input:
    user_scores = []
    for s in user_input.split():
        if s.isdigit():
            user_scores.append(int(s))

    if user_scores:
        user_report = analyze_scores(user_scores)
        print("\n你的成绩报告：")
        for key, value in user_report.items():
            print(f"  {key}：{value}")
