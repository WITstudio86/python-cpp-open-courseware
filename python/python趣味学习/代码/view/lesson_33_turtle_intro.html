# ============================================================
# 课程 33：认识 Turtle —— 让海龟动起来
# 学习目标：
#   1. 学习如何导入 turtle 模块
#   2. 学会使用 forward()、backward() 控制海龟前进后退
#   3. 学会使用 right()、left() 控制海龟转向
#   4. 用这些命令让海龟走出正方形
# ============================================================

import turtle

# ---------- 第一部分：创建海龟和画布 ----------

# 创建一个画布窗口
screen = turtle.Screen()
screen.title("第33课：认识 Turtle —— 让海龟动起来")
screen.bgcolor("white")  # 背景色设置为白色

# 创建一只小海龟
t = turtle.Turtle()
t.shape("turtle")   # 把画笔形状变成海龟的样子，更好看
t.color("blue")     # 海龟的颜色设为蓝色
t.pensize(3)        # 画笔粗细设为 3

# ---------- 第二部分：基础移动命令演示 ----------

# forward(100)：让海龟向前走 100 像素
t.forward(100)

# right(90)：让海龟向右转 90 度
t.right(90)

# 再向前走 100 像素
t.forward(100)

# left(90)：让海龟向左转 90 度（和 right 相反）
t.left(90)

# backward(50)：让海龟向后退 50 像素
t.backward(50)

# ---------- 第三部分：走出一个正方形 ----------

# 先把海龟移回起点附近，清空画布重新开始
t.clear()       # 清空画布上的图案
t.penup()       # 抬起画笔（移动时不画线）
t.goto(-50, -50)  # 移动到左下角的位置
t.pendown()     # 放下画笔（移动时开始画线）

# 正方形的特点：4 条边，每个角 90 度
# 走正方形：前进 → 右转90° → 重复4次

t.color("red")  # 换一个颜色画正方形
side_length = 100  # 正方形的边长

t.forward(side_length)  # 第 1 条边
t.right(90)             # 右转 90 度

t.forward(side_length)  # 第 2 条边
t.right(90)

t.forward(side_length)  # 第 3 条边
t.right(90)

t.forward(side_length)  # 第 4 条边
t.right(90)

# 恭喜！海龟走出了一个正方形！

# 隐藏海龟，让画面更干净
t.hideturtle()

# ---------- 保持窗口不关闭 ----------
turtle.done()
