def print_line(char="=", length=50):

def gate_3_quiz():
    """
    第三关：选择题关
    从题库中随机抽取 5 道 Python 选择题
    每题 10 分，至少答对 3 题过关
    返回：本关得分
    """
    global total_score

    print_title("📝 第三关：Python 知识选择题")
    print()
    print("规则：系统随机出 5 道选择题，每题 10 分。")
    print("每题只有一个正确答案，请选择 A、B、C 或 D。")
    print("至少答对 3 题才能过关！")
    print()

    input("准备好了吗？按 Enter 键开始...")
    print()

    # 从题库中随机选 5 题（如果题库不足 5 题，就全选）
    selected = random.sample(question_bank, min(5, len(question_bank)))

    correct_count = 0
    score = 0
    wrong_questions = []  # 记录答错的题目，用于最后展示解析

    for i, q in enumerate(selected, 1):
        # 显示题目
        print(f"  【第 {i} 题】{q['question']}")
        print()
        for opt in q["options"]:
            print(f"    {opt}")
        print()

        # 获取答案
        user_answer = input("  请选择（A/B/C/D）：").strip().upper()

        # 将 A/B/C/D 转换成索引 0/1/2/3
        option_map = {"A": 0, "B": 1, "C": 2, "D": 3}
        if user_answer in option_map:
            user_index = option_map[user_answer]
        else:
            print("  ❌ 输入无效，请选 A/B/C/D，本题计 0 分\n")
            wrong_questions.append(q)
            continue

        # 判断对错
        if user_index == q["answer"]:
            print("  ✅ 回答正确！+10 分\n")
            correct_count += 1
            score += 10
        else:
            correct_letter = ["A", "B", "C", "D"][q["answer"]]
            print(f"  ❌ 回答错误！正确答案是 {correct_letter}\n")
            wrong_questions.append(q)

    # 显示错题解析
    if wrong_questions:
        print_line("-", 50)
        print("📖 错题解析：")
        for q in wrong_questions:
            correct_letter = ["A", "B", "C", "D"][q["answer"]]
            print(f"  题目：{q['question']}")
            print(f"  正确答案：{correct_letter}")
            print(f"  解析：{q['explanation']}")
            print()

    # 检查过关条件
    print_line("-", 50)
    print(f"你答对了 {correct_count} / {len(selected)} 题，本关得分：{score} 分")
    total_score += score
    level_scores["第三关"] = score

    if correct_count >= 3:
        print("🎉 恭喜！你通过了第三关！")
        return True
    else:
        print("😢 很遗憾，你没能通过第三关（需要至少答对 3 题）")
        return False

def main():
    global player_name, total_score, level_scores

    print_title("🏰 Python 闯关大挑战（下半场）")
    print()
    print("欢迎回到闯关挑战！今天将迎来最后一关和总结~")
    print()

    # 获取玩家姓名
    player_name = input("请输入你的名字：").strip()
    if not player_name:
        player_name = "小勇士"
    print(f"\n你好，{player_name}！让我们继续闯关！\n")
    time.sleep(1)

    # 重置分数（如果是独立的赛后程，上一课的分数需要手动输入）
    total_score = 0
    level_scores = {}
    previous_score = 0

    print("如果你已经完成了上一课（第一关和第二关），")
    try:
        previous_score = int(input("请输入你上半场的总分（如果没玩过请直接按 Enter）：").strip() or "0")
    except ValueError:
        previous_score = 0

    if previous_score > 0:
        total_score = previous_score
        print(f"\n已记录上半场得分：{previous_score} 分\n")

    # ---- 第三关 ----
    passed_3 = gate_3_quiz()

    # ---- 显示闯关状态 ----
    print()
    print_line("=", 50)

    print(f"  累计总分：{total_score} 分")
