import random

# ===== 阶段1：打招呼和自我介绍 =====
print("=" * 44)
print("  🤖 欢迎来到 Python 迷你对话机器人！")
print("=" * 44)
print()
print("🤖：你好呀！我是小P，一个用 Python 写的聊天机器人。")
print("🤖：今天我想和你聊聊天，了解一些关于你的事情~")
print()

# ===== 阶段2：问名字 =====
print("─" * 44)
print("  📛 让我们先认识一下")
print("─" * 44)

name = input("🤖：首先，你叫什么名字呀？\n👤：").strip().title()

# 如果没有输入名字，给一个默认的
if name == "":
    name = "神秘人"

print(f"🤖：{name}！这个名字真好听！")
print(f"🤖：你知道吗，你的名字有 {len(name)} 个字呢~")
print()

# ===== 阶段3：问年龄和相关计算 =====
print("─" * 44)
print("  🎂 关于年龄")
print("─" * 44)

try:
    age = int(input(f"🤖：{name}，你今年几岁了？\n👤："))
except ValueError:
    print("🤖：哎呀，请输入数字哦！这次我就当你10岁啦~")
    age = 10

current_year = 2024
birth_year = current_year - age
print(f"🤖：{age}岁！那你大概是 {birth_year} 年出生的~")
print(f"🤖：你比 Python（1991年出生）年轻了 {1991 - birth_year} 岁呢！")
print(f"🤖：再过 {100 - age} 年，你就100岁了！到时候再聊~")
print()

# ===== 阶段4：问喜好 =====
print("─" * 44)
print("  🍕 聊聊你的喜好")
print("─" * 44)

food = input("🤖：你最喜欢吃什么东西？\n👤：").strip()
if food == "":
    food = "好吃的"

color = input("🤖：最喜欢的颜色是什么呢？\n👤：").strip()
if color == "":
    color = "彩色"

print(f"🤖：{food}！听到我都流口水了~")
print(f"🤖：{color}色啊，很有品味！")

# 把喜欢的食物用有趣的方式处理
print(f"🤖：悄悄告诉你，{food} 倒过来念是「{food[::-1]}」，哈哈！")
print()

# ===== 阶段5：问爱好 =====
print("─" * 44)
print("  ⚽ 你的爱好和梦想")
print("─" * 44)

hobby = input("🤖：你平时最喜欢做什么？（爱好）\n👤：").strip()
if hobby == "":
    hobby = "玩"

dream = input("🤖：长大以后，你想做什么呢？\n👤：").strip()
if dream == "":
    dream = "一个厉害的人"

print(f"🤖：{hobby}！这个爱好真有趣！")
print(f"🤖：想成为{dream}，这个梦想太棒了！加油！")
print()

# ===== 阶段6：数学小挑战 =====
print("─" * 44)
print("  🧮 来个小挑战！")
print("─" * 44)

a = random.randint(1, 20)
b = random.randint(1, 20)
print(f"🤖：考你一道数学题：{a} + {b} = ?")

try:
    answer = int(input("👤："))
except ValueError:
    print("🤖：请输入数字哦~这次算你答对了！")
    answer = a + b

if answer == a + b:
    print(f"🤖：答对了！{a} + {b} = {a + b}，你真聪明！")
else:
    print(f"🤖：不对哦，{a} + {b} = {a + b}，下次加油！")
print()

# ===== 阶段7：心情检测 =====
print("─" * 44)
print("  😊 今天开心吗？")
print("─" * 44)

mood = input("🤖：用1-10分给今天的心情打个分吧！\n👤：").strip()
try:
    mood_score = int(mood)
except ValueError:
    mood_score = 8

if mood_score >= 8:
    print("🤖：哇，心情这么好！继续保持！😄")
elif mood_score >= 5:
    print("🤖：还不错，希望接下来的时间更开心！🙂")
else:
    print("🤖：没关系，不开心的事情都会过去的！💪")
print()

# ===== 阶段8：总结报告 =====
print("=" * 44)
print("  📊 你的个人档案")
print("=" * 44)

# 计算姓名的趣味信息
name_length = len(name)

print(f"""
  📛 姓名：{name}（{name_length}个字）
  🎂 年龄：{age} 岁（约{birth_year}年出生）
  🍕 喜欢的食物：{food.upper()}（倒过来：{food[::-1]}）
  🎨 喜欢的颜色：{color}
  ⚽ 爱好：{hobby}
  🌟 梦想：{dream}
  😊 心情指数：{mood_score}/10
""")

# 根据信息给出趣味评语
print("─" * 44)
print("  🤖 我的评语：")
print("─" * 44)

# 根据年龄给评语
if age <= 8:
    level = "初级小小编程师"
    advice = "你这么小就开始学编程，前途无量！"
elif age <= 10:
    level = "中级聪明编程师"
    advice = "正是学编程的黄金年龄，加油！"
elif age <= 12:
    level = "高级天才编程师"
    advice = "这个年龄学编程，理解能力超强！"
else:
    level = "超级大师编程师"
    advice = "学习永不止步，编程世界等你探索！"

print(f"🤖：你的编程等级：【{level}】")
print(f"🤖：{advice}")
print(f"🤖：记住，你的名字 {name} 倒过来是 {name[::-1]}，这是个秘密暗号！")
print()

# ===== 阶段9：彩蛋 =====
print("─" * 44)
print("  🎁 彩蛋时间！")
print("─" * 44)

# 随机彩蛋
eggs = [
    f"🤖：你知道吗？{name} 这个词在计算机里是这样存的：{str([ord(c) for c in name[:3]])}...",
    f"🤖：如果每天学1小时 Python，一年后你能写很厉害的程序！",
    f"🤖：悄悄告诉你，print('Hello World') 是每个程序员的第一个程序！",
    f"🤖：Python 的名字不是来自蟒蛇，而是来自一个喜剧节目哦！",
    f"🤖：你写的代码就像魔法咒语，能让电脑做任何事！",
]
print(random.choice(eggs))
print()

# ===== 阶段10：告别 =====
print("=" * 44)
print(f"🤖：{name}，和你聊天真的很开心！")
print("🤖：这就是我们前10节课的综合成果 —— ")
print("🤖：一个用 print、input、变量、数字、字符串、")
print("🤖：类型转换等知识做出来的对话机器人！")
print("🤖：你已经是一个真正的 Python 小小编程师了！")
print()
print("🤖：Python 的世界还有更多精彩等着你——")
print("🤖：条件判断、循环、列表、函数……")
print("🤖：我们下节课见！👋")
