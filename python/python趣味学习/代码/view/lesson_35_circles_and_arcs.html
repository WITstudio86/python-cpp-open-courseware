# ============================================================
# 课程 35：画圆和弧线
# 学习目标：
#   1. 学会用 circle(radius) 画圆
#   2. 学会用 circle(radius, angle) 画弧线（圆的一部分）
#   3. 学会 penup() 和 pendown() 配合使用，移动到不同位置画图
#   4. 能用圆和弧线组合出有趣的图案（笑脸、雪人）
# ============================================================

import turtle

screen = turtle.Screen()
screen.title("第35课：画圆和弧线")
screen.bgcolor("lightyellow")  # 淡黄色背景，像一张画纸

t = turtle.Turtle()
t.shape("turtle")
t.pensize(3)
t.speed(6)

# =====================================================
# 第一部分：circle() 的基本用法
# =====================================================

# --- 1.1 画一个完整的圆 ---
# circle(半径)：画一个半径为 radius 的圆
# 海龟当前位置是圆的"底部切点"，圆会画在海龟的左边

t.color("blue")
t.penup()
t.goto(-200, 150)  # 移动到左上角
t.pendown()

t.circle(50)  # 画半径为 50 的圆

# 在圆的旁边写个标签（可选，帮助理解）
t.penup()
t.goto(-200, 100)
t.pendown()
t.write("半径50的圆", font=("Arial", 10, "normal"))
t.penup()


# --- 1.2 画不同大小的圆 ---
t.color("red")
t.goto(-80, 130)
t.pendown()
t.circle(30)  # 小圆，半径 30

t.color("green")
t.goto(30, 110)
t.pendown()
t.circle(70)  # 大圆，半径 70

# 发现了吗？circle(半径) 中的数字越大，圆越大！


# =====================================================
# 第二部分：circle(radius, angle) 画弧线
# =====================================================

# circle(半径, 角度)：画圆的一部分（弧线）
# 角度 = 180 → 半圆；角度 = 90 → 四分之一圆

t.color("purple")
t.penup()
t.goto(-200, -20)
t.pendown()

t.circle(60, 180)  # 画半圆（180 度）

t.penup()
t.goto(-200, -30)
t.write("半圆 (180°)", font=("Arial", 9, "normal"))


t.color("orange")
t.penup()
t.goto(-50, 0)
t.pendown()

t.circle(60, 90)  # 画四分之一圆（90 度）

t.penup()
t.goto(-50, -50)
t.write("1/4圆 (90°)", font=("Arial", 9, "normal"))


t.color("brown")
t.penup()
t.goto(80, 30)
t.pendown()

t.circle(60, 270)  # 画四分之三圆（270 度）

t.penup()
t.goto(80, -30)
t.write("3/4圆 (270°)", font=("Arial", 9, "normal"))


# =====================================================
# 第三部分：画一个笑脸
# =====================================================

def draw_smiley(turtle_obj, x, y):
    """在位置 (x, y) 画一个笑脸"""
    t = turtle_obj

    # --- 画脸（大圆） ---
    t.penup()
    t.goto(x, y)
    t.pendown()
    t.color("black")
    t.pensize(3)
    t.circle(80)  # 脸，半径 80

    # --- 画左眼 ---
    t.penup()
    t.goto(x - 30, y + 100)  # 移动到左眼位置
    t.pendown()
    t.color("blue")
    t.begin_fill()      # 开始填充
    t.circle(10)         # 小圆当眼睛
    t.end_fill()         # 结束填充（眼睛变成蓝色实心圆）

    # --- 画右眼 ---
    t.penup()
    t.goto(x + 30, y + 100)  # 移动到右眼位置
    t.pendown()
    t.circle(10)
    t.end_fill()

    # --- 画嘴巴（半圆弧线） ---
    t.penup()
    t.goto(x - 30, y + 70)  # 移动到嘴巴起点
    t.pendown()
    t.color("red")
    t.pensize(4)
    t.setheading(-60)      # 调整海龟朝向，让弧线向下弯
    t.circle(35, 120)      # 画 120 度的弧线当微笑的嘴巴


# 清除之前画的演示图形
t.clear()

# 画一个笑脸！
draw_smiley(t, -100, -80)

# 再在旁边画一个小一点的笑脸
# 我们没有用类，直接手动调整位置和大小
t.penup()
t.goto(120, -100)
t.pendown()
t.color("black")
t.pensize(3)
t.circle(50)  # 小脸

t.penup()
t.goto(100, -40)
t.pendown()
t.color("blue")
t.begin_fill()
t.circle(7)   # 小眼睛
t.end_fill()

t.penup()
t.goto(140, -40)
t.pendown()
t.begin_fill()
t.circle(7)
t.end_fill()

t.penup()
t.goto(100, -60)
t.pendown()
t.color("red")
t.pensize(3)
t.setheading(-60)
t.circle(25, 120)


# =====================================================
# 第四部分：画雪人
# =====================================================

# 清除画布，重新开始画雪人
t.clear()
t.penup()
t.goto(0, -200)
t.pendown()
t.setheading(0)  # 重置朝向

# 雪人的特点：两个圆叠在一起，下面大上面小
# 还有眼睛、鼻子、帽子

# --- 雪人身体（下面的大圆） ---
t.color("black")
t.pensize(3)
t.penup()
t.goto(0, -180)
t.pendown()
t.circle(100)  # 大圆当身体

# --- 雪人头（上面的小圆） ---
t.penup()
t.goto(0, 20)
t.pendown()
t.circle(60)  # 小圆当头

# --- 雪人的眼睛 ---
t.color("black")
for eye_x in [-20, 20]:
    t.penup()
    t.goto(eye_x, 90)
    t.pendown()
    t.begin_fill()
    t.circle(5)
    t.end_fill()

# --- 雪人的鼻子（用一条短线代替胡萝卜鼻子） ---
t.color("orange")
t.pensize(4)
t.penup()
t.goto(0, 80)
t.pendown()
t.setheading(0)
t.forward(20)

# --- 雪人的帽子 ---
t.color("red")
t.pensize(2)
t.penup()
t.goto(-30, 130)
t.pendown()
t.setheading(0)
t.begin_fill()
t.forward(60)
t.left(90)
t.forward(15)
t.left(90)
t.forward(60)
t.left(90)
t.forward(15)
t.end_fill()

# 帽子的顶
t.penup()
t.goto(-15, 145)
t.pendown()
t.begin_fill()
t.left(90)
t.forward(30)
t.left(90)
t.forward(25)
t.left(90)
t.forward(30)
t.end_fill()

t.hideturtle()
turtle.done()
