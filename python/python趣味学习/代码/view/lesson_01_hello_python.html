# ===== 1. 让电脑跟你打招呼 =====
print("=" * 40)
print("你好，欢迎来到 Python 的世界！")
print("我是 Python，很高兴认识你！")
print("=" * 40)

# ===== 2. 让电脑做计算（比人算得快多了！）=====
print("\n--- 我是超级计算器 ---")
print("123 + 456 =", 123 + 456)
print("999 × 888 =", 999 * 888)
print("1000 - 357 =", 1000 - 357)
print("2 的 100 次方 =", 2 ** 100)  # 超大的数字，电脑瞬间算出来！

# ===== 3. 让电脑画图 —— 字符爱心 =====

print("   ♥       ♥   ")
print("    ♥     ♥    ")
print("     ♥   ♥     ")
print("      ♥ ♥      ")
print("       ♥       ")

# ===== 4. 让电脑画一个大三角形（金字塔）=====
print("\n--- 金字塔 ---")
print("      *")
print("     ***")
print("    *****")
print("   *******")
print("  *********")
print(" ***********")
print("*************")

# ===== 5. 有趣的小发现 =====
print("\n--- 有趣的小实验 ---")
# 文字乘以数字会重复！
print("小狗" * 10)     # 汪汪汪... 10次！
print("★" * 20)        # 星星满天！

# 逗号会自动加空格
print("苹果", "香蕉", "橘子", "葡萄")  # 自动空格分隔
