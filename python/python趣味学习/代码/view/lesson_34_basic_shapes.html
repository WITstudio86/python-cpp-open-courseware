# ============================================================
# 课程 34：画基本图形
# 学习目标：
#   1. 学会用 for 循环简化重复代码
#   2. 理解正多边形的画法：边数越多越接近圆
#   3. 掌握角度公式：每次转的角度 = 360 ÷ 边数
#   4. 用变量控制边长，轻松修改图形大小
# ============================================================

import turtle

# ---------- 第一部分：准备工作 ----------

screen = turtle.Screen()
screen.title("第34课：画基本图形")
screen.bgcolor("white")

t = turtle.Turtle()
t.shape("turtle")
t.pensize(3)
t.speed(5)  # 速度设为 5（1 最慢，10 最快，0 瞬间完成）

# ---------- 第二部分：用 for 循环画正方形 ----------
# 还记得第 33 课我们写了 8 行代码才画出正方形吗？
# 现在用 for 循环，只需要 3 行！

t.color("red")
t.penup()
t.goto(-200, 100)  # 移到左边位置
t.pendown()

# for 循环：range(4) 表示"重复 4 次"
# 下面的代码会让海龟走 4 次"前进 + 右转 90°"
for i in range(4):
    t.forward(100)  # 边长 100
    t.right(90)     # 90° = 360 ÷ 4

# 看！4 行就画完了一个正方形，是不是比之前简洁多了？


# ---------- 第三部分：画三角形 ----------
# 三角形的角度：360 ÷ 3 = 120（每次转 120 度）

t.color("blue")
t.penup()
t.goto(-50, 100)  # 移动位置，不和正方形重叠
t.pendown()

for i in range(3):
    t.forward(100)
    t.right(120)  # 360 ÷ 3 = 120


# ---------- 第四部分：画五边形 ----------
# 五边形的角度：360 ÷ 5 = 72

t.color("green")
t.penup()
t.goto(120, 100)
t.pendown()

for i in range(5):
    t.forward(80)
    t.right(72)  # 360 ÷ 5 = 72


# ---------- 第五部分：画六边形 ----------
# 六边形的角度：360 ÷ 6 = 60

t.color("purple")
t.penup()
t.goto(-200, -80)
t.pendown()

for i in range(6):
    t.forward(80)
    t.right(60)  # 360 ÷ 6 = 60


# ---------- 第六部分：画八边形 ----------
# 八边形的角度：360 ÷ 8 = 45
# 边数越多，每个角越小，图形越接近圆

t.color("orange")
t.penup()
t.goto(50, -80)
t.pendown()

for i in range(8):
    t.forward(60)
    t.right(45)  # 360 ÷ 8 = 45


# ---------- 神奇的发现 ----------

# 我们来总结一下规律：
#   正方形：4 条边，每次转 360 ÷ 4 = 90 度
#   三角形：3 条边，每次转 360 ÷ 3 = 120 度
#   五边形：5 条边，每次转 360 ÷ 5 = 72 度
#   六边形：6 条边，每次转 360 ÷ 6 = 60 度
#   八边形：8 条边，每次转 360 ÷ 8 = 45 度
#
# 规律：每次转的角度 = 360 ÷ 边数
# 边数越多，图形越圆！

t.hideturtle()
turtle.done()
