import random  # 导入随机数模块


# ── 骰子图案字典 ──────────────────────────────
# 用多行字符串画出 1~6 点的骰子图案
DICE_ART = {
    1: """
┌─────────┐
│         │
│    ●    │
│         │
└─────────┘""",
    2: """
┌─────────┐
│  ●      │
│         │
│      ●  │
└─────────┘""",
    3: """
┌─────────┐
│  ●      │
│    ●    │
│      ●  │
└─────────┘""",
    4: """
┌─────────┐
│  ●   ●  │
│         │
│  ●   ●  │
└─────────┘""",
    5: """
┌─────────┐
│  ●   ●  │
│    ●    │
│  ●   ●  │
└─────────┘""",
    6: """
┌─────────┐
│  ●   ●  │
│  ●   ●  │
│  ●   ●  │
└─────────┘""",
}


def roll_dice():
    """模拟掷一次骰子，返回 1~6 的随机数"""
    result = random.randint(1, 6)  # 随机生成 1 到 6 的整数
    return result


def show_dice(number):
    """根据数字显示对应的骰子图案"""
    print(DICE_ART.get(number, "未知点数"))  # 从字典中取出图案并打印


def main():
    print("=" * 40)
    print("        🎲  欢迎来到骰子模拟器  🎲")
    print("=" * 40)
    print("输入'摇'来掷骰子，输入'退出'结束游戏。")

    while True:
        # 让用户输入指令
        command = input("\n👉 请输入指令（摇 / 退出）: ").strip()

        if command == "退出":
            print("感谢使用骰子模拟器，再见！👋")
            break  # 跳出循环，结束程序

        elif command == "摇":
            # 掷骰子
            dice_number = roll_dice()
            print(f"\n🎯 骰子点数：{dice_number} 点")
            show_dice(dice_number)

            # 询问是否继续
            again = input("\n还摇吗？(y/n): ").strip().lower()
            if again != "y":
                print("感谢使用骰子模拟器，再见！👋")
                break  # 跳出循环，结束程序

        else:
            print("⚠️  输入有误，请输入'摇'或'退出'哦！")


# ── 程序入口 ──────────────────────────────────
if __name__ == "__main__":
    main()
