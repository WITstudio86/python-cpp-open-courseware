### 步骤2：核心讲解（15-20分钟）

#### 2.1 列表操作综合复习

这是之前学过的知识的综合应用。快速回顾：

```python
scores = []  # 创建一个空列表，就像拿了一个空篮子

# 添加成绩（往篮子里放鸡蛋）
scores.append(85)
scores.append(92)
scores.append(78)

# 查看有几个成绩（数一数篮子里有几个鸡蛋）
count = len(scores)  # → 3

# 访问某个成绩（拿出第几个鸡蛋）
first_score = scores[0]  # → 85（注意：从 0 开始！）

# 遍历所有成绩（把每个鸡蛋都拿出来看看）
for score in scores:
    print(score)
```

#### 2.2 循环录入数据

用 while 循环不断询问"还有成绩吗？"，直到用户说"没了"：

```python
scores = []  # 空篮子

print("📝 成绩录入系统")
print("输入成绩后按回车，输入 'done' 结束录入\n")

while True:
    user_input = input(f"请输入第 {len(scores) + 1} 个成绩：")
    
    if user_input.lower() == "done":
        break  # 退出循环
    
    # 尝试转成数字
    try:
        score = float(user_input)
        if 0 <= score <= 100:
            scores.append(score)
            print(f"  ✅ 已录入：{score} 分")
        else:
            print(f"  ⚠️  成绩应该在 0-100 之间，请重新输入")
    except ValueError:
        print(f"  ⚠️  '{user_input}' 不是有效的数字，请重新输入")
```

`try-except` 是一个保护机制：如果用户输入了"abc"这种不能转成数字的东西，程序不会崩溃，而是友好地提示。

#### 2.3 内置统计函数

```python
scores = [85, 92, 78, 95, 88]

# 求和 —— 把所有分数加起来
total = sum(scores)  # → 438

# 求个数
count = len(scores)  # → 5

# 求平均值 —— 总分除以人数
average = sum(scores) / len(scores)  # → 87.6

# 最高分 —— 谁考得最好
highest = max(scores)  # → 95

# 最低分 —— 谁考得最差
lowest = min(scores)  # → 78

# 排序 —— 从小到大（升序）
sorted_asc = sorted(scores)  # → [78, 85, 88, 92, 95]

# 排序 —— 从大到小（降序）
sorted_desc = sorted(scores, reverse=True)  # → [95, 92, 88, 85, 78]
```

生活比喻：**这些函数就像瑞士军刀上的不同工具**
- `sum()` 是加法器
- `max()` 是找最大值的探测器
- `min()` 是找最小值的探测器
- `sorted()` 是排序整理器
- `len()` 是计数器

#### 2.4 函数封装

为什么需要函数？想象你要重复用一段代码 5 次，如果每次都复制粘贴，代码会又长又乱。函数就是把一段代码"打包"起来，给它取个名字，以后叫它的名字就能用。

```python
# 定义一个函数：计算平均分
def calculate_average(scores):
    """计算并返回平均分"""
    if len(scores) == 0:
        return 0  # 防止除以 0
    return sum(scores) / len(scores)


# 定义一个函数：打印成绩报告
def print_report(scores):
    """打印一份漂亮的成绩报告"""
    print("\n" + "=" * 40)
    print("          📊 成绩统计报告")
    print("=" * 40)
    print(f"  总人数：{len(scores)} 人")
    print(f"  平均分：{calculate_average(scores):.1f} 分")
    print(f"  最高分：{max(scores)} 分")
    print(f"  最低分：{min(scores)} 分")
    print("=" * 40)


# 调用函数
scores = [85, 92, 78, 95]
print_report(scores)  # 只需要一行！
```

函数的结构就像一个汉堡包：
- 上层面包：`def 函数名(参数):`
- 中间的肉：函数体（缩进的代码）
- 下层面包：`return` 返回值（可选）

### 基础版：成绩统计器

```python
"""
成绩统计器 —— 基础版
功能：录入成绩 → 自动生成统计报告
"""

# ── 第一步：录入成绩 ──
scores = []  # 用列表存储所有成绩

print("=" * 40)
print("         📊 成绩统计器")
print("=" * 40)
print("说明：逐个输入 0-100 之间的成绩")
print("      输入 'done' 结束录入\n")

while True:
    user_input = input(f"请输入第 {len(scores) + 1} 个成绩：").strip()
    
    # 检查是否结束录入
    if user_input.lower() == "done":
        break
    
    # 尝试将输入转换为数字
    try:
        score = float(user_input)
        # 检查成绩是否在合理范围内
        if 0 <= score <= 100:
            scores.append(score)
            print(f"  ✅ 已录入：{score} 分")
        else:
            print("  ⚠️  成绩应该在 0 到 100 之间，请重新输入！")
    except ValueError:
        print(f"  ⚠️  '{user_input}' 不是有效的数字，请重新输入！")

# 检查是否有数据
if len(scores) == 0:
    print("\n❌ 没有录入任何成绩，程序结束。")
    exit()

print(f"\n📋 成绩录入完成！共录入 {len(scores)} 个成绩。")

# ── 第二步：统计分析 ──

# 基本统计
total = sum(scores)           # 总分
average = total / len(scores) # 平均分
highest = max(scores)         # 最高分
lowest = min(scores)          # 最低分

# 排序（从高到低）
sorted_scores = sorted(scores, reverse=True)

# 及格统计
pass_count = 0
excellent_count = 0  # 90分以上
good_count = 0       # 80-89分
fail_count = 0       # 不及格

for s in scores:
    if s >= 60:
        pass_count += 1
    if s >= 90:
        excellent_count += 1
    elif s >= 80:
        good_count += 1
    elif s < 60:
        fail_count += 1

pass_rate = pass_count / len(scores) * 100

# ── 第三步：生成报告 ──

print("\n" + "=" * 45)
print("              📊 成绩统计报告")
print("=" * 45)

# 基本信息
print(f"  📝 参考人数：{len(scores)} 人")
print(f"  📈 总分：{total} 分")
print(f"  📊 平均分：{average:.1f} 分")
print(f"  🥇 最高分：{highest} 分")
print(f"  📉 最低分：{lowest} 分")
print(f"  📏 分差（最高-最低）：{highest - lowest} 分")

# 去掉最高和最低后的平均分（比赛常用）
if len(scores) >= 3:
    trimmed_scores = sorted(scores)[1:-1]  # 去头去尾
    trimmed_avg = sum(trimmed_scores) / len(trimmed_scores)
    print(f"  🎯 去掉最高最低后平均分：{trimmed_avg:.1f} 分")

print(f"\n  ── 成绩分布 ──")
print(f"  🌟 优秀（90-100）：{excellent_count} 人")
print(f"  👍 良好（80-89）：{good_count} 人")
print(f"  📖 中等（70-79）：{len(scores) - excellent_count - good_count - fail_count - pass_count + (len(scores) - pass_count)} 人")
print(f"  ✅ 及格率：{pass_rate:.1f}%（{pass_count}/{len(scores)}）")
print(f"  💪 不及格（<60）：{fail_count} 人")

# 柱状图
print(f"\n  ── 简易柱状图 ──")
for label, count, symbol in [
    ("优秀 90+", excellent_count, "🌟"),
    ("良好 80+", good_count, "👍"),
    ("中等 70+", len([s for s in scores if 70 <= s < 80]), "📖"),
    ("及格 60+", len([s for s in scores if 60 <= s < 70]), "⚠️"),
    ("不及格 ", fail_count, "💪")
]:
    bar = "█" * count
    print(f"  {label}：{bar} {count}人 {symbol}")

# 排名榜（前 5 名）
print(f"\n  ── 排行榜（前5名）──")
for rank, s in enumerate(sorted_scores[:5], 1):
    medal = ["🥇", "🥈", "🥉"][rank - 1] if rank <= 3 else f"  {rank}"
    print(f"  {medal} {s} 分")

print("=" * 45)
print("\n📋 报告生成完毕！")
```

### 进阶版：函数封装的成绩统计器

```python
"""
成绩统计器 —— 进阶版（函数封装）
使用函数组织代码，让程序结构更清晰
"""

def input_scores():
    """录入成绩，返回成绩列表"""
    scores = []
    print("=" * 35)
    print("     📊 成绩统计器（进阶版）")
    print("=" * 35)
    print("逐个输入成绩，输入 'done' 结束\n")
    
    while True:
        inp = input(f"第 {len(scores) + 1} 个成绩：").strip()
        if inp.lower() == "done":
            break
        try:
            s = float(inp)
            if 0 <= s <= 100:
                scores.append(s)
                print(f"  ✅ 已录入：{s}")
            else:
                print("  ⚠️  请输入 0-100 之间的分数")
        except ValueError:
            print(f"  ⚠️  '{inp}' 不是有效数字")
    return scores


def get_grade(score):
    """根据分数返回等级和符号"""
    if score >= 90:
        return "优秀", "🌟"
    elif score >= 80:
        return "良好", "👍"
    elif score >= 70:
        return "中等", "📖"
    elif score >= 60:
        return "及格", "⚠️"
    else:
        return "不及格", "💪"


def analyze_scores(scores):
    """分析成绩，返回一个包含所有统计信息的字典"""
    if not scores:
        return None
    
    total = sum(scores)
    avg = total / len(scores)
    
    # 分段统计
    distribution = {"优秀": 0, "良好": 0, "中等": 0, "及格": 0, "不及格": 0}
    for s in scores:
        grade_name, _ = get_grade(s)
        distribution[grade_name] += 1
    
    pass_count = len([s for s in scores if s >= 60])
    
    return {
        "count": len(scores),
        "total": total,
        "average": avg,
        "highest": max(scores),
        "lowest": min(scores),
        "sorted": sorted(scores, reverse=True),
        "distribution": distribution,
        "pass_count": pass_count,
        "pass_rate": pass_count / len(scores) * 100
    }


def print_report(stats):
    """打印成绩统计报告"""
    if not stats:
        print("没有数据！")
        return
    
    print("\n" + "=" * 45)
    print("            📊 成绩统计报告")
    print("=" * 45)
    print(f"  📝 总人数：{stats['count']} 人")
    print(f"  📈 总分：{stats['total']} 分")
    print(f"  📊 平均分：{stats['average']:.1f} 分")
    print(f"  🥇 最高分：{stats['highest']} 分")
    print(f"  📉 最低分：{stats['lowest']} 分")
    print(f"  ✅ 及格率：{stats['pass_rate']:.1f}%")
    
    print(f"\n  ── 成绩分布 ──")
    for grade, count in stats['distribution'].items():
        bar = "█" * count
        print(f"  {grade}：{bar} {count}人")
    
    print(f"\n  ── 排行榜 ──")
    for rank, s in enumerate(stats['sorted'][:10], 1):
        grade_name, emoji = get_grade(s)
        print(f"  {rank:2d}. {s:6.1f} 分 {emoji} {grade_name}")
    
    print("=" * 45)


def save_report(stats, filename="report.txt"):
    """保存报告到文件"""
    with open(filename, "w", encoding="utf-8") as f:
        f.write("成绩统计报告\n")
        f.write("=" * 40 + "\n")
        f.write(f"总人数：{stats['count']} 人\n")
        f.write(f"平均分：{stats['average']:.1f}\n")
        f.write(f"最高分：{stats['highest']}\n")
        f.write(f"最低分：{stats['lowest']}\n")
        f.write(f"及格率：{stats['pass_rate']:.1f}%\n")
    print(f"\n💾 报告已保存到 {filename}")


# ── 主程序 ──
def main():
    # 1. 录入成绩
    scores = input_scores()
    
    if len(scores) == 0:
        print("❌ 没有录入任何成绩，程序结束。")
        return
    
    # 2. 分析
    stats = analyze_scores(scores)
    
    # 3. 打印报告
    print_report(stats)
    
    # 4. 询问是否保存
    save_choice = input("\n是否保存报告到文件？(y/n)：")
    if save_choice.lower() == "y":
        save_report(stats)


# 程序入口
if __name__ == "__main__":
    main()
```

## 📋 小结

### 本节学了什么
- **列表综合操作**：`.append()` 动态添加数据，`len()` 获取长度，配合循环做批量处理
- **内置统计函数**：`sum()`、`max()`、`min()`、`sorted()` 一行代码搞定统计
- **函数封装**：用 `def` 把功能打包，让代码更清晰、更容易复用
- **格式化报告**：用 f-string 做对齐和保留小数，让输出像一份专业的统计报告
- **try-except**：保护程序，防止用户输入错误导致崩溃
