# 第32节：综合练习：制作工具函数库

## 🎯 教学目标
- 学生能够回顾并运用函数的所有知识（def、参数、return）
- 学生能够将之前学过的功能封装成可复用的函数
- 学生能够理解模块的概念，学会使用 `import` 导入自己写的模块
- 学生能够创建自己的工具函数库文件

## 📝 重点概念
- **模块（module）**：一个 .py 文件就是一个模块，里面的函数可以被其他程序导入使用
- **import**：导入模块的关键字，就像"借工具"
- **工具库**：把自己常用的函数收集在一个文件里，方便以后重复使用
- **代码复用**：一次编写，多次使用，是编程的核心思想

## 📖 讲解内容

### 步骤1：引入（5分钟）

"学完了函数四节课，我们来做个大总结！"

**情景**：小明写了很多程序，每个程序都需要一些相同的功能——算平均分、打印分隔线、判断奇偶数...每次都重新写一遍？太累了！

**解决方案**：把常用的函数收集到一个文件里，变成一个**工具库**。以后哪个程序需要，就直接 `import` 进来用。

**类比**：
- 你有一个工具箱，里面有锤子、螺丝刀、扳手
- 每次修东西不需要重新造工具，直接从工具箱里拿
- `import my_tools` 就是"打开我的工具箱"
- `my_tools.add(1, 2)` 就是"从工具箱里拿出 add 这个工具来用"

### 步骤2：核心讲解（15-20分钟）

#### 2.1 创建工具库文件

创建一个新文件 `my_tools.py`，把常用的函数放进去：

```python
# my_tools.py —— 我的 Python 工具库

def print_line(char="-", length=30):
    """打印分隔线"""
    print(char * length)

def add(a, b):
    """加法"""
    return a + b

def is_even(n):
    """判断偶数"""
    return n % 2 == 0

def get_average(numbers):
    """计算平均值"""
    if len(numbers) == 0:
        return 0
    return sum(numbers) / len(numbers)

def grade(score):
    """根据分数返回评级"""
    if score >= 90:
        return "优秀"
    elif score >= 80:
        return "良好"
    elif score >= 60:
        return "及格"
    else:
        return "不及格"
```

#### 2.2 导入和使用工具库

```python
# main.py —— 主程序
import my_tools  # 导入我们的工具库

# 使用工具库里的函数
my_tools.print_line("=", 40)
print("   成绩管理系统")
my_tools.print_line("=", 40)

score = 95
print(f"成绩：{score}，评级：{my_tools.grade(score)}")

scores = [88, 92, 76, 95, 83]
avg = my_tools.get_average(scores)
print(f"平均分：{avg}")

# 判断
n = 7
if my_tools.is_even(n):
    print(f"{n} 是偶数")
else:
    print(f"{n} 是奇数")
```

#### 2.3 import 的几种方式

```python
# 方式一：导入整个模块
import my_tools
my_tools.add(1, 2)  # 需要用 模块名.函数名() 调用

# 方式二：导入特定函数
from my_tools import add, grade
add(1, 2)   # 可以直接用函数名
grade(95)

# 方式三：导入所有函数（不推荐，容易重名）
from my_tools import *
add(1, 2)
```

**推荐方式一**：`import 模块名`，清晰知道函数来自哪个模块。

#### 2.4 为什么要把代码分到不同文件？

| 好处 | 说明 |
|------|------|
| 复用 | 工具函数一次写好，到处能用 |
| 清晰 | 主程序简洁，工具单独管 |
| 维护 | 改一个地方，所有程序都受益 |
| 分享 | 可以把 my_tools.py 分享给同学 |

## 💻 代码示例

```python
# ========== 第32节：综合练习：制作工具函数库 ==========
# 
# 本节需要两个文件：
# 1. my_tools.py —— 工具函数库
# 2. lesson_32_main.py —— 主程序（使用工具库）

# ==========================================
# 以下是 my_tools.py 的内容（通常单独保存）
# ==========================================

"""
my_tools.py —— 我的 Python 工具库

这个文件收集了我常用的函数，像一个工具箱。
其他程序可以通过 import 来使用这些工具。

用法：
    import my_tools
    my_tools.add(1, 2)
"""

# --- 显示类工具 ---

def print_line(char="-", length=30):
    """打印一条分隔线"""
    print(char * length)

def print_title(title, char="=", length=30):
    """打印一个带框的标题"""
    print(char * length)
    print(f"  {title}")
    print(char * length)

def show_menu(options):
    """显示菜单，options 是一个列表"""
    print_line()
    for i, option in enumerate(options, 1):
        print(f"  {i}. {option}")
    print_line()

# --- 数学类工具 ---

def add(a, b):
    """加法"""
    return a + b

def subtract(a, b):
    """减法"""
    return a - b

def multiply(a, b):
    """乘法"""
    return a * b

def divide(a, b):
    """除法，自动处理除零错误"""
    if b == 0:
        return "错误：不能除以0"
    return a / b

def is_even(n):
    """判断是否是偶数"""
    return n % 2 == 0

def is_odd(n):
    """判断是否是奇数"""
    return n % 2 != 0

def power(base, exp):
    """计算 base 的 exp 次方"""
    return base ** exp

# --- 列表类工具 ---

def get_average(numbers):
    """计算列表中数字的平均值"""
    if len(numbers) == 0:
        return 0
    return sum(numbers) / len(numbers)

def get_max_min(numbers):
    """返回列表的最大值和最小值"""
    return max(numbers), min(numbers)

def count_pass(scores, pass_line=60):
    """统计及格人数"""
    count = 0
    for s in scores:
        if s >= pass_line:
            count += 1
    return count

# --- 判断类工具 ---

def grade(score):
    """根据分数返回评级"""
    if score < 0 or score > 100:
        return "无效分数"
    if score >= 90:
        return "优秀"
    elif score >= 80:
        return "良好"
    elif score >= 60:
        return "及格"
    else:
        return "不及格"

def is_passing(score, pass_line=60):
    """判断是否及格"""
    return score >= pass_line

# --- 字符串类工具 ---

def reverse_string(s):
    """反转字符串"""
    return s[::-1]

def count_char(s, char):
    """统计字符串中某个字符出现的次数"""
    return s.count(char)

# --- 字典类工具 ---

def find_in_dict(d, key, default=None):
    """安全地在字典中查找"""
    return d.get(key, default)

# ==========================================
# 以下是主程序的内容
# ==========================================

print("=" * 40)
print("    🧰 我的 Python 工具函数库展示")
print("=" * 40)

# --- 导入工具库 ---
# 如果 my_tools.py 在同一个文件夹下，就可以这样导入
# import my_tools

# 因为本文件包含了 my_tools 的函数，我们直接测试
print("\n=== 1. 显示类工具 ===")
print_title("工具库演示", char="~")

show_menu(["查看成绩", "计算平均分", "判断奇偶", "退出"])

print("\n=== 2. 数学类工具 ===")
print(f"3 + 5 = {add(3, 5)}")
print(f"10 - 4 = {subtract(10, 4)}")
print(f"6 × 8 = {multiply(6, 8)}")
print(f"15 ÷ 3 = {divide(15, 3)}")
print(f"10 ÷ 0 = {divide(10, 0)}")

print(f"\n7 是偶数吗？{is_even(7)}")
print(f"8 是偶数吗？{is_even(8)}")
print(f"2 的 10 次方 = {power(2, 10)}")

print("\n=== 3. 列表类工具 ===")
scores = [95, 88, 92, 76, 45, 83, 90, 67]
print(f"成绩：{scores}")
print(f"平均分：{get_average(scores):.1f}")
hi, lo = get_max_min(scores)
print(f"最高分：{hi}，最低分：{lo}")
print(f"及格人数：{count_pass(scores)}")
print(f"优秀人数（>=90）：{count_pass(scores, 90)}")

print("\n=== 4. 判断类工具 ===")
for s in [95, 85, 72, 55, -5]:
    print(f"  {s}分 → {grade(s)}, {'及格' if is_passing(s) else '不及格'}")

print("\n=== 5. 字符串类工具 ===")
text = "hello world"
print(f"'{text}' 反转：{reverse_string(text)}")
print(f"'{text}' 中 'l' 的数量：{count_char(text, 'l')}")

print("\n=== 6. 字典类工具 ===")
info = {"姓名": "小明", "年龄": 12}
print(f"查找'姓名'：{find_in_dict(info, '姓名', '未知')}")
print(f"查找'身高'：{find_in_dict(info, '身高', '未记录')}")

# --- 7. 综合演示：用工具库做一个成绩管理程序 ---
print("\n" + "=" * 40)
print("    📊 成绩管理系统（用工具库搭建）")
print("=" * 40)

all_scores = {
    "小明": [95, 88, 92],
    "小红": [78, 85, 90],
    "小刚": [65, 72, 58],
    "小丽": [100, 98, 96]
}

print_title("全班成绩报告")
for name, scores in all_scores.items():
    avg = get_average(scores)
    g = grade(avg)
    print(f"  {name}：{scores} → 平均 {avg:.1f} 分 ({g})")

# 计算全班总平均
all = []
for scores in all_scores.values():
    all.extend(scores)
print(f"\n全班总平均：{get_average(all):.1f} 分")
print(f"全班最高：{max(all)} 分")
print(f"全班最低：{min(all)} 分")
print(f"全班总人数：{len(all_scores)} 人")
```
