# 第40节：项目二 —— 单词翻译本

## 🎯 教学目标
- 学生能够使用 Python 字典（dict）存储键值对数据
- 学生能够实现字典的增删查改基本操作
- 学生能够使用文件读写将数据保存到文件并从文件读取
- 学生能够综合运用 while 循环和 if-elif 分支制作菜单系统

## 📝 重点概念
- **字典（dict）**：用 `{键: 值}` 方式存储配对的容器，就像一本真正的字典——英文单词（键）对应中文翻译（值）
- **键值对（Key-Value Pair）**：字典的基本单位，"apple" 是键，"苹果" 是值，它们是配对的搭档
- **文件读写**：`open()` 打开文件，`.write()` 写入，`.read()` 读取，让数据"活过"一次运行
- **菜单驱动循环**：用 while 循环展示一个菜单，用户选择功能，就像餐厅点菜一样

#### 2.1 字典的创建和基本操作

**创建一个字典**：

```python
# 创建一个英中翻译字典
word_dict = {
    "apple": "苹果",
    "banana": "香蕉",
    "cat": "猫",
    "dog": "狗",
    "book": "书"
}
```

生活比喻：字典就像一排带标签的储物柜。每个柜子有一个标签（键），里面放着东西（值）。

**字典的四种基本操作**：

| 操作 | 代码 | 比喻 |
|------|------|------|
| **添加** | `word_dict["hello"] = "你好"` | 新开一个储物柜，贴上"hello"标签，放进去"你好" |
| **查询** | `word_dict["apple"]` → "苹果" | 按照标签找到柜子，拿出里面的东西 |
| **修改** | `word_dict["apple"] = "苹果🍎"` | 找到贴"apple"标签的柜子，把里面的东西换成新的 |
| **删除** | `del word_dict["cat"]` | 把贴"cat"标签的柜子整个拆掉 |

**安全查询** —— 如果查的单词不存在怎么办？

```python
# 方式一：用 in 判断（推荐！）
if "orange" in word_dict:
    print(word_dict["orange"])
else:
    print("查无此词！")

# 方式二：用 .get() 方法
result = word_dict.get("orange", "未找到")  # 找不到就返回"未找到"
print(result)
```

**遍历字典**：

```python
# 遍历所有单词和翻译
for english, chinese in word_dict.items():
    print(f"{english} → {chinese}")

# 只遍历单词
for english in word_dict.keys():
    print(english)

# 只遍历翻译
for chinese in word_dict.values():
    print(chinese)
```

#### 2.2 文件读写基础

生活比喻：**文件操作就像用一个记事本**

| 操作 | 代码 | 比喻 |
|------|------|------|
| 打开（写模式） | `file = open("words.txt", "w")` | 翻开记事本，准备写字（w = write） |
| 写入内容 | `file.write("hello,你好\n")` | 在记事本上写一行 |
| 关闭文件 | `file.close()` | 合上记事本，保存好 |
| 打开（读模式） | `file = open("words.txt", "r")` | 翻开记事本，准备阅读（r = read） |
| 读取全部 | `content = file.read()` | 一口气读完整个记事本 |

**更好的写法 —— with 语句**（自动关闭文件，推荐使用）：

```python
# 写入文件
with open("words.txt", "w", encoding="utf-8") as f:
    for english, chinese in word_dict.items():
        f.write(f"{english},{chinese}\n")  # 用逗号分隔

# 读取文件
word_dict = {}
with open("words.txt", "r", encoding="utf-8") as f:
    for line in f:
        english, chinese = line.strip().split(",")
        word_dict[english] = chinese
```

`encoding="utf-8"` 很重要！它确保中文不会变乱码。

## 💻 代码示例

### 基础版：单词翻译本

```python
"""
单词翻译本 —— 基础版
功能：添加、查询、显示、保存单词
"""

# 预装单词（就像新字典自带的一些常用词）
word_dict = {
    "apple": "苹果",
    "banana": "香蕉",
    "cat": "猫",
    "dog": "狗",
    "hello": "你好",
    "book": "书"
}

print("=" * 35)
print("      📖 我的单词翻译本 📖")
print("=" * 35)

# 主循环
while True:
    # 显示菜单
    print("\n" + "─" * 25)
    print("  请选择功能：")
    print("  1️⃣  添加单词")
    print("  2️⃣  查询单词")
    print("  3️⃣  显示全部单词")
    print("  4️⃣  保存到文件")
    print("  5️⃣  从文件加载")
    print("  6️⃣  退出程序")
    print("─" * 25)
    
    choice = input("👉 请输入数字（1-6）：")
    
    if choice == "1":
        # ── 添加单词 ──
        english = input("请输入英文单词：").strip().lower()
        if english == "":
            print("⚠️  单词不能为空！")
            continue  # 跳回循环开头
        
        chinese = input("请输入中文翻译：").strip()
        if chinese == "":
            print("⚠️  翻译不能为空！")
            continue
        
        # 检查是否已存在
        if english in word_dict:
            print(f"⚠️  「{english}」已经存在，翻译为：{word_dict[english]}")
            confirm = input("是否覆盖？(y/n)：")
            if confirm.lower() != "y":
                print("已取消添加")
                continue
        
        word_dict[english] = chinese
        print(f"✅ 添加成功：{english} → {chinese}")
        
    elif choice == "2":
        # ── 查询单词 ──
        word = input("请输入要查询的英文单词：").strip().lower()
        if word in word_dict:
            print(f"🔍 {word} → {word_dict[word]}")
        else:
            print(f"❌ 抱歉，单词本里还没有「{word}」")
            add_now = input("要现在添加它吗？(y/n)：")
            if add_now.lower() == "y":
                chinese = input("请输入中文翻译：").strip()
                word_dict[word] = chinese
                print(f"✅ 已添加：{word} → {chinese}")
                
    elif choice == "3":
        # ── 显示全部单词 ──
        if len(word_dict) == 0:
            print("📭 单词本是空的，快去添加单词吧！")
        else:
            print(f"\n📋 单词本（共 {len(word_dict)} 个单词）")
            print("═" * 30)
            # 按字母顺序排列表格更好看
            for i, (eng, chn) in enumerate(
                sorted(word_dict.items()), 1
            ):
                print(f"  {i:2d}. {eng:<15s} → {chn}")
            print("═" * 30)
            
    elif choice == "4":
        # ── 保存到文件 ──
        filename = "my_words.txt"
        with open(filename, "w", encoding="utf-8") as f:
            for eng, chn in word_dict.items():
                f.write(f"{eng},{chn}\n")
        print(f"💾 成功保存 {len(word_dict)} 个单词到 {filename}！")
        
    elif choice == "5":
        # ── 从文件加载 ──
        filename = "my_words.txt"
        try:
            loaded = {}
            with open(filename, "r", encoding="utf-8") as f:
                for line in f:
                    line = line.strip()
                    if line:  # 跳过空行
                        eng, chn = line.split(",", 1)  # 只按第一个逗号分割
                        loaded[eng] = chn
            word_dict = loaded
            print(f"📂 成功加载！现在单词本有 {len(word_dict)} 个单词")
        except FileNotFoundError:
            print(f"❌ 文件 {filename} 不存在，请先保存单词本！")
            
    elif choice == "6":
        # ── 退出 ──
        print("\n👋 感谢使用单词翻译本，好好学习，天天向上！")
        break
        
    else:
        print("❓ 输入错误，请输入 1-6 之间的数字")
```

### 进阶版：带听写模式的单词翻译本

```python
"""
单词翻译本 —— 进阶版
新增：删除单词、听写模式、统计功能
"""

import random

word_dict = {
    "apple": "苹果",
    "banana": "香蕉",
    "cat": "猫",
    "dog": "狗",
    "hello": "你好",
    "book": "书",
    "sun": "太阳",
    "moon": "月亮"
}

print("=" * 35)
print("    📖 单词翻译本（进阶版）📖")
print("=" * 35)

while True:
    print("\n" + "─" * 25)
    print("  1. 添加单词    4. 删除单词")
    print("  2. 查询单词    5. 单词听写")
    print("  3. 显示全部    6. 保存/加载")
    print("  7. 退出")
    print("─" * 25)
    
    choice = input("👉 请选择：")
    
    if choice == "1":
        eng = input("英文：").strip().lower()
        chn = input("中文：").strip()
        if eng and chn:
            word_dict[eng] = chn
            print(f"✅ 已添加：{eng} → {chn}")
            
    elif choice == "2":
        word = input("查询单词：").strip().lower()
        result = word_dict.get(word, "未找到")
        print(f"🔍 {word} → {result}")
        
    elif choice == "3":
        if not word_dict:
            print("📭 单词本为空")
        else:
            print(f"\n📋 共 {len(word_dict)} 个单词：")
            for i, (e, c) in enumerate(sorted(word_dict.items()), 1):
                print(f"  {i}. {e} → {c}")
                
    elif choice == "4":
        word = input("要删除的单词：").strip().lower()
        if word in word_dict:
            del word_dict[word]
            print(f"🗑️  已删除「{word}」")
        else:
            print(f"❌ 未找到「{word}」")
            
    elif choice == "5":
        # ── 听写模式 ──
        if len(word_dict) < 3:
            print("⚠️  单词太少，至少需要 3 个单词才能听写！")
            continue
            
        print("\n🎧 单词听写模式！")
        print("规则：看到中文，写出英文")
        print("输入'结束'可以提前退出\n")
        
        words = list(word_dict.items())  # 转成列表
        random.shuffle(words)  # 打乱顺序
        
        score = 0
        total = min(5, len(words))  # 最多考5个
        
        for i in range(total):
            eng, chn = words[i]
            answer = input(f"第{i+1}题：「{chn}」的英文是？").strip().lower()
            
            if answer == "结束":
                print("提前结束听写")
                break
            elif answer == eng:
                print("  ✅ 正确！")
                score += 1
            else:
                print(f"  ❌ 错误！正确答案是：{eng}")
        
        print(f"\n📊 听写成绩：{score}/{total}")
        if score == total:
            print("🏆 满分！你太厉害了！")
        elif score >= total / 2:
            print("👍 不错，继续加油！")
        else:
            print("💪 再加把劲，多复习几遍！")
            
    elif choice == "6":
        print("  1. 保存  2. 加载")
        sub = input("  👉 选择：")
        if sub == "1":
            with open("my_words.txt", "w", encoding="utf-8") as f:
                for e, c in word_dict.items():
                    f.write(f"{e},{c}\n")
            print("💾 已保存！")
        elif sub == "2":
            try:
                word_dict.clear()
                with open("my_words.txt", "r", encoding="utf-8") as f:
                    for line in f:
                        if line.strip():
                            e, c = line.strip().split(",", 1)
                            word_dict[e] = c
                print(f"📂 已加载 {len(word_dict)} 个单词")
            except FileNotFoundError:
                print("❌ 文件不存在")
                
    elif choice == "7":
        print("👋 再见！记得常复习单词哦~")
        break
        
    else:
        print("❓ 请输入 1-7")
```
