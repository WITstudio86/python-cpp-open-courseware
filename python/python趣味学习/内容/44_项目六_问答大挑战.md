# 第44节：项目六 —— 问答大挑战

## 🎯 教学目标
- 学生能够使用列表+字典的组合结构存储题库数据
- 学生能够用 `for` 循环逐题出题并接收用户回答
- 学生能够实现答案判断、计分和反馈系统
- 学生能够根据得分给出个性化的评语
- 学生能够独立完成一个完整的问答游戏项目

## 📝 重点概念
- **列表+字典组合**：`[{"question": "...", "answer": "..."}, ...]`，列表里每个元素是一个字典，字典里存题目、答案等信息
- **遍历出题**：用 `for` 循环依次访问题库中的每一道题
- **答案判断**：比较用户输入和正确答案（注意大小写、去掉空格）
- **计分反馈**：答对加分 + 即时鼓励，答错告知正确答案
- **评语系统**：根据最终得分划分等级，给出对应评语

### 步骤2：核心讲解（15-20分钟）

#### 2.1 题库数据结构：列表+字典

```python
# 题库是一个列表，里面每道题是一个字典
questions = [
    {
        "question": "Python 是一种什么类型的语言？",
        "options": ["A. 编译型", "B. 解释型", "C. 标记型", "D. 机器型"],
        "answer": "B",
        "explanation": "Python 是解释型语言，代码一行一行地被解释执行，不需要先编译。"
    },
    {
        "question": "下面哪个关键字用来定义函数？",
        "options": ["A. func", "B. define", "C. def", "D. function"],
        "answer": "C",
        "explanation": "在 Python 中，用 def 关键字来定义函数，比如 def hello():。"
    },
    # ... 更多题目
]
```

为什么这样设计？
- **列表**保证题目有顺序，可以按顺序出题
- **字典**让每道题的信息井井有条：题目、选项、答案、解释各归各位
- 要加新题？往列表里 append 一个新字典就行，非常方便

#### 2.2 遍历出题

```python
for i, q in enumerate(questions, start=1):
    print(f"\n第 {i} 题：{q['question']}")
    
    # 显示选项
    for option in q['options']:
        print(f"  {option}")
    
    # 获取用户答案
    user_answer = input("请选择（A/B/C/D）：").strip().upper()
    
    # 判断对错
    if user_answer == q['answer']:
        print("✅ 正确！")
    else:
        print(f"❌ 错误！正确答案是 {q['answer']}")
```

`enumerate(questions, start=1)` 同时给出序号和内容。想象你拿着一本练习册，`enumerate` 帮你自动标好页码"第 1 页、第 2 页..."。

#### 2.3 答案判断技巧

用户输入可能千奇百怪："a"、"A"、" A "、"a "，所以要处理：

```python
user_answer = input("请选择：").strip().upper()
# .strip() 去掉首尾空格：" A " → "A"
# .upper() 转成大写："a" → "A"
# 这样不管用户输入 "a"、"A"、" A " 都能正确匹配
```

#### 2.4 计分系统

```python
score = 0  # 一开始 0 分

```

#### 2.5 评语系统

根据得分给出不同的评语，让学生感到有趣：

```python
def get_comment(score, total):
    """根据得分率返回评语"""
    rate = score / total  # 得分率（0.0 到 1.0）
    
    if rate == 1.0:
        return "🏆 满分！你是 Python 知识之王/女王！请收下我的膝盖！"
    elif rate >= 0.9:
        return "🌟 非常优秀！你对 Python 的了解已经相当厉害了！"
    elif rate >= 0.8:
        return "👍 很不错！你已经是一个合格的 Python 小达人了！"
    elif rate >= 0.7:
        return "📖 良好！继续加油，再多学一点就能更厉害！"
    elif rate >= 0.6:
        return "⚠️ 刚刚及格，还有很多知识等着你去探索哦！"
    elif rate >= 0.4:
        return "💪 还需要多多练习！失败是成功之母，别灰心！"
    else:
        return "🌱 刚刚起步呢！没关系，每个高手都是从零开始的！"

print(get_comment(score, total))
```

生活比喻：**评语系统就像游乐场的奖品柜**
- 得满分 → 抱走最大号的毛绒玩具
- 得 80 分 → 换一个中型奖品
- 得 60 分 → 一个小挂件
- 低于 40 分 → "继续加油"贴纸一张
- 不管多少分，都有鼓励！

questions = [

    {

        "answer": "C",

    },
    {
        "question": "下面哪个是合法的变量名？",

        "question": "列表 [1, 2, 3] 的长度是多少？",

    }
]


def run_quiz(questions):
    """运行问答游戏"""
    print("\n" + "=" * 45)
    print("        🧠 Python 知识问答大挑战！")
    print("=" * 45)
    print(f"  题库共有 {len(questions)} 道题，每题 1 分")
    print("  输入 A、B、C 或 D 来作答")
    print("=" * 45)
    
    score = 0  # 得分
    
    # 逐题出题
    for i, q in enumerate(questions, start=1):
        print(f"\n{'─' * 40}")
        print(f"📝 第 {i} 题（{score}/{i-1}）：{q['question']}")
        print(f"{'─' * 40}")
        
        # 显示选项
        for opt in q["options"]:
            print(f"    {opt}")
        
        # 获取用户答案（带输入验证）
        while True:
            user_answer = input("\n👉 你的答案：").strip().upper()
            if user_answer in ["A", "B", "C", "D"]:
                break
            print("⚠️  请输入 A、B、C 或 D 哦～")
        
        # 判断对错

            score += 1
            print("✅ 回答正确！+1 分！太厉害了！")
        else:
            print(f"❌ 回答错误！正确答案是 {q['answer']}")
        
        # 显示解释
        print(f"💡 {q['explanation']}")
        
        # 每题之间给一点喘息
        if i < len(questions):
            input("\n按回车键继续下一题...")
    
    # ── 最终结果 ──
    print("\n\n" + "=" * 45)
    print("              🏁 答题结束！")
    print("=" * 45)
    print(f"\n📊 最终得分：{score} / {len(questions)}")
    print(f"📈 正确率：{score / len(questions) * 100:.1f}%")
    
    # 评语
    rate = score / len(questions)
    print("\n" + "─" * 30)
    if rate == 1.0:
        print("🏆 评价：满分！你是 Python 知识天才！")
        print("   老师都忍不住要为你鼓掌！👏👏👏")
    elif rate >= 0.9:
        print("🌟 评价：非常优秀！你的 Python 学得特别好！")
        print("   再努力一把就是满分啦！")
    elif rate >= 0.8:
        print("👍 评价：很不错！已经是合格的 Python 小达人了！")
        print("   继续保持这份热情！")
    elif rate >= 0.7:
        print("📖 评价：良好！你的基础挺扎实的。")
        print("   有些地方还需要巩固一下哦～")
    elif rate >= 0.6:
        print("⚠️ 评价：刚好及格，还有很多知识等着你！")
        print("   翻翻之前的笔记，再多练习几遍吧！")
    elif rate >= 0.4:
        print("💪 评价：还需要加油！但这正是进步的开始。")

def get_comment(rate):
    """根据得分率返回评语"""
    if rate == 1.0:
        return "🏆 满分！你简直是知识百科全书的化身！"
    elif rate >= 0.9:
        return "🌟 几乎完美！你的知识储备量令人惊叹！"
    elif rate >= 0.8:
        return "👍 非常棒！你已经超越了大多数同龄人！"
    elif rate >= 0.6:
        return "📖 还不错！还有一些知识点等着你去攻克！"
    elif rate >= 0.4:
        return "💪 还有进步空间，多复习几遍这些知识点吧！"
    else:
        return "🌱 刚开始嘛，失败是成功的妈妈，继续加油！"


def run_advanced_quiz():
    """运行进阶版问答游戏"""
    print("\n" + "=" * 50)
    print("         🧠 Python 知识问答大挑战（进阶版）")
    print("=" * 50)
    
    # 选择难度
    print("\n请选择难度：")
    print("  1. 简单（基础语法）")
    print("  2. 中等（数据结构）")
    print("  3. 困难（综合应用）")
    print("  4. 混合（所有难度混合）")
    
    difficulty = input("👉 请选择（1-4）：").strip()
    
    pool = create_question_pool()
    
    if difficulty == "1":
        questions = pool["简单"]
        diff_name = "简单"
    elif difficulty == "2":
        questions = pool["中等"]
        diff_name = "中等"
    elif difficulty == "3":
        questions = pool["困难"]
        diff_name = "困难"
    elif difficulty == "4":
        questions = pool["简单"] + pool["中等"] + pool["困难"]
        diff_name = "混合"
    else:
        print("默认选择：混合模式")
        questions = pool["简单"] + pool["中等"] + pool["困难"]
        diff_name = "混合"
    
    # 打乱题目顺序
    random.shuffle(questions)
    
    print(f"\n📋 难度：{diff_name} | 共 {len(questions)} 道题")
    print("─" * 50)
    
    # 准备就绪
    input("按回车键开始答题...")
    
    # 计时开始
    start_time = time.time()
    
    score = 0
    wrong_list = []  # 错题本
    
    # 出题循环
    for i, q in enumerate(questions, start=1):
        print(f"\n{'─' * 45}")
        print(f"📝 第 {i}/{len(questions)} 题：{q['question']}")
        print(f"{'─' * 45}")
        
        for opt in q["options"]:
            print(f"    {opt}")
        
        # 获取答案
        while True:
            ans = input("\n👉 你的答案：").strip().upper()
            if ans in ["A", "B", "C", "D"]:
                break
            print("⚠️  请输入 A、B、C 或 D")
        
        # 判断
        if ans == q["answer"]:
            score += 1
            print("✅ 正确！+1 分！")
        else:
            print(f"❌ 错误！正确答案是 {q['answer']}")
            wrong_list.append(q)  # 加入错题本
        
        print(f"💡 {q['explanation']}")
    
    # 计时结束
    elapsed = time.time() - start_time
    
    # ── 结果展示 ──
    print("\n\n" + "=" * 50)
    print("              🏁 答题结束！")
    print("=" * 50)
    
    rate = score / len(questions)
    print(f"\n📊 难度：{diff_name}")
    print(f"📊 得分：{score} / {len(questions)}")
    print(f"📈 正确率：{rate * 100:.1f}%")
    print(f"⏱️  用时：{elapsed:.0f} 秒")
    print(f"\n{get_comment(rate)}")
    
    # 错题本
    if wrong_list:
        print(f"\n📕 错题本（{len(wrong_list)} 道需要复习）：")
        print("─" * 45)
        for i, q in enumerate(wrong_list, 1):
            print(f"\n{i}. {q['question']}")
            print(f"   正确答案：{q['answer']}")
            print(f"   {q['explanation']}")
    else:
        print("\n📗 错题本：空空如也！全部答对，太厉害了！")
    
    # 保存成绩
    save_choice = input("\n💾 是否保存成绩到文件？(y/n)：")
    if save_choice.lower() == "y":
        with open("quiz_result.txt", "w", encoding="utf-8") as f:
            f.write(f"问答大挑战成绩\n")
            f.write(f"难度：{diff_name}\n")
            f.write(f"得分：{score}/{len(questions)}\n")
            f.write(f"正确率：{rate*100:.1f}%\n")
            f.write(f"用时：{elapsed:.0f} 秒\n")
        print("✅ 成绩已保存！")
    
    print("\n🙏 感谢参与！记得复习错题本哦～")


if __name__ == "__main__":
    run_advanced_quiz()
