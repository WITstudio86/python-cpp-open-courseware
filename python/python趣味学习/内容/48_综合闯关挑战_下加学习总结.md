#### 2.2 猜数字游戏的算法思路

"猜数字游戏的核心逻辑很简单，来看这个流程图："

```
1. 生成秘密数字（1-100）
2. 设定剩余次数 = 7
3. 创建空列表记录猜历史
4. 循环（当剩余次数 > 0）：
   a. 显示还剩几次机会
   b. 询问玩家猜什么数字
   c. 如果猜对了 → "恭喜！" → 结束游戏
   d. 如果猜大了 → "大了！小一点"
   e. 如果猜小了 → "小了！大一点"
   f. 剩余次数 - 1
   g. 记录本次猜测
5. 如果次数用完了还没猜中 → 公布答案
```

"这个流程里最重要的就是 while 循环——因为我们不确定玩家猜几次能中，只知道最多 7 次。"

```python
import random

def guessing_game():
    """猜数字游戏：7次机会猜中1-100的随机数"""
    print()
    print("-" * 50)
    print("  🎯 子任务A：猜数字挑战")
    print("-" * 50)
    
    secret = random.randint(1, 100)
    chances = 7
    history = []  # 记录猜过的数字
    
    print("  我想了一个 1-100 之间的数字，你能猜到吗？")
    print(f"  你有 {chances} 次机会哦！")
    print()
    
    while chances > 0:
        print(f"⏳ 剩余机会：{chances}")
        
        try:
            guess = int(input("  你猜的数字是："))
        except ValueError:
            print("  ⚠️ 请输入一个整数！")
            continue
        
        history.append(guess)
        
        if guess == secret:
            print()
            print(f"  🎉🎉🎉 恭喜你猜对了！答案就是 {secret}！")
            print(f"  你总共猜了 {len(history)} 次")
            print(f"  你的猜历史：{history}")
            print()
            return True
        
        elif guess > secret:
            print("  📉 大了！往小猜猜")
        else:
            print("  📈 小了！往大猜猜")
        
        chances -= 1
        print()
    
    # 机会用完
    print(f"  😢 机会用完了！答案是 {secret}")
    print(f"  你的猜历史：{history}")
    print()
    return False
```

```python
def score_analyzer():
    """成绩分析小程序"""
    print()
    print("-" * 50)
    print("  📊 子任务B：成绩分析")
    print("-" * 50)
    print()
    
    scores = []
    num_students = 5
    
    print(f"  请输入 {num_students} 个同学的成绩：")
    for i in range(num_students):
        while True:
            try:
                score = float(input(f"  第{i+1}个同学的成绩："))
                if 0 <= score <= 100:
                    scores.append(score)
                    break
                else:
                    print("    ⚠️ 成绩应在 0-100 之间！")
            except ValueError:
                print("    ⚠️ 请输入一个数字！")
    
    print()
    print("=" * 50)
    print("  📊 成绩分析报告")
    print("=" * 50)
    
    # 计算各项指标
    average = sum(scores) / len(scores)
    highest = max(scores)
    lowest = min(scores)
    
    # 排序
    sorted_scores = sorted(scores, reverse=True)
    
    # 评级
    if average >= 90:
        grade = "🌟 优秀"
    elif average >= 70:
        grade = "👍 良好"
    elif average >= 60:
        grade = "📚 及格"
    else:
        grade = "💪 需要努力"
    
    print(f"  原始成绩：{scores}")
    print(f"  排名（从高到低）：{sorted_scores}")
    print(f"  平均分：{average:.1f}")
    print(f"  最高分：{highest}")
    print(f"  最低分：{lowest}")
    print(f"  整体评级：{grade}")
    print("=" * 50)
    print()
    
    return True
```
