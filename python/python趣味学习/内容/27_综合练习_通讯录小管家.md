# 第27节：综合练习：通讯录小管家

## 🎯 教学目标
- 学生能够将列表和字典组合使用（列表中存放字典）
- 学生能够遍历列表中的字典进行查找
- 学生能够对通讯录进行添加、查找、删除、显示操作
- 学生能够使用 while 循环菜单构建交互式程序

## 📝 重点概念
- **嵌套结构**：列表里放字典，每个字典代表一条记录
- **菜单驱动**：while True + if/elif 实现功能选择
- **综合运用**：列表的增删遍历 + 字典的键值访问

## 📖 讲解内容

### 步骤1：引入（5分钟）

"我们已经学了列表和字典。列表能装很多东西，字典能存储成对的信息。那如果把字典放到列表里呢？"

**场景**：手机通讯录！
- 每个联系人是一个字典：`{"姓名": "小明", "电话": "13800138000"}`
- 整个通讯录是一个列表：`[{...}, {...}, {...}]`

"今天我们就用列表+字典做一个真正的通讯录程序！"

### 步骤2：核心讲解（15-20分钟）

#### 2.1 列表里放字典

```python
# 每个联系人是一个字典
contact1 = {"姓名": "小明", "电话": "13800138001"}
contact2 = {"姓名": "小红", "电话": "13800138002"}

# 通讯录是包含多个字典的列表
contacts = [contact1, contact2]

# 更常见的写法：直接写
contacts = [
    {"姓名": "小明", "电话": "13800138001"},
    {"姓名": "小红", "电话": "13800138002"},
    {"姓名": "小刚", "电话": "13800138003"}
]
```

#### 2.2 遍历通讯录

```python
contacts = [
    {"姓名": "小明", "电话": "13800138001"},
    {"姓名": "小红", "电话": "13800138002"},
]

for contact in contacts:
    print(f"姓名：{contact['姓名']}，电话：{contact['电话']}")
```

#### 2.3 查找联系人

```python
def search_contact(contacts, name):
    """根据姓名查找联系人"""
    for contact in contacts:
        if contact["姓名"] == name:
            return contact
    return None  # 没找到

result = search_contact(contacts, "小明")
if result:
    print("找到了！电话：", result["电话"])
else:
    print("通讯录中没有这个人。")
```

#### 2.4 添加和删除

```python
# 添加联系人
new_contact = {"姓名": "新朋友", "电话": "13900001111"}
contacts.append(new_contact)

# 删除联系人
name_to_delete = "小明"
for i, contact in enumerate(contacts):
    if contact["姓名"] == name_to_delete:
        contacts.pop(i)
        print("删除成功！")
        break
```

### 步骤3：动手演示（10分钟）

**通讯录小管家完整程序**：

```python
contacts = []

while True:
    print("\n=== 📞 通讯录小管家 ===")
    print("1. 添加联系人")
    print("2. 查找联系人")
    print("3. 显示全部联系人")
    print("4. 删除联系人")
    print("5. 退出")
    
    choice = input("请选择 (1-5): ")
    
    if choice == "1":
        name = input("姓名: ")
        phone = input("电话: ")
        contacts.append({"姓名": name, "电话": phone})
        print("✅ 添加成功！")
    
    elif choice == "2":
        name = input("输入要查找的姓名: ")
        found = None
        for c in contacts:
            if c["姓名"] == name:
                found = c
                break
        if found:
            print(f"📞 {found['姓名']}: {found['电话']}")
        else:
            print("❌ 未找到该联系人。")
    
    elif choice == "3":
        if len(contacts) == 0:
            print("通讯录为空。")
        else:
            for i, c in enumerate(contacts, 1):
                print(f"  {i}. {c['姓名']}: {c['电话']}")
    
    elif choice == "4":
        name = input("输入要删除的联系人姓名: ")
        deleted = False
        for i, c in enumerate(contacts):
            if c["姓名"] == name:
                contacts.pop(i)
                deleted = True
                break
        if deleted:
            print("✅ 删除成功！")
        else:
            print("❌ 未找到该联系人。")
    
    elif choice == "5":
        print("再见！")
        break
```

## 💻 代码示例

```python
# ========== 第27节：综合练习：通讯录小管家 ==========

# --- 1. 数据结构：列表里放字典 ---
print("=== 通讯录数据结构 ===")

contacts = [
    {"姓名": "小明", "电话": "13800138001", "关系": "同学"},
    {"姓名": "小红", "电话": "13800138002", "关系": "朋友"},
    {"姓名": "王老师", "电话": "13800138003", "关系": "老师"},
]

print(f"通讯录共有 {len(contacts)} 个联系人")
for c in contacts:
    print(f"  {c['姓名']:6s} | {c['电话']:15s} | {c['关系']}")

# --- 2. 查找联系人 ---
print("\n=== 查找联系人 ===")

def find_contact(contacts, name):
    """在通讯录中查找联系人，返回该联系人的字典，找不到返回 None"""
    for contact in contacts:
        if contact["姓名"] == name:
            return contact
    return None

# 测试查找
result = find_contact(contacts, "小明")
if result:
    print(f"找到了！{result['姓名']}的电话是 {result['电话']}")
else:
    print("没有这个人。")

result = find_contact(contacts, "小李")
if result:
    print(f"找到了！{result['姓名']}的电话是 {result['电话']}")
else:
    print("通讯录中没有'小李'")

# --- 3. 综合项目：通讯录小管家 ---
print("\n" + "=" * 40)
print("    📞 通讯录小管家")
print("=" * 40)

# 预置几个联系人
contacts = [
    {"姓名": "小明", "电话": "13800138001"},
    {"姓名": "小红", "电话": "13800138002"},
]

while True:
    print("\n--- 功能菜单 ---")
    print("  1️⃣  添加联系人")
    print("  2️⃣  查找联系人")
    print("  3️⃣  显示全部联系人")
    print("  4️⃣  删除联系人")
    print("  5️⃣  修改联系人")
    print("  6️⃣  退出程序")
    print("-" * 20)
    
    choice = input("请选择功能 (1-6): ").strip()
    
    if choice == "1":
        # 添加联系人
        print("\n--- 添加联系人 ---")
        name = input("请输入姓名: ").strip()
        phone = input("请输入电话: ").strip()
        
        # 检查是否已存在
        existing = find_contact(contacts, name)
        if existing:
            print(f"⚠️ '{name}' 已存在！电话: {existing['电话']}")
            confirm = input("是否覆盖？(y/n): ").strip().lower()
            if confirm == "y":
                existing["电话"] = phone
                print("✅ 已更新！")
        else:
            contacts.append({"姓名": name, "电话": phone})
            print(f"✅ 已添加 {name}！")
    
    elif choice == "2":
        # 查找联系人
        print("\n--- 查找联系人 ---")
        name = input("请输入要查找的姓名: ").strip()
        result = find_contact(contacts, name)
        if result:
            print(f"📞 找到联系人：")
            print(f"   姓名: {result['姓名']}")
            print(f"   电话: {result['电话']}")
        else:
            print(f"❌ 通讯录中没有「{name}」")
    
    elif choice == "3":
        # 显示全部
        print("\n--- 全部联系人 ---")
        if len(contacts) == 0:
            print("📭 通讯录是空的，快去添加联系人吧！")
        else:
            print(f"共有 {len(contacts)} 个联系人：\n")
            print(f"{'序号':<6}{'姓名':<10}{'电话':<15}")
            print("-" * 31)
            for i, c in enumerate(contacts, 1):
                print(f"{i:<6}{c['姓名']:<10}{c['电话']:<15}")
    
    elif choice == "4":
        # 删除联系人
        print("\n--- 删除联系人 ---")
        name = input("请输入要删除的联系人姓名: ").strip()
        deleted = False
        for i, c in enumerate(contacts):
            if c["姓名"] == name:
                confirm = input(f"确认删除「{name}」？(y/n): ").strip().lower()
                if confirm == "y":
                    contacts.pop(i)
                    print(f"✅ 已删除「{name}」")
                else:
                    print("已取消。")
                deleted = True
                break
        if not deleted:
            print(f"❌ 通讯录中没有「{name}」")
    
    elif choice == "5":
        # 修改联系人
        print("\n--- 修改联系人 ---")
        name = input("请输入要修改的联系人姓名: ").strip()
        result = find_contact(contacts, name)
        if result:
            print(f"当前信息：{result['姓名']} - {result['电话']}")
            new_phone = input("请输入新电话（直接回车保持不变）: ").strip()
            if new_phone:
                result["电话"] = new_phone
                print("✅ 修改成功！")
            else:
                print("未做修改。")
        else:
            print(f"❌ 通讯录中没有「{name}」")
    
    elif choice == "6":
        print("再见！👋")
        break
    
    else:
        print("⚠️ 请输入 1-6 之间的数字。")
```

## ✏️ 课堂练习

### 基础练习
1. **简单通讯录**：创建包含 3 个联系人的通讯录（列表+字典），遍历并打印每个联系人的信息。
2. **查找功能**：在上面通讯录中，让用户输入姓名，查找并显示该联系人的电话。
3. **添加功能**：让用户输入新联系人的姓名和电话，添加到通讯录中。

### 挑战练习
1. **完整通讯录**：实现完整的通讯录程序，包含添加、查找、显示、删除、修改五个功能，用 while 循环驱动菜单。
2. **增强通讯录**：给每个联系人增加"关系"（家人/朋友/同学）和"生日"字段，实现按关系筛选显示的功能。
