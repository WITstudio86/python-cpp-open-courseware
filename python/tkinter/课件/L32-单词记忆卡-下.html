def change_category(*args):
    """切换分类时的处理：重置所有状态"""

    cat = category_var.get()

    unmastered = []

def next_card():

current_word_list = []     # 当前分类下的所有单词列表
is_front = True            # 正面（英文）= True / 背面（中文）= False

mastered = []              # 已掌握的单词列表
unmastered = []            # 未掌握的单词列表

# ========================================
# 4. 功能函数
# ========================================

def change_category(*args):
    """切换分类时的处理：重置所有状态"""
    global current_word_list, mastered, unmastered

    cat = category_var.get()
    current_word_list = list(word_bank[cat].keys())  # 获取该分类的所有单词

    # 重置掌握/未掌握列表
    mastered = []
    unmastered = []

    # 更新统计
    update_stats()

    # 抽取第一张卡片
    next_card()


def next_card():
    """抽取下一张记忆卡"""
    global current_word, current_meaning, is_front

    if not current_word_list:
        return

    # 根据复习模式决定从哪个池子里抽词
    if review_var.get() == "unmastered" and unmastered:
        # "只复习未掌握"模式：从未掌握的单词中筛选属于当前分类的
        pool = [w for w in unmastered if w in current_word_list]
        if not pool:
            # 如果当前分类的单词都已掌握，就全部随机
            pool = current_word_list
    else:
        pool = current_word_list

    # 随机选一个单词
    current_word = random.choice(pool)
    current_meaning = all_words[current_word]

    # 显示正面（英文）
    is_front = True
    word_label.config(text=current_word, fg="#3a2a1a")
    flip_btn.config(text="🔍 翻面看释义")

    # 更新状态指示
    update_status_display()


def flip():
    """翻面：切换正/反面显示"""
    global is_front

    if is_front:
        # 翻到背面 → 显示中文释义
        word_label.config(text=current_meaning, fg="#8b4513")
        flip_btn.config(text="↩ 翻回正面")
        is_front = False
    else:
        # 翻回正面 → 显示英文单词
        word_label.config(text=current_word, fg="#3a2a1a")
        flip_btn.config(text="🔍 翻面看释义")
        is_front = True


def mark(status):
    """
    标记当前单词的掌握状态
    status: "mastered" = 已掌握, "unmastered" = 未掌握
    """
    if status == "mastered":
        if current_word not in mastered:
            mastered.append(current_word)
        if current_word in unmastered:
            unmastered.remove(current_word)
    else:
        if current_word not in unmastered:
            unmastered.append(current_word)
        if current_word in mastered:
            mastered.remove(current_word)

    # 更新统计和状态显示
    update_stats()
    update_status_display()

    # 自动跳到下一张卡片
    next_card()


def update_status_display():
    """更新当前单词的状态指示"""
    if current_word in mastered:
        status_label.config(text="✅ 已掌握", fg="#27ae60")
    elif current_word in unmastered:
        status_label.config(text="❌ 未掌握", fg="#e74c3c")
    else:
        status_label.config(
            text=f"📚 共 {len(current_word_list)} 个单词",
            fg="#aaa"
        )


def update_stats():
    """更新学习统计显示"""
    total = len(current_word_list)
    # 计算当前分类中已掌握的单词数
    learned = len([w for w in mastered if w in current_word_list])
    percent = int(learned / total * 100) if total > 0 else 0

    stats_label.config(
        text=f"📊 已掌握 {learned} / {total}（{percent}%）"
    )

改错时刻

这三处最容易错

❌ command=say_hello()

✅ command=say_hello
加了括号，函数会立刻执行，点击时反而不动。

❌ geometry("500*400")

✅ geometry("500x400")
中间必须是英文字母 x。

❌ 函数里直接 count += 1

✅ 先写 global count
漏了 global，改不到外面的变量。

这一步运行后记住的坑
创建主窗口标题栏换成了这一步写的文字。窗口大小写成 "宽x高"，中间是英文字母 x，不是星号。
功能函数窗口还能打开。绿底这些新行已经出现在窗口里。函数里要改外面的变量，第一行先写 global。漏了会报 UnboundLocalError。
mark窗口还能打开。绿底这些新行已经出现在窗口里。def 下面的函数体必须缩进。少缩进一格，运行就会报错。
界面布局控件排队出现在窗口里，不再叠在同一个点。同一个父容器里不要又 pack 又 grid，否则会直接报错。
分类选择行窗口顶部多了菜单，点开能看到命令。IntVar、StringVar 必须写在 Tk() 之后，不然会报太早创建变量。
复习模式选择出现一组单选，一次只能选中一个。command 后面写函数名，不要加括号。加了括号会马上执行。
卡片区域控件排队出现在窗口里，不再叠在同一个点。同一个父容器里不要又 pack 又 grid，否则会直接报错。
