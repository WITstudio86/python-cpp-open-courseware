import tkinter as tk

# ========================================
# 第 16 节：涂鸦板 —— 用鼠标画出你的作品
# ========================================

root = tk.Tk()
root.title("涂鸦板")
root.geometry("550x550")
root.configure(bg="#f0f0f0")

# ---------- 画笔设置 ----------
brush_color = "black"   # 当前画笔颜色，默认黑色
brush_size = 4          # 当前画笔大小（直径，单位：像素）

# ---------- 创建画布 ----------
# Canvas 是 tkinter 的绘图区域，可以在上面画圆、线、矩形等
# width/height 设置画布大小，bg 设置背景色为白色
canvas = tk.Canvas(root, width=500, height=400, bg="white",
                   highlightthickness=1, highlightbackground="#cccccc")
canvas.pack(pady=(10, 5))

# ---------- 鼠标事件处理 ----------
def draw(event):
    """
    鼠标拖拽时的绘图函数
    当用户按住鼠标左键并在画布上移动时，tkinter 会自动调用此函数
    event 参数包含鼠标的坐标信息：
      - event.x：鼠标在画布上的水平坐标（从左到右增大）
      - event.y：鼠标在画布上的垂直坐标（从上到下增大）
    """
    r = brush_size / 2  # 圆的半径 = 画笔大小的一半
    # create_oval(x1, y1, x2, y2) 画一个椭圆
    # (x1, y1) 是椭圆外接矩形的左上角，(x2, y2) 是右下角
    # 宽高相等时椭圆就是正圆，用来模拟画笔的笔触
    canvas.create_oval(
        event.x - r, event.y - r,  # 左上角：以鼠标位置为中心向左上偏移 r
        event.x + r, event.y + r,  # 右下角：以鼠标位置为中心向右下偏移 r
        fill=brush_color,           # 圆的填充颜色
        outline=brush_color         # 圆的边框颜色（与填充色一致，看不出边框）
    )

# 将 <B1-Motion> 事件绑定到 draw 函数
# 含义：当用户"按住鼠标左键（B1）并在画布上拖动（Motion）"时，调用 draw
# tkinter 会自动把鼠标坐标通过 event 参数传给 draw 函数
canvas.bind("<B1-Motion>", draw)

# ---------- 启动主循环 ----------
# mainloop() 让窗口持续显示，监听用户的鼠标操作
root.mainloop()
运行后你该看到： 画布上多出新的图形。
坑： bind 的事件名要带尖括号，例如 "<Return>"、"<Button-1>"。

         bg="#f0f0f0", fg="#333333").grid(row=0, column=0, padx=(0, 5))

# 颜色列表：每个元素是 (按钮上显示的文字, 实际颜色值)
colors = [
    ("⚫ 黑", "black"), ("🔴 红", "red"), ("🔵 蓝", "blue"),
    ("🟢 绿", "green"), ("🟠 橙", "orange"), ("🟣 紫", "purple")
]

def change_color(color):
    """切换画笔颜色"""
    global brush_color
    brush_color = color

# 为每种颜色创建一个按钮
for i, (text, color) in enumerate(colors):
    # lambda c=color: change_color(c)
    # ⚠️ 重要：必须用 c=color 来捕获当前循环的 color 值
    # 如果写成 lambda: change_color(color)，所有按钮都会使用循环最后的颜色值
    btn = tk.Button(color_frame, text=text,
                    command=lambda c=color: change_color(c),
                    font=("Microsoft YaHei", 10),
                    bg="#ffffff", fg="#333333",
                    relief="flat", padx=8, pady=3,
                    cursor="hand2")
    btn.grid(row=0, column=i+1, padx=3)

                      font=("Microsoft YaHei", 12, "bold"),
                      bg="#ffffff", fg="#333333",
                      width=3, relief="groove")
size_label.grid(row=0, column=1, padx=5)

def increase_size():
    """增大画笔尺寸"""
    global brush_size
    if brush_size < 20:  # 限制最大尺寸为 20
        brush_size += 2
        size_label.config(text=str(brush_size))  # 更新标签上的数字

def decrease_size():
    """减小画笔尺寸"""
    global brush_size
    if brush_size > 2:  # 限制最小尺寸为 2
        brush_size -= 2
        size_label.config(text=str(brush_size))

# 变小按钮
small_btn = tk.Button(size_frame, text="➖ 变小",
                      command=decrease_size,
                      font=("Microsoft YaHei", 10),
                      bg="#ffffff", fg="#333333",
                      relief="flat", padx=10, pady=3,
                      cursor="hand2")
small_btn.grid(row=0, column=2, padx=5)

# 变大按钮
big_btn = tk.Button(size_frame, text="➕ 变大",
                    command=increase_size,
                    font=("Microsoft YaHei", 10),
                    bg="#ffffff", fg="#333333",
                    relief="flat", padx=10, pady=3,
                    cursor="hand2")
big_btn.grid(row=0, column=3, padx=5)

改错时刻

这三处最容易错

❌ command=say_hello()

✅ command=say_hello
加了括号，函数会立刻执行，点击时反而不动。

❌ geometry("500*400")

✅ geometry("500x400")
中间必须是英文字母 x。

❌ 函数里直接 count += 1

✅ 先写 global count
漏了 global，改不到外面的变量。
