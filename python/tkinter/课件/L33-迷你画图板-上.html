# 画笔颜色和粗细
pen_color = "black"
pen_width = 2

# 铅笔工具用的「上一个点」坐标
last_x = None
last_y = None

# 直线工具用的「起始点」坐标

def on_mouse_press(event):

    global last_x, last_y, start_x, start_y

    current_tool = tool.get()

    if current_tool == "pencil":
        # 铅笔：记录起始点，准备开始画线
        last_x = event.x
        last_y = event.y

    elif current_tool == "line":
        # 直线：记录起点
        start_x = event.x
        start_y = event.y


def on_mouse_drag(event):
    """鼠标拖拽（按住移动）时的处理"""
    global last_x, last_y, preview_line_id

    current_tool = tool.get()

    if current_tool == "pencil":
        # 铅笔：从上一点到当前点画一条小线段
        if last_x is not None and last_y is not None:
            canvas.create_line(
                last_x, last_y,           # 起点（上一个位置）
                event.x, event.y,         # 终点（当前位置）
                fill=pen_color,
                width=pen_width,
                capstyle=tk.ROUND,        # 圆头线端，让笔迹更流畅
                smooth=True               # 平滑处理
            )
            # 更新「上一个点」为当前位置
            last_x = event.x
            last_y = event.y

            start_x, start_y,
            event.x, event.y,
            fill="gray",
            dash=(5, 5),   # 虚线样式：5px 实线 + 5px 空白
            width=1
        )

改错时刻

这三处最容易错

❌ command=say_hello()

✅ command=say_hello
加了括号，函数会立刻执行，点击时反而不动。

❌ geometry("500*400")

✅ geometry("500x400")
中间必须是英文字母 x。

❌ 函数里直接 count += 1

✅ 先写 global count
漏了 global，改不到外面的变量。

这一步运行后记住的坑
创建主窗口标题栏换成了这一步写的文字。窗口大小写成 "宽x高"，中间是英文字母 x，不是星号。
状态变量窗口还能打开。绿底这些新行已经出现在窗口里。IntVar、StringVar 必须写在 Tk() 之后，不然会报太早创建变量。
鼠标事件处理函数窗口还能打开。绿底这些新行已经出现在窗口里。函数里要改外面的变量，第一行先写 global。漏了会报 UnboundLocalError。
on_mouse_drag画布上多出新的图形。函数里要改外面的变量，第一行先写 global。漏了会报 UnboundLocalError。
辅助功能窗口还能打开。绿底这些新行已经出现在窗口里。函数里要改外面的变量，第一行先写 global。漏了会报 UnboundLocalError。
界面布局控件排队出现在窗口里，不再叠在同一个点。同一个父容器里不要又 pack 又 grid，否则会直接报错。
工具栏出现一组单选，一次只能选中一个。同一个父容器里不要又 pack 又 grid，否则会直接报错。
颜色选择标签窗口里多了按钮。点一下，看它有没有反应。command 后面写函数名，不要加括号。加了括号会马上执行。
Canvas 画布窗口里多了一块画布。bind 的事件名要带尖括号，例如 "<Return>"、"<Button-1>"。
