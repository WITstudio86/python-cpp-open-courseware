# 大字体的时间显示——科技感拉满！
time_label = tk.Label(
    root,
    text="00:00:00",
    font=("Consolas", 72, "bold"),  # 等宽字体，72号，加粗
    fg="#00FF00",                       # 亮绿色文字（前景色）
    bg="#1a1a2e"                         # 深色背景
)
💡 等宽字体：每个数字宽度一样，时钟跳秒时数字不会左右晃动！推荐 Consolas、Menlo、Courier New。

颜色切换 + 12/24 小时制

两个按钮，两种玩法——让时钟更有个性！

# 颜色循环：列表 + 取余（%）实现无限循环
colors = ["#00FF00", "#FF6B6B", "#4ECDC4", "#FFE66D"]
color_index = 0

def change_color():
    global color_index
    color_index = (color_index + 1) % len(colors)
    time_label.config(fg=colors[color_index])

# 12/24 小时制：IntVar 存模式，动态选格式
is_24h = tk.IntVar(value=1)

def toggle_mode():
    is_24h.set(1 - is_24h.get())

def update_time():
    fmt = "%H" if is_24h.get() else "%I"
    now = time.strftime(f"{fmt}:%M:%S")
    time_label.config(text=now)
    root.after(1000, update_time)

🔄 取余 % 的循环魔法

(3+1) % 4 = 0，走到末尾自动回到开头。这就是颜色能无限循环的秘密！

🕐 %H vs %I

%H = 0~23（24 小时）
%I = 1~12（12 小时）
下午 3 点 → %H 显示 15，%I 显示 03
