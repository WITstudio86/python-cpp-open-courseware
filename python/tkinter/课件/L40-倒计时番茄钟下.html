# ==================== 核心函数 ====================

def format_time(seconds):
    """把秒数格式化为 MM:SS 字符串
    例如：1500 → "25:00"，90 → "01:30"
    """
    m = seconds // 60              # 整除得到分钟数
    s = seconds % 60               # 取余得到秒数
    return f"{m:02d}:{s:02d}"     # :02d 表示至少两位，不足前面补 0


def countdown():
    """倒计时核心函数：每秒减 1 秒，直到 0
    使用 after() 定时循环，不会阻塞窗口
    """
    global remaining, is_running, after_id

    # 如果被暂停了，不再继续（退出递归）
    if not is_running:
        return

    if remaining > 0:
        remaining -= 1             # 减 1 秒
        time_label.config(text=format_time(remaining))
        # 1 秒（1000 毫秒）后再次调用自己
        after_id = root.after(1000, countdown)
    else:
        # 时间到！
        time_up()


def time_up():
    """番茄钟时间到的处理：响铃 + 统计 + 弹窗 + 恢复状态"""
    global is_running, is_paused, remaining, today_count

    # 重置运行状态
    is_running = False
    is_paused = False
    remaining = total_seconds
    time_label.config(text=format_time(remaining))

    # 番茄数 +1
    today_count += 1
    count_label.config(text=f"🍅 今日番茄：{today_count} 个")

    # 播放提示音
    play_sound()

    # 弹出完成提示窗口

        f"「{task_name}」完成！\n\n"
        f"今日已完成 {today_count} 个番茄 🍅\n"
        "休息 5 分钟，准备下一个吧！"
    )

    # 恢复按钮状态
    start_btn.config(state="normal")
    pause_btn.config(state="disabled")
    task_label.config(text="📌 等待开始...")

def play_sound():
    """播放完成提示音（跨平台兼容）
    Windows：使用 winsound.Beep() 蜂鸣
    Mac/Linux：使用 root.bell() 系统响铃
    """
    if sys.platform == "win32":
        # Windows 系统
        try:
            import winsound
            # 连续蜂鸣 3 次：频率 1000Hz，持续 300 毫秒
            for _ in range(3):
                winsound.Beep(1000, 300)
        except:
            # 如果 winsound 不可用，回退到 bell()
            root.bell()
    else:
        # Mac / Linux 系统
        # 连续响铃 3 次
        for _ in range(3):
            root.bell()
            root.update()          # 刷新事件，让 bell 生效


def start():
    """开始倒计时（全新开始 / 从暂停恢复）"""
    global is_running, is_paused, remaining

    # 获取并显示任务名称
    task_name = task_entry.get().strip()
    # 如果用户没输入或还是默认提示文字，给个默认名称
    if task_name == "" or task_name == "例如：背古诗":
        task_name = "专注中..."
    task_label.config(text=f"📌 {task_name}")

    if is_paused:
        # 从暂停恢复，不重置时间
        is_paused = False
    else:
        # 全新开始，重置时间到 25:00
        remaining = total_seconds
        time_label.config(text=format_time(remaining))

    is_running = True

    # 更新按钮状态
    start_btn.config(state="disabled")       # 运行中，开始按钮禁用
    pause_btn.config(text="⏸️ 暂停", state="normal")  # 启用暂停按钮

    countdown()  # 启动倒计时循环


def pause():
    """暂停 / 继续切换"""
    global is_running, is_paused

    if is_paused:
        # 当前已暂停 → 继续运行
        is_paused = False
        is_running = True
        pause_btn.config(text="⏸️ 暂停")
        countdown()  # 重新开始倒计时循环
    else:
        # 当前运行中 → 暂停
        is_paused = True
        is_running = False   # 停止倒计时循环
        pause_btn.config(text="▶️ 继续")


def reset():
    """重置番茄钟：恢复时间、清除状态、恢复按钮"""
    global is_running, is_paused, remaining, after_id

    is_running = False
    is_paused = False
    remaining = total_seconds

    # 取消计划中的 after() 调用（重要！否则会继续倒计时）
    if after_id is not None:
        root.after_cancel(after_id)
        after_id = None

    # 恢复界面
    time_label.config(text=format_time(remaining))
    task_label.config(text="📌 等待开始...")

    # 恢复按钮状态
    start_btn.config(state="normal")         # 开始按钮恢复可用
    pause_btn.config(text="⏸️ 暂停", state="disabled")  # 暂停按钮禁用

# ==================== 核心函数 ====================

def format_time(seconds):
    """把秒数格式化为 MM:SS 字符串
    例如：1500 → "25:00"，90 → "01:30"
    """
    m = seconds // 60              # 整除得到分钟数
    s = seconds % 60               # 取余得到秒数
    return f"{m:02d}:{s:02d}"     # :02d 表示至少两位，不足前面补 0


def countdown():
    """倒计时核心函数：每秒减 1 秒，直到 0
    使用 after() 定时循环，不会阻塞窗口
    """
    global remaining, is_running, after_id

    # 如果被暂停了，不再继续（退出递归）
    if not is_running:
        return

    if remaining > 0:
        remaining -= 1             # 减 1 秒
        time_label.config(text=format_time(remaining))
        # 1 秒（1000 毫秒）后再次调用自己
        after_id = root.after(1000, countdown)
    else:
        # 时间到！
        time_up()


def time_up():
    """番茄钟时间到的处理：响铃 + 统计 + 弹窗 + 恢复状态"""
    global is_running, is_paused, remaining, today_count

    # 重置运行状态
    is_running = False
    is_paused = False
    remaining = total_seconds
    time_label.config(text=format_time(remaining))

    # 番茄数 +1
    today_count += 1
    count_label.config(text=f"🍅 今日番茄：{today_count} 个")

    # 播放提示音
    play_sound()

    # 弹出完成提示窗口
    task_name = task_entry.get().strip() or "专注中..."
    messagebox.showinfo(
        "⏰ 时间到！",
        f"「{task_name}」完成！\n\n"
        f"今日已完成 {today_count} 个番茄 🍅\n"
        "休息 5 分钟，准备下一个吧！"
    )

    # 恢复按钮状态
    start_btn.config(state="normal")
    pause_btn.config(state="disabled")
    task_label.config(text="📌 等待开始...")


def play_sound():
    """播放完成提示音（跨平台兼容）
    Windows：使用 winsound.Beep() 蜂鸣
    Mac/Linux：使用 root.bell() 系统响铃

    if sys.platform == "win32":
        # Windows 系统

            import winsound
            # 连续蜂鸣 3 次：频率 1000Hz，持续 300 毫秒

            root.bell()
            root.update()          # 刷新事件，让 bell 生效

这三处最容易错

❌ command=say_hello()

✅ command=say_hello
加了括号，函数会立刻执行，点击时反而不动。

❌ geometry("500*400")

✅ geometry("500x400")
中间必须是英文字母 x。

❌ 函数里直接 count += 1

✅ 先写 global count
漏了 global，改不到外面的变量。
