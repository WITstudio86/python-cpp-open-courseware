import tkinter as tk

# 第 5 节：综合小练 —— 打招呼机器人

root = tk.Tk()
root.title("打招呼机器人 🤖")
root.geometry("500x450")
root.configure(bg="#e0f7fa")

# --- 标题 Label ---
title = tk.Label(
    root,
    text="🤖 打招呼机器人",
    font=("微软雅黑", 22, "bold"),
    fg="#006064",
    bg="#e0f7fa"
)
title.pack(pady=20)

# --- 机器人头像（用大号 emoji Label 代替）---

    root,
    text="🤖",
    font=("微软雅黑", 60),
    bg="#e0f7fa"
)
avatar.pack(pady=5)

# --- 提示 ---
prompt = tk.Label(
    root,
    text="机器人：你叫什么名字呀？",
    font=("微软雅黑", 14),
    fg="#00838f",
    bg="#e0f7fa"
)
prompt.pack(pady=10)

# --- 输入框 ---
name_entry = tk.Entry(
    root,
    font=("微软雅黑", 15),
    width=20,
    justify="center",
    bd=2,
    relief="solid"
)
name_entry.pack(pady=5)

# --- 回复显示 Label（动态更新） ---
reply = tk.Label(
    root,
    text="",
    font=("微软雅黑", 16, "bold"),
    fg="#004d40",
    bg="#e0f7fa",
    wraplength=380
)
reply.pack(pady=15)

root.mainloop()

改错时刻

这三处最容易错

❌ command=say_hello()

✅ command=say_hello
加了括号，函数会立刻执行，点击时反而不动。

❌ geometry("500*400")

✅ geometry("500x400")
中间必须是英文字母 x。

❌ print(“你好”)

✅ print("你好")
中文引号不是 Python 字符串，一运行就语法错误。
