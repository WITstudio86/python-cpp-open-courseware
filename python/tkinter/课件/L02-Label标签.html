    bg="#f5f0e8"                     # 背景色和窗口一致
)
title_label.pack(pady=15)  # pady 设置上下间距

# --- 公告内容 Label：不同样式来区分不同类型 ---

# 公告 1：通知类 —— 红色文字，醒目

    root,
    text="【通知】明天下午 3 点有编程比赛！",

    fg="#c0392b",                    # 红色文字，表示重要通知

)
notice.pack(pady=5)

# 公告 2：活动类 —— 蓝色斜体，轻松活泼
activity = tk.Label(
    root,
    text="【活动】周末一起去科技馆参观～",
    font=("微软雅黑", 14, "italic"),  # 斜体表示活动
    fg="#2980b9",                     # 蓝色文字
    bg="#f5f0e8"
)
activity.pack(pady=5)

# 公告 3：提醒类 —— 带黄色背景，像便利贴！
reminder = tk.Label(
    root,
    text="⚠️ 别忘了带你的学生证！",
    font=("微软雅黑", 13),
    fg="#e67e22",                    # 橙色文字

reminder.pack(pady=10)

root.mainloop()
运行后你该看到： 控件排队出现在窗口里，不再叠在同一个点。

# --- 用 StringVar 控制的动态 Label ---
# StringVar 就像一个"魔术变量"，改变它的值，Label 自动更新！
var_text = tk.StringVar()            # 创建一个 StringVar 对象

    root,

    bg="#f5f0e8"
)

# 按钮：点击后修改 StringVar 的值
def change_text():
    """修改 var_text 的值，Label 上的文字会自动更新"""
    var_text.set("✅ 文字已经改变！StringVar 让 Label 活起来了！")

change_btn = tk.Button(
    root,
    text="点我修改文字",

改错时刻

这三处最容易错

❌ command=say_hello()

✅ command=say_hello
加了括号，函数会立刻执行，点击时反而不动。

❌ geometry("500*400")

✅ geometry("500x400")
中间必须是英文字母 x。

❌ print(“你好”)

✅ print("你好")
中文引号不是 Python 字符串，一运行就语法错误。

标题 Label：大号字体，深色控件排队出现在窗口里，不再叠在同一个点。同一个父容器里不要又 pack 又 grid，否则会直接报错。
公告内容 Label控件排队出现在窗口里，不再叠在同一个点。同一个父容器里不要又 pack 又 grid，否则会直接报错。
带边框的公告区域控件排队出现在窗口里，不再叠在同一个点。同一个父容器里不要又 pack 又 grid，否则会直接报错。
用 StringVar 控制的动态窗口里多了按钮。点一下，看它有没有反应。command 后面写函数名，不要加括号。加了括号会马上执行。
