import tkinter as tk

# ========================================
# 第 9 节：Radiobutton + Checkbutton —— 喜好收集器
# ========================================

root = tk.Tk()
root.title("喜好收集器")
root.geometry("500x600")
root.configure(bg="#fce4ec")

# --- 标题 ---
title = tk.Label(
    root,
    text="📋 喜好收集器",
    font=("微软雅黑", 20, "bold"),
    fg="#880e4f",
    bg="#fce4ec"
)
title.pack(pady=15)

subtitle = tk.Label(
    root,
    text="告诉我你的喜好，让我更了解你～",
    font=("微软雅黑", 12),
    fg="#ad1457",
    bg="#fce4ec"
)
subtitle.pack()

root.mainloop()
运行后你该看到： 标题栏换成了这一步写的文字。
坑： 窗口大小写成 "宽x高"，中间是英文字母 x，不是星号。

import tkinter as tk

# ========================================
# 第 9 节：Radiobutton + Checkbutton —— 喜好收集器
# ========================================

root = tk.Tk()
root.title("喜好收集器")
root.geometry("500x600")
root.configure(bg="#fce4ec")

# --- 标题 ---
title = tk.Label(
    root,
    text="📋 喜好收集器",
    font=("微软雅黑", 20, "bold"),
    fg="#880e4f",
    bg="#fce4ec"
)
title.pack(pady=15)

subtitle = tk.Label(
    root,
    text="告诉我你的喜好，让我更了解你～",
    font=("微软雅黑", 12),
    fg="#ad1457",
    bg="#fce4ec"
)
subtitle.pack()

# ============= 单选题：性别 =============
gender_frame = tk.Frame(root, bg="#ffffff", relief="groove", bd=2, padx=20, pady=15)
gender_frame.pack(pady=15, padx=40, fill="x")

tk.Label(
    gender_frame,
    text="① 你的性别（单选）",
    font=("微软雅黑", 14, "bold"),
    fg="#c2185b",
    bg="#ffffff"
).pack(anchor="w")

gender_var = tk.StringVar(value="")  # ← StringVar 存储选中的值

tk.Radiobutton(
    gender_frame,
    text="👦 男生",
    variable=gender_var,  # ← 绑定到同一个变量
    value="男生",         # ← 选中时，gender_var 的值
    font=("微软雅黑", 13),
    bg="#ffffff",
    activebackground="#fce4ec"
).pack(anchor="w", pady=3)

tk.Radiobutton(
    gender_frame,
    text="👧 女生",
    variable=gender_var,  # ← 同一个变量！
    value="女生",
    font=("微软雅黑", 13),
    bg="#ffffff",
    activebackground="#fce4ec"
).pack(anchor="w", pady=3)

tk.Radiobutton(
    gender_frame,
    text="🤫 保密",
    variable=gender_var,
    value="保密",
    font=("微软雅黑", 13),
    bg="#ffffff",
    activebackground="#fce4ec"
).pack(anchor="w", pady=3)

root.mainloop()
运行后你该看到： 出现一组单选，一次只能选中一个。
坑： IntVar、StringVar 必须写在 Tk() 之后，不然会报太早创建变量。

改错时刻

这三处最容易错

❌ command=say_hello()

✅ command=say_hello
加了括号，函数会立刻执行，点击时反而不动。

❌ geometry("500*400")

✅ geometry("500x400")
中间必须是英文字母 x。

❌ print(“你好”)

✅ print("你好")
中文引号不是 Python 字符串，一运行就语法错误。

这一步运行后记住的坑
标题标题栏换成了这一步写的文字。窗口大小写成 "宽x高"，中间是英文字母 x，不是星号。
单选题：性别出现一组单选，一次只能选中一个。IntVar、StringVar 必须写在 Tk() 之后，不然会报太早创建变量。
多选题：兴趣爱好出现可以勾选的方框，可以同时选多个。IntVar、StringVar 必须写在 Tk() 之后，不然会报太早创建变量。
结果显示区控件排队出现在窗口里，不再叠在同一个点。同一个父容器里不要又 pack 又 grid，否则会直接报错。
提交按钮窗口里多了按钮。点一下，看它有没有反应。command 后面写函数名，不要加括号。加了括号会马上执行。
