### 知识点 2：已掌握 / 未掌握标记

用两个列表来记录：

```python
mastered = []    # 已掌握的单词
unmastered = []  # 还没掌握的单词

def mark_mastered():
    """把当前单词标记为已掌握"""
    if current_word not in mastered:
        mastered.append(current_word)
        if current_word in unmastered:
            unmastered.remove(current_word)

def mark_unmastered():
    """把当前单词标记为未掌握"""
    if current_word not in unmastered:
        unmastered.append(current_word)
        if current_word in mastered:
            mastered.remove(current_word)
```

## 完整代码

```python
import tkinter as tk
import random

root = tk.Tk()
root.title("单词记忆卡 📖")
root.geometry("520x580")
root.configure(bg="#fff8e7")

# ===== 分类词库 =====
word_bank = {
    "🐾 动物": {
        "cat":"猫","dog":"狗","elephant":"大象",
        "lion":"狮子","tiger":"老虎","rabbit":"兔子",
        "bird":"鸟","fish":"鱼","monkey":"猴子","horse":"马",
    },
    "🍎 水果": {
        "apple":"苹果","banana":"香蕉","orange":"橙子",
        "grape":"葡萄","watermelon":"西瓜","strawberry":"草莓",
        "mango":"芒果","peach":"桃子",
    },
    "🏫 学校": {
        "book":"书","pen":"笔","desk":"课桌",
        "teacher":"老师","student":"学生","classroom":"教室",
        "homework":"作业","exam":"考试",
    },
    "🌈 自然": {
        "sun":"太阳","moon":"月亮","star":"星星",
        "rain":"雨","snow":"雪","wind":"风",
        "ocean":"海洋","mountain":"山","river":"河流",
    },
}

# 将所有单词展开到一个大字典（用于查询释义）
all_words = {}
for category, words in word_bank.items():
    all_words.update(words)

# ===== 状态变量 =====
current_word = ""
current_meaning = ""
current_word_list = []  # 当前分类下的单词列表
is_front = True
mastered = []
unmastered = []

# ===== 功能函数 =====
def change_category(*args):
    """切换分类时重置"""
    global current_word_list, mastered, unmastered
    cat = category_var.get()
    current_word_list = list(word_bank[cat].keys())
    mastered = []
    unmastered = []
    update_stats()
    next_card()

def next_card():
    global current_word, current_meaning, is_front
    if not current_word_list:
        return

    if review_var.get() == "unmastered" and unmastered:
        pool = [w for w in unmastered if w in current_word_list]
        if not pool:
            pool = current_word_list
    else:
        pool = current_word_list

    current_word = random.choice(pool)
    current_meaning = all_words[current_word]
    is_front = True
    word_label.config(text=current_word, fg="#3a2a1a")
    flip_btn.config(text="🔍 翻面看释义")
    update_status()

def flip():
    global is_front
    if is_front:
        word_label.config(text=current_meaning, fg="#8b4513")
        flip_btn.config(text="↩ 翻回正面")
        is_front = False
    else:
        word_label.config(text=current_word, fg="#3a2a1a")
        flip_btn.config(text="🔍 翻面看释义")
        is_front = True

def mark(event_type):
    """标记掌握/未掌握"""
    if event_type == "mastered":
        if current_word not in mastered:
            mastered.append(current_word)
        if current_word in unmastered:
            unmastered.remove(current_word)
    else:
        if current_word not in unmastered:
            unmastered.append(current_word)
        if current_word in mastered:
            mastered.remove(current_word)
    update_stats()
    next_card()

def update_status():
    status = ""
    if current_word in mastered:
        status = " ✅ 已掌握"
    elif current_word in unmastered:
        status = " ❌ 未掌握"
    status_label.config(text=f"📚 {len(current_word_list)} 词{status}")

def update_stats():
    total = len(current_word_list)
    learned = len([w for w in mastered if w in current_word_list])
    percent = int(learned / total * 100) if total > 0 else 0
    stats_label.config(
        text=f"已掌握 {learned}/{total}（{percent}%）")

# ===== 界面布局 =====

# 标题
tk.Label(root, text="📖 单词记忆卡", font=("微软雅黑", 22, "bold"),
         bg="#fff8e7", fg="#5a3e2b").pack(pady=(20,5))

# 分类选择
cat_frame = tk.Frame(root, bg="#fff8e7")
cat_frame.pack(pady=5)
tk.Label(cat_frame, text="分类：", font=("微软雅黑", 12),
         bg="#fff8e7").pack(side=tk.LEFT)

category_var = tk.StringVar()
category_var.set("🐾 动物")
category_var.trace("w", change_category)
cat_menu = tk.OptionMenu(cat_frame, category_var, *word_bank.keys())
cat_menu.config(font=("微软雅黑", 11))
cat_menu.pack(side=tk.LEFT)

# 复习模式
review_var = tk.StringVar()
review_var.set("all")
tk.Radiobutton(cat_frame, text="全部", variable=review_var,
               value="all", bg="#fff8e7", font=("微软雅黑", 10),
               command=next_card).pack(side=tk.LEFT, padx=(15,5))
tk.Radiobutton(cat_frame, text="只复习未掌握", variable=review_var,
               value="unmastered", bg="#fff8e7", font=("微软雅黑", 10),
               command=next_card).pack(side=tk.LEFT)

# 卡片区域
card_frame = tk.Frame(root, bg="#fffef5", relief="solid", bd=2,
                      width=340, height=180)
card_frame.pack(pady=15)
card_frame.pack_propagate(False)

word_label = tk.Label(card_frame, text="选择分类开始",
                      font=("微软雅黑", 28, "bold"),
                      bg="#fffef5", fg="#3a2a1a")
word_label.pack(expand=True)

# 状态
status_label = tk.Label(root, text="", font=("微软雅黑", 11),
                        bg="#fff8e7", fg="#aaa")
status_label.pack()

# 统计
stats_label = tk.Label(root, text="", font=("微软雅黑", 11),
                       bg="#fff8e7", fg="#7d9d6e")
stats_label.pack()

# 按钮区
btn_frame = tk.Frame(root, bg="#fff8e7")
btn_frame.pack(pady=15)

flip_btn = tk.Button(btn_frame, text="🔍 翻面看释义", command=flip,
                     font=("微软雅黑", 13), bg="#f0c060",
                     padx=15, pady=6, relief="flat", cursor="hand2")
flip_btn.pack(side=tk.LEFT, padx=5)

tk.Button(btn_frame, text="✅ 掌握", command=lambda: mark("mastered"),
          font=("微软雅黑", 13), bg="#7d9d6e", fg="white",
          padx=15, pady=6, relief="flat", cursor="hand2").pack(side=tk.LEFT, padx=5)

tk.Button(btn_frame, text="❌ 未掌握", command=lambda: mark("unmastered"),
          font=("微软雅黑", 13), bg="#e07060", fg="white",
          padx=15, pady=6, relief="flat", cursor="hand2").pack(side=tk.LEFT, padx=5)

tk.Button(btn_frame, text="➡ 下一张", command=next_card,
          font=("微软雅黑", 13), bg="#60a0c0", fg="white",
          padx=15, pady=6, relief="flat", cursor="hand2").pack(side=tk.LEFT, padx=5)

# 启动
change_category()
next_card()

root.mainloop()
```
