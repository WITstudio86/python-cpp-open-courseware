## 知识点讲解

### 知识点 1：Entry 输入任务名称 📝

`Entry` 是 tkinter 的**单行文本输入框**，就像一个"填空题"的空格，让用户可以输入文字。

```python
# 创建一个 Entry 控件
task_entry = tk.Entry(root, width=25, font=("微软雅黑", 13))
task_entry.pack(pady=10)

# 常用方法
task_entry.get()          # 获取用户输入的文字（返回字符串）
task_entry.delete(0, tk.END)  # 清空输入框
task_entry.insert(0, "默认文字")  # 插入默认文字
```

在我们的番茄钟里，用户先输入任务名称（比如"背古诗"、"做数学卷子"），然后点开始，任务名称就显示在计时器上方，让自己知道当前在做什么。

```python
def start_timer():
    task_name = task_entry.get().strip()
    if task_name == "":
        task_name = "专注中..."  # 如果用户没输入，给个默认名称
    task_label.config(text=f"📌 {task_name}")
```

> 💡 **.strip()** 的作用：去掉用户输入首尾的空格。比如用户输入了 `"  背单词  "`，`.strip()` 后变成 `"背单词"`。

### 知识点 2：完成提示音 🔔

任务完成了，得有个"叮叮叮"的提醒！不同操作系统播放声音的方法不一样：

```python
import sys

def play_sound():
    """播放完成提示音（跨平台）"""
    if sys.platform == "win32":
        # Windows 系统：用 winsound 模块（Python 内置）
        import winsound
        winsound.Beep(1000, 500)  # 频率1000Hz，持续500毫秒
    else:
        # Mac / Linux：用 print("\a") 触发系统响铃
        print("\a")
```

`print("\a")` 中的 `\a` 是一个特殊字符，叫"响铃符"（Bell character）。在大多数终端里，它会发出"叮"的一声。如果从终端启动了 Python 程序，这个声音就能被播放出来。

> 🍎 在 Mac 上运行 tkinter 程序时，`print("\a")` 的声音可能不明显。一个更通用的替代方案是用 `root.bell()` —— tkinter 自带的响铃方法，无需任何额外导入！

```python
# 更简单的方式：tkinter 自带的响铃
root.bell()  # 发出系统提示音
```

### 知识点 3：统计今日完成番茄数 🍅

用一个变量来记录今天完成了多少个番茄。每完成一个 25 分钟，变量加 1，同时在界面上更新显示。

```python
# 全局变量记录番茄数
today_pomodoros = 0

def finish_pomodoro():
    """番茄钟时间到！"""
    global today_pomodoros
    today_pomodoros += 1
    count_label.config(text=f"🍅 今日番茄：{today_pomodoros} 个")
    root.bell()  # 响铃提醒
```

把番茄数显示在窗口上，每完成一个就自动 +1，看着数字一个个累加，满满的成就感！🍅🍅🍅

### 知识点 4：按钮状态的切换管理 🎮

番茄钟有三种状态：**运行中**、**暂停中**、**空闲中**。不同状态下，按钮的显示文字和功能都不一样。这就像一个"有限状态机"：

| 当前状态 | 开始按钮 | 暂停按钮 | 重置按钮 |
|---------|---------|---------|---------|
| 空闲（初始） | 可以按 | 禁用 | 可以按 |
| 运行中 | 禁用 | 可以按 | 可以按 |
| 暂停中 | 可以按（继续） | 禁用 | 可以按 |

用 `state` 参数控制按钮是否可点击：

```python
# state="normal" 表示按钮可以点击（默认）
# state="disabled" 表示按钮变灰，不能点击

start_btn.config(state="disabled")   # 禁用开始按钮
pause_btn.config(state="normal")     # 启用暂停按钮
```

完整的状态切换逻辑：

```python
is_running = False   # 是否正在倒计时
is_paused = False    # 是否暂停中

def start():
    """开始倒计时"""
    global is_running, is_paused
    is_running = True
    is_paused = False
    # 更新按钮状态
    start_btn.config(state="disabled")
    pause_btn.config(state="normal")
    reset_btn.config(state="normal")
    countdown()  # 启动倒计时

def pause():
    """暂停 / 继续"""
    global is_running, is_paused
    if is_paused:
        # 当前已暂停 → 点一下继续
        is_paused = False
        pause_btn.config(text="⏸️ 暂停")
        countdown()  # 继续倒计时
    else:
        # 当前运行中 → 点一下暂停
        is_paused = True
        pause_btn.config(text="▶️ 继续")
        # 不再调用 countdown()，倒计时自然停止

def reset():
    """重置番茄钟"""
    global is_running, is_paused
    is_running = False
    is_paused = False
    # 恢复时间
    time_label.config(text="25:00")
    # 恢复按钮状态
    start_btn.config(state="normal")
    pause_btn.config(state="disabled")
```

## 完整代码

```python
import tkinter as tk
import sys

# ========================================
# 第 40 节：项目 11 —— 倒计时番茄钟（下）
# 知识点：Entry输入任务名、完成提示音、今日番茄数统计、按钮状态切换
# ========================================

# ---------- 全局变量 ----------
WORK_MIN = 25          # 工作时间：25 分钟
total_seconds = WORK_MIN * 60  # 总秒数
remaining = total_seconds      # 剩余秒数
is_running = False             # 是否正在运行
is_paused = False              # 是否暂停中
after_id = None                # after() 的 ID，用于取消定时
today_count = 0                # 今日番茄数

# ---------- 创建窗口 ----------
root = tk.Tk()
root.title("🍅 番茄钟")
root.geometry("480x520")
root.configure(bg="#fff8e1")

# ---------- 标题 ----------
title_label = tk.Label(root, text="🍅 番茄工作法",
                       font=("微软雅黑", 22, "bold"),
                       bg="#fff8e1", fg="#e65100")
title_label.pack(pady=(20, 5))

# ---------- 任务输入区 ----------
input_frame = tk.Frame(root, bg="#fff8e1")
input_frame.pack(pady=10)

tk.Label(input_frame, text="任务名称：", bg="#fff8e1",
         font=("微软雅黑", 13)).pack(side=tk.LEFT)

task_entry = tk.Entry(input_frame, width=22,
                      font=("微软雅黑", 14), justify="center")
task_entry.insert(0, "例如：背古诗")  # 提示文字
task_entry.pack(side=tk.LEFT, padx=5)

# ---------- 当前任务显示 ----------
task_label = tk.Label(root, text="📌 等待开始...",
                      font=("微软雅黑", 14),
                      bg="#fff8e1", fg="#555555")
task_label.pack(pady=5)

# ---------- 倒计时显示 ----------
time_label = tk.Label(root, text="25:00",
                      font=("Consolas", 64, "bold"),
                      bg="#1b1b1b", fg="#ff6f00",
                      width=8)
time_label.pack(pady=15)

# ---------- 今日番茄数 ----------
count_label = tk.Label(root, text="🍅 今日番茄：0 个",
                       font=("微软雅黑", 13, "bold"),
                       bg="#fff8e1", fg="#bf360c")
count_label.pack(pady=5)


# ==================== 核心逻辑 ====================

def format_time(seconds):
    """把秒数格式化为 MM:SS 字符串"""
    m = seconds // 60
    s = seconds % 60
    return f"{m:02d}:{s:02d}"


def countdown():
    """倒计时核心函数：每秒减 1，直到 0"""
    global remaining, is_running, after_id

    if not is_running:
        return  # 如果被暂停了，就不再继续

    if remaining > 0:
        remaining -= 1
        time_label.config(text=format_time(remaining))
        # 1 秒后再次调用自己
        after_id = root.after(1000, countdown)
    else:
        # 时间到！
        time_up()


def time_up():
    """番茄钟时间到：响铃 + 统计 + 恢复状态"""
    global is_running, is_paused, remaining, today_count

    is_running = False
    is_paused = False
    remaining = total_seconds  # 重置时间
    time_label.config(text=format_time(remaining))

    # 番茄数 +1
    today_count += 1
    count_label.config(text=f"🍅 今日番茄：{today_count} 个")

    # 播放提示音
    play_sound()

    # 弹窗提醒
    from tkinter import messagebox
    task_name = task_entry.get().strip() or "专注中..."
    messagebox.showinfo("⏰ 时间到！",
                        f"「{task_name}」完成！\n\n"
                        f"今日已完成 {today_count} 个番茄 🍅\n"
                        "休息 5 分钟，准备下一个吧！")

    # 恢复按钮状态
    start_btn.config(state="normal")
    pause_btn.config(state="disabled")
    task_label.config(text="📌 等待开始...")


def play_sound():
    """播放完成提示音"""
    if sys.platform == "win32":
        try:
            import winsound
            for _ in range(3):
                winsound.Beep(1000, 300)
        except:
            root.bell()
    else:
        # Mac / Linux：连续响铃 3 次
        for _ in range(3):
            root.bell()
            root.update()  # 让 bell 生效


def start():
    """开始倒计时"""
    global is_running, is_paused, remaining

    # 更新任务名称显示
    task_name = task_entry.get().strip()
    if task_name == "" or task_name == "例如：背古诗":
        task_name = "专注中..."
    task_label.config(text=f"📌 {task_name}")

    if is_paused:
        # 从暂停恢复，不需要重置时间
        is_paused = False
    else:
        # 全新开始，重置时间
        remaining = total_seconds
        time_label.config(text=format_time(remaining))

    is_running = True
    start_btn.config(state="disabled")
    pause_btn.config(text="⏸️ 暂停", state="normal")

    countdown()


def pause():
    """暂停 / 继续"""
    global is_running, is_paused

    if is_paused:
        # 当前暂停中 → 继续
        is_paused = False
        is_running = True
        pause_btn.config(text="⏸️ 暂停")
        countdown()
    else:
        # 当前运行中 → 暂停
        is_paused = True
        is_running = False
        pause_btn.config(text="▶️ 继续")


def reset():
    """重置番茄钟"""
    global is_running, is_paused, remaining, after_id

    is_running = False
    is_paused = False
    remaining = total_seconds

    # 取消计划中的 after 调用
    if after_id is not None:
        root.after_cancel(after_id)
        after_id = None

    time_label.config(text=format_time(remaining))
    task_label.config(text="📌 等待开始...")
    start_btn.config(state="normal")
    pause_btn.config(text="⏸️ 暂停", state="disabled")


# ==================== 按钮区域 ====================

btn_frame = tk.Frame(root, bg="#fff8e1")
btn_frame.pack(pady=15)

start_btn = tk.Button(btn_frame, text="▶️ 开始专注",
                      command=start,
                      font=("微软雅黑", 13, "bold"),
                      bg="#ff9800", fg="white",
                      activebackground="#f57c00",
                      relief="flat", padx=20, pady=8,
                      cursor="hand2")
start_btn.pack(side=tk.LEFT, padx=8)

pause_btn = tk.Button(btn_frame, text="⏸️ 暂停",
                      command=pause,
                      font=("微软雅黑", 13, "bold"),
                      bg="#ffcc02", fg="#333",
                      activebackground="#ffc107",
                      relief="flat", padx=20, pady=8,
                      cursor="hand2",
                      state="disabled")  # 初始禁用
pause_btn.pack(side=tk.LEFT, padx=8)

reset_btn = tk.Button(btn_frame, text="🔄 重置",
                      command=reset,
                      font=("微软雅黑", 13, "bold"),
                      bg="#bdbdbd", fg="#333",
                      activebackground="#9e9e9e",
                      relief="flat", padx=20, pady=8,
                      cursor="hand2")
reset_btn.pack(side=tk.LEFT, padx=8)

# ---------- 底部提示 ----------
tip = tk.Label(root,
               text="💡 输入任务名 → 点「开始专注」→ 25 分钟后响铃提醒",
               font=("微软雅黑", 10),
               bg="#fff8e1", fg="#888888")
tip.pack(pady=15)

root.mainloop()
```

## 小贴士
- `after_cancel(after_id)` 可以取消一个计划中的 `after()` 调用，在重置功能中很重要
- `Entry` 获取的内容是**字符串**，计算前记得转换类型（比如 `int()`）
- `root.bell()` 是 tkinter 自带的响铃，不需要任何 import，跨平台通用
- 按钮的 `state="disabled"` 和 `state="normal"` 是管理按钮是否可点击的重要工具
- 用**全局变量**（如 `is_running`、`is_paused`）管理状态是一个简单好用的方法，但要记得在函数里加 `global` 声明
- 📌 番茄工作法的核心：25 分钟专注 + 5 分钟休息 = 1 个番茄。坚持使用，专注力会越来越强！🍅
