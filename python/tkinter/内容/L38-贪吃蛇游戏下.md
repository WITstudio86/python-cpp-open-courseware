## 知识点讲解

### 1. 随机生成食物 —— `random` 模块

食物的位置不能固定，每次被吃掉后要随机出现在另一个地方。Python 的 `random` 模块可以帮助我们：

```python
import random

# 随机生成食物的坐标（以 20 像素为步长，保证对齐蛇的网格）
def create_food():
    global food_x, food_y
    food_x = random.randint(0, 29) * 20  # 窗口宽 600，600÷20=30 格
    food_y = random.randint(0, 29) * 20  # 窗口高 600，600÷20=30 格
    canvas.coords(food, food_x, food_y, food_x + 20, food_y + 20)
```

`random.randint(0, 29)` 会在 0 到 29 之间随机选一个整数。乘以 20 后，食物坐标就是 0, 20, 40, ..., 580，正好对齐蛇的移动网格。

> 注意：如果食物随机到了蛇身上，需要重新生成！后面会处理。

### 2. 吃食物 —— 碰撞检测 + 变长 + 加分

判断蛇头是否碰到食物，就看蛇头的坐标是否等于食物的坐标：

```python
def eat_food():
    global score, food_x, food_y
    head_x, head_y = snake[0]  # 蛇头的坐标

    if head_x == food_x and head_y == food_y:
        # 加分！
        score += 10
        score_label.config(text=f"得分：{score}")

        # 重新生成食物
        create_food()

        # 蛇变长：在尾部添加一节（下一帧移动时自动保留）
        return True
    return False
```

蛇变长的秘诀是：**如果吃到了食物，就不删除蛇尾**。在原本的移动逻辑中，我们每次都会从 `snake` 列表末尾 `pop()` 掉最后一节。吃到食物时跳过这一步，蛇就"长"了一节！

```python
def move():
    # ... 计算新蛇头位置 ...
    snake.insert(0, (new_x, new_y))  # 头部前进

    if not eat_food():
        snake.pop()  # 没吃到食物 → 删除尾部（保持长度不变）
    # 吃到食物 → 不删除尾部 → 蛇变长一节！
```

### 3. 得分系统 —— Label 显示分数

用 `Label` 来显示分数，每次吃到食物后更新：

```python
score = 0
score_label = tk.Label(root, text="得分：0", font=("Arial", 16, "bold"),
                        bg="#2e7d32", fg="white")
score_label.pack(pady=5)
```

每次吃到食物后，`score += 10`，然后 `score_label.config(text=f"得分：{score}")` 更新显示。

### 4. 碰撞检测 —— 撞墙 / 撞自己

游戏结束有两种情况：

```python
def check_collision(head_x, head_y):
    # 情况1：撞墙
    if head_x < 0 or head_x >= 600 or head_y < 0 or head_y >= 600:
        return True

    # 情况2：撞到自己（蛇头碰到了身体的任何一节）
    if (head_x, head_y) in snake[1:]:
        return True

    return False
```

在 `move()` 函数中，每次移动后都调用 `check_collision()`：

```python
def move():
    # ... 移动蛇 ...

    if check_collision(snake[0][0], snake[0][1]):
        game_over()
        return  # 停止移动

    root.after(150, move)  # 没撞就继续
```

### 5. 游戏结束 + 重新开始

游戏结束时显示提示，并提供一个"重新开始"按钮：

```python
def game_over():
    canvas.create_text(300, 300, text="游戏结束！",
                       font=("Arial", 36, "bold"), fill="red",
                       tags="gameover")

    # 显示重新开始按钮
    restart_button = tk.Button(root, text="重新开始",
                               font=("Arial", 14), command=restart_game)
    restart_button.place(x=250, y=340)

def restart_game():
    global snake, direction, score, food_x, food_y
    snake = [(300, 300), (280, 300), (260, 300)]  # 重置蛇
    direction = "Right"  # 重置方向
    score = 0
    score_label.config(text="得分：0")
    canvas.delete("snake")  # 清除旧蛇
    canvas.delete("gameover")  # 清除文字
    draw_snake()  # 重新画蛇
    create_food()
    move()  # 重新开始移动
```

## 完整代码

```python
import tkinter as tk
import random

# ========================================
# 第 38 节：贪吃蛇游戏（下）—— 完整版
# 上接第 37 节：蛇的移动 + 方向控制
# ========================================

root = tk.Tk()
root.title("贪吃蛇大作战 🐍")
root.geometry("600x670")
root.configure(bg="#2e7d32")
root.resizable(False, False)

# ---------- 得分显示 ----------
score = 0
score_label = tk.Label(root, text="得分：0", font=("Arial", 16, "bold"),
                        bg="#2e7d32", fg="white")
score_label.pack(pady=5)

# ---------- 游戏画布 ----------
canvas = tk.Canvas(root, width=600, height=600, bg="#1b5e20", highlightthickness=0)
canvas.pack()

# ---------- 蛇的数据 ----------
snake = [(300, 300), (280, 300), (260, 300)]  # 每个元素是 (x, y)
direction = "Right"  # 初始方向：向右

def draw_snake():
    """在画布上画出整条蛇"""
    canvas.delete("snake")
    for i, (x, y) in enumerate(snake):
        color = "#4caf50" if i == 0 else "#81c784"  # 蛇头深绿，身体浅绿
        canvas.create_rectangle(x, y, x + 20, y + 20,
                                fill=color, outline="#1b5e20", tags="snake")

# ---------- 食物 ----------
food_x, food_y = 0, 0
food = canvas.create_rectangle(0, 0, 20, 20, fill="#ff5252",
                                outline="#d32f2f", tags="food")

def create_food():
    """在空白位置随机生成食物"""
    global food_x, food_y
    while True:
        food_x = random.randint(0, 29) * 20
        food_y = random.randint(0, 29) * 20
        # 确保食物不在蛇身上
        if (food_x, food_y) not in snake:
            break
    canvas.coords(food, food_x, food_y, food_x + 20, food_y + 20)

# ---------- 方向控制 ----------
def change_direction(new_dir):
    global direction
    # 禁止反向（比如正在向右走不能立刻向左）
    opposites = {"Up": "Down", "Down": "Up", "Left": "Right", "Right": "Left"}
    if new_dir != opposites.get(direction):
        direction = new_dir

root.bind("<Up>", lambda e: change_direction("Up"))
root.bind("<Down>", lambda e: change_direction("Down"))
root.bind("<Left>", lambda e: change_direction("Left"))
root.bind("<Right>", lambda e: change_direction("Right"))

# ---------- 碰撞检测 ----------
def check_collision(head_x, head_y):
    """检测蛇头是否撞墙或撞到自己"""
    # 撞墙
    if head_x < 0 or head_x >= 600 or head_y < 0 or head_y >= 600:
        return True
    # 撞自己
    if (head_x, head_y) in snake[1:]:
        return True
    return False

# ---------- 吃食物 ----------
def eat_food():
    """如果蛇头碰到食物，加分并返回 True"""
    global score
    head_x, head_y = snake[0]
    if head_x == food_x and head_y == food_y:
        score += 10
        score_label.config(text=f"得分：{score}")
        create_food()
        return True
    return False

# ---------- 移动蛇 ----------
running = True

def move():
    """蛇的每一步移动"""
    global running
    if not running:
        return

    head_x, head_y = snake[0]
    if direction == "Right":
        new_head = (head_x + 20, head_y)
    elif direction == "Left":
        new_head = (head_x - 20, head_y)
    elif direction == "Down":
        new_head = (head_x, head_y + 20)
    elif direction == "Up":
        new_head = (head_x, head_y - 20)

    # 碰撞检测
    if check_collision(new_head[0], new_head[1]):
        game_over()
        return

    # 蛇头前进
    snake.insert(0, new_head)

    # 吃食物判断
    if not eat_food():
        snake.pop()  # 没吃到 → 去掉尾部（长度不变）

    draw_snake()
    root.after(150, move)  # 150ms 后再次移动

# ---------- 游戏结束 ----------
def game_over():
    global running
    running = False
    canvas.create_text(300, 270, text="游戏结束！",
                       font=("Arial", 40, "bold"), fill="#ff5252",
                       tags="gameover")
    canvas.create_text(300, 320, text=f"最终得分：{score}",
                       font=("Arial", 20), fill="white",
                       tags="gameover")

    restart_btn = tk.Button(root, text="↻ 重新开始",
                            font=("Arial", 15, "bold"),
                            bg="#4caf50", fg="white",
                            activebackground="#66bb6a",
                            command=restart_game)
    restart_btn.place(x=240, y=350)

def restart_game():
    global snake, direction, score, running
    # 清除画布
    canvas.delete("snake")
    canvas.delete("gameover")

    # 重置数据
    snake = [(300, 300), (280, 300), (260, 300)]
    direction = "Right"
    score = 0
    running = True
    score_label.config(text="得分：0")

    # 清除按钮（遍历销毁）
    for widget in root.place_slaves():
        widget.destroy()

    draw_snake()
    create_food()
    move()

# ---------- 启动游戏 ----------
draw_snake()
create_food()
move()
root.mainloop()
```

## 小贴士
- 🐍 蛇变长的关键：吃到食物时 **不 pop() 尾巴**，蛇就自然多了一节
- 🔄 `after()` 的延迟时间决定了蛇的移动速度——越小越快
- 🚫 反向禁止很重要：如果蛇能立刻回头，会撞到自己身体的第二节
- 🎯 食物的随机生成要检查是否落在蛇身上，否则食物会"藏"在蛇里
- 📦 `canvas.delete("snake")` + `draw_snake()` 是重绘的标准套路，比逐个移动每个方块简单得多
