## 知识点讲解

### 知识点 1：Entry 单行输入框

`Entry` 是 tkinter 的**单行文本输入框**，用户可以输入一行文字。非常适合用来输入账号、密码、网址等信息。

```python
# 创建 Entry 输入框
entry = tk.Entry(root, width=30)  # width 是字符宽度（不是像素）
entry.pack()

# 获取输入框中的文字
text = entry.get()

# 清空输入框
entry.delete(0, tk.END)

# 插入默认文字
entry.insert(0, "请输入...")

# 密码模式（输入的字符显示为 *）
pwd_entry = tk.Entry(root, show="*")
```

| 方法 | 说明 | 例子 |
|------|------|------|
| `.get()` | 获取输入框中的文字 | `username = entry.get()` |
| `.delete(0, tk.END)` | 清空输入框 | `entry.delete(0, tk.END)` |
| `.insert(0, "文字")` | 在开头插入文字 | `entry.insert(0, "请输入")` |
| `show="*"` | 密码模式（回显为 *） | `tk.Entry(root, show="*")` |

**用 Label + Entry 组合制作表单：**

```python
# 每个输入项用一个 Frame 包装 Label 和 Entry
frame = tk.Frame(root)
tk.Label(frame, text="网站：", width=8).pack(side="left")
site_entry = tk.Entry(frame, width=30)
site_entry.pack(side="left")
frame.pack(pady=5)
```

### 知识点 2：base64 简单加密

`base64` 是 Python 内置的一个编码模块，可以把任意数据"翻译"成一串字母和数字的混合字符。**它不算真正的加密**，但对于初学者来说，是一个很好的"隐藏密码"的入门方式。

> 把它理解成"暗号"——把明文变成一串看不懂的乱码，只有知道解码方法的人才能还原。

```python
import base64

# 加密：把字符串变成 base64 编码
original = "mypassword123"
encoded = base64.b64encode(original.encode("utf-8")).decode("utf-8")
print(encoded)  # 输出：bXlwYXNzd29yZDEyMw==

# 解密：把 base64 编码还原成原始字符串
decoded = base64.b64decode(encoded.encode("utf-8")).decode("utf-8")
print(decoded)  # 输出：mypassword123
```

**关键步骤拆解：**

```python
# 加密三步走：
# ① 字符串 → bytes        encode("utf-8")
# ② bytes → base64        b64encode()
# ③ bytes → 字符串         decode("utf-8")

# 解密三步走：
# ① 字符串 → bytes        encode("utf-8")
# ② base64 → bytes        b64decode()
# ③ bytes → 字符串         decode("utf-8")
```

> 为什么要来回转？因为 base64 操作的是**字节（bytes）**，不是字符串（str）。所以加密前要把字符串编码成字节，加密后要把字节解码回字符串才能显示。

| 函数 | 作用 | 输入 | 输出 |
|------|------|------|------|
| `b64encode(data)` | 加密/编码 | bytes | bytes |
| `b64decode(data)` | 解密/解码 | bytes | bytes |

### 知识点 3：JSON 文件存储

`JSON`（JavaScript Object Notation）是一种轻量级的数据格式，用键值对的方式存储数据。Python 的 `json` 模块可以轻松地把字典/列表保存到文件中，也能从文件中读取回来。

```python
import json

# 准备数据（Python 字典）
data = [
    {"site": "bilibili.com", "user": "xiaoming", "pwd": "bXlwYXNz"},
    {"site": "zhihu.com",    "user": "xiaoming", "pwd": "c2VjcmV0"},
]

# 保存到 JSON 文件
with open("passwords.json", "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=2)
```

保存后的 `passwords.json` 文件内容：

```json
[
  {
    "site": "bilibili.com",
    "user": "xiaoming",
    "pwd": "bXlwYXNz"
  },
  {
    "site": "zhihu.com",
    "user": "xiaoming",
    "pwd": "c2VjcmV0"
  }
]
```

**读取 JSON 文件：**

```python
# 从 JSON 文件读取数据
try:
    with open("passwords.json", "r", encoding="utf-8") as f:
        data = json.load(f)
except FileNotFoundError:
    data = []  # 文件不存在就用空列表
```

| 函数 | 作用 | 方向 |
|------|------|------|
| `json.dump(data, file)` | 把数据写入文件 | 内存 → 文件 |
| `json.load(file)` | 从文件读取数据 | 文件 → 内存 |

**常用参数：**
- `ensure_ascii=False`：允许保存中文（默认会把中文转成 `\uxxxx` 编码）
- `indent=2`：缩进 2 格，让 JSON 文件排版好看，方便人类阅读

## 完整代码

```python
import tkinter as tk
from tkinter import messagebox
import base64
import json
import os

# ========================================
# 第 35 节：密码管理器（上）
# 知识点：Entry + base64加密 + JSON文件存储
# ========================================

# ---------- 数据文件路径 ----------
DATA_FILE = "passwords.json"

# ---------- 数据操作函数 ----------
def load_data():
    """从 JSON 文件加载密码数据，文件不存在则返回空列表"""
    if os.path.exists(DATA_FILE):
        with open(DATA_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    return []

def save_data(data):
    """将密码数据保存到 JSON 文件"""
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

def encrypt(text):
    """用 base64 加密：明文 → 密文"""
    return base64.b64encode(text.encode("utf-8")).decode("utf-8")

def decrypt(text):
    """用 base64 解密：密文 → 明文"""
    return base64.b64decode(text.encode("utf-8")).decode("utf-8")

# ---------- 主窗口 ----------
root = tk.Tk()
root.title("密码管理器")
root.geometry("500x550")
root.configure(bg="#f5f7fa")

# ---------- 标题 ----------
tk.Label(root, text="🔐 密码管理器",
         font=("Microsoft YaHei", 20, "bold"),
         bg="#f5f7fa", fg="#2c3e50").pack(pady=15)

# ---------- 输入区域 ----------
input_frame = tk.Frame(root, bg="#ffffff", relief="groove", bd=1)
input_frame.pack(pady=10, padx=30, fill="x")

tk.Label(input_frame, text="添加新密码",
         font=("Microsoft YaHei", 13, "bold"),
         bg="#ffffff", fg="#2c3e50").pack(pady=(10, 5))

# 网站输入行
row1 = tk.Frame(input_frame, bg="#ffffff")
tk.Label(row1, text="网　站：", font=("Microsoft YaHei", 11),
         bg="#ffffff", width=8, anchor="e").pack(side="left")
site_entry = tk.Entry(row1, width=30, font=("Microsoft YaHei", 11))
site_entry.pack(side="left", padx=5)
row1.pack(pady=5)

# 账号输入行
row2 = tk.Frame(input_frame, bg="#ffffff")
tk.Label(row2, text="账　号：", font=("Microsoft YaHei", 11),
         bg="#ffffff", width=8, anchor="e").pack(side="left")
user_entry = tk.Entry(row2, width=30, font=("Microsoft YaHei", 11))
user_entry.pack(side="left", padx=5)
row2.pack(pady=5)

# 密码输入行
row3 = tk.Frame(input_frame, bg="#ffffff")
tk.Label(row3, text="密　码：", font=("Microsoft YaHei", 11),
         bg="#ffffff", width=8, anchor="e").pack(side="left")
pwd_entry = tk.Entry(row3, width=30, font=("Microsoft YaHei", 11), show="*")
pwd_entry.pack(side="left", padx=5)
row3.pack(pady=5)

# ---------- 按钮区域 ----------
def add_password():
    """获取输入，加密密码，保存到 JSON 文件"""
    site = site_entry.get().strip()
    user = user_entry.get().strip()
    pwd = pwd_entry.get().strip()

    if not site or not user or not pwd:
        messagebox.showwarning("提示", "请填写完整信息！")
        return

    # 加载已有数据
    data = load_data()

    # 添加新记录
    data.append({
        "site": site,
        "user": user,
        "pwd": encrypt(pwd)  # 密码加密后保存！
    })

    # 保存到文件
    save_data(data)

    # 清空输入框
    site_entry.delete(0, tk.END)
    user_entry.delete(0, tk.END)
    pwd_entry.delete(0, tk.END)

    messagebox.showinfo("成功", f"「{site}」的密码已加密保存！")
    refresh_list()

# 添加按钮
add_btn = tk.Button(input_frame, text="🔒 加密保存",
                    command=add_password,
                    font=("Microsoft YaHei", 12, "bold"),
                    bg="#3498db", fg="white",
                    activebackground="#2980b9",
                    relief="flat", padx=20, pady=6,
                    cursor="hand2")
add_btn.pack(pady=15)

# ---------- 密码列表区域 ----------
tk.Label(root, text="📋 已保存的密码",
         font=("Microsoft YaHei", 13, "bold"),
         bg="#f5f7fa", fg="#2c3e50").pack(pady=(10, 5))

# 列表框架
list_frame = tk.Frame(root, bg="#f5f7fa")
list_frame.pack(pady=5, padx=30, fill="both", expand=True)

# 使用 Listbox 展示已保存的网站
pwd_listbox = tk.Listbox(
    list_frame,
    font=("Microsoft YaHei", 11),
    height=8,
    selectmode="single"
)
pwd_listbox.pack(side="left", fill="both", expand=True)

# 滚动条
scrollbar = tk.Scrollbar(list_frame)
scrollbar.pack(side="right", fill="y")
pwd_listbox.config(yscrollcommand=scrollbar.set)
scrollbar.config(command=pwd_listbox.yview)

# ---------- 查看详情按钮 ----------
def show_detail():
    """选中列表中的一项后，解密并显示详细信息"""
    selection = pwd_listbox.curselection()
    if not selection:
        messagebox.showwarning("提示", "请先在左侧列表中选择一项！")
        return

    data = load_data()
    item = data[selection[0]]
    decrypted_pwd = decrypt(item["pwd"])

    messagebox.showinfo(
        f"🔍 {item['site']}",
        f"网站：{item['site']}\n账号：{item['user']}\n密码：{decrypted_pwd}"
    )

detail_btn = tk.Button(root, text="🔍 查看详情",
                       command=show_detail,
                       font=("Microsoft YaHei", 11),
                       bg="#27ae60", fg="white",
                       activebackground="#219a52",
                       relief="flat", padx=15, pady=5,
                       cursor="hand2")
detail_btn.pack(pady=5)

# ---------- 刷新列表函数 ----------
def refresh_list():
    """从文件读取数据，更新 Listbox 显示"""
    pwd_listbox.delete(0, tk.END)  # 先清空列表
    data = load_data()
    for item in data:
        # 列表中显示网站和账号
        pwd_listbox.insert(tk.END, f"{item['site']}  ({item['user']})")

# 程序启动时加载已有数据
refresh_list()

root.mainloop()
```

## 小贴士
- base64 是**编码**，不是真正的加密。懂 base64 的人可以直接解码。但对于初学者项目来说，它已经足够"隐藏"密码了。
- JSON 文件的 `ensure_ascii=False` 很重要——不加这个参数，中文会被存成 `网站` 这样的格式，人看不懂。
- 读取文件时记得处理 `FileNotFoundError`（文件不存在的情况），否则程序第一次运行会报错。
- `Entry` 的 `show="*"` 只会改变显示效果，`get()` 依然能拿到真正的密码文字。
- 真实项目中的密码管理器应该使用 `hashlib`（哈希）或 `cryptography`（对称加密）库，base64 只是入门教学用的。
