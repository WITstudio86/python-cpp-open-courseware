## 知识点讲解

### 回顾上一节
上一节我们成功接入了天气 API，能查到温度、湿度、风速等信息。但界面还有点简陋 —— 所有信息挤在一个 Label 里，不够好看。今天来给它「美颜」一下！

### 知识点 1：卡片式布局 —— 用 Frame 分区

把天气信息分成多个「卡片」，每个卡片显示一类信息：

```python
# 温度卡片
temp_frame = tk.Frame(root, bg="white", relief="solid", bd=1)
temp_frame.pack(...)

temp_label = tk.Label(temp_frame, text=f"{temp}°C",
                      font=("微软雅黑", 36, "bold"), ...)
temp_label.pack()
```

用多个 Frame 把不同信息分组，整体看起来就像一张张卡片，非常清晰。

### 知识点 2：天气 emoji 对照表

根据天气描述文字，返回对应的 emoji：

```python
def get_weather_emoji(desc):
    """根据天气描述返回对应的 emoji"""
    desc_lower = desc.lower()
    if "晴" in desc_lower or "sunny" in desc_lower or "clear" in desc_lower:
        return "☀️"
    elif "云" in desc_lower or "cloud" in desc_lower:
        return "⛅"
    elif "雨" in desc_lower or "rain" in desc_lower:
        return "🌧️"
    elif "雪" in desc_lower or "snow" in desc_lower:
        return "❄️"
    elif "雾" in desc_lower or "mist" in desc_lower or "fog" in desc_lower:
        return "🌫️"
    elif "风" in desc_lower or "wind" in desc_lower:
        return "💨"
    else:
        return "🌤️"  # 默认
```

### 知识点 3：多城市切换 —— Listbox 选城市

添加一个常用城市列表，点一下就能切换查询：

```python
# 城市列表
cities = ["北京", "上海", "广州", "深圳", "杭州", "成都", "Tokyo", "London", "New York"]

city_listbox = tk.Listbox(root, height=6, ...)
for city in cities:
    city_listbox.insert(tk.END, city)

def on_city_select(event):
    """选中城市后自动查询"""
    selection = city_listbox.curselection()
    if selection:
        city = city_listbox.get(selection[0])
        entry.delete(0, tk.END)
        entry.insert(0, city)
        search()  # 自动查询

city_listbox.bind("<<ListboxSelect>>", on_city_select)
```

### 知识点 4：美化界面技巧

几个让界面更好看的小技巧：

1. **用大字号显示温度**：`font=("微软雅黑", 48, "bold")`
2. **给不同信息配不同颜色**：温度用橙色、湿度用蓝色、风速用绿色
3. **加分隔线**：用细 Frame 或 Label 做视觉分隔
4. **圆角卡片效果**：给 Frame 设背景色 + padding

## 完整代码

```python
import tkinter as tk
import requests

root = tk.Tk()
root.title("天气预报查询 🌤️")
root.geometry("550x600")
root.configure(bg="#e8f4fd")

# ===== 天气 emoji 对照 =====
def get_emoji(desc):
    d = desc.lower()
    if any(w in d for w in ["晴","sunny","clear"]):
        return "☀️"
    elif any(w in d for w in ["云","cloud","overcast"]):
        return "⛅"
    elif any(w in d for w in ["雨","rain","drizzle","shower"]):
        return "🌧️"
    elif any(w in d for w in ["雪","snow"]):
        return "❄️"
    elif any(w in d for w in ["雾","mist","fog","haze"]):
        return "🌫️"
    else:
        return "🌤️"

# ===== API 调用 =====
def get_weather(city):
    try:
        url = f"https://wttr.in/{city}?format=j1"
        r = requests.get(url, timeout=5)
        r.raise_for_status()
        return r.json()
    except:
        return None

# ===== 查询功能 =====
def search(city_name=None):
    if city_name is None:
        city_name = entry.get().strip()
    if not city_name:
        return

    # 显示加载状态
    city_label.config(text="查询中...")
    root.update()

    data = get_weather(city_name)
    if data is None:
        city_label.config(text="❌ 网络错误")
        return

    try:
        c = data["current_condition"][0]
        temp = c["temp_C"]
        feels = c["FeelsLikeC"]
        desc = c["weatherDesc"][0]["value"]
        humidity = c["humidity"]
        wind = c["windspeedKmph"]
        wind_dir = c["winddir16Point"]

        emoji = get_emoji(desc)

        city_label.config(text=f"{emoji} {city_name}")
        temp_label.config(text=f"{temp}°C")
        feels_label.config(text=f"体感 {feels}°C")
        desc_label.config(text=desc)
        humidity_label.config(text=f"💧 湿度\n{humidity}%")
        wind_label.config(text=f"💨 风速\n{wind} km/h\n{wind_dir}")
    except:
        city_label.config(text="❌ 查询失败")

def on_city_select(event):
    sel = city_listbox.curselection()
    if sel:
        city = city_listbox.get(sel[0])
        entry.delete(0, tk.END)
        entry.insert(0, city)
        search(city)

# ===== 界面布局 =====

# 顶部标题
tk.Label(root, text="🌤️ 天气预报", font=("微软雅黑", 22, "bold"),
         bg="#e8f4fd", fg="#1a5276").pack(pady=(20,5))

# 输入行
input_frame = tk.Frame(root, bg="#e8f4fd")
input_frame.pack(pady=5)

entry = tk.Entry(input_frame, font=("微软雅黑", 14), width=15, justify="center")
entry.pack(side=tk.LEFT, padx=(0,5))
entry.bind("<Return>", lambda e: search())

tk.Button(input_frame, text="🔍 查询", command=search,
          font=("微软雅黑", 12), bg="#3498db", fg="white",
          padx=15, relief="flat").pack(side=tk.LEFT)

# 常用城市列表
tk.Label(root, text="📍 常用城市", font=("微软雅黑", 10),
         bg="#e8f4fd", fg="#7f8c8d").pack(pady=(15,2))

city_listbox = tk.Listbox(root, font=("微软雅黑", 11), height=4,
                          bg="white", relief="solid", bd=1,
                          selectbackground="#3498db")
city_listbox.pack(pady=5)

cities = ["北京", "上海", "广州", "深圳", "杭州", "成都", "Tokyo", "London"]
for c in cities:
    city_listbox.insert(tk.END, c)
city_listbox.bind("<<ListboxSelect>>", on_city_select)

# ==== 天气信息卡片区 ====

# 城市名 + 天气描述
city_label = tk.Label(root, text="请输入城市名",
                      font=("微软雅黑", 18, "bold"),
                      bg="#e8f4fd", fg="#2c3e50")
city_label.pack(pady=(15,5))

desc_label = tk.Label(root, text="",
                      font=("微软雅黑", 14),
                      bg="#e8f4fd", fg="#7f8c8d")
desc_label.pack()

# 温度（大字号）
temp_label = tk.Label(root, text="--°C",
                      font=("微软雅黑", 52, "bold"),
                      bg="#e8f4fd", fg="#e67e22")
temp_label.pack(pady=(5,0))

feels_label = tk.Label(root, text="",
                       font=("微软雅黑", 11),
                       bg="#e8f4fd", fg="#bdc3c7")
feels_label.pack()

# 湿度 + 风速（并排卡片）
cards_frame = tk.Frame(root, bg="#e8f4fd")
cards_frame.pack(pady=20)

# 湿度卡片
humidity_card = tk.Frame(cards_frame, bg="white", relief="solid", bd=1,
                         padx=25, pady=15)
humidity_card.pack(side=tk.LEFT, padx=10)

humidity_label = tk.Label(humidity_card, text="💧 湿度\n--%",
                          font=("微软雅黑", 14), bg="white", fg="#3498db")
humidity_label.pack()

# 风速卡片
wind_card = tk.Frame(cards_frame, bg="white", relief="solid", bd=1,
                     padx=25, pady=15)
wind_card.pack(side=tk.LEFT, padx=10)

wind_label = tk.Label(wind_card, text="💨 风速\n-- km/h",
                      font=("微软雅黑", 14), bg="white", fg="#27ae60")
wind_label.pack()

root.mainloop()
```
