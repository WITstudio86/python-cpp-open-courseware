## 本节课成品

一个能倒计时的番茄钟 ⏲️——输入工作时间（比如 25 分钟），点击开始，就能看到进度环慢慢变小、数字一秒一秒减少，直到归零！

## 知识点讲解

### 1. 输入框 Entry —— 让用户设置时间

`Entry` 是单行文本输入框，用户可以在这里输入数字：

```python
# 创建输入框
entry = tk.Entry(root, font=("Arial", 14), width=5, justify="center")
entry.pack()
entry.insert(0, "25")  # 默认显示 25 分钟

# 获取用户输入的值
minutes = int(entry.get())  # get() 返回字符串，用 int() 转成数字
```

`justify="center"` 让输入的文字居中显示，`width=5` 限制输入框宽度。

### 2. 倒计时逻辑 —— `after()` 每秒减 1

倒计时的核心思路：用一个变量存储剩余的**总秒数**，然后每 1000 毫秒（1 秒）调用一次函数，把秒数减 1：

```python
total_seconds = 0

def start_timer():
    global total_seconds
    minutes = int(entry.get())  # 从输入框获取分钟数
    total_seconds = minutes * 60  # 转换成总秒数
    countdown()  # 开始倒计时

def countdown():
    global total_seconds
    if total_seconds > 0:
        total_seconds -= 1  # 减 1 秒
        update_display()  # 更新界面显示
        root.after(1000, countdown)  # 1 秒后再来一次
    else:
        # 时间到！
        time_label.config(text="时间到！")
```

`after(1000, countdown)` 的意思是：**1000 毫秒之后，调用 `countdown` 函数**。这样每秒调用一次，就实现了倒计时。

### 3. 显示剩余时间 —— MM:SS 格式

把总秒数转换成"分钟:秒钟"的格式：

```python
def update_display():
    minutes = total_seconds // 60  # 整除得到分钟
    seconds = total_seconds % 60   # 取余得到秒钟
    time_str = f"{minutes:02d}:{seconds:02d}"  # 02d 表示两位数，不足补 0
    time_label.config(text=time_str)

    # 同时更新进度环
    update_progress()
```

`:02d` 是格式化的技巧：如果数字不满两位，前面自动补 0。比如 5 秒显示为 `05`。

### 4. Canvas 画弧 —— 制作进度环

`create_arc()` 可以在 Canvas 上画一段弧线，像是一个被咬了一口的甜甜圈 🍩：

```python
# 创建进度环（初始是完整的圆）
progress_arc = canvas.create_arc(50, 50, 250, 250,
                                  start=90, extent=359.9,
                                  outline="#ff6b6b", width=20,
                                  style="arc")

def update_progress():
    """根据剩余时间更新进度环的角度"""
    total = int(entry.get()) * 60  # 总秒数
    progress = total_seconds / total  # 剩余比例（0.0 ~ 1.0）
    extent = 359.9 * progress  # 剩余角度
    canvas.itemconfig(progress_arc, extent=extent)
```

参数说明：
| 参数 | 含义 |
|------|------|
| `50, 50, 250, 250` | 弧的外接矩形（左上角 x,y 到右下角 x,y） |
| `start=90` | 起始角度（90 度 = 12 点钟方向） |
| `extent=359.9` | 弧的角度跨度（接近 360 度 = 完整圆） |
| `outline="#ff6b6b"` | 弧线的颜色 |
| `width=20` | 弧线的粗细 |
| `style="arc"` | 弧线模式（只画弧线，不填色） |

随着 `total_seconds` 减少，`extent` 也会变小，弧就越来越短，形成一个"倒计时环"的效果。

## 完整代码

```python
import tkinter as tk

# ========================================
# 第 39 节：倒计时番茄钟（上）
# ========================================

root = tk.Tk()
root.title("番茄钟 · 倒计时")
root.geometry("420x540")
root.configure(bg="#fafafa")
root.resizable(False, False)

# ---------- 标题 ----------
title = tk.Label(root, text="🍅 番茄钟", font=("Arial", 24, "bold"),
                  bg="#fafafa", fg="#e74c3c")
title.pack(pady=(20, 5))

# ---------- 输入区 ----------
input_frame = tk.Frame(root, bg="#fafafa")
input_frame.pack(pady=10)

tk.Label(input_frame, text="工作时间：", font=("Arial", 13),
         bg="#fafafa").pack(side=tk.LEFT)

entry = tk.Entry(input_frame, font=("Arial", 16), width=4,
                 justify="center", relief="solid", bd=1)
entry.pack(side=tk.LEFT, padx=5)
entry.insert(0, "25")

tk.Label(input_frame, text="分钟", font=("Arial", 13),
         bg="#fafafa").pack(side=tk.LEFT)

# ---------- 进度环画布 ----------
canvas = tk.Canvas(root, width=300, height=300, bg="#fafafa",
                    highlightthickness=0)
canvas.pack(pady=10)

# 底部装饰圆（浅灰色底环）
canvas.create_arc(40, 40, 260, 260, start=90, extent=359.9,
                   outline="#e0e0e0", width=22, style="arc")

# 进度弧（红色，随倒计时减少）
progress_arc = canvas.create_arc(40, 40, 260, 260, start=90, extent=359.9,
                                  outline="#ff6b6b", width=22, style="arc")

# 时间显示（叠在圆心）
time_label = tk.Label(canvas, text="25:00", font=("Arial", 42, "bold"),
                       bg="#fafafa", fg="#333333")
canvas.create_window(150, 150, window=time_label)

# ---------- 状态提示 ----------
status_label = tk.Label(root, text="准备开始", font=("Arial", 12),
                         bg="#fafafa", fg="#888888")
status_label.pack(pady=5)

# ---------- 按钮区 ----------
btn_frame = tk.Frame(root, bg="#fafafa")
btn_frame.pack(pady=10)

def start_timer():
    global total_seconds, running, original_total
    if running:
        return
    try:
        minutes = int(entry.get())
        if minutes <= 0:
            status_label.config(text="请输入大于 0 的数字！", fg="#e74c3c")
            return
    except ValueError:
        status_label.config(text="请输入有效的数字！", fg="#e74c3c")
        return

    total_seconds = minutes * 60
    original_total = total_seconds
    running = True
    status_label.config(text="工作中… 专注！💪", fg="#2ecc71")
    entry.config(state="disabled")  # 计时中禁止修改
    start_btn.config(state="disabled")
    countdown()

def countdown():
    global total_seconds, running
    if total_seconds > 0 and running:
        total_seconds -= 1

        # 更新 MM:SS 显示
        m = total_seconds // 60
        s = total_seconds % 60
        time_label.config(text=f"{m:02d}:{s:02d}")

        # 更新进度弧
        progress = total_seconds / original_total
        canvas.itemconfig(progress_arc, extent=359.9 * progress)

        root.after(1000, countdown)
    elif total_seconds <= 0:
        running = False
        time_label.config(text="00:00")
        canvas.itemconfig(progress_arc, extent=0)
        status_label.config(text="时间到！休息一下吧 🎉", fg="#e74c3c")
        entry.config(state="normal")
        start_btn.config(state="normal")

def reset_timer():
    global total_seconds, running
    running = False
    total_seconds = 0
    time_label.config(text="25:00")
    canvas.itemconfig(progress_arc, extent=359.9)
    status_label.config(text="准备开始", fg="#888888")
    entry.config(state="normal")
    start_btn.config(state="normal")

start_btn = tk.Button(btn_frame, text="▶ 开始计时", font=("Arial", 14, "bold"),
                       bg="#2ecc71", fg="white", width=10,
                       activebackground="#27ae60",
                       command=start_timer)
start_btn.pack(side=tk.LEFT, padx=5)

reset_btn = tk.Button(btn_frame, text="↺ 重置", font=("Arial", 14),
                       bg="#ecf0f1", fg="#333333", width=8,
                       activebackground="#dfe4e5",
                       command=reset_timer)
reset_btn.pack(side=tk.LEFT, padx=5)

# ---------- 全局变量 ----------
total_seconds = 0
original_total = 25 * 60
running = False

root.mainloop()
```

## 小贴士
- ⏱️ `after()` 不是 `sleep()` ——它不会卡住界面，是在"后台"等待，到了时间再执行
- 🍩 `create_arc` 的 `extent` 设为 359.9 而不是 360，因为 360 度会被 tkinter 当成"无弧"（也就是满圆反而什么都不画）
- 📝 `entry.get()` 返回的是**字符串**，要用 `int()` 转换成数字
- 🚫 计时中应该禁用 Entry 和开始按钮，防止用户误操作导致 bug
- 🔄 `after()` 调用自己形成递归——这种模式叫"定时循环"，是 tkinter 动画的核心技巧
