## 知识点讲解

### 回顾上一节
上一节我们学会了用 Pillow 加载和显示一张图片。但那只是一个"单张图片查看器"——真正的电子相册要能翻页、能自动播放！

### 知识点 1：管理多张图片 —— 列表 + 索引

相册里有多张照片，我们用一个列表来存，用一个变量 `current_index` 记录当前显示的是第几张。

```python
# 创建多张不同颜色的占位图片
photos = []  # 存储 Image 对象

colors = ["#E74C3C", "#3498DB", "#2ECC71", "#F39C12", "#9B59B6"]
texts = ["日出 🌅", "大海 🌊", "森林 🌲", "沙漠 🏜️", "花海 🌸"]

for i in range(5):
    img = Image.new("RGB", (450, 300), color=colors[i])
    draw = ImageDraw.Draw(img)
    draw.rectangle([5, 5, 444, 294], outline="white", width=2)
    draw.text((150, 120), texts[i], fill="white")
    draw.text((170, 160), f"第 {i+1} 张", fill="#DDDDDD")
    photos.append(img)

current_index = 0  # 当前显示第几张（从 0 开始）
```

### 知识点 2：翻页功能 —— prev 和 next

```python
def show_photo():
    """根据 current_index 显示对应图片"""
    img = photos[current_index]
    # 重置索引显示文字（避免文字重叠）
    # 将 PIL Image 转为 tk PhotoImage
    tk_img = ImageTk.PhotoImage(img)
    img_label.configure(image=tk_img)
    img_label.image = tk_img
    # 更新页码显示
    page_label.configure(text=f"{current_index + 1} / {len(photos)}")

def next_photo():
    """下一张"""
    global current_index
    current_index = (current_index + 1) % len(photos)  # 循环
    show_photo()

def prev_photo():
    """上一张"""
    global current_index
    current_index = (current_index - 1) % len(photos)  # 循环
    show_photo()
```

**重点讲解**：`% len(photos)` 实现了循环翻页！
- 在最后一张点"下一张"→ 回到第一张
- 在第一张点"上一张"→ 跳到最后一张

### 知识点 3：自动播放 —— after() 定时器

`after(毫秒, 函数)` 是 tkinter 的定时器方法，让指定函数在延迟一定时间后自动执行。

```python
def auto_play():
    """自动播放：每隔 2 秒翻到下一张"""
    next_photo()
    # 2 秒 = 2000 毫秒后再调用自己
    root.after(2000, auto_play)

# 开始自动播放
def start_auto():
    global playing
    playing = True
    auto_play()

# 停止自动播放
def stop_auto():
    global playing
    playing = False
```

**关于 after() 的细节**：
- `after(2000, auto_play)` → 2000 毫秒后调用 `auto_play` 函数
- `auto_play` 调用完 `next_photo()` 后又用 `after` 安排下一次调用 → 形成循环
- 需要停止时，不再调用 `after` 即可

### 知识点 4：缩略图预览

在底部显示一排小缩略图，点击哪个就跳到哪张：

```python
def create_thumbnail(pil_img, size=(80, 60)):
    """生成缩略图（PIL Image → tk PhotoImage）"""
    thumb = pil_img.copy()
    thumb.thumbnail(size)  # 等比缩放
    return ImageTk.PhotoImage(thumb)

# 为每张图片创建缩略图按钮
for i, photo in enumerate(photos):
    thumb = create_thumbnail(photo)
    btn = tk.Button(thumb_frame, image=thumb,
                    command=lambda idx=i: jump_to(idx))
    btn.image = thumb  # 保存引用
    btn.pack(side=tk.LEFT, padx=3)
```

## 完整代码

```python
import tkinter as tk
from PIL import Image, ImageTk, ImageDraw

# ========================================
# 第 28 节：项目 5 — 电子相册（完整版）
# ========================================

root = tk.Tk()
root.title("电子相册 📸")
root.geometry("650x600")
root.configure(bg="#1a1a2e")

# ===== 生成 5 张不同颜色的占位图片 =====
photos = []
colors = ["#E74C3C", "#3498DB", "#2ECC71", "#F39C12", "#9B59B6"]
texts = ["日出 🌅", "大海 🌊", "森林 🌲", "沙漠 🏜️", "花海 🌸"]

for i in range(5):
    img = Image.new("RGB", (480, 320), color=colors[i])
    draw = ImageDraw.Draw(img)
    draw.rectangle([5, 5, 474, 314], outline="white", width=2)
    draw.text((180, 120), texts[i], fill="white")
    draw.text((190, 170), f"Photo {i+1}", fill="#DDDDDD")
    photos.append(img)

current_index = 0
playing = False
play_job = None  # 保存 after 的 job id，用于取消自动播放

# ===== 功能函数 =====

def show_photo():
    """刷新当前图片显示"""
    tk_img = ImageTk.PhotoImage(photos[current_index])
    img_label.configure(image=tk_img)
    img_label.image = tk_img
    page_label.configure(text=f"{current_index + 1} / {len(photos)}")

def next_photo():
    global current_index
    current_index = (current_index + 1) % len(photos)
    show_photo()

def prev_photo():
    global current_index
    current_index = (current_index - 1) % len(photos)
    show_photo()

def jump_to(index):
    """跳到指定索引的图片"""
    global current_index
    current_index = index
    show_photo()

def auto_play():
    """自动播放"""
    global playing, play_job
    if playing:
        next_photo()
        play_job = root.after(2000, auto_play)  # 2 秒后再次调用

def toggle_play():
    """切换自动播放状态"""
    global playing, play_job
    if playing:
        playing = False
        if play_job:
            root.after_cancel(play_job)  # 取消定时任务
        play_btn.configure(text="▶ 自动播放")
    else:
        playing = True
        play_btn.configure(text="⏸ 停止播放")
        auto_play()

# ===== 界面布局 =====

# 标题
tk.Label(root, text="📷 我的电子相册", font=("微软雅黑", 20, "bold"),
         fg="white", bg="#1a1a2e").pack(pady=(15, 5))

# 图片显示区域
tk_img = ImageTk.PhotoImage(photos[0])
img_label = tk.Label(root, image=tk_img, bg="#1a1a2e")
img_label.image = tk_img
img_label.pack(pady=10)

# 页码显示
page_label = tk.Label(root, text=f"1 / {len(photos)}",
                      font=("微软雅黑", 12), fg="#cccccc", bg="#1a1a2e")
page_label.pack()

# 翻页按钮
nav_frame = tk.Frame(root, bg="#1a1a2e")
nav_frame.pack(pady=10)

tk.Button(nav_frame, text="◀ 上一张", command=prev_photo,
          font=("微软雅黑", 12), bg="#16213e", fg="white",
          padx=15).pack(side=tk.LEFT, padx=5)

play_btn = tk.Button(nav_frame, text="▶ 自动播放", command=toggle_play,
                     font=("微软雅黑", 12), bg="#0f3460", fg="white",
                     padx=15)
play_btn.pack(side=tk.LEFT, padx=5)

tk.Button(nav_frame, text="下一张 ▶", command=next_photo,
          font=("微软雅黑", 12), bg="#16213e", fg="white",
          padx=15).pack(side=tk.LEFT, padx=5)

# 缩略图预览条
thumb_frame = tk.Frame(root, bg="#1a1a2e")
thumb_frame.pack(pady=15)

for i, photo in enumerate(photos):
    thumb_img = photo.copy()
    thumb_img.thumbnail((80, 55))
    tk_thumb = ImageTk.PhotoImage(thumb_img)
    btn = tk.Button(thumb_frame, image=tk_thumb,
                    command=lambda idx=i: jump_to(idx),
                    bg="#16213e", relief="flat")
    btn.image = tk_thumb
    btn.pack(side=tk.LEFT, padx=4)

root.mainloop()
```

## 小贴士
- 🔄 `% len(photos)` 是循环翻页的关键，取余数让索引始终在合法范围内
- ⏱️ `after()` 是 tkinter 的定时器，非常适合做轮播、动画、倒计时
- 🖼️ 缩略图用 `thumbnail()` 等比缩放，不会变形
- ⚠️ 每个 Button 里的 `lambda idx=i` 是必须的 —— 如果直接写 `lambda: jump_to(i)`，所有按钮都会跳到同一个值（闭包问题）
