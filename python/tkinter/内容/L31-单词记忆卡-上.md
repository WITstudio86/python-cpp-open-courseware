## 知识点讲解

### 什么是单词记忆卡？
你用过纸质的单词卡片吗？正面写英文，背面写中文。今天我们用 Python 做一个电子版的！

### 知识点 1：字典 dict —— 单词和释义的对应关系

Python 的**字典（dict）**是一种用「键-值对」存储数据的数据结构。非常适合存"单词 → 释义"这种一一对应的关系。

```python
# 字典 = {键: 值, 键: 值, ...}
word_dict = {
    "apple": "苹果 🍎",
    "book": "书 📖",
    "cat": "猫 🐱",
    "dog": "狗 🐕",
    "elephant": "大象 🐘",
    "flower": "花 🌸",
    "garden": "花园 🌻",
    "house": "房子 🏠",
    "ice cream": "冰淇淋 🍦",
    "jungle": "丛林 🌴",
}
```

**字典的基本操作**：
```python
# 查单词
print(word_dict["apple"])  # 输出：苹果 🍎

# 获取所有单词（键）
words = list(word_dict.keys())  # ['apple', 'book', 'cat', ...]

# 获取所有释义（值）
meanings = list(word_dict.values())  # ['苹果 🍎', '书 📖', ...]
```

### 知识点 2：random.choice() —— 随机抽单词

`random` 是 Python 内置的随机数模块。`random.choice(列表)` 从列表中随机返回一个元素。

```python
import random

words = ["apple", "book", "cat", "dog"]
word = random.choice(words)  # 随机选一个，比如 "cat"
```

结合字典使用：
```python
current_word = random.choice(list(word_dict.keys()))
current_meaning = word_dict[current_word]
```

### 知识点 3：显示单词 —— Label 文字更新

用 Label 显示当前单词，用 `config()` 方法动态更新文字：

```python
word_label = tk.Label(root, text="apple", font=("微软雅黑", 32, "bold"))
word_label.pack()

# 换一个单词
word_label.config(text="banana")
```

### 知识点 4：翻面效果 —— 状态切换

用一个变量 `is_front` 记录当前是正面还是背面：

```python
is_front = True  # True = 正面（英文），False = 背面（中文）

def flip():
    """翻面：切换正面/背面"""
    global is_front
    if is_front:
        # 显示背面：中文释义
        word_label.config(text=current_meaning)
        flip_btn.config(text="翻回正面 ↩")
        is_front = False
    else:
        # 显示正面：英文单词
        word_label.config(text=current_word)
        flip_btn.config(text="翻面看释义 🔄")
        is_front = True
```

### 知识点 5：下一张 —— 换单词

换一个单词时，自动翻回正面：

```python
def next_card():
    """切换到下一个随机单词"""
    global current_word, current_meaning, is_front
    current_word = random.choice(list(word_dict.keys()))
    current_meaning = word_dict[current_word]
    is_front = True
    word_label.config(text=current_word)
    flip_btn.config(text="翻面看释义 🔄")
```

## 完整代码

```python
import tkinter as tk
import random

# ========================================
# 第 31 节：项目 7 — 单词记忆卡 v1
# ========================================

root = tk.Tk()
root.title("单词记忆卡 🃏")
root.geometry("450x420")
root.configure(bg="#fff8e7")

# 单词库（字典）
word_dict = {
    "apple": "苹果 🍎",
    "book": "书 📖",
    "cat": "猫 🐱",
    "dog": "狗 🐕",
    "elephant": "大象 🐘",
    "flower": "花 🌸",
    "garden": "花园 🌻",
    "house": "房子 🏠",
    "ice cream": "冰淇淋 🍦",
    "jungle": "丛林 🌴",
    "king": "国王 👑",
    "lion": "狮子 🦁",
    "moon": "月亮 🌙",
    "night": "夜晚 🌃",
    "ocean": "海洋 🌊",
}

# 当前状态
current_word = ""
current_meaning = ""
is_front = True  # True=正面（英文），False=背面（中文）

def next_card():
    """随机抽取一张新卡片"""
    global current_word, current_meaning, is_front
    current_word = random.choice(list(word_dict.keys()))
    current_meaning = word_dict[current_word]
    is_front = True
    word_label.config(text=current_word, fg="#3a2a1a")
    flip_btn.config(text="🔍 翻面看释义")
    status_label.config(text=f"📚 词库共 {len(word_dict)} 个单词")

def flip():
    """翻面切换正/反面"""
    global is_front
    if is_front:
        word_label.config(text=current_meaning, fg="#8b4513")
        flip_btn.config(text="↩ 翻回正面")
        is_front = False
    else:
        word_label.config(text=current_word, fg="#3a2a1a")
        flip_btn.config(text="🔍 翻面看释义")
        is_front = True

# ===== 界面布局 =====

tk.Label(root, text="📖 单词记忆卡", font=("微软雅黑", 22, "bold"),
         bg="#fff8e7", fg="#5a3e2b").pack(pady=(25,10))

# 卡片区域（模拟卡片）
card_frame = tk.Frame(root, bg="#fffef5", relief="solid", bd=2,
                      width=320, height=180)
card_frame.pack(pady=15)
card_frame.pack_propagate(False)  # 不让内容改变 Frame 大小

word_label = tk.Label(card_frame, text="点击下方按钮开始",
                      font=("微软雅黑", 28, "bold"),
                      bg="#fffef5", fg="#3a2a1a")
word_label.pack(expand=True)

# 状态标签
status_label = tk.Label(root, text=f"📚 词库共 {len(word_dict)} 个单词",
                        font=("微软雅黑", 11), bg="#fff8e7", fg="#aaa")
status_label.pack()

# 按钮区
btn_frame = tk.Frame(root, bg="#fff8e7")
btn_frame.pack(pady=20)

flip_btn = tk.Button(btn_frame, text="🔍 翻面看释义", command=flip,
                     font=("微软雅黑", 14), bg="#f0c060",
                     padx=20, pady=8, relief="flat", cursor="hand2")
flip_btn.pack(side=tk.LEFT, padx=10)

tk.Button(btn_frame, text="➡ 下一张", command=next_card,
          font=("微软雅黑", 14), bg="#7d9d6e", fg="white",
          padx=20, pady=8, relief="flat", cursor="hand2").pack(side=tk.LEFT, padx=10)

# 启动时自动抽第一张
next_card()

root.mainloop()
```

## 小贴士
- 📖 字典的 `keys()` 方法获取所有键（单词），`values()` 获取所有值（释义）
- 🎲 `random.choice()` 从列表中随机选一个，非常适合做抽查
- 🔄 用 `global` 在函数里修改外部变量（current_word, is_front 等）
- 🃏 翻面效果 = 一个状态变量 + `config()` 更新文字，简单但巧妙！
