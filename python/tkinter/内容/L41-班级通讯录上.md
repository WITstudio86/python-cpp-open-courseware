## 知识点讲解

### 知识点 1：Treeview —— tkinter 里的表格 📊

`ttk.Treeview` 是 tkinter 的主题扩展（ttk）中的一个控件，可以显示**表格数据**。它就像一个迷你版的 Excel，有行有列，可以选中、滚动。

```python
from tkinter import ttk

# 创建 Treeview 表格
tree = ttk.Treeview(root, columns=("col1", "col2", "col3"), show="headings")
```

| 参数 | 说明 | 示例 |
|------|------|------|
| `columns` | 定义列名（元组），供代码中引用 | `("name", "phone", "qq")` |
| `show` | 显示什么：`"headings"` 只显示表头，`"tree"` 显示树形，`"tree headings"` 都显示 | `show="headings"` |

> 💡 **show="headings"** 是表格模式最常用的设置。如果写成 `show="tree"` 会出现一个展开箭头，适合树形结构（文件夹浏览），但不适合表格。

### 知识点 2：设置表头 heading 和列宽

创建表格后，要用 `heading()` 给每列设置**显示名称**，用 `column()` 设置**列宽**。

```python
# 设置表头文字
tree.heading("name", text="姓名")     # "name" 是列名，"姓名" 是显示的文字
tree.heading("phone", text="电话")
tree.heading("qq", text="QQ")
tree.heading("note", text="备注")

# 设置列宽
tree.column("name", width=100)        # 姓名列 100 像素宽
tree.column("phone", width=130)       # 电话列 130 像素宽
tree.column("qq", width=120)          # QQ 列 120 像素宽
tree.column("note", width=200)        # 备注列 200 像素宽
```

**列名 vs 显示名：**
- `columns=("name", "phone")` 中的 `"name"` 是**内部列名**，代码里用它
- `heading("name", text="姓名")` 中的 `"姓名"` 是**显示文字**，用户在界面上看到的

### 知识点 3：插入数据 insert()

用 `insert()` 方法向表格中添加一行数据：

```python
# 语法：tree.insert(父节点, 位置, values=数据元组)
tree.insert("", tk.END, values=("张三", "13800001111", "12345678", "同桌"))
#           ↑父节点  ↑末尾插入   ↑ 这行的数据内容
```

- 第一个参数 `""`：表示插入到根级别（表格模式都是 `""`）
- 第二个参数 `tk.END`：表示插入到末尾（也可以写 `0` 插到最前面）
- `values=`：一个元组，值的顺序要和 `columns` 定义的顺序一致

```python
# 批量插入示例数据
sample_data = [
    ("张三", "13800001111", "12345678", "同桌"),
    ("李四", "13800002222", "23456789", "班长"),
    ("王五", "13800003333", "34567890", "学习委员"),
]

for item in sample_data:
    tree.insert("", tk.END, values=item)
```

### 知识点 4：多信息录入 —— 用多个 Entry 输入 📝

通讯录需要录入多条信息（姓名、电话、QQ、备注），我们可以创建 4 个 Entry 输入框，配合 Label 说明，排成一个小表单。

```python
# 创建标签和输入框的框架
form_frame = tk.Frame(root)

# 姓名
tk.Label(form_frame, text="姓名：").grid(row=0, column=0)
name_entry = tk.Entry(form_frame, width=15)
name_entry.grid(row=0, column=1)

# 电话
tk.Label(form_frame, text="电话：").grid(row=0, column=2)
phone_entry = tk.Entry(form_frame, width=18)
phone_entry.grid(row=0, column=3)

# QQ
tk.Label(form_frame, text="QQ：").grid(row=1, column=0)
qq_entry = tk.Entry(form_frame, width=15)
qq_entry.grid(row=1, column=1)

# 备注
tk.Label(form_frame, text="备注：").grid(row=1, column=2)
note_entry = tk.Entry(form_frame, width=18)
note_entry.grid(row=1, column=3)
```

然后用一个"添加"按钮，点击时收集所有 Entry 的内容，用 `insert()` 写入 Treeview：

```python
def add_contact():
    """收集输入 → 插入 Treeview → 清空输入框"""
    name = name_entry.get().strip()
    phone = phone_entry.get().strip()
    qq = qq_entry.get().strip()
    note = note_entry.get().strip()

    if name == "":
        return  # 姓名不能为空

    tree.insert("", tk.END, values=(name, phone, qq, note))

    # 清空所有输入框，准备输入下一个
    name_entry.delete(0, tk.END)
    phone_entry.delete(0, tk.END)
    qq_entry.delete(0, tk.END)
    note_entry.delete(0, tk.END)
```

### 知识点 5：删除选中的数据行 🗑️

Treeview 有一个 `selection()` 方法，返回当前选中的行的 ID。拿到 ID 后，用 `delete()` 把它删掉。

```python
def delete_contact():
    """删除 Treeview 中当前选中的行"""
    selected = tree.selection()           # 获取所有选中的行 ID
    if selected:                          # 如果有选中的行
        for item in selected:
            tree.delete(item)             # 逐行删除
```

> ⚠️ 当前版本是"直接删除"，第 42 节课我们会加上确认弹窗，防止手滑误删。

### 知识点 6：Scrollbar 滚动条 📜

当通讯录里的联系人越来越多，表格高度不够用时，需要加一个**滚动条**。Treeview 和 Scrollbar 像一个"最佳拍档"：

```python
# 第一步：创建滚动条
scrollbar = ttk.Scrollbar(root, orient=tk.VERTICAL, command=tree.yview)

# 第二步：把滚动条和 Treeview 关联
tree.configure(yscrollcommand=scrollbar.set)

# 第三步：放置滚动条
scrollbar.pack(side=tk.RIGHT, fill=tk.Y)
```

把它们连起来的逻辑：
- `command=tree.yview`：用户拖滚动条时，Treeview 跟着滚动
- `yscrollcommand=scrollbar.set`：Treeview 内容变化时，滚动条滑块跟着调整大小

`orient=tk.VERTICAL` 是垂直滚动条（上下滚动），`tk.HORIZONTAL` 是水平滚动条（左右滚动）。

## 完整代码

```python
import tkinter as tk
from tkinter import ttk

# ========================================
# 第 41 节：项目 12 —— 班级通讯录（上）
# 知识点：Treeview 表格、列定义、插入数据、多 Entry 录入、Scrollbar
# ========================================

root = tk.Tk()
root.title("班级通讯录")
root.geometry("650x500")
root.configure(bg="#e8f0fe")  # 浅蓝色背景

# ---------- 标题 ----------
title_label = tk.Label(root, text="📒 班级通讯录",
                       font=("微软雅黑", 20, "bold"),
                       bg="#e8f0fe", fg="#1a237e")
title_label.pack(pady=(15, 10))

# ---------- 录入表单区域 ----------
form_frame = tk.Frame(root, bg="#e8f0fe")
form_frame.pack(pady=10)

# 第一行：姓名 + 电话
tk.Label(form_frame, text="姓名：", bg="#e8f0fe",
         font=("微软雅黑", 12)).grid(row=0, column=0, sticky="e", padx=5, pady=3)
name_entry = tk.Entry(form_frame, width=14, font=("微软雅黑", 11))
name_entry.grid(row=0, column=1, padx=5, pady=3)

tk.Label(form_frame, text="电话：", bg="#e8f0fe",
         font=("微软雅黑", 12)).grid(row=0, column=2, sticky="e", padx=5, pady=3)
phone_entry = tk.Entry(form_frame, width=18, font=("微软雅黑", 11))
phone_entry.grid(row=0, column=3, padx=5, pady=3)

# 第二行：QQ + 备注
tk.Label(form_frame, text="QQ：", bg="#e8f0fe",
         font=("微软雅黑", 12)).grid(row=1, column=0, sticky="e", padx=5, pady=3)
qq_entry = tk.Entry(form_frame, width=14, font=("微软雅黑", 11))
qq_entry.grid(row=1, column=1, padx=5, pady=3)

tk.Label(form_frame, text="备注：", bg="#e8f0fe",
         font=("微软雅黑", 12)).grid(row=1, column=2, sticky="e", padx=5, pady=3)
note_entry = tk.Entry(form_frame, width=18, font=("微软雅黑", 11))
note_entry.grid(row=1, column=3, padx=5, pady=3)


def add_contact():
    """收集表单数据，添加到 Treeview 表格中"""
    name = name_entry.get().strip()
    phone = phone_entry.get().strip()
    qq = qq_entry.get().strip()
    note = note_entry.get().strip()

    # 姓名不能为空
    if name == "":
        return

    # 插入到 Treeview
    tree.insert("", tk.END, values=(name, phone, qq, note))

    # 清空输入框
    name_entry.delete(0, tk.END)
    phone_entry.delete(0, tk.END)
    qq_entry.delete(0, tk.END)
    note_entry.delete(0, tk.END)

    # 让姓名输入框重新获得焦点
    name_entry.focus_set()


# 添加按钮
add_btn = tk.Button(form_frame, text="➕ 添加联系人",
                    command=add_contact,
                    font=("微软雅黑", 11, "bold"),
                    bg="#4caf50", fg="white",
                    activebackground="#388e3c",
                    relief="flat", padx=15, pady=5,
                    cursor="hand2")
add_btn.grid(row=0, column=4, rowspan=2, padx=15, sticky="ns")

# ---------- Treeview 表格区域 ----------
# 定义列（内部列名）
columns = ("name", "phone", "qq", "note")

# 创建 Treeview 表格
tree = ttk.Treeview(root, columns=columns, show="headings", height=10)

# 设置表头文字
tree.heading("name", text="姓名")
tree.heading("phone", text="电话")
tree.heading("qq", text="QQ")
tree.heading("note", text="备注")

# 设置列宽
tree.column("name", width=100)
tree.column("phone", width=130)
tree.column("qq", width=120)
tree.column("note", width=180)

# 创建垂直滚动条
scrollbar = ttk.Scrollbar(root, orient=tk.VERTICAL, command=tree.yview)
tree.configure(yscrollcommand=scrollbar.set)

# 放置 Treeview 和滚动条
tree.pack(pady=10, padx=20, fill=tk.BOTH, expand=True, side=tk.LEFT)
scrollbar.pack(side=tk.RIGHT, fill=tk.Y)

# ---------- 预置示例数据 ----------
sample_data = [
    ("张三", "13800001111", "12345678", "同桌"),
    ("李四", "13800002222", "23456789", "班长"),
    ("王五", "13800003333", "34567890", "学习委员"),
    ("赵六", "13800004444", "45678901", "体育委员"),
    ("陈七", "13800005555", "56789012", "文艺委员"),
]

for item in sample_data:
    tree.insert("", tk.END, values=item)

# ---------- 删除按钮 ----------
def delete_contact():
    """删除 Treeview 中当前选中的行"""
    selected = tree.selection()
    if selected:
        for item in selected:
            tree.delete(item)


delete_btn = tk.Button(root, text="🗑️ 删除选中",
                       command=delete_contact,
                       font=("微软雅黑", 11, "bold"),
                       bg="#f44336", fg="white",
                       activebackground="#d32f2f",
                       relief="flat", padx=20, pady=5,
                       cursor="hand2")
delete_btn.pack(pady=(0, 5))

# ---------- 底部提示 ----------
tip = tk.Label(root,
               text="💡 在输入框中填写信息 → 点「添加」→ 选中某行点「删除」",
               font=("微软雅黑", 9),
               bg="#e8f0fe", fg="#666666")
tip.pack(pady=(0, 10))

root.mainloop()
```

## 小贴士
- `ttk.Treeview` 需要从 `tkinter` 导入 `ttk`：`from tkinter import ttk`
- `show="headings"` 是表格模式的关键设置，不要漏掉
- `values=` 元组的顺序必须和 `columns` 定义的顺序**完全一致**，否则数据会对错列
- `tree.selection()` 返回的是选中行的 ID 列表，可以多选，所以用 `for` 循环删除
- Scrollbar 的 `command=tree.yview` 和 Treeview 的 `yscrollcommand=scrollbar.set` 是**互相绑定的**，缺一个滚动条就不能正常工作
