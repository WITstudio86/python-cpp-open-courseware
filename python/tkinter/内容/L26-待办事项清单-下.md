## 知识点讲解

### 回顾上一节
上一节我们用 Listbox + Entry + Button 做出了一个能添加和删除任务的待办清单。但它有两个遗憾：
1. 没法标记"这件事做完了"
2. 关掉程序后，任务就消失了

今天我们来解决这两个问题！

### 知识点 1：标记完成 —— 给已完成任务加删除线

Listbox 有一个 `itemconfig(index, **options)` 方法，可以修改某一项的样式：

```python
# 把第 i 项的前景色改成灰色
listbox.itemconfig(i, fg="gray")

# 把第 i 项的前景色改成灰色，并加删除线
listbox.itemconfig(i, fg="gray")
# 注意：tkinter 的 Listbox 本身不支持删除线,
# 我们用改变前景色 + 在文字前加 "✓" 来标记完成
```

更好的做法：选中的项前面加上 ✓，并且变灰：

```python
def mark_done():
    selection = listbox.curselection()  # 获取选中项的索引元组
    if selection:
        i = selection[0]  # 取第一个选中的索引
        task = listbox.get(i)
        if not task.startswith("✓ "):  # 如果还没被标记
            listbox.delete(i)
            listbox.insert(i, "✓ " + task)
            listbox.itemconfig(i, fg="gray")
```

### 知识点 2：保存到文件 —— open() + write()

把 Listbox 里的所有任务存到 txt 文件中：

```python
def save_tasks():
    with open("tasks.txt", "w", encoding="utf-8") as f:
        # 获取 Listbox 中所有项
        all_tasks = listbox.get(0, tk.END)
        for task in all_tasks:
            f.write(task + "\n")
    print("保存成功！")
```

**重点讲解**：
- `open("tasks.txt", "w")` → `"w"` 是 write 模式，会覆盖原有内容
- `encoding="utf-8"` → 支持中文，一定要加！
- `with ... as f:` → 自动关闭文件，不用手动 f.close()
- `listbox.get(0, tk.END)` → 获取从第 0 项到最后一项的所有内容

### 知识点 3：启动时加载 —— open() + readlines()

程序启动时，自动把文件里的任务读出来，填到 Listbox 里：

```python
def load_tasks():
    try:
        with open("tasks.txt", "r", encoding="utf-8") as f:
            lines = f.readlines()
            for line in lines:
                task = line.strip()  # 去掉末尾的换行符
                if task:  # 跳过空行
                    listbox.insert(tk.END, task)
                    # 如果是已完成的任务，恢复灰色
                    if task.startswith("✓ "):
                        last_idx = listbox.size() - 1
                        listbox.itemconfig(last_idx, fg="gray")
    except FileNotFoundError:
        # 文件不存在就跳过（第一次运行没有文件很正常）
        pass
```

**重点讲解**：
- `try...except FileNotFoundError` → 第一次运行时没有 tasks.txt，不能报错
- `readlines()` → 把文件每一行读成一个列表元素
- `strip()` → 去掉字符串两端的空白字符（包括 `\n` 换行符）

### 知识点 4：窗口关闭时自动保存

我们还可以让程序在关闭窗口时自动保存：

```python
def on_closing():
    save_tasks()
    root.destroy()

root.protocol("WM_DELETE_WINDOW", on_closing)
```

`protocol("WM_DELETE_WINDOW", 函数)` → 拦截窗口关闭事件，先执行我们的函数。

## 完整代码

```python
import tkinter as tk

# ========================================
# 第 26 节：项目 4 — 待办事项清单（完整版）
# ========================================

root = tk.Tk()
root.title("待办事项清单")
root.geometry("500x500")
root.configure(bg="#f5f0e8")

# ===== 功能函数 =====

def add_task():
    """添加新任务"""
    task = entry.get().strip()
    if task:
        listbox.insert(tk.END, task)
        entry.delete(0, tk.END)

def delete_task():
    """删除选中任务"""
    selection = listbox.curselection()
    if selection:
        listbox.delete(selection[0])

def mark_done():
    """标记选中任务为完成"""
    selection = listbox.curselection()
    if selection:
        i = selection[0]
        task = listbox.get(i)
        if not task.startswith("✓ "):
            listbox.delete(i)
            listbox.insert(i, "✓ " + task)
            listbox.itemconfig(i, fg="gray")

def save_tasks():
    """保存所有任务到文件"""
    with open("tasks.txt", "w", encoding="utf-8") as f:
        all_tasks = listbox.get(0, tk.END)
        for task in all_tasks:
            f.write(task + "\n")

def load_tasks():
    """启动时从文件加载任务"""
    try:
        with open("tasks.txt", "r", encoding="utf-8") as f:
            for line in f.readlines():
                task = line.strip()
                if task:
                    listbox.insert(tk.END, task)
                    if task.startswith("✓ "):
                        listbox.itemconfig(listbox.size() - 1, fg="gray")
    except FileNotFoundError:
        pass

def on_closing():
    """关闭窗口时自动保存"""
    save_tasks()
    root.destroy()

# ===== 界面布局 =====

# 标题
title_label = tk.Label(root, text="📝 我的待办清单", 
                       font=("微软雅黑", 20, "bold"),
                       bg="#f5f0e8", fg="#5a4a3a")
title_label.pack(pady=15)

# 输入框 + 添加按钮
input_frame = tk.Frame(root, bg="#f5f0e8")
input_frame.pack(pady=5)

entry = tk.Entry(input_frame, font=("微软雅黑", 14), width=30)
entry.pack(side=tk.LEFT, padx=(0, 5))
entry.bind("<Return>", lambda e: add_task())  # 回车也能添加

add_btn = tk.Button(input_frame, text="➕ 添加", command=add_task,
                    font=("微软雅黑", 12), bg="#7d9d6e", fg="white",
                    padx=10)
add_btn.pack(side=tk.LEFT)

# 任务列表
listbox = tk.Listbox(root, font=("微软雅黑", 13), width=45, height=12,
                     selectmode=tk.SINGLE, bg="#fffaf0",
                     selectbackground="#c8d6af")
listbox.pack(pady=15)

# 操作按钮
btn_frame = tk.Frame(root, bg="#f5f0e8")
btn_frame.pack(pady=10)

tk.Button(btn_frame, text="✅ 标记完成", command=mark_done,
          font=("微软雅黑", 12), bg="#f0c060", padx=10).pack(side=tk.LEFT, padx=5)

tk.Button(btn_frame, text="🗑️ 删除", command=delete_task,
          font=("微软雅黑", 12), bg="#e07060", fg="white", padx=10).pack(side=tk.LEFT, padx=5)

tk.Button(btn_frame, text="💾 保存", command=save_tasks,
          font=("微软雅黑", 12), bg="#60a0c0", fg="white", padx=10).pack(side=tk.LEFT, padx=5)

# 启动时加载已有任务
load_tasks()

# 关闭窗口时自动保存
root.protocol("WM_DELETE_WINDOW", on_closing)

root.mainloop()
```

## 小贴士
- 🗂️ 文件保存在程序运行的同一目录下
- 📝 编码一定要写 `encoding="utf-8"`，否则中文会乱码
- ⚠️ `try...except` 是处理"文件不存在"的最佳方式，不要用 if 判断文件是否存在
- 🔄 `protocol("WM_DELETE_WINDOW", ...)` 能让你在窗口关闭前做一些事情（比如保存）
- 💡 现在你已经能做出一个有"持久存储"的程序了！数据不会因为关掉程序就消失
