## 知识点讲解

### 知识点 1：复习控件全家福 📋

在动手之前，我们先快速回顾一下阶段一学过的所有控件：

| 控件 | 中文名 | 一句话总结 |
|------|--------|-----------|
| `Label` | 标签 | 显示文字或图片，最基础的控件 |
| `Entry` | 输入框 | 让用户输入单行文字 |
| `Radiobutton` | 单选按钮 | 多个选项中只能选一个 |
| `Checkbutton` | 多选按钮 | 多个选项中可以选多个（或不选） |
| `Combobox` | 下拉框 | 点击展开列表，选一个 |
| `Button` | 按钮 | 点击触发动作，做事情 |

```python
# 各控件速览示例
import tkinter as tk
from tkinter import ttk

root = tk.Tk()
root.title("控件全家福")

# Label - 显示文字
tk.Label(root, text="我是标签 Label 👋").pack()

# Entry - 输入框
tk.Entry(root).pack()

# Radiobutton - 单选（需要用变量绑定）
gender_var = tk.StringVar(value="男")
tk.Radiobutton(root, text="男", variable=gender_var, value="男").pack()
tk.Radiobutton(root, text="女", variable=gender_var, value="女").pack()

# Checkbutton - 多选（每个需要自己的变量）
code_var = tk.BooleanVar()
tk.Checkbutton(root, text="编程 💻", variable=code_var).pack()

# Combobox - 下拉框
grade_combo = ttk.Combobox(root, values=["三年级", "四年级", "五年级", "六年级"])
grade_combo.pack()

# Button - 按钮
tk.Button(root, text="点我！").pack()

root.mainloop()
```

### 知识点 2：用变量绑定控件的值 🔗

要把控件的值「拿出来用」，我们需要给控件绑定一个**变量**：

| 变量类型 | 配合使用的控件 | 说明 |
|---------|-------------|------|
| `tk.StringVar()` | Entry、Radiobutton、Combobox | 存文字 |
| `tk.IntVar()` | Radiobutton | 存整数（男=1，女=0 这样） |
| `tk.BooleanVar()` | Checkbutton | 存 True/False |

```python
# 变量绑定示例
name_var = tk.StringVar()        # 姓名用 StringVar
gender_var = tk.IntVar(value=1)  # 性别用 IntVar，1=男，0=女
code_var = tk.BooleanVar()       # 是否喜欢编程

# Entry 绑定
tk.Entry(root, textvariable=name_var)

# Radiobutton 绑定同一个变量
tk.Radiobutton(root, text="男 👦", variable=gender_var, value=1)
tk.Radiobutton(root, text="女 👧", variable=gender_var, value=0)

# Checkbutton 绑定各自变量
tk.Checkbutton(root, text="编程 💻", variable=code_var)

# 读取值
print(name_var.get())   # → 输入框里的文字
print(gender_var.get()) # → 1 或 0
print(code_var.get())   # → True 或 False
```

### 知识点 3：用 .get() 读取所有值并汇总 📝

这是本节课的核心！点击「提交」按钮后，我们要：

1. 用每个变量的 `.get()` 方法读出用户填的内容
2. 把读到的内容拼成一段漂亮的文字
3. 更新到一个 Label 上显示出来

```python
def submit():
    """读取所有控件的值，汇总显示"""
    # 1. 读取姓名
    name = name_var.get()
    
    # 2. 读取性别（IntVar 的 1/0 转换成 男/女）
    gender = "男 👦" if gender_var.get() == 1 else "女 👧"
    
    # 3. 读取爱好（检查每个 BooleanVar）
    hobbies = []
    if code_var.get():
        hobbies.append("编程 💻")
    if draw_var.get():
        hobbies.append("画画 🎨")
    if sport_var.get():
        hobbies.append("运动 ⚽")
    if music_var.get():
        hobbies.append("音乐 🎵")
    hobby_text = "、".join(hobbies) if hobbies else "暂无"
    
    # 4. 读取年级
    grade = grade_var.get()
    
    # 5. 拼成汇总文字
    summary = f"📋 个人信息汇总\n\n"
    summary += f"👤 姓名：{name}\n"
    summary += f"🚻 性别：{gender}\n"
    summary += f"🎯 爱好：{hobby_text}\n"
    summary += f"📚 年级：{grade}\n"
    
    # 6. 更新到结果 Label
    result_label.config(text=summary)
```

## 完整代码

```python
"""
第 18 节 · 阶段综合 —— 个人信息小管家
阶段一收官之作，综合运用 Label、Entry、Radiobutton、Checkbutton、Combobox、Button
"""

import tkinter as tk
from tkinter import ttk


def submit_info():
    """点击「提交」按钮后：读取所有控件的值，汇总到结果标签"""
    # 1. 读取姓名
    name = name_var.get().strip()
    if not name:
        result_label.config(text="⚠️ 请先输入姓名再提交哦～")
        return

    # 2. 读取性别：IntVar 的 1 = 男，0 = 女
    gender = "男 👦" if gender_var.get() == 1 else "女 👧"

    # 3. 读取爱好：检查每个 BooleanVar 是否为 True
    hobbies = []
    if hobby_code_var.get():
        hobbies.append("编程 💻")
    if hobby_draw_var.get():
        hobbies.append("画画 🎨")
    if hobby_sport_var.get():
        hobbies.append("运动 ⚽")
    if hobby_music_var.get():
        hobbies.append("音乐 🎵")
    hobby_text = "、".join(hobbies) if hobbies else "暂无爱好"

    # 4. 读取年级：Combobox 的 StringVar
    grade = grade_var.get()
    if not grade:
        grade = "未选择"

    # 5. 拼成一段美观的汇总信息
    summary = f"""📋━━━ 个人信息汇总 ━━━📋

👤 姓名：{name}
🚻 性别：{gender}
🎯 爱好：{hobby_text}
📚 年级：{grade}

━━━━━━━━━━━━━━━━━━━━━━"""

    # 6. 更新结果标签
    result_label.config(text=summary)


def clear_form():
    """清空表单，恢复初始状态"""
    name_var.set("")
    gender_var.set(1)
    hobby_code_var.set(False)
    hobby_draw_var.set(False)
    hobby_sport_var.set(False)
    hobby_music_var.set(False)
    grade_var.set("")
    result_label.config(text="📝 请填写上方信息，然后点击「提交」按钮～")


# ==================== 创建窗口 ====================
root = tk.Tk()
root.title("个人信息小管家 📋")
root.geometry("550x580")
root.resizable(False, False)

# ==================== 定义变量 ====================
name_var = tk.StringVar()           # 姓名
gender_var = tk.IntVar(value=1)     # 性别：1=男, 0=女
hobby_code_var = tk.BooleanVar()    # 爱好：编程
hobby_draw_var = tk.BooleanVar()    # 爱好：画画
hobby_sport_var = tk.BooleanVar()   # 爱好：运动
hobby_music_var = tk.BooleanVar()   # 爱好：音乐
grade_var = tk.StringVar()          # 年级

# ==================== 标题区域 ====================
title_label = tk.Label(
    root,
    text="📋 个人信息小管家",
    font=("Microsoft YaHei", 20, "bold"),
    fg="#4A90D9"
)
title_label.pack(pady=15)

subtitle_label = tk.Label(
    root,
    text="阶段一收官之作 · 控件全家福实战 🎉",
    font=("Microsoft YaHei", 11),
    fg="#888888"
)
subtitle_label.pack(pady=(0, 15))

# ==================== 表单区域 ====================
form_frame = tk.LabelFrame(root, text=" 📝 请填写你的信息 ", font=("Microsoft YaHei", 12, "bold"),
                           fg="#333333", padx=15, pady=10)
form_frame.pack(padx=30, pady=5, fill="x")

# --- 姓名 ---
tk.Label(form_frame, text="👤 姓名：", font=("Microsoft YaHei", 11)).grid(
    row=0, column=0, sticky="w", pady=8, padx=(0, 10))
entry_name = tk.Entry(form_frame, textvariable=name_var, font=("Microsoft YaHei", 11),
                      width=20)
entry_name.grid(row=0, column=1, sticky="w", pady=8)

# --- 性别 ---
tk.Label(form_frame, text="🚻 性别：", font=("Microsoft YaHei", 11)).grid(
    row=1, column=0, sticky="w", pady=8, padx=(0, 10))
gender_frame = tk.Frame(form_frame)
gender_frame.grid(row=1, column=1, sticky="w", pady=8)
tk.Radiobutton(gender_frame, text="男 👦", variable=gender_var, value=1,
               font=("Microsoft YaHei", 11)).pack(side="left", padx=(0, 20))
tk.Radiobutton(gender_frame, text="女 👧", variable=gender_var, value=0,
               font=("Microsoft YaHei", 11)).pack(side="left")

# --- 爱好 ---
tk.Label(form_frame, text="🎯 爱好：", font=("Microsoft YaHei", 11)).grid(
    row=2, column=0, sticky="w", pady=8, padx=(0, 10))
hobby_frame = tk.Frame(form_frame)
hobby_frame.grid(row=2, column=1, sticky="w", pady=8)
tk.Checkbutton(hobby_frame, text="编程 💻", variable=hobby_code_var,
               font=("Microsoft YaHei", 10)).grid(row=0, column=0, sticky="w", padx=(0, 15))
tk.Checkbutton(hobby_frame, text="画画 🎨", variable=hobby_draw_var,
               font=("Microsoft YaHei", 10)).grid(row=0, column=1, sticky="w", padx=(0, 15))
tk.Checkbutton(hobby_frame, text="运动 ⚽", variable=hobby_sport_var,
               font=("Microsoft YaHei", 10)).grid(row=1, column=0, sticky="w", padx=(0, 15))
tk.Checkbutton(hobby_frame, text="音乐 🎵", variable=hobby_music_var,
               font=("Microsoft YaHei", 10)).grid(row=1, column=1, sticky="w", padx=(0, 15))

# --- 年级 ---
tk.Label(form_frame, text="📚 年级：", font=("Microsoft YaHei", 11)).grid(
    row=3, column=0, sticky="w", pady=8, padx=(0, 10))
grade_combo = ttk.Combobox(form_frame, textvariable=grade_var, font=("Microsoft YaHei", 11),
                           values=["三年级 📗", "四年级 📘", "五年级 📙", "六年级 📕"],
                           state="readonly", width=18)
grade_combo.grid(row=3, column=1, sticky="w", pady=8)

# ==================== 按钮区域 ====================
button_frame = tk.Frame(root)
button_frame.pack(pady=15)

submit_btn = tk.Button(button_frame, text="✅ 提交信息", command=submit_info,
                       font=("Microsoft YaHei", 12, "bold"),
                       bg="#4A90D9", fg="white", activebackground="#357ABD",
                       activeforeground="white", width=12, height=1,
                       cursor="hand2")
submit_btn.pack(side="left", padx=10)

clear_btn = tk.Button(button_frame, text="🔄 清空重填", command=clear_form,
                      font=("Microsoft YaHei", 12, "bold"),
                      bg="#E8E8E8", fg="#333333", activebackground="#D0D0D0",
                      width=12, height=1, cursor="hand2")
clear_btn.pack(side="left", padx=10)

# ==================== 结果展示区域 ====================
result_frame = tk.LabelFrame(root, text=" 📊 信息汇总 ", font=("Microsoft YaHei", 12, "bold"),
                             fg="#333333", padx=15, pady=10)
result_frame.pack(padx=30, pady=5, fill="both", expand=True)

result_label = tk.Label(
    result_frame,
    text="📝 请填写上方信息，然后点击「提交」按钮～",
    font=("Microsoft YaHei", 12),
    fg="#555555",
    justify="left",
    anchor="w"
)
result_label.pack(anchor="w", expand=True)

# ==================== 底部信息 ====================
footer_label = tk.Label(
    root,
    text="🎓 阶段一 · 完美收官！ 下一站：阶段二 · 电子时钟项目 ⏰",
    font=("Microsoft YaHei", 10),
    fg="#AAAAAA"
)
footer_label.pack(pady=(5, 10))

# ==================== 启动主循环 ====================
root.mainloop()
```

## 小贴士

- 🧵 **变量绑定是桥梁**：`textvariable` 和 `variable` 负责把控件和 Python 变量连在一起，`.get()` 负责取出来，`.set()` 负责放进去。
- 🔑 **记得 import ttk**：`Combobox` 在 `tkinter.ttk` 里，别忘了 `from tkinter import ttk`！
- 📐 **布局规划**：做复杂界面时，先用 `Frame` 把不同区域分组，再用 `grid` 或 `pack` 在里面布局，结构更清晰。
- ✅ **善用 readonly**：给 Combobox 加上 `state="readonly"`，可以防止用户乱输入，只能从列表里选。
- 🎨 **善用 emoji**：Label 里加 emoji 能让界面更活泼，小朋友更喜欢！
