import tkinter as tk
from tkinter import messagebox

# ========================================
# 第 10 节：趣味问答小测验
# 知识点：grid 布局 + Radiobutton + Button + messagebox
# ========================================

# 1. 创建主窗口
root = tk.Tk()
root.title("趣味问答小测验")           # 设置窗口标题
root.geometry("500x450")               # 设置窗口大小：宽500，高450
root.configure(bg="#f5f0ff")           # 设置背景色：浅紫色

# 2. 准备题目数据
#    使用列表嵌套字典存储题目信息
#    每道题包含：题目文字(question)、4个选项(options)、正确答案(answer)
questions = [
    {
        "question": "1. Python 中创建 tkinter 窗口使用什么方法？",
        "options": ["A. Tk()", "B. Window()", "C. Screen()", "D. Frame()"],
        "answer": "A"          # 正确答案是 A
    },
    {
        "question": "2. 下面哪个方法用来设置窗口标题？",
        "options": ["A. name()", "B. title()", "C. header()", "D. label()"],
        "answer": "B"          # 正确答案是 B
    },
    {
        "question": "3. mainloop() 的作用是什么？",
        "options": ["A. 关闭窗口", "B. 设置窗口大小", "C. 让窗口持续显示", "D. 创建按钮"],
        "answer": "C"          # 正确答案是 C
    }
]

# 3. 创建一个列表，用来存放每道题对应的 StringVar（记录用户选择）
vars_list = []

# 4. 使用 grid 布局逐题放置控件
#    每道题占 2 行：题目文字 1 行，选项按钮 1 行
for i, q in enumerate(questions):
    # 4a. 显示题目文字（跨越 4 列，因为每题有 4 个选项）
    q_label = tk.Label(
        root,
        text=q["question"],
        font=("Microsoft YaHei", 13, "bold"),  # 字体设置
        bg="#f5f0ff",                           # 背景色与窗口一致
        fg="#333333",                           # 文字颜色
        anchor="w",                             # 文字左对齐
        justify="left"
    )
    q_label.grid(
        row=i * 2, column=0, columnspan=4,     # 跨4列显示
        sticky="w", padx=30, pady=(15, 5)      # 左对齐，外边距
    )

    # 4b. 为这道题创建一个 StringVar，用来记录用户选择了哪个选项
    var = tk.StringVar(value="")  # 初始值为空字符串，表示未选择
    vars_list.append(var)          # 存入列表，后面评分时会用到

    # 4c. 创建 4 个 Radiobutton（单选按钮），每个对应一个选项
    for j, option in enumerate(q["options"]):
        rb = tk.Radiobutton(
            root,
            text=option,                # 显示的选项文字（如 "A. Tk()"）
            variable=var,               # 同一道题的选项共用同一个变量，实现互斥
            value=option[0],            # 选中时 var 的值设为选项首字母（A/B/C/D）
            font=("Microsoft YaHei", 11),
            bg="#f5f0ff",               # 背景色
            fg="#555555",               # 文字颜色
            selectcolor="#d5c8f0",     # 选中时的圆点颜色
            anchor="w"                  # 文字左对齐
        )
        rb.grid(
            row=i * 2 + 1, column=j,
            sticky="w",
            padx=(30 if j == 0 else 5, 5),  # 第一个选项左距大一些
            pady=5
        )


# 5. 提交按钮的回调函数：检查答案并计算得分
def submit():
    """遍历每道题，比对用户选择和正确答案，计算得分"""
    score = 0
    for i, q in enumerate(questions):
        # vars_list[i].get() 获取用户在第 i 题选的值
        if vars_list[i].get() == q["answer"]:
            score += 1

    total = len(questions)

    # 根据得分显示不同的评语
    if score == total:
        comment = "太棒了，全部正确！🎉"
    else:
        comment = f"不错哦，对了 {score} 题，继续加油！💪"

    # 使用 messagebox 弹出成绩窗口
    messagebox.showinfo(
        "测验结果",
        f"你答对了 {score} / {total} 题！\n\n{comment}"
    )


# 6. 创建"提交答案"按钮
submit_btn = tk.Button(
    root,
    text="📝 提交答案",
    command=submit,                      # 点击时调用 submit 函数
    font=("Microsoft YaHei", 14, "bold"),
    bg="#7c4dff",                        # 按钮背景色：紫色
    fg="white",                          # 按钮文字颜色：白色
    activebackground="#651fff",          # 鼠标按下时的背景色
    relief="flat",                       # 扁平样式（无边框凸起）
    padx=30, pady=8,                     # 内边距
    cursor="hand2"                       # 鼠标悬停时显示手指光标
)
# 按钮放在所有题目下方，跨 4 列居中
submit_btn.grid(
    row=len(questions) * 2, column=0, columnspan=4,
    pady=30
)

# 7. 底部提示文字
tip = tk.Label(
    root,
    text="💡 选好答案后点击「提交答案」查看得分",
    font=("Microsoft YaHei", 10),
    bg="#f5f0ff",
    fg="#888888"
)
tip.grid(
    row=len(questions) * 2 + 1, column=0, columnspan=4,
    pady=(0, 10)
)

# 8. 启动主循环（必须放在代码最后！）
root.mainloop()
